"""Sampled metadata-only RSS watchdog. No subprocesses, rlimits or VSize cap.

Not a continuous physical-memory theorem. Cgroup controls remain authoritative.
The one persistent thread exists before and throughout both collector snapshots.
"""
import os
import resource
import signal
import threading
import time

BUDGET = 512 * 2**20
INTERVAL = 0.02
MAX_SAMPLES = 8192
MAX_LINEAGE = 256

class GuardError(RuntimeError):
    pass


def inherited_limits():
    return {name:list(resource.getrlimit(getattr(resource, name))) for name in
            sorted(dir(resource)) if name.startswith('RLIMIT_')}


def check_limits(limits, expected=None):
    if expected is not None and limits != expected:
        raise GuardError('inherited resource limits changed; no bypass')
    for name in ('RLIMIT_AS', 'RLIMIT_DATA'):
        if limits.get(name) != [resource.RLIM_INFINITY, resource.RLIM_INFINITY]:
            raise GuardError('unexpected inherited restriction: '+name)


class Proc:
    """Direct procfs only; status TGID and stat identity/RSS, never thread sum."""
    def read(self, path):
        with open(path) as stream:
            return stream.read()

    def identity(self, pid):
        text = self.read(f'/proc/{pid}/stat')
        leader, tail = text.split(' (', 1)
        fields = tail.rsplit(') ', 1)[1].split()
        status = dict(line.split(':', 1) for line in self.read(f'/proc/{pid}/status').splitlines() if ':' in line)
        if int(leader) != pid or int(status['Tgid']) != pid or int(status['Pid']) != pid:
            raise GuardError('unknown non-TGID identity')
        # stat field22=starttime,23=vsize,24=rss; state starts field3.
        result = dict(pid=pid, ppid=int(fields[1]), starttime=int(fields[19]),
                      rss_bytes=int(fields[21])*os.sysconf('SC_PAGE_SIZE'))
        again = self.read(f'/proc/{pid}/stat').rsplit(') ', 1)[1].split()
        if int(again[19]) != result['starttime']:
            raise GuardError('PID reuse during proc observation')
        if result['rss_bytes'] < 0 or result['starttime'] <= 0:
            raise GuardError('invalid proc identity/RSS')
        return result

    def absent(self, path):
        try:
            os.stat(path)
        except FileNotFoundError:
            return True
        return False

    def optional(self, pid):
        try:
            return self.identity(pid)
        except FileNotFoundError:
            if self.absent(f'/proc/{pid}'):
                return None
            raise GuardError('unknown live-child visibility')

    def children(self, pid):
        result = set()
        for tid in os.listdir(f'/proc/{pid}/task'):
            if not tid.isdecimal():
                raise GuardError('invalid task identity')
            try:
                raw = self.read(f'/proc/{pid}/task/{tid}/children')
            except FileNotFoundError:
                if self.absent(f'/proc/{pid}/task/{tid}'):
                    continue
                raise GuardError('unknown live task children')
            for child in raw.split():
                if not child.isdecimal() or int(child) <= 0:
                    raise GuardError('invalid child identity')
                result.add(int(child))
        return result

    def cancel(self, pid, starttime):
        # Open the stable handle BEFORE rechecking identity; reuse never signalled.
        try:
            fd = os.pidfd_open(pid)
        except ProcessLookupError:
            return 'disappeared'
        try:
            observed = self.optional(pid)
            if observed is None:
                return 'disappeared'
            if observed['starttime'] != starttime:
                return 'reused-not-signalled'
            signal.pidfd_send_signal(fd, signal.SIGKILL)
            return 'verified-owned-signalled'
        finally:
            os.close(fd)


class Guard:
    def __init__(self, proc=None, pid=None, notify=None):
        self.proc = proc or Proc()
        self.pid = os.getpid() if pid is None else pid
        self.notify = notify or (lambda: os.kill(self.pid, signal.SIGUSR1))
        self.stop_event = threading.Event()
        self.ready = threading.Event()
        self.samples = []
        self.owned = {}
        self.lineage = []
        self.peak = 0
        self.error = None
        self.cancelled = []
        self.thread = None
        self.started = None
        self.stopped = None

    def remember(self, row, parent):
        pid = row['pid']
        if pid in self.owned:
            if self.owned[pid] != row['starttime']:
                raise GuardError('PID reuse: '+str(pid))
        else:
            if len(self.owned) >= MAX_LINEAGE:
                raise GuardError('lineage record limit')
            self.owned[pid] = row['starttime']
            self.lineage.append(dict(pid=pid, starttime=row['starttime'], parent=parent))

    def sample(self):
        live = {}
        pending = [(self.pid, None)] + [(p, 'retained-descendant') for p in self.owned if p != self.pid]
        while pending:
            pid, parent = pending.pop()
            if pid in live:
                continue
            row = self.proc.optional(pid)
            if row is None:
                if pid == self.pid:
                    raise GuardError('collector disappeared')
                continue
            if pid not in self.owned and parent is not None and row['ppid'] != parent:
                raise GuardError('unknown live-child lineage')
            self.remember(row, parent)
            live[pid] = row
            try:
                children = self.proc.children(pid)
            except FileNotFoundError:
                if self.proc.optional(pid) is not None:
                    raise GuardError('unknown live-child visibility')
                children = set()
            pending.extend((child, pid) for child in children)
        total = sum(row['rss_bytes'] for row in live.values())
        self.peak = max(self.peak, total)
        if len(self.samples) >= MAX_SAMPLES:
            raise GuardError('sample record limit')
        self.samples.append(dict(monotonic=time.monotonic(),rss_bytes=total,
                                 processes=sorted(live),process_rss=live))
        if total > BUDGET:
            raise GuardError('sampled aggregate metadata RSS exceeds 512MiB')

    def check(self):
        if self.error:
            raise GuardError(self.error)

    def cancel_owned(self):
        for pid, start in list(self.owned.items()):
            if pid == self.pid:
                continue
            try:
                outcome = self.proc.cancel(pid, start)
            except (OSError, ValueError, GuardError) as exc:
                outcome = 'cancellation-error: '+str(exc)
            self.cancelled.append(dict(pid=pid,starttime=start,outcome=outcome))

    def loop(self):
        try:
            while True:
                self.sample()
                self.ready.set()
                if self.stop_event.wait(INTERVAL):
                    self.sample()  # final observation before monitor exit
                    break
        except Exception as exc:
            self.error = type(exc).__name__+': '+str(exc)
            self.cancel_owned()
            self.ready.set()
            self.notify()

    def start(self):
        self.started = time.monotonic()
        self.thread = threading.Thread(target=self.loop, name='ResourceMeta1G-resident-guard', daemon=True)
        self.thread.start()
        self.ready.wait(2)
        if not self.ready.is_set():
            raise GuardError('monitor failed to become ready')
        self.check()

    def finish(self, failed=False):
        self.stop_event.set()
        self.thread.join(2)
        if self.thread.is_alive():
            raise GuardError('monitor failed to stop')
        self.stopped = time.monotonic()
        if failed:
            self.cancel_owned()
        return dict(kind='sampled aggregate PROCESS RSS; TGID once; metadata only; not continuous',
                    budget_bytes=BUDGET,interval_seconds=INTERVAL,max_samples=MAX_SAMPLES,
                    max_lineage=MAX_LINEAGE,peak_sampled_rss_bytes=self.peak,
                    samples=self.samples,lineage=self.lineage,error=self.error,
                    cancellations=self.cancelled,started=self.started,stopped=self.stopped,
                    cgroup_peak='separate authoritative go-controls receipts; N/A if no scope')

"""Pure-testable finite metadata operations. No traversal of unlisted files.

New implementation, not a byte-identical claim about the old walker. All
filesystem operations are injectable; tests supply an entirely synthetic FS.
"""
import os
import stat
import hashlib

PUBLIC = ('/usr/bin', '/usr/lib', '/usr/local/lib')
PREFIX = ('/lib', '/lib64', '/usr/lib64')
SSL = ('/usr/lib/ssl/openssl.cnf', '/etc/ssl/openssl.cnf')
CONFIG = ('/etc/ld.so.cache', '/etc/ld.so.preload', '/etc/localtime') + SSL
DIR = os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC
FILE = os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC | os.O_NONBLOCK


def under(path, roots):
    return any(path == r or path.startswith(r + '/') for r in roots)


def lexical(path):
    if not isinstance(path, str) or not path.startswith('/') or len(path) >= 4096 or not path.isascii():
        raise ValueError('path syntax')
    if any(p in ('', '.', '..', 'site-packages', 'dist-packages') or p.startswith('.') or
           any(ord(c) < 33 or ord(c) > 126 for c in p) for p in path.split('/')[1:]):
        raise ValueError('path components')
    return path


def permitted(original, endpoint, final=False):
    # SSL is an exact two-endpoint grammar, not an /etc/ssl subtree grant.
    if original in SSL:
        return endpoint in SSL
    if original == '/etc/localtime':
        return endpoint == original or under(endpoint, ('/usr/share/zoneinfo',))
    if original in ('/etc/ld.so.cache', '/etc/ld.so.preload'):
        return endpoint == original  # no config redirect into arbitrary public data
    return under(endpoint, PUBLIC if final else PUBLIC + PREFIX)


def identity(s):
    return [getattr(s, k) for k in ('st_dev', 'st_ino', 'st_mode', 'st_uid', 'st_gid',
            'st_nlink', 'st_size', 'st_mtime_ns', 'st_ctime_ns')]


def owned(s, symlink=False):
    if s.st_uid != 0 or s.st_gid != 0 or (not symlink and s.st_mode & 0o022):
        raise ValueError('ownership/mode')
    if symlink and s.st_nlink != 1:
        raise ValueError('symlink link count')


def acquire(path, attempts, fs, limit=16*2**20, check=lambda: None, admit=lambda physical, s: None):
    """Return bytes/metadata after checking every held FD and named entry.

    Directories stay open across link restarts and through the final read.
    Rechecks detect observed races; root/kernel adversaries remain outside the
    inherited boundary. This does not prove historical open-to-inode identity.
    """
    row = dict(path=path, status='ATTEMPT', chain=[], symlinks=[])
    attempts.append(row)
    fds, entries, descriptors = [], [], []
    try:
        lexical(path)
        if not (under(path, PUBLIC + PREFIX) or path in CONFIG):
            raise ValueError('outside explicit paths')
        todo, done, seen = path.split('/')[1:], [], set()
        def root():
            fd = fs.open('/', DIR)
            fds.append(fd)
            s = fs.fstat(fd)
            owned(s)
            if not stat.S_ISDIR(s.st_mode):
                raise ValueError('root type')
            descriptors.append((fd, identity(s)))
            row['chain'].append(dict(path='/', identity=identity(s)))
            return fd
        parent = root()
        while todo:
            check()
            name = todo.pop(0)
            current = '/' + '/'.join(done + [name])
            row['failed_component'] = current
            s = fs.stat(name, dir_fd=parent, follow_symlinks=False)
            link = stat.S_ISLNK(s.st_mode)
            owned(s, link)
            entries.append((parent, name, identity(s)))
            item = dict(path=current, identity=identity(s))
            row['chain'].append(item)
            if link:
                target = fs.readlink(name, dir_fd=parent)
                item['target'] = target
                row['symlinks'].append(item.copy())
                if identity(s) != identity(fs.stat(name, dir_fd=parent, follow_symlinks=False)):
                    raise ValueError('symlink drift')
                state = (current, tuple(todo))
                if state in seen or len(row['symlinks']) > 16:
                    raise ValueError('symlink loop/depth')
                seen.add(state)
                combined = target if target.startswith('/') else '/' + '/'.join(done + [target])
                resolved = os.path.normpath(combined + ('/' + '/'.join(todo) if todo else ''))
                lexical(resolved)
                if not permitted(path, resolved):
                    raise ValueError('symlink endpoint outside exact grammar: ' + resolved)
                todo, done, parent = resolved.split('/')[1:], [], root()
                continue
            if todo and not stat.S_ISDIR(s.st_mode):
                raise ValueError('non-directory ancestor')
            if not todo and not permitted(path, current, final=True):
                raise ValueError('final endpoint outside exact grammar')
            if not todo and not (stat.S_ISDIR(s.st_mode) or stat.S_ISREG(s.st_mode)):
                raise ValueError('unsupported endpoint')
            fd = fs.open(name, DIR if stat.S_ISDIR(s.st_mode) else FILE, dir_fd=parent)
            fds.append(fd)
            if identity(s) != identity(fs.fstat(fd)):
                raise ValueError('entry/FD drift')
            descriptors.append((fd, identity(s)))
            if todo:
                done.append(name)
                parent = fd
                continue
            raw = None
            if stat.S_ISREG(s.st_mode):
                if s.st_nlink != 1 or s.st_size < 0 or s.st_size > limit:
                    raise ValueError('regular link/size bound')
                admit(current, s)  # bound unique files/bytes BEFORE reading/hashing
                chunks, count = [], 0
                while True:
                    check()
                    part = fs.read(fd, min(65536, s.st_size + 1 - count))
                    if not part:
                        break
                    count += len(part)
                    if count > s.st_size:
                        raise ValueError('file grew')
                    chunks.append(part)
                if count != s.st_size:
                    raise ValueError('file shrank')
                raw = b''.join(chunks)
            for d, before in descriptors:
                if before != identity(fs.fstat(d)):
                    raise ValueError('descriptor drift after read')
            for d, n, before in entries:
                if before != identity(fs.stat(n, dir_fd=d, follow_symlinks=False)):
                    raise ValueError('chain entry drift after read')
            row.update(status='DIRECTORY' if raw is None else 'PINNED', physical=current,
                       identity=identity(s), stable_fd=True, failed_component=None)
            if raw is not None:
                row.update(size=len(raw), sha256=hashlib.sha256(raw).hexdigest())
            return raw, row
        raise ValueError('empty endpoint')
    except (OSError, ValueError) as exc:
        row.update(status='ABSENT' if isinstance(exc, FileNotFoundError) else 'ERROR',
                   error_type=type(exc).__name__, error=str(exc), errno=getattr(exc, 'errno', None))
        return None, row
    finally:
        for fd in reversed(fds):
            fs.close(fd)


def pyc_header(raw):
    if len(raw) < 16:
        return dict(classification='REJECTED_SHORT_HEADER', header_hex=raw.hex())
    flags = int.from_bytes(raw[4:8], 'little')
    row = dict(header_hex=raw[:16].hex(), magic_hex=raw[:4].hex(), flags=flags,
               executed_bytecode_equality='OPEN; no marshal/body evaluation')
    if flags not in (0, 1, 3):
        row['classification'] = 'REJECTED_FLAGS'
    elif flags == 0:
        row.update(classification='TIMESTAMP', timestamp=int.from_bytes(raw[8:12], 'little'),
                   source_size=int.from_bytes(raw[12:16], 'little'))
    else:
        row.update(classification='CHECKED_HASH' if flags == 3 else 'UNCHECKED_HASH',
                   source_hash_hex=raw[8:16].hex(), source_hash_binding='UNVERIFIED')
    return row


def association(plan, outcomes):
    stale, incomplete = [], []
    for item in plan['paths']:
        path = item['path']
        if item['classification'] == 'OWN_PROC_METADATA_NOT_REACQUIRED':
            continue
        result = outcomes[path]
        old = item.get('old_pin')
        expected_status = 'DIRECTORY' if item['classification'] == 'DIRECTORY_METADATA' else 'PINNED'
        if result['status'] != expected_status:
            incomplete.append(path)
        known_chain = item.get('old_chain', [])
        fresh = {r['path']:r['identity'] for r in result.get('chain', [])}
        chain_drift = any(fresh.get(r['path']) != r['identity'] for r in known_chain)
        if chain_drift:
            stale.append(path)
        if old and (result['status'] != 'PINNED' or
                    any(result.get(k) != old[k] for k in ('sha256', 'size', 'identity')) or
                    result.get('physical') != old['physical']):
            stale.append(path)
    return dict(oldtrace_association='STALE' if stale else 'REUSED_MATCHING_AVAILABLE_PINS_ONLY',
                drift_paths=sorted(set(stale)), incomplete_paths=incomplete,
                eligible_candidate=not stale and not incomplete,
                open_fd_to_later_inode='OPEN even when all later hashes/identities match',
                automatic_retrace=False)


def correspondence(modules, outcomes):
    rows = []
    for name, module in sorted(modules.items()):
        row = dict(module=name, source=module.get('file'), cache=module.get('cached'),
                   executed_bytecode_equality='OPEN')
        source, cache = (outcomes.get(module.get(k), {}) for k in ('file', 'cached'))
        row.update(source_status=source.get('status', 'NO_PATH'), cache_status=cache.get('status', 'NO_PATH'))
        header = cache.get('pyc')
        if header:
            row['header'] = header
            if header['classification'] == 'TIMESTAMP' and source.get('status') == 'PINNED':
                row['timestamp_size_match'] = (header['timestamp'] == (source['identity'][7] // 10**9) & 0xffffffff
                                               and header['source_size'] == source['size'] & 0xffffffff)
        rows.append(row)
    return rows


def candidate(plan, outcomes, state):
    # Logical aliases are literal plan paths with complete fresh chains. Every
    # byte clone counts independently. No basename/SONAME inference.
    files, logicals, total = [], set(), 0
    for item in plan['paths']:
        if item['classification'] != 'FILE_CANDIDATE':
            continue
        row = outcomes[item['path']]
        if row['status'] != 'PINNED':
            continue
        for logical in (item['path'], row['physical']):
            if logical in logicals:
                continue
            logicals.add(logical)
            total += row['size']
            files.append(dict(logical=logical, physical=row['physical'], size=row['size'],
                              sha256=row['sha256'], mode=0o555 if row['identity'][2] & 0o111 else 0o444))
    return dict(manifest=dict(plan['candidate_template'], files=files), status='PROSPECTIVE_INPUT_DATA_ONLY',
                copied_bytes=total, eligible_candidate=state['eligible_candidate'] and total <= 64*2**20 and 0 < len(files) <= 256,
                guest_copy_cap=64*2**20, runtime_approval=None, security_approval=None,
                tool_dependencies=plan['tool_dependencies'])

def parse_elf(raw):
    global ELFS
    if raw[:4]!=b'\x7fELF': return None
    ELFS+=1
    require_data(ELFS<=64,'ELF count bound')
    require_data(raw[:6]==b'\x7fELF\x02\x01' and len(raw)>=64,'ELF64 little-endian required')
    require_data(struct.unpack_from('<H',raw,18)[0]==62,'x86_64 ELF required')
    phoff=struct.unpack_from('<Q',raw,32)[0]
    phsize,phnum=struct.unpack_from('<HH',raw,54)
    require_data(phsize==56 and phnum<=256 and phoff+phnum*phsize<=len(raw),'ELF program header bounds')
    loads=[];dynamic=[];interp=[]
    def string(table,start):
        require_data(0<=start<len(table),'ELF string offset')
        end=table.find(b'\0',start)
        require_data(end>=start and end-start<4096,'ELF string bound')
        return table[start:end].decode('ascii')
    for i in range(phnum):
        p=struct.unpack_from('<IIQQQQQQ',raw,phoff+i*phsize)
        typ,flags,off,va,pa,sz,mem,align=p
        require_data(off+sz<=len(raw),'ELF segment bounds')
        if typ==1: loads.append((va,off,sz))
        if typ==2:
            require_data(sz%16==0 and sz//16<=65536,'ELF dynamic bound')
            dynamic=[struct.unpack_from('<qQ',raw,o) for o in range(off,off+sz,16)]
        if typ==3: interp.append(string(raw[off:off+sz],0))
    tags={}
    for tag,val in dynamic:
        if tag==0:break
        tags.setdefault(tag,[]).append(val)
    ans=dict(interpreter=interp,needed=[],soname=[],rpath=[],runpath=[])
    if 5 in tags:
        require_data(len(tags[5])==1 and len(tags.get(10,[]))==1,'ELF dynamic string table shape')
        va=tags[5][0];sz=tags[10][0]
        match=[off+va-base for base,off,n in loads if base<=va and va+sz<=base+n]
        require_data(len(match)==1 and match[0]+sz<=len(raw),'ELF string table mapping')
        table=raw[match[0]:match[0]+sz]
        for tag,name in ((1,'needed'),(14,'soname'),(15,'rpath'),(29,'runpath')):
            ans[name]=[string(table,i) for i in tags.get(tag,[])]
    return ans

    state = association(SBF_FINISH_PLAN, OUTCOMES)
    RESULT.update(association=state, modules=correspondence(SBF_FINISH_PLAN['modules'], OUTCOMES),
                  candidate=candidate(SBF_FINISH_PLAN, OUTCOMES, state))
    check_limits(inherited_limits(), INITIAL_LIMITS)
    MONITOR.check()
    require_data(set(MONITOR.owned) == {os.getpid()}, 'unexpected collector descendant')
    RESULT['status'] = 'FINITE_METADATA_INPUT_DATA_ONLY'
    FAILED = not RESULT['candidate']['eligible_candidate']
except BaseException as exc:
    RESULT['error'] = dict(type=type(exc).__name__, message=str(exc), errno=getattr(exc,'errno',None))
finally:
    signal.signal(signal.SIGUSR1, signal.SIG_IGN)
    guard = MONITOR.finish(failed=FAILED) if MONITOR.thread is not None else dict(error='guard not started')
    RESULT.update(guard=guard, outcomes=OUTCOMES, attempts=ATTEMPTS, files=FILES,
                  unique_hashed_bytes=TOTAL, elf_count=ELFS,
                  inherited_limits_before=INITIAL_LIMITS, inherited_limits_after=inherited_limits(),
                  wall_seconds=time.monotonic()-START,
                  peak_rss_bytes=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss*1024,
                  swap=resource.getrusage(resource.RUSAGE_SELF).ru_nswap,
                  completeness='OPEN; source/runtime effects/conditional dlopen/locale/NSS/private guest/control/LSM/cancellation/parentdeath',
                  own_proc='unchanged guard metadata only; no maps/tracee/probe collection',
                  cgroup_reservations='UNOBSERVED; no admission or execution permission')
    if guard['error']:
        FAILED = True
        RESULT['status'] = 'UNKNOWN'
    export = json.dumps(RESULT, sort_keys=True, separators=(',',':')).encode() + b'\n'
    if len(export) > 4*2**20:
        raise GuardError('output over 4MiB; no eligible candidate')
    sys.stdout.buffer.write(export)
    sys.stdout.buffer.flush()
    signal.alarm(0)
raise SystemExit(78 if FAILED else 0)
