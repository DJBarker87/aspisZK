import R1Z.PV_RE_RetryOutcomes
set_option autoImplicit false
noncomputable section
namespace R1Z.PV_RE_RetryTerminal
open Aeneas Aeneas.Std Result ProverRwFull PV_RE_RetryStep
variable [ExternalTypes] [ExternalFuns]
variable {C : Type} (entropy : Entropy C) (pub : Public)
  (compiled : Compiled) (v : transport.Variant) (hash : Hash)
  (store : Store) (state : C) (k : Usize)

theorem attempt_failure (active : ¬ k >= prover.ATTEMPT_CAP)
    (secret : Secrets) (after : C) (e : Error)
    (h : entropy.call_mut state k = .ok (.Ok secret,after))
    (body : prover.prove_masked_attempt pub compiled v hash secret store = .fail e) :
    prover.retry_attempt entropy pub compiled v hash store state k = .fail e := by
  rw [unfold_retry]
  unfold step
  rw [if_neg active, h]
  change (prover.prove_masked_attempt pub compiled v hash secret store >>= _) = _
  rw [body]
  rfl

theorem attempt_diverges (active : ¬ k >= prover.ATTEMPT_CAP)
    (secret : Secrets) (after : C)
    (h : entropy.call_mut state k = .ok (.Ok secret,after))
    (body : prover.prove_masked_attempt pub compiled v hash secret store = .div) :
    prover.retry_attempt entropy pub compiled v hash store state k = .div := by
  rw [unfold_retry]
  unfold step
  rw [if_neg active, h]
  change (prover.prove_masked_attempt pub compiled v hash secret store >>= _) = _
  rw [body]
  rfl

/-- Every non-rejection error terminates the complete retry call. -/
theorem nonretry_error (active : ¬ k >= prover.ATTEMPT_CAP)
    (secret : Secrets) (after : C) (nextStore : Store) (e : verifier.Error)
    (notRejected : e ≠ .Rejected)
    (h : entropy.call_mut state k = .ok (.Ok secret,after))
    (body : prover.prove_masked_attempt pub compiled v hash secret store = .ok (.Err e,nextStore)) :
    prover.retry_attempt entropy pub compiled v hash store state k = .ok (.Err e,nextStore) := by
  rw [unfold_retry]
  unfold step
  rw [if_neg active, h]
  change (prover.prove_masked_attempt pub compiled v hash secret store >>= _) = _
  rw [body]
  cases e <;> try rfl
  exact False.elim (notRejected rfl)

end R1Z.PV_RE_RetryTerminal
