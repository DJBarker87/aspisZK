abbrev Prepared := aspis_r1_cost_probe.verifier.Prepared
abbrev Accumulator := aspis_r1_cost_probe.verifier.Accumulator

def field (p : Prepared) (a : Accumulator) : Fin 6 → CheckpointBytesE.RawE :=
  ![p.gamma,p.kappa,p.alpha.val[0]'(by simp [p.alpha.property]),
    p.alpha.val[1]'(by simp [p.alpha.property]),p.claim,a.rhs]

def offset : Fin 6 → Nat := ![176,208,240,272,632,664]

/-- Sparse execution of the six wide sections; field bytes are not unfolded. -/
theorem dispatch (p : Prepared) (a : Accumulator) (hp hu : Array U8 32#usize)
    (section : Fin 6) (i : Usize) (bound : i.val < 32) :
    body_byte p a hp hu (Usize.ofNat (offset section+i.val)
      (by fin_cases section <;> simp [offset] <;> scalar_tac)) =
      e_byte (field p a section) i := by
  have cast : Usize.ofNat i.val (by scalar_tac) = i := by
    apply UScalar.eq_of_val_eq; simp
  fin_cases section <;>
    simp (disch := scalar_tac) [body_byte,offset,field,sub_exact,div_exact,rem_exact,
      Array.index_usize,Array.make,p.alpha.property,Nat.add_div,Nat.add_mod,
      Nat.div_eq_of_lt bound,Nat.mod_eq_of_lt bound,cast]

/-- All six source-dispatched wide sections have exact C5 field bytes. -/
theorem c5_bytes (p : Prepared) (a : Accumulator) (hp hu : Array U8 32#usize)
    (section : Fin 6) (canonical : CanonicalWide (field p a section))
    (i : Usize) (bound : i.val < 32) :
    ∃ v, body_byte p a hp hu (Usize.ofNat (offset section+i.val)
      (by fin_cases section <;> simp [offset] <;> scalar_tac)) = ok v ∧
      (OraclePortCheckpointAbiC5.encodeE (exactE (field p a section) canonical))[i.val]? =
        some (AspisV8R19.SqueezeOracleBridge.decodeByte v) := by
  rw [dispatch p a hp hu section i bound]
  exact e_byte_c5 _ canonical i bound

end AspisR1.CheckpointBytesSparseE
