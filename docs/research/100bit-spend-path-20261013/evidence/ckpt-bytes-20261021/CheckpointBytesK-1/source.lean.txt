/-- The exact source K selector inherits totality and bytes from the proved
nested writer; no successful call is supplied as a premise. -/
theorem k_byte_c5 (q : RawK) (canonical : Canonical q) (i : Usize) (bound : i.val < 16) :
    ∃ v, k_byte q i = ok v ∧
      (OraclePortCheckpointAbiC5.encodeK (exactK q canonical))[i.val]? =
        some (AspisV8R19.SqueezeOracleBridge.decodeByte v) := by
  obtain ⟨out,run,bytes⟩ := CheckpointP6KBytes.encode_k_c5 q canonical
  have inside : i.val < out.val.length := by simpa [out.property] using bound
  refine ⟨out.val[i.val],?_,?_⟩
  · simp [k_byte,run,Array.index_usize,inside]
  · rw [← bytes]
    simp [bytesOf,inside]

#print axioms k_byte_c5
