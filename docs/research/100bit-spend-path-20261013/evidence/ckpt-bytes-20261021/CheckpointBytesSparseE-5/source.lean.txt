attribute [local irreducible] body_byte e_byte
abbrev Prepared := aspis_r1_cost_probe.verifier.Prepared
abbrev Accumulator := aspis_r1_cost_probe.verifier.Accumulator

/-- One symbolic interval; no encoded field or checkpoint vector is reduced. -/
theorem gamma_dispatch (p : Prepared) (a : Accumulator) (hp hu : Array U8 32#usize)
    (i : Usize) (bound : i.val < 32) :
    body_byte p a hp hu (Usize.ofNat (176+i.val) (by scalar_tac)) = e_byte p.gamma i := by
  let at1 := Usize.ofNat (176+i.val) (by scalar_tac)
  have value : at1.val = 176+i.val := by simp [at1]
  have h8 : ¬ at1 < 8#usize := by scalar_tac
  have h16 : ¬ at1 < 16#usize := by scalar_tac
  have h48 : ¬ at1 < 48#usize := by scalar_tac
  have h80 : ¬ at1 < 80#usize := by scalar_tac
  have h112 : ¬ at1 < 112#usize := by scalar_tac
  have h176 : ¬ at1 < 176#usize := by scalar_tac
  have h304 : at1 < 304#usize := by scalar_tac
  change body_byte p a hp hu at1 = _
  simp only [body_byte,if_neg h8,if_neg h16,if_neg h48,if_neg h80,
    if_neg h112,if_neg h176,if_pos h304]
  simp (disch := scalar_tac) [Array.index_usize,Array.make,p.alpha.property,
    sub_exact,div_exact,rem_exact,value,Nat.div_eq_of_lt bound,Nat.mod_eq_of_lt bound]
  congr 1
  apply UScalar.eq_of_val_eq
  simp

#print axioms gamma_dispatch
end AspisR1.CheckpointBytesSparseE
