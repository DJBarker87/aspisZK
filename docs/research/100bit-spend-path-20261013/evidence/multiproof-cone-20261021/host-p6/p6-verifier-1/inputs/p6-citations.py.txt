#!/usr/bin/env python3
"""Pin individual frozen functions; distinguish extraction from semantic consumers."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[2]
EVIDENCE = ROOT / 'docs/research/100bit-spend-path-20261013/evidence/multiproof-cone-20261021'
BASE = 'crates/aspis-r1-cost-probe/src/'

def cut(file, marker):
    text = (ROOT / (BASE + file)).read_text()
    start = text.index(marker)
    end = text.index('{', start) + 1
    depth = 1
    while depth:
        depth += (text[end] == '{') - (text[end] == '}')
        end += 1
    body = text[start:end]
    return body, {'path': BASE + file, 'marker': marker,
                  'line': text[:start].count('\n') + 1,
                  'sha256': hashlib.sha256(body.encode()).hexdigest()}

FUNCTIONS = [
    ('candidate.rs', 'fn batch_tree<', 'MultiproofConeSlotState.slotState', None),
    ('multiproof.rs', 'pub fn verify_in_place<', 'MultiproofConePhysicalVerify.accepted_verify', 'reference.verify_owned'),
    ('multiproof.rs', 'pub fn binary_parent<', 'MultiproofConePhysicalProgram.writer', 'reference.write_parent'),
    ('multiproof.rs', 'fn occupied_binary_parent<', 'MultiproofConePhysicalProgram.writer', 'reference.write_parent'),
    ('multiproof.rs', 'fn sibling_binary_parent<', 'MultiproofConePhysicalProgram.writer', 'reference.write_parent'),
    ('candidate.rs', 'fn batch_leaf<', 'MultiproofConeLeafFrozen.batch_leaves', None),
    ('multiproof.rs', 'pub fn node_count(', 'MultiproofConeCountPhysicalLevels.eighteen_count', None),
    ('candidate.rs', 'fn batch_keys(', 'MultiproofConeKeyAdapters', None),
    ('candidate.rs', 'fn sort_batch_keys(', 'MultiproofConeSortTotal', 'sort_reference.sort_rows'),
]

rows = []
for file, marker, consumer, reference in FUNCTIONS:
    _, pin = cut(file, marker)
    rows.append({**pin, 'semantic_consumer': consumer,
                 'source_function_extracted': False,
                 'owned_reference_extraction': reference,
                 'note': 'A reference extraction is not extraction of the shipped function.'})
_, batch = cut('candidate.rs', 'fn batch_tree<')
rows.append({'path': batch['path'], 'parent_function_sha256': batch['sha256'],
             'marker': 'nodes[32 * count..].chunks_exact(8).any',
             'semantic_consumer': 'MultiproofConePadding.padding',
             'source_function_extracted': False, 'owned_reference_extraction': None,
             'note': 'Inline padding expression; no standalone shipped padding function.'})

callbacks = []
for marker in ['pub fn binary_parent<', 'fn occupied_binary_parent<', 'fn sibling_binary_parent<']:
    body, _ = cut('multiproof.rs', marker)
    if not body.startswith('pub '):
        body = 'pub(crate) ' + body
    callbacks.append(body)
generated = '// Generated by p6-citations.py; exact callback bodies, visibility only.\n' + \
    'use aspis_core::Digest32;\nconst BINARY_NODE_TAG:u8=0x12;\nconst MAX_BATCH:usize=16;\n' + '\n'.join(callbacks) + '\n'
out = ROOT / 'tools/multiproof-cone-20261021/src/p6_callbacks.rs'
if '--check' in sys.argv:
    assert out.read_text() == generated
else:
    out.write_text(generated)
ledger = {'schema': 'multiproof-p6-function-citations-v1',
          'source_revision': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
          'functions': rows,
          'callback_copy_sha256': hashlib.sha256(generated.encode()).hexdigest(),
          'reference_extraction_receipts': ['host/extraction/charon-5-inputs.json', 'host/extraction/aeneas-5-inputs.json'],
          'source_batch_certificate_supplied': False,
          'whole_verifier_certificate_supplied': False,
          'stage_instantiation_permitted': False}
if '--check' in sys.argv:
    old = json.loads((EVIDENCE / 'p6-function-citations.json').read_text())
    ledger['source_revision'] = old['source_revision']
    assert old == ledger
else:
    (EVIDENCE / 'p6-function-citations.json').write_text(json.dumps(ledger, indent=2) + '\n')
print('P6 per-function source citations checked; whole-source certificates remain open')
