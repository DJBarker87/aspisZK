theorem word_bytes (x : U32) :
    bytesOf (core.num.U32.to_le_bytes x).val=le32 x.val := by
  have layout : bytesOf (core.num.U32.to_le_bytes x).val =
      [AspisV8R19.SqueezeOracleBridge.decodeByte (UScalar.mk (x.bv.setWidth 8)),
       AspisV8R19.SqueezeOracleBridge.decodeByte (UScalar.mk ((x.bv >>> 8).setWidth 8)),
       AspisV8R19.SqueezeOracleBridge.decodeByte
         (UScalar.mk ((((x.bv >>> 8).setWidth 24) >>> 8).setWidth 8)),
       AspisV8R19.SqueezeOracleBridge.decodeByte
         (UScalar.mk ((((((x.bv >>> 8).setWidth 24) >>> 8).setWidth 16) >>> 8).setWidth 8))] := rfl
  rw [layout]
  simp only [le32,List.range_succ,List.range_zero,List.map_append,List.map_cons,
    List.map_nil,List.nil_append,List.cons_append,List.append_nil]
  congr 1 <;> apply Fin.ext <;>
    simp only [AspisV8R19.SqueezeOracleBridge.decodeByte,
      UScalar.val,BitVec.toNat_setWidth,BitVec.toNat_ushiftRight,
      Nat.shiftRight_eq_div_pow] <;> omega

/-- Entire nested emitted K writer equals the existing canonical C5 byte
encoder. Canonicality is a reached-field/caller obligation; writer execution
itself was proved on every raw U32 input. -/
theorem encode_k_c5 (q : RawK) (canonical : OraclePortCkptAbi4Decode.Canonical q) :
    ∃ out, encode_k q=ok out ∧
      bytesOf out.val=OraclePortCheckpointAbiC5.encodeK (exactK q canonical) := by
  obtain ⟨out,run,layout⟩ := CheckpointP6EncodeK.encode_k_total q
  refine ⟨out,run,?_⟩
  rw [layout,OraclePortCkptAbi5Codec.c5_k_bytes]
  simp only [CheckpointP6EncodeK.bytes,bytesOf,List.map_append]
  change bytesOf (core.num.U32.to_le_bytes q.c0.a).val ++
    bytesOf (core.num.U32.to_le_bytes q.c0.b).val ++
    bytesOf (core.num.U32.to_le_bytes q.c1.a).val ++
    bytesOf (core.num.U32.to_le_bytes q.c1.b).val = _
  rw [word_bytes,word_bytes,word_bytes,word_bytes]
  simp [OraclePortCkptAbi4Decode.words,List.append_assoc]
