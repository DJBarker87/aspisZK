import MultiproofSortRef.Funs

set_option autoImplicit false
namespace AspisR1.MultiproofConeSortStep
open Aeneas Aeneas.Std Result
open MultiproofSortRef

def StepPost (keys : Array U64 16#usize) (at : Usize) : sort_reference.InsertStep → Prop
  | .Done out => out=keys ∧ (at.val=0 ∨ keys.val[at.val-1]!≤keys.val[at.val]!)
  | .Next out lower => lower.val+1=at.val ∧
      keys.val[lower.val]!>keys.val[at.val]! ∧
      out=(keys.set lower keys.val[at.val]!).set at keys.val[lower.val]!

/-- Full checked atomic poststate: stop reason, both overwritten cells and
all untouched cells, and the strictly decreasing cursor. -/
theorem insert_step_total (keys : Array U64 16#usize) (at : Usize) (bound : at.val<16) :
    ∃ out, sort_reference.insert_step keys at=ok out ∧ StepPost keys at out := by
  have spec : sort_reference.insert_step keys at ⦃ out => StepPost keys at out ⦄ := by
    unfold sort_reference.insert_step
    step* <;> simp_all [StepPost] <;> scalar_tac
  exact WP.spec_imp_exists spec

#print axioms insert_step_total
end AspisR1.MultiproofConeSortStep
