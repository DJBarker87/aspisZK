/-- The input is defined from this SAME successful old reader run, itself
obtained from the native run. This is a model value function, not a runtime
extractor or a claim about extractor cost. -/
noncomputable def successfulInput (f : Slice U8) (out : C)
    (hbytes : 2048 ≤ f.val.length)
    (h : weights.s2_grouped_inactive_pair_loop0 {start:=0#usize,«end»:=64#usize} f
      (Array.repeat 12#usize aspis_core.field.wide_exact.WideExact.ZERO)=.ok out)
    (n : Nat) : M :=
  if hn : n<64 then
    let oldRun := native_zero_coefficient_success_old f out hbytes h
    let reads := AspisR1GroupedPairActualFieldOpsV3.ActualLoop0SuccessReadback.loop0_success_parser_reads
      f {start:=0#usize,«end»:=64#usize}
      (Array.repeat 12#usize AspisR1GroupedPairActualFieldOpsV3.field.WideExact.ZERO)
      (toOld out) (by decide) rfl oldRun n (by omega) hn
    decode (Classical.choose (Classical.choose_spec reads))
  else 0

/-- The SAME accepted native zero-seeded64 coefficient loop supplies EVERY
strict old read needed by the existing semantic-prefix theorem. No hfields,
initial canonicality, old loop success or coefficient meaning is assumed. -/
theorem native_zero_coefficient_success_prefix (f : Slice U8) (out : C)
    (hbytes : 2048 ≤ f.val.length)
    (h : weights.s2_grouped_inactive_pair_loop0 {start:=0#usize,«end»:=64#usize} f
      (Array.repeat 12#usize aspis_core.field.wide_exact.WideExact.ZERO)=.ok out) :
    DecodedMeaning (toOld out) (PrefixMeaning (successfulInput f out hbytes h) (fun _ => 0) 64) ∧
    CanonicalAccumulatorArray (toOld out) ∧
    (∀ j : Usize,j.val<64 → ∃ x : S,decodeField f j=.ok x ∧
      decode x=successfulInput f out hbytes h j.val) := by
  let oldRun := native_zero_coefficient_success_old f out hbytes h
  have hfields : ∀ j : Usize,j.val<64 → ∃ x : S,decodeField f j=.ok x ∧
      decode x=successfulInput f out hbytes h j.val := by
    intro j hj
    let reads := AspisR1GroupedPairActualFieldOpsV3.ActualLoop0SuccessReadback.loop0_success_parser_reads
      f {start:=0#usize,«end»:=64#usize}
      (Array.repeat 12#usize AspisR1GroupedPairActualFieldOpsV3.field.WideExact.ZERO)
      (toOld out) (by decide) rfl oldRun j.val (by omega) hj
    let rest := Classical.choose_spec reads
    let input := Classical.choose rest
    have hrest := Classical.choose_spec rest
    rcases hrest with ⟨raw,start,next,finish,selected,hidx,hread,hfield,hi,hn,hf,hs,hdecoded⟩
    have hjEq : Classical.choose reads=j := UScalar.eq_of_val_eq hidx
    have hr : decodeField f j=.ok input := by
      calc
        decodeField f j=decodeField f (Classical.choose reads) :=
          congrArg (decodeField f) hjEq.symm
        _ = .ok input := hread
    refine ⟨input,hr,?_⟩
    simp only [successfulInput,dif_pos hj] <;> rfl
  have hseedCan : CanonicalAccumulatorArray
      (Array.repeat 12#usize AspisR1GroupedPairActualFieldOpsV3.field.WideExact.ZERO) := by
    intro k
    change AspisR1GroupedPairActualFieldOpsV3.ActualFieldComplete.Canonical
      ((List.replicate 12 AspisR1GroupedPairActualFieldOpsV3.field.WideExact.ZERO)[k.val]'(by simpa using k.isLt))
    rw [List.getElem_replicate]
    exact AspisR1GroupedPairActualFieldOpsV3.NativeZeroReadbackReuseV1.zero_seed_canonical
  have hseedMeaning : DecodedMeaning
      (Array.repeat 12#usize AspisR1GroupedPairActualFieldOpsV3.field.WideExact.ZERO)
      (PrefixMeaning (successfulInput f out hbytes h) (fun _ => 0) 0) :=
    AspisR1GroupedPairActualFieldOpsV3.NativeZeroReadbackReuseV1.zero_seed_meaning
  obtain ⟨result,hrun,hmeaning,hcanonical⟩ :=
    AspisR1GroupedPairActualFieldOpsV3.ActualFieldComplete.actual_loop0_whole_prefix
      f (successfulInput f out hbytes h) (fun _ => 0)
      {start:=0#usize,«end»:=64#usize}
      (Array.repeat 12#usize AspisR1GroupedPairActualFieldOpsV3.field.WideExact.ZERO)
      rfl rfl hseedMeaning hseedCan hfields
  have he : result=toOld out := Result.ok.inj (hrun.symm.trans oldRun)
  subst result
  exact ⟨hmeaning,hcanonical,hfields⟩
