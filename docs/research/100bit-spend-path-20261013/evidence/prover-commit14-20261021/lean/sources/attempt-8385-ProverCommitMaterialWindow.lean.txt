/-! The new material reference uses exactly the prior global cursor/cache
carrier. The window theorem retains the global hash, never resets it. -/
set_option autoImplicit false
namespace R1Z.ProverCommitMaterialWindow
open Aeneas Aeneas.Std Result AspisR1ProverMaterial14
open R1Z.ProverCommitMaterialRecursion (Hash State)
open R1Z.ProverCommitStreamSource (Bytes32)
abbrev OldState := R1Z.ProverCommitStepLoop.State

def toOldState (s : State) : OldState := ⟨s.seed,s.counter,s.block,s.word⟩
def fromOldState (s : OldState) : State := ⟨s.seed,s.counter,s.block,s.word⟩
theorem state_roundtrip (s : State) : fromOldState (toOldState s) = s := by cases s; rfl
theorem old_state_roundtrip (s : OldState) : toOldState (fromOldState s) = s := by cases s; rfl

theorem next_bridge (hash : Hash) (domain : Slice U8) (state : State) :
    (do
      let (word,after) ← prover_commit_reference.next hash domain state
      ok (word,toOldState after)) =
      AspisR1ProverStep8.prover_commit_reference.next hash domain (toOldState state) := by
  simp only [prover_commit_reference.next,
    AspisR1ProverStep8.prover_commit_reference.next, toOldState, bind_assoc_eq, bind_tc_ok]
  congr 1
  funext triple
  rcases triple with ⟨counter, block, word⟩
  simp only [Aeneas.Std.uncurry_apply_pair, bind_assoc_eq, bind_tc_ok, toOldState]

theorem next_success_from_old (hash : Hash) (domain : Slice U8) (state : State)
    (word : U32) (after : OldState)
    (success : AspisR1ProverStep8.prover_commit_reference.next hash domain (toOldState state) = ok (word,after)) :
    ∃ fresh, prover_commit_reference.next hash domain state = ok (word,fresh) ∧ toOldState fresh = after := by
  have bridge := next_bridge hash domain state
  rw [success] at bridge
  cases run : prover_commit_reference.next hash domain state with
  | fail e => simp only [run, bind_tc_fail] at bridge; cases bridge
  | div => simp only [run, bind_tc_div] at bridge; cases bridge
  | ok pair =>
    rcases pair with ⟨value,fresh⟩
    simp only [run, bind_tc_ok] at bridge
    cases bridge
    exact ⟨fresh, rfl, rfl⟩

open R1Z.ProverCommitOwnedDecoder (blockHash valid position)

theorem next_local_support (left right : Hash) (domain : Slice U8) (state : State)
    (same : state.word = 8#usize →
      left (ProverCommitMaterialSupport.blockSlices domain state.seed state.counter) =
      right (ProverCommitMaterialSupport.blockSlices domain state.seed state.counter)) :
    prover_commit_reference.next left domain state = prover_commit_reference.next right domain state := by
  by_cases needed : state.word = 8#usize
  · have frame := same needed
    unfold ProverCommitMaterialSupport.blockSlices at frame
    simp only [prover_commit_reference.next, lift, needed, if_true, bind_tc_ok, frame]
  · simp only [prover_commit_reference.next, if_neg needed]

/-- Any bounded scalar prefix in the indexed material extraction can use a
local window only when it agrees with the retained global block answers.
The mapped global cursor advances through rejected candidates too. -/
theorem finite_window {blocks : Nat} (answers : Fin blocks → R0Z.FsStatement.OracleAnswer)
    (right : Hash) (domain : Slice U8) (capacity : blocks < 2^64)
    (start finish r : Nat) (endbound : finish ≤ blocks*8)
    (same : ∀ (seed : Bytes32) (counter : U64),
      start ≤ 8*counter.val → 8*counter.val < finish →
      blockHash answers (ProverCommitMaterialSupport.blockSlices domain seed counter) =
        right (ProverCommitMaterialSupport.blockSlices domain seed counter))
    (state : State) (before : valid blocks (toOldState state))
    (lower : start ≤ position (toOldState state))
    (upper : position (toOldState state)+r ≤ finish) :
    ProverCommitMaterialSupport.finite (blockHash answers) domain r state =
      ProverCommitMaterialSupport.finite right domain r state := by
  induction r generalizing state with
  | zero => rfl
  | succ r ih =>
    obtain ⟨word, after, oldnext⟩ := ProverCommitOwnedDecoder.next_total answers domain (toOldState state) before (by omega)
    obtain ⟨fresh, next, mapped⟩ := next_success_from_old _ domain state word after oldnext
    have progress := ProverCommitOwnedDecoder.next_progress answers domain after word capacity before oldnext
    rw [← mapped] at progress
    have eq := next_local_support (blockHash answers) right domain state (by
      intro needed
      have slot : state.word.val = 8 := by rw [needed]; scalar_tac
      have pos : position (toOldState state) = 8*state.counter.val := by
        unfold position toOldState; simp only; rw [slot]; omega
      apply same state.seed state.counter <;> omega)
    rw [ProverCommitMaterialSupport.finite, ProverCommitMaterialSupport.finite, ← eq, next]
    simp only [bind_tc_ok, Aeneas.Std.uncurry_apply_pair]
    split_ifs
    · rfl
    · exact ih fresh progress.1 (by omega) (by omega)
