This is one family of source coverage, not an all-family execution join. -/

theorem iterator_more (attempts : I32) (running : attempts.val < 16) :
    ∃ after : core.ops.range.Range I32,
      core.iter.range.IteratorRange.next core.iter.range.StepI32
        {start := attempts, «end» := 16#i32} = ok (some attempts, after) ∧
      after.start.val = attempts.val + 1 ∧ after.end = 16#i32 := by
  have hmax : attempts.val + 1 ≤ I32.max := by scalar_tac
  simp only [core.iter.range.IteratorRange.next, core.iter.range.StepI32,
    core.iter.range.IScalarStep, core.cmp.PartialOrdI32,
    core.cmp.impls.PartialOrdI32.lt, core.clone.CloneI32,
    core.clone.impls.CloneI32.clone, liftFun1, liftFun2, bind_tc_ok]
  simp [running, core.iter.range.IScalarStep.forward_checked, hmax]

theorem iterator_done (attempts : I32) (stopped : 16 ≤ attempts.val) :
    core.iter.range.IteratorRange.next core.iter.range.StepI32
      {start := attempts, «end» := 16#i32} =
      ok (none, {start := attempts, «end» := 16#i32}) := by
  simp [core.iter.range.IteratorRange.next, core.iter.range.StepI32,
    core.iter.range.IScalarStep, core.cmp.PartialOrdI32,
    core.cmp.impls.PartialOrdI32.lt, not_lt.mpr stopped]

abbrev Sample := core.result.Result aspis_core.field.M31
  state_only_hiding.StateOnlyMaskBuildError

def finite (hash : Hash) (domain : Slice U8) : Nat → State →
    Result (State × Option (Sample × State))
  | 0, state => ok (state, none)
  | n+1, state => do
    let (word, after) ← prover_commit_reference.next hash domain state
    let candidate ← lift (word &&& 2147483647#u32)
    if candidate < aspis_core.field.P then
      ok (after, some (.Ok candidate, after))
    else finite hash domain n after

theorem loop_finite (hash : Hash) (domain : Slice U8) (n : Nat)
    (state : State) (attempts : I32)
    (remaining : attempts.val + (n : Int) = 16) :
    prover_commit_reference.limb_caller_loop
      {start := attempts, «end» := 16#i32} hash domain state =
      finite hash domain n state := by
  induction n generalizing state attempts with
  | zero =>
    have stopped : 16 ≤ attempts.val := by omega
    rw [prover_commit_reference.limb_caller_loop, loop]
    simp only [prover_commit_reference.limb_caller_loop.body,
      iterator_done attempts stopped, bind_tc_ok, Aeneas.Std.uncurry_apply_pair,
      finite]
  | succ n ih =>
    have running : attempts.val < 16 := by omega
    obtain ⟨after, heq, hstart, hend⟩ := iterator_more attempts running
    rw [prover_commit_reference.limb_caller_loop, loop]
    simp only [prover_commit_reference.limb_caller_loop.body, heq,
      bind_tc_ok, Aeneas.Std.uncurry_apply_pair, finite]
    cases hn : prover_commit_reference.next hash domain state with
    | fail error => rfl
    | div => rfl
    | ok pair =>
      rcases pair with ⟨word, state'⟩
      simp only [bind_tc_ok, Aeneas.Std.uncurry_apply_pair, lift]
      split_ifs
      · rfl
      · have hafter : after = {start := after.start, «end» := 16#i32} := by
          cases after
          simp_all
        rw [hafter]
        exact ih state' _ (by omega)

theorem caller_sixteen (hash : Hash) (domain : Slice U8) (state : State) :
    prover_commit_reference.limb_caller hash domain state = (do
      let (after, pending) ← finite hash domain 16 state
      match pending with
      | none => ok (.Err .MaskSampleExhausted, after)
      | some pair => ok pair) := by
  unfold prover_commit_reference.limb_caller
  rw [loop_finite hash domain 16 state 0#i32 (by decide)]
  rfl


def blockSlices (domain : Slice U8) (seed : Aeneas.Std.Array U8 32#usize) (counter : U64) : Slice (Slice U8) :=
  Aeneas.Std.Array.to_slice (Aeneas.Std.Array.make 3#usize [domain,
    Aeneas.Std.Array.to_slice seed, Aeneas.Std.Array.to_slice (core.num.U64.to_le_bytes counter)])

def SameFrames (left right : Hash) (domain : Slice U8) : Prop :=
  ∀ seed counter, left (blockSlices domain seed counter) = right (blockSlices domain seed counter)

theorem next_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (state : State) : prover_commit_reference.next left domain state = prover_commit_reference.next right domain state := by
  have frame := same state.seed state.counter
  unfold blockSlices at frame
  simp only [prover_commit_reference.next, lift]
  by_cases h : state.word = 8#usize
  · simp only [h, if_true, bind_tc_ok, frame]
  · simp only [h, if_false]

theorem finite_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (n : Nat) (state : State) : finite left domain n state = finite right domain n state := by
  induction n generalizing state with
  | zero => rfl
  | succ n ih =>
    simp only [finite, next_support left right domain same, ih]

theorem scalar_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (state : State) : prover_commit_reference.limb_caller left domain state = prover_commit_reference.limb_caller right domain state := by
  rw [caller_sixteen, caller_sixteen, finite_support left right domain same]

theorem tower_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (state : State) : prover_commit_reference.qm31_caller left domain state = prover_commit_reference.qm31_caller right domain state := by
  simp only [prover_commit_reference.qm31_caller, scalar_support left right domain same]

open R1Z.ProverCommitMaterialRecursion (Values QValues)
theorem collection_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (n : Nat) (values : Values) (state : State) :
    ProverCommitMaterialRecursion.finite left domain n values state =
      ProverCommitMaterialRecursion.finite right domain n values state := by
  induction n generalizing values state with
  | zero => rfl
  | succ n ih =>
    simp only [ProverCommitMaterialRecursion.finite, scalar_support left right domain same, ih]

theorem raw_collection_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (count : Usize) (values : Values) (state : State) :
    prover_commit_reference.collect_m31_owned left domain state count values =
      prover_commit_reference.collect_m31_owned right domain state count values := by
  rw [ProverCommitMaterialRecursion.extracted_finite left domain count.val count values state rfl,
    ProverCommitMaterialRecursion.extracted_finite right domain count.val count values state rfl]
  exact collection_support left right domain same _ values state

theorem qm_collection_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (n : Nat) (values : QValues) (state : State) :
    ProverCommitMaterialRecursion.qmFinite left domain n values state =
      ProverCommitMaterialRecursion.qmFinite right domain n values state := by
  induction n generalizing values state with
  | zero => rfl
  | succ n ih =>
    simp only [ProverCommitMaterialRecursion.qmFinite, tower_support left right domain same, ih]

theorem columns_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (n : Nat) (columns : alloc.vec.Vec Values) (state : State)
    (index dependent : Usize) (active : Aeneas.Std.Array Bool 1024#usize) :
    ProverCommitMaterialRecursion.columnsFinite left domain state columns index n dependent active =
      ProverCommitMaterialRecursion.columnsFinite right domain state columns index n dependent active := by
  induction n generalizing state columns index with
  | zero =>
    rw [ProverCommitMaterialRecursion.columnsFinite, ProverCommitMaterialRecursion.columnsFinite]
    simp only [if_true]
  | succ n ih =>
    rw [ProverCommitMaterialRecursion.columnsFinite, ProverCommitMaterialRecursion.columnsFinite]
    simp only [Nat.add_eq_zero_iff, Nat.one_ne_zero, and_false, if_false,
      raw_collection_support left right domain same, Nat.add_sub_cancel, ih]

theorem padding_support (left right : Hash) (domain : Slice U8) (same : SameFrames left right domain)
    (n : Nat) (values : QValues) (state : State)
    (row dependent : Usize) (active : Aeneas.Std.Array Bool 1024#usize) :
    ProverCommitMaterialRecursion.paddingFinite left domain state values row n dependent active =
      ProverCommitMaterialRecursion.paddingFinite right domain state values row n dependent active := by
  induction n generalizing state values row with
  | zero =>
    rw [ProverCommitMaterialRecursion.paddingFinite, ProverCommitMaterialRecursion.paddingFinite]
    simp only [if_true]
  | succ n ih =>
    rw [ProverCommitMaterialRecursion.paddingFinite, ProverCommitMaterialRecursion.paddingFinite]
    simp only [Nat.add_eq_zero_iff, Nat.one_ne_zero, and_false, if_false,
      tower_support left right domain same, Nat.add_sub_cancel, ih]
