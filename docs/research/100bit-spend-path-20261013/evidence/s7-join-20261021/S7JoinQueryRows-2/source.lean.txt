/-! Total basis/64-row/terms portion of one production-shaped query. The
opening/fold and rhs/power/next caller effects are not included here. -/
set_option autoImplicit false
noncomputable section
namespace AspisR1.S7JoinQueryRows
open Aeneas Aeneas.Std Result
open S7Kani2AccumulatorLoop (K BatchedWeights)
open S7Kani3CallerTrace (PushTrace Sum initialRaw initialRow initialSum)
open S7JoinRowsTrace

/-- Raw initialization is phase-safe even for arbitrary stored U32 limbs. -/
theorem initial_progress (old : S8OwnedP6Finish.Weights) (hl : old.val.length=64) :
    Progress 0 0 (⟨initialRaw old,0#usize⟩ : BatchedWeights) := by
  refine ⟨rfl,by simpa only [initialRaw,List.length_map] using hl,?_⟩
  intro i hi
  have hi' : i<old.val.length := by omega
  let q := old.val[i]
  let x : Fin 4 → U64 := fun k => core.convert.num.FromU64U32.from
    (S8OwnedP6ValueModel.coordinate q k)
  refine ⟨x,?_,?_⟩
  · change (old.val.map initialRow)[i]?=some (S7Kani2OwnedRowRefinement.makeRow x)
    rw [List.getElem?_map,List.getElem?_eq_getElem hi']
    rfl
  · intro k
    simp only [Nat.not_lt_zero,if_false,S7Kani3PhaseInvariant.Phase,
      Nat.zero_mod,Nat.zero_mul,Nat.add_zero,x,ExtractedCurrentCM31Mul.widen_value]
    scalar_tac

def query_rows (output : alloc.vec.Vec K) (state : BatchedWeights)
    (node : U32) (power : K) (next : Usize) : Result BatchedWeights := do
  let basis ← S7JoinProductionBasis.basis_owned node
  let result ← S7Kani2AccumulatorLoop.accumulate_rows_owned backend output state
    basis power next 64#usize 0#usize
  let terms ← result.2.terms+1#usize
  ok {result.2 with terms:=terms}

/-- Basis success, canonicality, every push, both row traces' bounded index
computations, and terms advancement are derived, rather than assumed. -/
theorem query_rows_exact (old : S8OwnedP6Finish.Weights) (output : alloc.vec.Vec K)
    (state : BatchedWeights) (sum : Sum) (node : U32) (power : K) (next : Usize)
    (t : Nat) (hout : output.val.length=64) (hn : next.val≤22) (ht : t<22)
    (hc : S7JoinBackendCarriers.Canonical power)
    (history : PushTrace old state sum) (hp : Progress t 0 state) :
    ∃ after sumOut, query_rows output state node power next=ok after ∧
      PushTrace old after sumOut ∧ Progress (t+1) 0 after ∧
      ∀ i, i<64 → ∀ k : Fin 4,
        (sumOut i k : ZMod AspisCircleGroupOrder.P)=(sum i k : ZMod AspisCircleGroupOrder.P)+
          (S7Kani2OwnedRowRefinement.powerLimb power k).val*
            AspisCircleTensorBinding.naturalLineValue (S7JoinBasisMachine.decode node) i := by
  obtain ⟨basis,hbasis,hcanon,hmeaning⟩ := S7JoinBasisCanonical.total_exact node
  obtain ⟨out,sumOut,hrun,hist,hprogress,hsum⟩ := rows_exact old output state sum basis power next
    0#usize 64 t hout hn (by decide) hc hcanon history hp
  obtain ⟨terms,hinc,hterms⟩ := S7Kani2AccumulatorLoop.increment out.terms (by
    have he := hprogress.1
    omega)
  have ht' : terms.val=t+1 := by rw [hterms,hprogress.1]
  refine ⟨{out with terms:=terms},sumOut,?_,.counter hist terms,
    next_query_progress t out terms ht' hprogress,?_⟩
  · simp only [query_rows,hbasis,hrun,bind_tc_ok]
    change (do let terms ← out.terms+1#usize; ok {out with terms:=terms})=ok _
    rw [hinc,bind_tc_ok]
  · intro i hi k
    rw [hsum i k,if_pos (by change (0#usize).val≤i ∧ i<64; simp; exact hi),
      Nat.cast_add,Nat.cast_mul]
    have hm := hmeaning (⟨i,hi⟩ : Fin 64)
    change (sum i k : ZMod AspisCircleGroupOrder.P)+
      (S7Kani2OwnedRowRefinement.powerLimb power k).val*
        S7JoinBasisMachine.decode (S7JoinBasisBlock.cell basis i)=_
    rw [hm]

/-- The first query needs no pre-existing PushTrace or row-phase premise. -/
theorem initial_query_rows (old : S8OwnedP6Finish.Weights) (output : alloc.vec.Vec K)
    (node : U32) (power : K) (next : Usize)
    (hold : old.val.length=64) (hout : output.val.length=64) (hn : next.val≤22)
    (hc : S7JoinBackendCarriers.Canonical power) :
    ∃ after sumOut,
      query_rows output ⟨initialRaw old,0#usize⟩ node power next=ok after ∧
      PushTrace old after sumOut ∧ Progress 1 0 after := by
  obtain ⟨after,sumOut,hrun,hist,hphase,_⟩ :=
    query_rows_exact old output _ (initialSum old) node power next 0 hout hn (by decide)
      hc .initial (initial_progress old hold)
  exact ⟨after,sumOut,hrun,hist,hphase⟩
