70: fn supported_loader(owner: &Pubkey) -> bool {
71:     owner == &bpf_loader::id()
72:         || owner == &bpf_loader_upgradeable::id()
73:         || owner == &loader_v4::id()
74: }
75: 
76: fn require_verifier_program(account: &AccountInfo<'_>, selected: &Pubkey) -> ProgramResult {
77:     if account.key != selected
78:         || !account.executable
79:         || account.is_signer
80:         || account.is_writable
81:         || !supported_loader(account.owner)
82:     {
83:         return Err(PoolV1ProgramError::InvalidVerifierProgramAccount.into());
84:     }
85:     Ok(())
86: }

88: /// R1 buffer ownership does not authenticate the selected verifier. Registry
89: /// selection still does that; only its separate checkpoint must be owned by it.
90: fn proof_transport_verifier(
91:     verifier_program: &AccountInfo<'_>,
92:     proof: &AccountInfo<'_>,
93:     verifier_checkpoint: Option<&AccountInfo<'_>>,
94: ) -> Result<Pubkey, ProgramError> {
95:     #[cfg(feature = "r1-split-checkpoint")]
96:     {
97:         let checkpoint = verifier_checkpoint.ok_or(ProgramError::NotEnoughAccountKeys)?;
98:         if proof.key == verifier_program.key
99:             || proof.executable
100:             || proof.is_signer
101:             || proof.is_writable
102:             || checkpoint.owner != verifier_program.key
103:             || checkpoint.executable
104:             || checkpoint.is_signer
105:             || checkpoint.is_writable
106:             || checkpoint.key == proof.key
107:             || checkpoint.key == verifier_program.key
108:         {
109:             return Err(PoolV1ProgramError::InvalidVerifierProofAccount.into());
110:         }
