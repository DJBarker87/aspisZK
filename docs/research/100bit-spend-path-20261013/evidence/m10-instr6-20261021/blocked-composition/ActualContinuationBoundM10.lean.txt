/-! Whole extractor work composition. Recovery/point arrays and the byte bank
are eager; public failure and private aborts still have the same returned
value. All later word reads use retained typed arrays. The sole component
premise is the fixed-shape public image inverse/check bound. -/
set_option autoImplicit false
noncomputable section
namespace AspisR1.ActualContinuationBoundM10
open AspisWideTower AspisV8R15.ExactTowerBase SpendRelation
open AspisPool.AlgorithmicCircleDecoderV7
open CanonicalCutEnrichedTranscript (Words)
open RawRecordedEagerExtractorReadsBin (RunView Coins continueWithWords)
open GSPostProcessWorkM10

def received (words : Words) : Fin 29 → InitialWord WideExact :=
  LiteralTypedWords.word ExactBaseProjection.embed (algebraMap QM31Exact WideExact) words.c1 words.c2

def continuation (image : PublicImage) (pub : R0P.Public M31Exact) (words : Words) (coins : Coins) :
    Option Wspend × Nat :=
  ExtractorStagesM10.continuation
    (GSPointBatchWorkM10.inputs (received words) (fun _ => words.final))
    (fun inputs => GSByteBankWorkM10.run inputs coins)
    (GSPostProcessCompleteM10.eval image pub (received words))

attribute [local irreducible] GSPointBatchWorkM10.inputs GSByteBankWorkM10.run
  GSPostProcessCompleteM10.eval GSPointBatchWorkM10.budget GSByteBankWorkM10.budget
  GSPostProcessCompleteM10.budget continueWithWords GSByteJointSpendExtractor.extract
  GSConcreteByteDecoderBank.run BinEagerWordsReads.words
  RawRecordedPublicSpendExtractorBin.extract RawCutWordsFromRunPublicValidBin.runDecodedPublic

/-- The isolated local GS value bridge. Its direct focused proof is retained
in the obstruction receipts; no resource, allocation or log bound is assumed. -/
def GSValueRefinement (image : PublicImage) : Prop := ∀ pub words coins,
  (continuation image pub words coins).1 = continueWithWords pub words coins

@[never_extract,noinline] def continuationBudget (C : Nat) : Nat :=
  GSPointBatchWorkM10.budget+(GSByteBankWorkM10.budget+1)+
    (GSPostProcessCompleteM10.budget C+1)+1000

attribute [local irreducible] continuationBudget

theorem continuation_work (image : PublicImage) (C : Nat) (hi : PublicImageBound image C)
    (pub : R0P.Public M31Exact) (hp : pub.nextPairIndex < 2^64) (words : Words) (coins : Coins) :
    (continuation image pub words coins).2 ≤ continuationBudget C := by
  unfold continuation continuationBudget
  apply ExtractorStagesM10.continuation_work
  · exact GSPointBatchWorkM10.inputs_work _ _
  · exact GSByteBankWorkM10.work _ coins
  · intro out ho
    exact GSPostProcessCompleteM10.work image C hi pub hp (received words) out
      (GSByteBankWorkM10.bounded _ coins out ho)


#print axioms continuation_work
end AspisR1.ActualContinuationBoundM10
