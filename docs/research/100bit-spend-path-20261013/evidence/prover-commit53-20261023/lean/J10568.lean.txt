/-! Complete inverse fibre transform for arbitrary canonical wide inputs.
No domain point or channel result is assumed correct by this local theorem. -/
set_option autoImplicit false
noncomputable section

open ProverCommitWideOps48 ProverCommitMix48

def phi {K : Type} [Field K] (x y : K) (h : Fin 4 → K) : Fin 4 → K :=
  ![h 0+y*h 1+x*h 2+x*y*h 3,
    h 0-y*h 1+x*h 2-x*y*h 3,
    h 0-y*h 1-x*h 2+x*y*h 3,
    h 0+y*h 1-x*h 2-x*y*h 3]

theorem phi_0 {K : Type} [Field K] (x y : K) (h : Fin 4 → K) :
    phi x y h 0=h 0+y*h 1+x*h 2+x*y*h 3 := rfl

theorem phi_1 {K : Type} [Field K] (x y : K) (h : Fin 4 → K) :
    phi x y h 1=h 0-y*h 1+x*h 2-x*y*h 3 := rfl

theorem phi_2 {K : Type} [Field K] (x y : K) (h : Fin 4 → K) :
    phi x y h 2=h 0-y*h 1-x*h 2+x*y*h 3 := rfl

theorem phi_3 {K : Type} [Field K] (x y : K) (h : Fin 4 → K) :
    phi x y h 3=h 0+y*h 1-x*h 2-x*y*h 3 := rfl

attribute [local irreducible] value source sourceArray

theorem exact_result (x y : Raw) (hx : canonical x) (hy : canonical y)
    (h : Nat → Raw) (hh : ∀ i, i<4 → canonical (h i)) :
    ∃ out : Nat → Raw,
      encoder53.phi_inverse (source x) (source y) (sourceArray 4#usize h)=
        ok (sourceArray 4#usize out) ∧
      (∀ i, i<4 → canonical (out i)) ∧
      ∀ i : Fin 4, value (out i.val)=phi (value x) (value y) (fun i => value (h i.val)) i := by
  obtain ⟨a,ha,hca,hda⟩ := ProverCommitWideOps48.mul_exact (y) (h 1) (hy) (hh 1 (by decide))
  obtain ⟨b,hb,hcb,hdb⟩ := ProverCommitWideOps48.mul_exact (x) (h 2) (hx) (hh 2 (by decide))
  obtain ⟨xy,hxy,hcxy,hdxy⟩ := ProverCommitWideOps48.mul_exact (x) (y) (hx) (hy)
  obtain ⟨c,hc,hcc,hdc⟩ := ProverCommitWideOps48.mul_exact (xy) (h 3) (hcxy) (hh 3 (by decide))
  obtain ⟨t0,ht0,hct0,hdt0⟩ := ProverCommitWideOps48.add_exact (h 0) (a) (hh 0 (by decide)) (hca)
  obtain ⟨t1,ht1,hct1,hdt1⟩ := ProverCommitWideOps48.add_exact (t0) (b) (hct0) (hcb)
  obtain ⟨o0,ho0,hco0,hdo0⟩ := ProverCommitWideOps48.add_exact (t1) (c) (hct1) (hcc)
  obtain ⟨t2,ht2,hct2,hdt2⟩ := ProverCommitWideSub53.sub_exact (h 0) (a) (hh 0 (by decide)) (hca)
  obtain ⟨t3,ht3,hct3,hdt3⟩ := ProverCommitWideOps48.add_exact (t2) (b) (hct2) (hcb)
  obtain ⟨o1,ho1,hco1,hdo1⟩ := ProverCommitWideSub53.sub_exact (t3) (c) (hct3) (hcc)
  obtain ⟨t4,ht4,hct4,hdt4⟩ := ProverCommitWideSub53.sub_exact (t2) (b) (hct2) (hcb)
  obtain ⟨o2,ho2,hco2,hdo2⟩ := ProverCommitWideOps48.add_exact (t4) (c) (hct4) (hcc)
  obtain ⟨t5,ht5,hct5,hdt5⟩ := ProverCommitWideSub53.sub_exact (t0) (b) (hct0) (hcb)
  obtain ⟨o3,ho3,hco3,hdo3⟩ := ProverCommitWideSub53.sub_exact (t5) (c) (hct5) (hcc)
  let out : Nat → Raw := fun i => match i with | 0 => o0 | 1 => o1 | 2 => o2 | _ => o3
  refine ⟨out,?_,?_,?_⟩
  · simp only [encoder53.phi_inverse,index 4#usize h 0#usize (by decide),
      index 4#usize h 1#usize (by decide),index 4#usize h 2#usize (by decide),
      index 4#usize h 3#usize (by decide),
      show (0#usize).val=0 from rfl,show (1#usize).val=1 from rfl,
      show (2#usize).val=2 from rfl,show (3#usize).val=3 from rfl,ha,hb,hxy,hc,ht0,ht1,ho0,ht2,ht3,ho1,ht4,ho2,ht5,ho3,bind_tc_ok]
    unfold sourceArray
    rfl
  · intro i hi
    interval_cases i
    · exact hco0
    · exact hco1
    · exact hco2
    · exact hco3
  · intro i
    fin_cases i
    · change value o0=phi (value x) (value y) (fun j => value (h j.val)) 0
      rw [phi_0]
      change value o0=value (h 0)+value y*value (h 1)+value x*value (h 2)+value x*value y*value (h 3)
      rw [hdo0,hdt1,hdt0,hda,hdb,hdc,hdxy]
      rfl
    · change value o1=phi (value x) (value y) (fun j => value (h j.val)) 1
      rw [phi_1]
      change value o1=value (h 0)-value y*value (h 1)+value x*value (h 2)-value x*value y*value (h 3)
      rw [hdo1,hdt3,hdt2,hda,hdb,hdc,hdxy]
      rfl
    · change value o2=phi (value x) (value y) (fun j => value (h j.val)) 2
      rw [phi_2]
      change value o2=value (h 0)-value y*value (h 1)-value x*value (h 2)+value x*value y*value (h 3)
      rw [hdo2,hdt4,hdt2,hda,hdb,hdc,hdxy]
      rfl
    · change value o3=phi (value x) (value y) (fun j => value (h j.val)) 3
      rw [phi_3]
      change value o3=value (h 0)+value y*value (h 1)-value x*value (h 2)-value x*value y*value (h 3)
      rw [hdo3,hdt5,hdt0,hda,hdb,hdc,hdxy]
      rfl
