noncomputable section

abbrev QM := field.QM31
abbrev Arr := Array QM 10#usize
abbrev Claims := Array QM 84#usize
abbrev Statement := AspisR1TerminalDecoder.PublicBinding.Statement
abbrev Public := pool_v1.pair_forest_semantic_terminal.SemanticPublic
abbrev Error := AspisR1.CompositionPartsTail.Error
abbrev Canonical := AspisR1.ExtractedMaskQM31Exact.CurrentCanonicalQM
abbrev decode := AspisR1.ExtractedMaskQM31Exact.decode
abbrev Output := QM × Array QM 16#usize × Arr × QM

abbrev run := AspisR1.TerminalCallable53.run

theorem run_success (point zc : Arr) (s : Statement) (claims : Claims)
    (lambda chi theta mu : QM)
    (hpoint : CanonicalArray point) (hzc : CanonicalArray zc)
    (hclaims : CanonicalArray claims)
    (hlambda : Canonical lambda) (hchi : Canonical chi)
    (htheta : Canonical theta) (hmu : Canonical mu)
    {H : Type} (D : AspisR1PublicPreparation.aspis_core.Digest32 H)
    (hash : H) (c1 : Array U8 32#usize)
    (outTranscript : AspisR1PublicPreparation.r1_wire.SemanticTranscript H)
    (hprep : AspisR1PublicPreparation.r1_semantic.public_transcript D
      (AspisR1PublicPreparation.PreparationBinding.preparationPublic s)
      hash c1 = .ok (.Ok outTranscript)) :
    ∃ (z : Array QM 16#usize) (mask : Arr) (g terminal : QM),
      run (copyFromStatement s) claims point lambda chi theta zc mu =
        .ok (.Ok (terminal,z,mask,g)) ∧ Canonical terminal ∧
      decode terminal = R0P.SemSource.terminalValue (exactPublic (copyFromStatement s))
        (decode lambda) (decode chi) (decode theta) (decode mu)
        (AspisR1.SelectorMarginalModel.modelPoint zc) TerminalLaneModel53.Basis
        (TerminalClaimModel53.claims claims) (AspisR1.SelectorMarginalModel.modelPoint point) := by
  obtain ⟨openings,selectors,mask,helper,composition,g,active,
    hz,hsucc,hxor,hbox,hhelper,hcomposition,hcompCan,hgCan,hactiveCan,hactiveValue,hvalue⟩ :=
    AspisR1.CompositionDetailed53.run_success point s claims lambda chi theta
      hpoint hclaims hlambda hchi htheta D hash c1 outTranscript hprep
  obtain ⟨helper',hhelper',hcell,hhelperCan⟩ :=
    AspisR1SelectedMaskProbe.selected_claim_success claims 0#usize 26#usize
      (by decide) (by decide) hclaims
  have heq : helper' = helper := Result.ok.inj (hhelper'.symm.trans hhelper)
  have hh : Canonical helper := heq ▸ hhelperCan
  obtain ⟨terminal,hterminal,hterminalCan,hterminalValue⟩ :=
    AspisR1.TerminalPartsArithmeticExact.run_exact zc point composition helper active mu
      hzc hpoint hcompCan hh hactiveCan hmu
  have hhelperValue : decode helper = TerminalClaimModel53.claims claims 0 26 := by
    rw [← heq,hcell]
    rfl
  have hopenings := TerminalClaimModel53.openings_eq claims openings hz hsucc hxor
  have hlanes := TerminalLaneModel53.lanes_eq (copyFromStatement s) point openings
    selectors helper lambda chi hpoint hbox
  change decode composition = ∑ i : Fin 29, decode theta ^ i.val *
    TerminalLaneModel53.lanes (copyFromStatement s) point openings selectors helper lambda chi i at hvalue
  rw [hlanes,hopenings] at hvalue
  simp only [TerminalLaneModel53.decode,CompositionDetailed53.decode,decode] at hvalue hhelperValue hactiveValue
  rw [hhelperValue] at hvalue
  have hactive : decode active = R0P.SemSource.activeAt
      (AspisR1.SelectorMarginalModel.modelPoint point) := by
    change AspisR1.ExtractedMaskQM31Exact.decode active = _
    rw [hactiveValue,TerminalLaneModel53.selectors_eq point selectors hpoint hbox,
      AspisR1.SelectorMarginalModel.pointSelector_eq_selAt]
    rfl
  refine ⟨openings.z,mask,g,terminal,?_,hterminalCan,?_⟩
  · simp only [run,TerminalCallable53.run,hcomposition,bind_tc_ok,hterminal]
  · simp only [TerminalPartsArithmeticExact.decode,decode] at hterminalValue hactive ⊢
    rw [hterminalValue,hvalue,hhelperValue,hactive,
      TerminalClaimModel53.decodedPoint_eq,TerminalClaimModel53.decodedPoint_eq]
    rfl

#print axioms run_success
end
end AspisR1.TerminalModel53
