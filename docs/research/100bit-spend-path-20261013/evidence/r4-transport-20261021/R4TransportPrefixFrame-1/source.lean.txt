import RawSamplerSuffixSourceFrames
import RawSamplerSuffixCorrespondence

/-! Exact preimage test for ordinary suffix rows. Agreement of the sampled
field values cannot replace agreement of the preceding duplex state. -/
set_option autoImplicit false
namespace AspisR1PublicPreparation.R4TransportPrefixFrame
open Aeneas Aeneas.Std Result
open AspisR1
open AspisV8R19.DuplexFrames AspisV8R19.SourceDuplexStep
open FS FS2 FS2.Duplex
open TranscriptStep SourceDigestAddresses

/-- All three supplied data components are already equal. The remaining
prefix obligation is exactly the 32-byte carried-state equality. -/
theorem opening_frame_eq_iff (modelState : State) (sourceState : Bytes32)
    (tag : U8) (n : U32) (payload : Slice U8) :
    (bytes modelState ++ [0, decodeByte tag] ++ PublicInitFrame.lengthBytes n.val ++ rawBytes payload =
      flatten (CallerStep.openingAbsorbParts sourceState tag n payload)) ↔
      modelState = decodeState sourceState := by
  rw [SamplerSuffixSourceFrames.opening_absorb_frame]
  constructor
  · intro h
    have ht := congrArg (List.take 32) h
    simp only [List.append_assoc] at ht
    rw [List.take_left' (bytes_length modelState),
      List.take_left' (bytes_length (decodeState sourceState))] at ht
    exact List.ofFn_injective ht
  · intro h
    rw [h]

/-- The byte preimage of every ordinary opening row is the actual source
absorb preimage as soon as its state, label and encoded payload agree.
Applies equally before query row31; it asserts no scan-state transport. -/
theorem absorption_preimage (P : LiteralTranscript.Prefix) (i : Nat)
    (b : SelectedTranscriptParams.Message) (sourceState : Bytes32)
    (tag : U8) (n : U32) (payload : Slice U8)
    (hstate : UnifiedSampler.stateOf SelectedTranscriptParams.params
      (LiteralTranscript.erasePrefix P) = decodeState sourceState)
    (hlabel : SelectedTranscriptParams.params.lbl i = decodeByte tag)
    (hbytes : SelectedTranscriptParams.params.enc b = PublicInitFrame.lengthBytes n.val ++ rawBytes payload) :
    (RawSamplerSuffixCorrespondence.absorption P i b).toBytes =
      flatten (CallerStep.openingAbsorbParts sourceState tag n payload) := by
  unfold RawSamplerSuffixCorrespondence.absorption UnifiedSampler.absorb
  rw [hstate, hlabel, absorbA, Addr.toBytes_ofBytes,
    SamplerSuffixSourceFrames.opening_absorb_frame, hbytes]
  rfl

#print axioms opening_frame_eq_iff
#print axioms absorption_preimage
end AspisR1PublicPreparation.R4TransportPrefixFrame
