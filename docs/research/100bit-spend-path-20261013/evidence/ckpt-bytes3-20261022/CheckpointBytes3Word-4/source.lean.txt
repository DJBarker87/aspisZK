def pack (a b c d : U8) : U32 :=
  UScalar.mk ((a.bv.setWidth 32) ||| ((b.bv.setWidth 32) <<< 8) |||
    ((c.bv.setWidth 32) <<< 16) ||| ((d.bv.setWidth 32) <<< 24))

theorem word_total (b : Slice U8) (i : Usize)
    (bound : i.val + 4 ≤ b.val.length) :
    word_spec b i = ok (pack
      (b.val[i.val]'(by omega)) (b.val[i.val+1]'(by omega))
      (b.val[i.val+2]'(by omega)) (b.val[i.val+3]'(by omega))) := by
  have maxLen := b.property
  have capacity : b.val.length < 2 ^ System.Platform.numBits := by
    have := b.property
    change b.val.length ≤ 2 ^ System.Platform.numBits - 1 at this
    have : 0 < 2 ^ System.Platform.numBits := Nat.two_pow_pos _
    omega
  unfold word_spec
  dsimp only [lift,HAdd.hAdd,instHAddUScalarResult,HShiftLeft.hShiftLeft,instHShiftLeftUScalarIScalarResult]
  simp [Slice.index_usize, UScalar.cast, UScalar.add, UScalar.tryMk,
    UScalar.tryMkOpt, UScalar.shiftLeft_IScalar, UScalar.shiftLeft,
    pack, show i.val < b.val.length by omega,
    show i.val+1 < b.val.length by omega,
    show i.val+2 < b.val.length by omega,
    show i.val+3 < b.val.length by omega,
    show i.val+1 < 2 ^ System.Platform.numBits by omega,
    show i.val+2 < 2 ^ System.Platform.numBits by omega,
    show i.val+3 < 2 ^ System.Platform.numBits by omega]

#print axioms word_total
end AspisR1.CheckpointBytes3Word
