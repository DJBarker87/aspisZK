def rawLimbs (l : QM31Limbs) : RawK :=
  kWords (M31ParserRefinement.valueToU32 (l 0)) (M31ParserRefinement.valueToU32 (l 1))
    (M31ParserRefinement.valueToU32 (l 2)) (M31ParserRefinement.valueToU32 (l 3))

def rawK (x : OraclePortCheckpointAbiC5.K) : RawK := rawLimbs (qm31ExactLimbEquiv.symm x)

theorem raw_ext (x y : RawK)
    (h0 : x.c0.a.val = y.c0.a.val) (h1 : x.c0.b.val = y.c0.b.val)
    (h2 : x.c1.a.val = y.c1.a.val) (h3 : x.c1.b.val = y.c1.b.val) : x = y := by
  rcases x with ⟨⟨a,b⟩,⟨c,d⟩⟩
  rcases y with ⟨⟨e,f⟩,⟨g,h⟩⟩
  have h0' : a = e := UScalar.eq_of_val_eq h0
  have h1' : b = f := UScalar.eq_of_val_eq h1
  have h2' : c = g := UScalar.eq_of_val_eq h2
  have h3' : d = h := UScalar.eq_of_val_eq h3
  cases h0'; cases h1'; cases h2'; cases h3'; rfl

theorem model_shape (bs : QM31Bytes) (w : Fin 4 → U32)
    (hw : ∀ j, (w j).val = (decodeWordLE (qm31LimbBytes bs j)).val) :
    (decodeQM31LE bs).map rawLimbs =
      if Canonical (kWords (w 0) (w 1) (w 2) (w 3)) then
        some (kWords (w 0) (w 1) (w 2) (w 3)) else none := by
  have hc : (∀ j : Fin 4, (decodeWordLE (qm31LimbBytes bs j)).val < m31Modulus) ↔
      Canonical (kWords (w 0) (w 1) (w 2) (w 3)) := by
    simp_rw [← hw]
    change (∀ j : Fin 4, (w j).val < 2147483647) ↔
      (w 0).val < 2147483647 ∧ (w 1).val < 2147483647 ∧
      (w 2).val < 2147483647 ∧ (w 3).val < 2147483647
    constructor
    · intro h; exact ⟨h 0,h 1,h 2,h 3⟩
    · intro h j
      fin_cases j
      · exact h.1
      · exact h.2.1
      · exact h.2.2.1
      · exact h.2.2.2
  unfold decodeQM31LE
  split
  next h =>
    rw [if_pos (hc.mp h),Option.map_some]
    apply congrArg some
    apply raw_ext <;> simp only [rawLimbs,kWords,M31ParserRefinement.valueToU32_val]
    · exact (hw 0).symm
    · exact (hw 1).symm
    · exact (hw 2).symm
    · exact (hw 3).symm
  next h => rw [if_neg (fun c => h (hc.mpr c)),Option.map_none]

def fixed (b : Slice U8) (width : b.val.length = 16) : QM31Bytes :=
  fun j => AspisV8R19.SqueezeOracleBridge.decodeByte (b.val[j.val]'(by omega))

def word (b : Slice U8) (width : b.val.length = 16) (j : Fin 4) : U32 :=
  pack (b.val[4*j.val]'(by omega)) (b.val[4*j.val+1]'(by omega))
    (b.val[4*j.val+2]'(by omega)) (b.val[4*j.val+3]'(by omega))

theorem word_model (b : Slice U8) (width : b.val.length = 16) (j : Fin 4) :
    (word b width j).val = (decodeWordLE (qm31LimbBytes (fixed b width) j)).val := by
  rw [word,pack_value]
  simp only [decodeWordLE,qm31LimbBytes,fixed,Fin.val_mk,limbByteIndex_val,
    AspisV8R19.SqueezeOracleBridge.decodeByte,Nat.zero_add,
    show 1+4*j.val = 4*j.val+1 by omega,
    show 2+4*j.val = 4*j.val+2 by omega,
    show 3+4*j.val = 4*j.val+3 by omega]

theorem fixed_decode (b : Slice U8) (width : b.val.length = 16) :
    (ExactLeafFieldCodec.decodeQM31 (bytesOf b.val)).map rawK =
      (decodeQM31LE (fixed b width)).map rawLimbs := by
  have fb : fixedBytes? 16 (bytesOf b.val) = some (fixed b width) := by
    unfold fixedBytes?
    rw [dif_pos (by simpa [bytesOf] using width)]
    rfl
  simp only [ExactLeafFieldCodec.decodeQM31,decodeQM31Bytes,fb,
    Option.bind_some,Option.map_map,Function.comp_def,rawK,Equiv.symm_apply_apply]

/-- Exact all-input Result equality, retaining canonicality rejection. -/
theorem decode_k_identification (b : Slice U8) (width : b.val.length = 16) :
    decode_k_spec b =
      ok ((ExactLeafFieldCodec.decodeQM31 (bytesOf b.val)).map rawK) := by
  rw [fixed_decode b width,model_shape (fixed b width) (word b width) (word_model b width)]
  have h0 : word_spec b 0#usize = ok (word b width 0) := by
    simpa [word] using word_total b 0#usize (by simp only [width]; scalar_tac)
  have h1 : word_spec b 4#usize = ok (word b width 1) := by
    simpa [word] using word_total b 4#usize (by simp only [width]; scalar_tac)
  have h2 : word_spec b 8#usize = ok (word b width 2) := by
    simpa [word] using word_total b 8#usize (by simp only [width]; scalar_tac)
  have h3 : word_spec b 12#usize = ok (word b width 3) := by
    simpa [word] using word_total b 12#usize (by simp only [width]; scalar_tac)
  rw [k_shape b _ _ _ _ h0 h1 h2 h3]
  split <;> rfl

theorem reject_k (b : Slice U8) (width : b.val.length = 16) :
    decode_k_spec b = ok none ↔ ExactLeafFieldCodec.decodeQM31 (bytesOf b.val) = none := by
  rw [decode_k_identification b width]
  simp

theorem successful_k (b : Slice U8) (width : b.val.length = 16) (q : RawK) :
    decode_k_spec b = ok (some q) ↔
      ∃ x, ExactLeafFieldCodec.decodeQM31 (bytesOf b.val) = some x ∧ rawK x = q := by
  rw [decode_k_identification b width]
  simp
