import CheckpointBytes3Word

set_option autoImplicit false
namespace AspisR1.CheckpointBytes3WordMath
open Aeneas Aeneas.Std Result
open CheckpointBytes3Word

theorem pack_append (a b c d : U8) :
    (pack a b c d).bv = d.bv ++ c.bv ++ b.bv ++ a.bv := by
  change (a.bv.setWidth 32) ||| ((b.bv.setWidth 32) <<< 8) |||
    ((c.bv.setWidth 32) <<< 16) ||| ((d.bv.setWidth 32) <<< 24) = _
  bv_decide

theorem pack_value (a b c d : U8) :
    (pack a b c d).val = a.val + 256*b.val + 65536*c.val + 16777216*d.val := by
  change (pack a b c d).bv.toNat = _
  rw [pack_append]
  simp only [BitVec.toNat_append]
  change ((d.val * 256 + c.val) * 256 + b.val) * 256 + a.val = _
  omega

#print axioms pack_append
#print axioms pack_value
end AspisR1.CheckpointBytes3WordMath
