struct Environment {outcomes:[u8;16],calls:usize,trace:[Kind;16],created:bool,draws:usize,raw:[[u8;3];16],derived:usize,build_fail:bool}
static mut ENV:Option<Environment>=None;
fn with_env<T>(f:impl FnOnce(&mut Environment)->T)->T {unsafe {f(ENV.as_mut().unwrap())}}
fn ask(kind:Kind)->io::Result<()> {
 with_env(|e| {let i=e.calls;e.trace[i]=kind;e.calls+=1;
 match e.outcomes[i] {0=>Ok(()),1=>Err(io::Error(io::ErrorKind::AlreadyExists)),_=>Err(io::Error(io::ErrorKind::Other))}})
}

#[cfg(kani)]
mod proofs {
 use super::*;
 fn setup()->Environment {Environment{outcomes:kani::any(),calls:0,trace:[Kind::Rng;16],created:false,draws:0,raw:kani::any(),derived:0,build_fail:kani::any()}}
 fn install(e:Environment) {unsafe{ENV=Some(e)}}
 fn read()->Environment {with_env(|e|*e)}
 fn expected(e:Environment)->(io::Result<bool>,Environment) {
  install(e);
  let result=(|| {match ask(Kind::Create) {Ok(())=>with_env(|e|e.created=true),Err(x) if x.kind()==io::ErrorKind::AlreadyExists=>return Ok(false),Err(x)=>return Err(x)}
   for kind in [Kind::Magic,Kind::Digest,Kind::Nonce,Kind::FileSync,Kind::DirectoryOpen,Kind::DirectorySync] {ask(kind)?;}
   Ok(true)})();(result,read())
 }
 fn same(a:Environment,b:Environment) {
  assert_eq!(a.calls,b.calls);assert_eq!(a.created,b.created);assert_eq!(a.draws,b.draws);assert_eq!(a.derived,b.derived);
  for i in 0..16 {assert_eq!(a.trace[i],b.trace[i]);assert_eq!(a.outcomes[i],b.outcomes[i]);}
 }
 #[kani::proof] #[kani::unwind(18)]
 fn reservation_all_paths() {
  let e=setup();install(e);
  let mut store=DurableStateOnlyMaskNonceStore{directory:(),last_error:Some(io::ErrorKind::Other)};
  let digest:[u8;32]=kani::any();let nonce:[u8;32]=kani::any();
  let result=store.reserve_fallible(digest,nonce);let after=read();
  let (model,final_state)=expected(e);assert_eq!(result,model);same(after,final_state);assert_eq!(store.last_error,None);
  assert_eq!(after.created,e.outcomes[0]==0);
  install(e);let mut store=DurableStateOnlyMaskNonceStore{directory:(),last_error:None};
  let accepted=store.reserve(digest,nonce);same(read(),after);
  match result {Ok(v)=>{assert_eq!(accepted,v);assert_eq!(store.last_error,None)},Err(err)=>{assert!(!accepted);assert_eq!(store.last_error,Some(err.kind()))}}
 }
 #[kani::proof] #[kani::unwind(18)]
 fn reservation_bytes_blind() {
  let e=setup();install(e);let mut a=DurableStateOnlyMaskNonceStore{directory:(),last_error:None};
  let ra=a.reserve_fallible(kani::any(),kani::any());let sa=read();
  install(e);let mut b=DurableStateOnlyMaskNonceStore{directory:(),last_error:None};
  let rb=b.reserve_fallible(kani::any(),kani::any());let sb=read();
  assert_eq!(ra,rb);same(sa,sb);
 }
 #[kani::proof] #[kani::unwind(100)]
 fn generate_failure_paths() {
  let mut e=setup();let fail:usize=kani::any();kani::assume(fail<16);
  for i in 0..16 {e.outcomes[i]=if i==fail {2}else{0};e.raw[i]=[0;3];}
  install(e);assert!(matches!(StateOnlyAttemptSecrets::generate(),Err(StateOnlyAttemptEntropyError::Os)));
  let after=read();assert_eq!(after.calls,fail+1);assert_eq!(after.draws,fail);assert!(!after.created);assert_eq!(after.derived,0);
 }

  assert!(matches!(StateOnlyAttemptSecrets::generate(),Err(StateOnlyAttemptEntropyError::RepeatedZeroBlock)));
  let after=read();assert_eq!(after.calls,16);assert_eq!(after.draws,16);assert!(!after.created);
 }
 #[kani::proof] #[kani::unwind(100)]
 fn build_reserves_before_derivation() {
  let e=setup();install(e);let nonce:[u8;32]=kani::any();
  let secrets=StateOnlyAttemptSecrets{mask_nonce:nonce,field_mask_entropy:Zeroizing::new(kani::any()),leaf_salt_seed:Zeroizing::new(kani::any())};
  let context=StateOnlyHidingContext{mask_nonce:nonce,statement_digest:kani::any(),mask_layout_fingerprint:1,layout_factor_fingerprint:2};

  let after=read();let success=(0..7).all(|i|e.outcomes[i]==0);
  assert_eq!(after.derived,if success {1}else{0});assert_eq!(after.created,e.outcomes[0]==0);
  if !success {assert!(matches!(result,Err(StateOnlyMaskBuildError::ReusedMaskNonce)));}
  if success && e.build_fail {assert!(matches!(result,Err(StateOnlyMaskBuildError::Sample)));assert!(after.created);}
 }
