const CLAIMS:usize=87;const LANES:usize=29;
pub const fn row_shape(row: u8) -> Option<(u8, usize)> {
    match row {
        0 | 1 | 3..=13 => Some((0, 0)),
        2 => Some((1, 32)),
        14 => Some((3, 32)),
        15..=24 => Some((2, 28 * 32)),
        25 => Some((4, (CLAIMS + LANES) * 32)),
        26 => Some((5, LANES * 32)),
        _ => None,
    }
}
pub fn record_range(row:u8)->(usize,usize) {
    let mut i=0u8;let mut at=32usize;
    while i<row {at+=5+row_shape(i).unwrap().1;i+=1;}
    (at,5+row_shape(row).unwrap().1)
}
