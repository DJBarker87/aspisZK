def initialize (bound : Nat) (fits : 65 ≤ bound) (seed : OracleAnswer)
    (attempt : Fin 8) (recording : Bool) (t : Tab bound) : LoggedMachine :=
  (initialMachine bound fits seed attempt recording t,[])

theorem source_law {A : Type} (bound : Nat) [Fintype (RomFiniteBin.Address bound)]
    (fits : 65 ≤ bound) (seed : OracleAnswer) (attempt : Fin 8) (recording : Bool)
    (continuation : Port LoggedMachine A) (test : Result (A × LoggedMachine) → ℚ) :
    mean (fun p : Input × Tab bound => test (continuation
      (initialize bound fits seed attempt recording (table bound fits seed p)))) =
    mean (fun t : Tab bound => test (continuation
      (initialize bound fits seed attempt recording t))) := by
  exact (QueryScanIdealHistory.mean_fintype_transport _ _ _).trans
    ((source_oracle_uniform bound fits seed
      (fun t => test (continuation (initialize bound fits seed attempt recording t)))).trans
      (QueryScanIdealHistory.mean_fintype_transport _ _ _))

theorem model_law {A : Type} (bound : Nat) [Fintype (RomFiniteBin.Address bound)]
    (fits : 65 ≤ bound) (seed : OracleAnswer) (attempt : Fin 8) (recording : Bool)
    (continuation : Port LoggedMachine A) (test : Result (A × LoggedMachine) → ℚ) :
    mean (fun p : Input × Tab bound => test (continuation
      (initialize bound fits seed attempt recording p.2))) =
    mean (fun t : Tab bound => test (continuation
      (initialize bound fits seed attempt recording t))) := by
  exact (QueryScanIdealHistory.mean_fintype_transport _ _ _).trans
    ((R1Z.CouplingR1PBin.mp_snd
      (fun t => test (continuation (initialize bound fits seed attempt recording t)))).trans
      (QueryScanIdealHistory.mean_fintype_transport _ _ _))

theorem initial_product (bound : Nat) (fits : 65 ≤ bound) (seed : OracleAnswer)
    (attempt : Fin 8) (recording : Bool) (p : Input × Tab bound) :
    initialize bound fits seed attempt recording (table bound fits seed p) =
    initialize bound fits seed attempt recording p.2 := by
  unfold initialize
  rw [ProverCommitMachineMarginal27.initial_product]
