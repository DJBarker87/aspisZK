import R0P.Zerocheck
import Mathlib.Algebra.BigOperators.Group.Finset.Pi
/-! Boolean-suffix pruning as a symbolic finite-product identity. No cube
or 1024-entry concrete universe is evaluated. Native bit/address binding is separate. -/
set_option autoImplicit false
noncomputable section
namespace R1Z.ProverCommitSparse50
open scoped BigOperators
variable {K : Type} [Field K]

def weight {I : Type} [Fintype I] (p : I → K) (b : I → Bool) : K :=
  ∏ i, if b i then p i else 1-p i

theorem eqwB_product (n : Nat) (p : Fin n → K) (b : Fin n → Bool) :
    R0P.Sumcheck.eqwB n p b=weight p b := by
  induction n with
  | zero => simp [R0P.Sumcheck.eqwB,weight]
  | succ n ih =>
    rw [R0P.Sumcheck.eqwB,ih]
    simp only [weight,Fin.prod_univ_succ,Fin.tail]

theorem boolean_weight {I : Type} [Fintype I] [DecidableEq I] (s b : I → Bool) :
    weight (fun i => if s i then (1:K) else 0) b=if b=s then 1 else 0 := by
  classical
  by_cases h : b=s
  · subst b
    rw [if_pos rfl]
    unfold weight
    apply Finset.prod_eq_one
    intro i hi
    cases s i <;> simp
  · rw [if_neg h]
    obtain ⟨i,hi⟩ : ∃ i, b i≠s i := Function.ne_iff.mp h
    apply Finset.prod_eq_zero (Finset.mem_univ i)
    cases hb : b i <;> cases hs : s i <;> simp_all

theorem prune {I J : Type} [Fintype I] [Fintype J] [DecidableEq I] [DecidableEq J]
    (p : I → K) (suffix : J → Bool) (f : (I → Bool) → (J → Bool) → K) :
    (∑ a : I → Bool, ∑ b : J → Bool,
      weight p a*weight (fun j => if suffix j then (1:K) else 0) b*f a b)=
      ∑ a : I → Bool, weight p a*f a suffix := by
  classical
  apply Finset.sum_congr rfl
  intro a ha
  simp only [boolean_weight]
  simp

#print axioms eqwB_product
#print axioms boolean_weight
#print axioms prune
end R1Z.ProverCommitSparse50
