#![allow(dead_code)]
const P:u32=2147483647;
const QUERY_COUNT:usize=22;
const QUERY_MAX_DRAWS:usize=64;
#[derive(Clone,Copy)]
struct QueryScan {
    accepted: [u32; QUERY_COUNT],
    accepted_len: usize,
    draws: usize,
}
fn qm31_div_pass(mut rank: [u32; 8]) -> ([u32; 8], u32) {
    let mut remainder = 0u64;
    let mut j = 8usize;
    while j > 0 {
        j -= 1;
        let dividend = (remainder << 32) | u64::from(rank[j]);
        rank[j] = (dividend / u64::from(P)) as u32;
        remainder = dividend % u64::from(P);
    }
    (rank, remainder as u32)
}
fn division_step(mut word:u32, mut remainder:u64)->(u32,u64) {let limb=&mut word;let dividend = (remainder << 32) | u64::from(*limb);
*limb = (dividend / u64::from(P)) as u32;
remainder = dividend % u64::from(P); (word,remainder)}
fn addition_step(mut word:u32, mut addend:u32)->(u32,u32) {let digit=&mut word;let sum = u64::from(*digit) + u64::from(addend);
*digit = (sum % u64::from(P)) as u32;
addend = (sum / u64::from(P)) as u32; (word,addend)}
impl QueryScan {fn step(&mut self,candidate:u32)->bool {if self.accepted_len == QUERY_COUNT || self.draws == QUERY_MAX_DRAWS {
return true;
}
self.draws += 1;
if !self.accepted[..self.accepted_len].contains(&candidate) {
self.accepted[self.accepted_len] = candidate;
self.accepted_len += 1;
} false}}
fn add_small_base_p(digits: &mut [u32; 8], mut addend: u32) -> bool {
    for digit in digits {
        let sum = u64::from(*digit) + u64::from(addend);
        *digit = (sum % u64::from(P)) as u32;
        addend = (sum / u64::from(P)) as u32;
    }
    debug_assert!(addend <= 1);
    addend != 0
}
fn low_base_p_digits(block: &[u8; 32]) -> ([u32; 8], u32) {
    let mut quotient: [u32; 8] =
        core::array::from_fn(|i| u32::from_le_bytes(block[4 * i..4 * i + 4].try_into().unwrap()));
    let digits = core::array::from_fn(|_| {
        let mut remainder = 0u64;
        for limb in quotient.iter_mut().rev() {
            let dividend = (remainder << 32) | u64::from(*limb);
            *limb = (dividend / u64::from(P)) as u32;
            remainder = dividend % u64::from(P);
        }
        remainder as u32
    });

    // After eight divisions, the quotient is at most 256 and fits in u32.
    debug_assert!(quotient[1..].iter().all(|&limb| limb == 0));
    debug_assert!(quotient[0] <= 256);
    (digits, quotient[0])
}

 fn division_limb() {
  let word:u32=kani::any();let r:u64=kani::any();kani::assume(r<u64::from(P));
  let (q,rem)=division_step(word,r);
  let dividend=r*4294967296+u64::from(word);
  assert_eq!(u64::from(q),dividend/u64::from(P));
  assert_eq!(rem,dividend%u64::from(P));
  assert!(rem<u64::from(P));
 }
 #[kani::proof]
 fn addition_limb() {
  let word:u32=kani::any();let a:u32=kani::any();kani::assume(word<P);kani::assume(a<=257);
  let (digit,carry)=addition_step(word,a);
  let sum=u64::from(word)+u64::from(a);
  assert_eq!(u64::from(digit),sum%u64::from(P));
  assert_eq!(u64::from(carry),sum/u64::from(P));
  assert!(digit<P);assert!(carry<=1);
 }
 // Every array write, stop branch and unchanged cell for one actual source word.
 fn scan_step<const N:usize>() {
  let accepted:[u32;22]=kani::any();let draws:usize=kani::any();let x:u32=kani::any();
  kani::assume(draws<=64);
  let mut actual=QueryScan{accepted,accepted_len:N,draws};
  let mut duplicate=false;
  for i in 0..N { if accepted[i]==x {duplicate=true;} }
  let stop=actual.step(x);
  assert_eq!(stop,N==22 || draws==64);
  let active=N<22 && draws<64;
  assert_eq!(actual.draws,draws+usize::from(active));
  assert_eq!(actual.accepted_len,N+usize::from(active && !duplicate));
  for i in 0..22 {
   assert_eq!(actual.accepted[i],if active && !duplicate && i==N {x} else {accepted[i]});
  }
 }
