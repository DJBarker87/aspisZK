theorem stored_power (u : Fin 262144) :
    AspisV7ExactOneFoldDomains.storedInitialFibrePoint20 u=
      g^(1024+4096*AspisV5FriBitReverse.reverseBits 18 u.val) := by
  rw [AspisV7ExactOneFoldDomains.storedInitialFibrePoint20_eq_zpow,
    AspisV6EncoderDistance.initialCircleExponent,
    AspisV7ExactOneFoldDomains.storedInitialNaturalIndex20_child_zero]
  have he : (2 : Int)^10*(2*(2*(AspisV5FriBitReverse.reverseFin 18 u).val)+1)=
      ((1024+4096*AspisV5FriBitReverse.reverseBits 18 u.val : Nat) : Int) := by
    simp only [AspisV5FriBitReverse.reverseFin,Nat.cast_add,Nat.cast_mul,Nat.cast_ofNat]
    ring
  rw [he,zpow_natCast]

theorem x_square (z : C) : X (z^2)=2*(X z)^2-1 := by
  rw [pow_two]
  change z.1.1*z.1.1-z.1.2*z.1.2=2*z.1.1^2-1
  have hn : z.1.1^2+z.1.2^2=1 := z.2
  linear_combination -hn

theorem doubled_circle (z : C) (n : Nat) :
    doubledFactor (X z) n=X (z^(2^n)) := by
  induction n with
  | zero => simp only [doubledFactor,pow_zero,pow_one]
  | succ n ih =>
    rw [doubledFactor,ih,←x_square,←pow_mul,pow_succ]

      ProverCommitNativeC49.value z=embed (AspisV7ExactOneFoldDomains.storedInitialFibrePoint20 u) := by
  obtain ⟨z,hz,hcz,hdz⟩ := ProverCommitEncoderPoint54.exact_result j
  refine ⟨z,hz,hcz,?_⟩
  rw [hdz,hj,generator_value,stored_power,map_pow]
