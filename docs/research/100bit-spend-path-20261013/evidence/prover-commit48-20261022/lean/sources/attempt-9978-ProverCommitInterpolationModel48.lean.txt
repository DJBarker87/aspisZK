theorem basis_map {K L : Type} [Field K] [Field L] (f : K →+* L) (i : Fin 28) :
    (Lagrange.basis Finset.univ (fun j : Fin 28 => (j.val : K)) i).map f=
      Lagrange.basis Finset.univ (fun j : Fin 28 => (j.val : L)) i := by
  simp only [Lagrange.basis,Lagrange.basisDivisor,map_prod,map_mul,map_C,map_sub,map_X,
    map_inv₀,map_natCast]

theorem contribution_basis (q : Old.Q) (i : Fin 28) :
    contribution q i.val=C (decode q)*Lagrange.basis Finset.univ (fun j : Fin 28 => (j.val : QM31Exact)) i := by
  have h := congrArg (Polynomial.map embed)
    (ProverCommitLagrangeProduct48.normalized_basis (K := ProverCommitM31Ops48.M) i)
  rw [Polynomial.map_mul,Polynomial.map_C,basis_map] at h
  rw [contribution,← h]
  simp only [map_mul]
  ring

theorem sum_range (values : Nat → Old.Q) (start n : Nat) :
    rowSum values start n=∑ k : Fin n, contribution (values (start+k.val)) (start+k.val) := by
  induction n generalizing start with
  | zero => simp only [rowSum,Fin.sum_univ_zero]
  | succ n ih =>
    rw [rowSum,Fin.sum_univ_succ,ih]
    simp only [Fin.val_zero,Nat.add_zero]
    congr 1
    apply Finset.sum_congr rfl
    intro k hk
    have he : (start+1)+k.val=start+(Fin.succ k).val := by
      change (start+1)+k.val=start+(k.val+1)
      omega
    rw [he]

theorem sum_model (values : Nat → Old.Q) :
    rowSum values 0 28=Lagrange.interpolate Finset.univ
      (fun j : Fin 28 => (j.val : QM31Exact)) (fun j => decode (values j.val)) := by
  rw [sum_range,Lagrange.interpolate_apply]
  simp only [Nat.zero_add]
  apply Finset.sum_congr rfl
  intro i hi
  exact contribution_basis (values i.val) i

theorem interpolate_model (values : Nat → Old.Q) :
    ∃ ys : Nat → Old.Q,
      interpolate (ProverCommitQArray48.array 28#usize values)=ok (ProverCommitQArray48.array 28#usize ys) ∧
      (∀ k, k < 28 → Canonical (ys k)) ∧
      qpoly ys=Lagrange.interpolate Finset.univ
        (fun j : Fin 28 => (j.val : QM31Exact)) (fun j => decode (values j.val)) := by
  obtain ⟨ys,hy,hcy,hdy⟩ := interpolate_value values
  exact ⟨ys,hy,hcy,hdy.trans (sum_model values)⟩
