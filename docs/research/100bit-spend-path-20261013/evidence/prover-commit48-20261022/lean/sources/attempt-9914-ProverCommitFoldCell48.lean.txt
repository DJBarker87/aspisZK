import R1Z.ProverCommitMix48
/-! Exact full-width primal four-child fold. The source expression is bound
by the native Kani fold-cell gate; all six arithmetic results are proved. -/
set_option autoImplicit false
noncomputable section
namespace R1Z.ProverCommitFoldCell48
open Aeneas Aeneas.Std Result AspisR1ProducerRelease47
open ProverCommitWideOps48 ProverCommitMix48
attribute [local irreducible] sourceArray source value

def polynomial (q : Nat → Raw) (a : Exact) : Exact :=
  value (q 0)+a*value (q 1)+a^2*value (q 2)+a^3*value (q 3)

theorem cell (q : Nat → Raw) (hq : ∀ j, j<4 → canonical (q j))
    (a : Raw) (ha : canonical a) :
    ∃ z : Raw, fold_cell (sourceArray 4#usize q) (source a)=ok (source z) ∧
      canonical z ∧ value z=polynomial q (value a) := by
  obtain ⟨m3,hm3,hcm3,hdm3⟩ := mul_exact a (q 3) ha (hq 3 (by decide))
  obtain ⟨s2,hs2,hcs2,hds2⟩ := add_exact (q 2) m3 (hq 2 (by decide)) hcm3
  obtain ⟨m2,hm2,hcm2,hdm2⟩ := mul_exact a s2 ha hcs2
  obtain ⟨s1,hs1,hcs1,hds1⟩ := add_exact (q 1) m2 (hq 1 (by decide)) hcm2
  obtain ⟨m1,hm1,hcm1,hdm1⟩ := mul_exact a s1 ha hcs1
  obtain ⟨z,hz,hcz,hdz⟩ := add_exact (q 0) m1 (hq 0 (by decide)) hcm1
  refine ⟨z,?_,hcz,?_⟩
  · rw [fold_cell,index _ _ 0#usize (by decide),index _ _ 1#usize (by decide),
      index _ _ 2#usize (by decide),index _ _ 3#usize (by decide)]
    simp only [bind_tc_ok,hm3,hs2,hm2,hs1,hm1,hz]
  · rw [hdz,hdm1,hds1,hdm2,hds2,hdm3,polynomial]
    ring

#print axioms cell
end R1Z.ProverCommitFoldCell48
