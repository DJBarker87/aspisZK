import R1Z.ProverCommitBasisPolynomial48
import R1Z.ProverCommitInterpolationIndices48
/-! Complete owned omitted-node factor loop. Polynomial, denominator,
canonicality and the exact degree schedule are propagated together. -/
set_option autoImplicit false
noncomputable section
namespace R1Z.ProverCommitInterpolationFactors48
open Aeneas Aeneas.Std Result AspisR1ProducerRelease47
open ProverCommitM31Ops48 ProverCommitM31Array48
open ProverCommitBasisPolynomial48 ProverCommitInterpolationSchedule48
open ProverCommitInterpolationIndices48
attribute [local irreducible] array poly polyProd denProd

theorem factors_value (i : U8) (hi : i.val < 28) (n : Nat)
    (xs : Nat → U32) (hxs : ∀ k, k < 28 → canonical (xs k))
    (den : U32) (hden : canonical den) (deg j left : U8)
    (hleft : left.val=n) (count : j.val+n=28)
    (hdeg : deg.val=degree i.val j.val)
    (space : ∀ k, k < 28 → deg.val+1 ≤ k → decode (xs k)=0) :
    ∃ (ys : Nat → U32) (denOut : U32),
      basis_factors (array 28#usize xs) den deg i j left=ok (array 28#usize ys,denOut) ∧
      (∀ k, k < 28 → canonical (ys k)) ∧ canonical denOut ∧
      poly ys=poly xs*polyProd i.val j.val n ∧
      decode denOut=decode den*denProd i.val j.val n := by
  induction n generalizing xs den deg j left with
  | zero =>
    have hz : left=0#u8 := by scalar_tac
    refine ⟨xs,den,?_,hxs,hden,?_,?_⟩
    · rw [basis_factors,hz];rfl
    · rw [polyProd,mul_one]
    · rw [denProd,mul_one]
  | succ n ih =>
    have hn : left≠0#u8 := by scalar_tac
    have hdec : (U8.wrapping_sub left 1#u8).val=n := by
      rw [ProverCommitBasisCounter47.sub_one left (by omega)];omega
    have hj : j.val < 28 := by omega
    have hjnext : (U8.wrapping_add j 1#u8).val=j.val+1 := next_val j hj
    by_cases same : i=j
    · subst j
      have hdnew : deg.val=degree i.val (U8.wrapping_add i 1#u8).val := by
        rw [hjnext,skipped_degree];exact hdeg
      obtain ⟨ys,dout,hy,hcy,hcdout,hpoly,hdenval⟩ := ih
        (xs := xs) (hxs := hxs) (den := den) (hden := hden) (deg := deg)
        (j := U8.wrapping_add i 1#u8) (left := U8.wrapping_sub left 1#u8)
        (hleft := hdec) (count := by rw [hjnext];omega) (hdeg := hdnew) (space := space)
      refine ⟨ys,dout,?_,hcy,hcdout,?_,?_⟩
      · rw [basis_factors]
        simp only [hn,if_false,if_true,lift,bind_tc_ok]
        exact hy
      · simpa only [hjnext,polyProd,if_true] using hpoly
      · simpa only [hjnext,denProd,if_true] using hdenval
    · have different : i.val≠j.val := fun he => same (UScalar.val_eq_imp i j he)
      have room : deg.val+2 ≤ 28 := by rw [hdeg];exact selected_room _ _ hi hj different
      have hwidth : (U8.wrapping_add deg 2#u8).val=deg.val+2 := by scalar_tac
      obtain ⟨basis,hbasis,hcbasis,hpolybasis,hspacebasis⟩ :=
        ProverCommitBasisPolynomial48.multiplication xs hxs (UScalar.cast .U32 j)
          (deg.val+2) (by omega) (U8.wrapping_add deg 2#u8) hwidth room
          (fun k hk hs => space k hk (by omega))
      obtain ⟨difference,hdiff,hcdiff,hddiff⟩ := sub_exact (UScalar.cast .U32 i) (UScalar.cast .U32 j)
        (cast32_canonical i) (cast32_canonical j)
      obtain ⟨denNext,hdenNext,hcdenNext,hdDenNext⟩ := mul_exact den difference
      have hdegNext : (U8.wrapping_add deg 1#u8).val=deg.val+1 := by scalar_tac
      have hdegree : (U8.wrapping_add deg 1#u8).val=
          degree i.val (U8.wrapping_add j 1#u8).val := by
        rw [hdegNext,hjnext,hdeg,selected_degree _ _ different]
      have hspace : ∀ k, k < 28 → (U8.wrapping_add deg 1#u8).val+1 ≤ k → decode (basis k)=0 := by
        intro k hk hs
        apply hspacebasis k hk
        rw [hdegNext] at hs
        omega
      obtain ⟨ys,dout,hy,hcy,hcdout,hpoly,hdenval⟩ := ih
        (xs := basis) (hxs := hcbasis) (den := denNext) (hden := hcdenNext)
        (deg := U8.wrapping_add deg 1#u8) (j := U8.wrapping_add j 1#u8)
        (left := U8.wrapping_sub left 1#u8) (hleft := hdec)
        (count := by rw [hjnext];omega) (hdeg := hdegree) (space := hspace)
      refine ⟨ys,dout,?_,hcy,hcdout,?_,?_⟩
      · rw [basis_factors]
        simp only [hn,if_false,same,lift,bind_tc_ok,hbasis,hdiff,hdenNext]
        exact hy
      · rw [hpoly,hpolybasis,cast32_decode,hjnext,polyProd,if_neg different]
        ring
      · rw [hdenval,hdDenNext,hddiff,cast32_decode,cast32_decode,hjnext,denProd,if_neg different]
        ring

#print axioms factors_value
end R1Z.ProverCommitInterpolationFactors48
