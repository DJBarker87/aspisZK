theorem empty_node_count (indices : Slice U32) (empty : Slice.len indices = 0#usize) :
    host_tree.node_count indices = .ok none := by
  simp only [host_tree.node_count,host_tree.check_indices,
    core.slice.Slice.is_empty,empty]
