/-- Full symbolic source-function equality for `prover.publish_compact`. -/
theorem erase_420 (ops other : ExternalFuns) :
    @prover.publish_compact _ (replace ops other) = @prover.publish_compact _ ops := by
  funext variant p
  delta prover.publish_compact
  simp only [keep_op12 ops other, keep_op51 ops other, keep_op234 ops other, keep_op242 ops other, keep_op243 ops other, keep_op244 ops other, keep_op245 ops other, erase_396 ops other, erase_405 ops other, erase_419 ops other]
  all_goals
    congr 1
  all_goals
    set_option pp.deepTerms true in
    trace_state
  all_goals with_reducible rfl
#print axioms erase_420
