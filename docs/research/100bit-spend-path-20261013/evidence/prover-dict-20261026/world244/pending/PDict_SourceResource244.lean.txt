/-- Error data is observed after consuming the event, never by rewinding it.
This is an actual extracted OWNED REFERENCE transition, not a native File law. -/
theorem error_advances (tape : Tape) (operation : U8) (reply : Reply)
    (small : tape.used.val < 16)
    (read : Array.index_usize tape.replies tape.used = .ok reply)
    (same : reply.operation=operation) :
    (do let out ← PDict_TapeSource244.world.next tape operation
        .ok (out.reply.map (fun r => r.error),out.tape.used.val,out.tape.replies)) =
      .ok (some reply.error,tape.used.val+1,tape.replies) := by
  rw [PDict_TapeCells244.matched tape operation reply small read same]
  simp only [bind_tc_ok,Option.map_some]
  rw [PDict_Advance244.value tape.used (by omega)]

theorem reference_total (tape : Tape) (operation : U8) :
    ∃ out, PDict_TapeSource244.world.next tape operation = .ok out := by
  have law := PDict_TapeCorrect244.reference_next tape operation
  cases run : PDict_TapeSource244.world.next tape operation with
  | ok out => exact ⟨out,run⟩
  | fail error => simp only [run,bind_tc_fail] at law
  | div => simp only [run,bind_tc_div] at law

#print axioms error_advances
#print axioms reference_total
end R1Z.PDict_SourceResource244
