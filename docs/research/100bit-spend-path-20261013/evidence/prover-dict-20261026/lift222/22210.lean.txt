theorem bind_congr {A B : Type} (r : Result A) (f g : A → Result B)
    (h : ∀ a, f a = g a) : (do let a ← r; f a) = (do let a ← r; g a) := by
  cases r with
  | ok a => exact h a
  | fail e => rfl
  | div => rfl
#print axioms bind_congr

theorem map_prod {A B C D : Type} (f : C → D) (p : A × B)
    (g : A → B → Result C) :
    resultMap f (Prod.rec g p) =
      Prod.rec (fun a b => resultMap f (g a b)) p := by cases p; rfl

theorem map_result {A B C D : Type} (f : C → D) (r : core.result.Result A B)
    (g : A → Result C) (h : B → Result C) :
    resultMap f (core.result.Result.rec g h r) =
      core.result.Result.rec (fun a => resultMap f (g a)) (fun b => resultMap f (h b)) r := by
  cases r <;> rfl

theorem map_control {A B C D : Type} (f : C → D) (r : core.ops.control_flow.ControlFlow A B)
    (g : B → Result C) (h : A → Result C) :
    resultMap f (core.ops.control_flow.ControlFlow.rec g h r) =
      core.ops.control_flow.ControlFlow.rec (fun b => resultMap f (g b)) (fun a => resultMap f (h a)) r := by
  cases r <;> rfl

#print axioms map_result
#print axioms map_control

theorem fallible (b : ExternalTypes) (F : Type) (ops : @ExternalFuns b)
    (s : @Store b) (sd nonce : Array U8 32#usize) :
    @aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.reserve_fallible
      (types b F) (selected b F ops) (store b F s) sd nonce =
    resultMap (pairStore b F) (@aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.reserve_fallible b ops s sd nonce) := by
  delta aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.reserve_fallible
  have hs :
      @aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.mk (types b F)
        (@aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.directory (types b F) (store b F s)) none =
      store b F (@aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.mk b
        (@aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.directory b s) none) := rfl
  rw [hs, PDict_FormatterStorePath.reservation_path]
  delta aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.reserve_fallible.match_1
  delta aspis_prover.state_only_entropy.DurableStateOnlyMaskNonceStore.reserve_fallible.match_2
  simp only [map_bind, map_prod, map_result, map_control, map_ok, pairStore, PDict_FormatterIOMap.io14, PDict_FormatterIOMap.io15, PDict_FormatterIOMap.io19, PDict_FormatterIOMap.io20, PDict_FormatterIOMap.io21, PDict_FormatterIOMap.io22, PDict_FormatterIOMap.io23, PDict_FormatterIOMap.io24, PDict_FormatterIOMap.io25, PDict_FormatterIOMap.io26, PDict_FormatterIOMap.io27, PDict_FormatterIOMap.io28, PDict_FormatterIOMap.io29, PDict_FormatterIOMap.io30, PDict_FormatterIOMap.io31, PDict_FormatterIOMap.io261, PDict_FormatterIOMap.sync_directory]
  repeat first
    | rfl
    | (apply bind_congr; intro x; (try dsimp only); (try simp only [map_bind, map_ok]))
    | (casesm* (_ × _), core.result.Result _ _, core.ops.control_flow.ControlFlow _ _, Bool <;>
        (try dsimp only) <;> (try simp only [map_bind, map_ok]))

#print axioms fallible
