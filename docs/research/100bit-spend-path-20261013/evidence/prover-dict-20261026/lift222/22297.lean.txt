theorem masked_store (ops : ExternalFuns) {A B : Type}
    (sa : aspis_prover.state_only_hiding.StateOnlyMaskNonceStore A)
    (sb : aspis_prover.state_only_hiding.StateOnlyMaskNonceStore B) (f : A → B)
    (hr : ∀ s sd n, sb.reserve (f s) sd n = resultMap (pairMap f) (sa.reserve s sd n))
    (public1 : aspis_statement.pool_v1.pair_forest_semantic_terminal.r0.Public)
    (compiled : aspis_statement.pool_v1.pair_forest_trace.PoolV1PairForestMergedC1CompilationV1)
    (variant : transport.Variant) (hash : Slice (Slice U8) → Result (Array U8 32#usize))
    (ctx : aspis_core.state_only_hiding.StateOnlyHidingContext) (binding : Array U8 32#usize)
    (attempt : aspis_prover.state_only_entropy.StateOnlyAttemptSecrets) (s : A) :
    @prover.prove_masked_with_attempt _ ops B sb public1 compiled variant hash ctx binding attempt (f s) =
    resultMap (pairMap f) (@prover.prove_masked_with_attempt _ ops A sa public1 compiled variant hash ctx binding attempt s) := by
  delta prover.prove_masked_with_attempt
  run_tac do
    let env ← getEnv
    for (n, _) in env.constants.toList do
      if "ProverRwFull.prover.prove_masked_with_attempt.match_".isPrefixOf n.toString then
        evalTactic (← `(tactic| try delta $(mkIdent n)))
  rw [PDict_MaterialStore222.material_store ops sa sb f hr]
  rw [bind_map, map_bind]
  apply PDict_FallibleTransport.bind_congr
  rintro ⟨r, s'⟩
  simp only [pairMap, Aeneas.Std.uncurry]
  repeat first
    | (rw [map_bind]; apply PDict_FallibleTransport.bind_congr; intro x;
        (try simp only [pairMap, Aeneas.Std.uncurry]))
    | (casesm* (_ × _), core.result.Result _ _, core.ops.control_flow.ControlFlow _ _, Bool <;>
        (try simp only [pairMap, Aeneas.Std.uncurry]))
    | (rw [apply_ite])
    | split
    | rfl
  all_goals simp only [map_bind, map_ok, pairMap]
  all_goals rfl

#print axioms masked_store
end R1Z.PDict_MaskedStore222
