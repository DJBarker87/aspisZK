/-! Real caller composition only. The extra state retains oracle queries,
answers, TLS and ordered observations, including rejected prefixes. The
public pre and post-reservation body are parameters with separate, visible
contracts. Neither contract is a whole-experiment join or a distribution law.
E1 and E2 remain the existing environment items; deterministic transport does
not require an error-free schedule, a ledger cap, or a random public nonce. -/
set_option autoImplicit false
noncomputable section
namespace R1Z.RealR1_OuterOwner
open Aeneas Aeneas.Std Result AspisR1PublicNonce41
open ProverCommitRetry34
abbrev LedgerState := ProverCommitLedger36.State
abbrev Bytes := ProverCommitOwnerZero42.Bytes
abbrev Raw := ProverCommitPublicNonce40.Raw
abbrev Secret := ProverCommitRetryOwner42.Secret
abbrev Pub := ProverCommitDomain34.Source
abbrev PubBytes := ProverCommitDomain34.SourceBytes

/-- Extra is not erased or reset on retry; it can contain the full ROM state. -/
abbrev Store (Extra : Type) := LedgerState × Extra
abbrev NoncePolicy (Extra : Type) := Nat → Extra → Bytes
abbrev Prefix (Extra : Type) := R0Z.FsStatement.OracleAnswer → PubBytes → Extra →
  Result (Status ProverCommitLedger36.Path × Extra)
abbrev Body (Extra Proof : Type) := Secret → PubBytes → Store Extra →
  Result (Status Proof × Store Extra)

def nativeEntropy {Extra : Type} (nonce : NoncePolicy Extra)
    (schedule : ProverCommitLedger36.Schedule) (tape : Nat → owner.Raw) :
    Entropy (Store Extra) Secret := fun index s => do
  let (result,after) ← ProverCommitRetryOwner42.nativeEntropy
    (fun _ => nonce index s.2) schedule tape index s.1
  ok (result,(after,s.2))

def modelEntropy {Extra : Type} (nonce : NoncePolicy Extra)
    (schedule : ProverCommitLedger36.Schedule) (tape : Nat → owner.Raw) :
    Entropy (Store Extra) Secret := fun index s => do
  let (result,after) ← ProverCommitRetryOwner42.modelEntropy
    (fun _ => nonce index s.2) schedule tape index s.1
  ok (result,(after,s.2))

theorem entropy_exact {Extra : Type} (nonce : NoncePolicy Extra)
    (schedule : ProverCommitLedger36.Schedule) (tape : Nat → owner.Raw)
    (index : Nat) (s : Store Extra) :
    nativeEntropy nonce schedule tape index s = modelEntropy nonce schedule tape index s := by
  unfold nativeEntropy modelEntropy
  rw [ProverCommitRetryOwner42.entropy_model]

/-- Public context/binding reads and layout checks occur before reservation.
The pre cannot inspect either private seed. Its address must be bound to
the source reservation_path by the native caller instance, not by this type.
Reservation failures map to Semantic, as ReusedMaskNonce via mask_error does.
The private mask/prover body is invoked only after successful reservation. -/
def reserved {Extra Proof : Type} (pre : Prefix Extra)
    (schedule : ProverCommitLedger36.Schedule) (unix : Bool)
    (body : Body Extra Proof) : Body Extra Proof := fun secret bytes s => do
  let (address,extra) ← pre secret.1 bytes s.2
  match address with
  | .Err e => ok (.Err e,(s.1,extra))
  | .Ok path =>
    let (result,after) := ProverCommitLedger36.reserve schedule unix path s.1
    match result with
    | .error _ => ok (.Err .Semantic,(after,extra))
    | .ok _ => body secret bytes (after,extra)

/-- Named open R2/R3 interface: complete returned proof/audit, errors and all
state, not merely acceptance or commitment-root equality. This is deliberately
strong and pointwise; an except-bad implementation instance must supply it
on the relevant coupled execution, not silently assert it for every ROM. -/
def BodyContract {Extra Proof : Type} (left right : Body Extra Proof) : Prop :=
  ∀ secret bytes s, left secret bytes s = right secret bytes s

def PrefixContract {Extra : Type} (left right : Prefix Extra) : Prop :=
  ∀ nonce bytes s, left nonce bytes s = right nonce bytes s

theorem reserved_exact {Extra Proof : Type}
    (leftPrefix rightPrefix : Prefix Extra) (pre : PrefixContract leftPrefix rightPrefix)
    (schedule : ProverCommitLedger36.Schedule) (unix : Bool)
    (left right : Body Extra Proof) (body : BodyContract left right) :
    BodyContract (reserved leftPrefix schedule unix left) (reserved rightPrefix schedule unix right) := by
  have hp : leftPrefix = rightPrefix := funext fun n => funext fun b => funext (pre n b)
  have hb : left = right := funext fun n => funext fun b => funext (body n b)
  rw [hp,hb]
  intro secret bytes s
  rfl

def selected {Extra Proof : Type} (s2 : Bool) (pub : Pub)
    (entropy : Entropy (Store Extra) Secret) (continuation : Body Extra Proof)
    (s : Store Extra) : Result (Step (Published Proof) × Store Extra) :=
  if s2 then eight (realAttempt pub entropy continuation) s
  else ok (.stop .Semantic,s)

/-- Real outer owner, including S2 admission, the actual extracted 64-byte
owner, serializer, public-pre reads, faithful reservation and eight retries.
Only the two stated callee contracts are retained. This is NOT C1: the
native experiment, E1 marginals and fixed bad-event mapping remain separate. -/
theorem outer_owner {Extra Proof : Type} (s2 : Bool) (pub : Pub)
    (nonce : NoncePolicy Extra) (schedule : ProverCommitLedger36.Schedule)
    (tape : Nat → owner.Raw) (unix : Bool)
    (leftPrefix rightPrefix : Prefix Extra) (pre : PrefixContract leftPrefix rightPrefix)
    (left right : Body Extra Proof) (body : BodyContract left right) (s : Store Extra) :
    selected s2 pub (nativeEntropy nonce schedule tape)
      (reserved leftPrefix schedule unix left) s =
    selected s2 pub (modelEntropy nonce schedule tape)
      (reserved rightPrefix schedule unix right) s := by
  have he : nativeEntropy nonce schedule tape = modelEntropy nonce schedule tape :=
    funext fun index => funext (entropy_exact nonce schedule tape index)
  have hc := reserved_exact leftPrefix rightPrefix pre schedule unix left right body
  have hb : reserved leftPrefix schedule unix left = reserved rightPrefix schedule unix right :=
    funext fun secret => funext fun bytes => funext (hc secret bytes)
  rw [he,hb]

theorem wrong_variant {Extra Proof : Type} (pub : Pub)
    (entropy : Entropy (Store Extra) Secret) (body : Body Extra Proof) (s : Store Extra) :
    selected false pub entropy body s = ok (.stop .Semantic,s) := rfl

theorem entropy_failure_stops {Extra Proof : Type} (pub : Pub)
    (entropy : Entropy (Store Extra) Secret) (body : Body Extra Proof)
    (s after : Store Extra) (index fuel : Nat) (e : ProverCommitRetry34.Error)
    (failed : entropy index s = ok (.Err e,after)) :
    run (realAttempt pub entropy body) (fuel+1) index s = ok (.stop e,after) := by
  rw [run,entropy_error pub entropy body index s after e failed]
  rfl

theorem reservation_failure_closed {Extra Proof : Type} (pre : Prefix Extra)
    (schedule : ProverCommitLedger36.Schedule) (unix : Bool)
    (left right : Body Extra Proof) (secret : Secret) (bytes : PubBytes)
    (s : Store Extra) (extra : Extra) (path : ProverCommitLedger36.Path)
    (after : LedgerState) (error : ProverCommitEnvironment35.ReservationError)
    (hp : pre secret.1 bytes s.2 = ok (.Ok path,extra))
    (hr : ProverCommitLedger36.reserve schedule unix path s.1 = (.error error,after)) :
    reserved pre schedule unix left secret bytes s = ok (.Err .Semantic,(after,extra)) ∧
    reserved pre schedule unix right secret bytes s = ok (.Err .Semantic,(after,extra)) := by
  constructor <;> simp only [reserved,hp,bind_tc_ok,hr]

/-- Arbitrary finite initial ledgers and additions; no ledger-size charge. -/
theorem duplicate_closed {Extra Proof : Type} (pre : Prefix Extra)
    (schedule : ProverCommitLedger36.Schedule) (unix : Bool) (body : Body Extra Proof)
    (secret : Secret) (bytes : PubBytes) (s : Store Extra) (extra : Extra)
    (path : ProverCommitLedger36.Path)
    (hp : pre secret.1 bytes s.2 = ok (.Ok path,extra))
    (occupied : path ∈ s.1.ledger) :
    reserved pre schedule unix body secret bytes s =
      ok (.Err .Semantic,((ProverCommitLedger36.reserve schedule unix path s.1).2,extra)) := by
  have hit := ProverCommitLedger36.duplicate_path schedule unix path s.1 occupied
  simp only [reserved,hp,bind_tc_ok,hit]
