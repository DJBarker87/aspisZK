abbrev eval := ActualExtractorStagesM10.eval
abbrev continuation := ActualExtractorStagesM10.continuation
abbrev continuationCells := ActualExtractorStagesM10.continuationCells

theorem value (run : RunView) (coins : Coins) :
    (eval run coins).1 = RawRecordedPublicSpendExtractorBin.extract run coins := ActualExtractorStagesM10.value run coins

theorem work (run : RunView) (coins : Coins) :
    (eval run coins).2 ≤ CostedAllocationLedgerM10.workConstant*(run.1.length+1) := ActualExtractorStagesM10.work run coins

theorem continuation_paid (pub : R0P.Public M31Exact) (words : Words) (coins : Coins) :
    continuationCells pub words coins ≤ (continuation pub words coins).2 := ActualExtractorStagesM10.continuation_paid pub words coins

@[irreducible] def cells (run : RunView) (coins : Coins) : Nat :=
  AllocationPublicStageM10.cells (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run
    (BinPublicRecoverySharedStageM10.stage run) (PublicRecoveryAllocationM10.stageCells run)
    (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).cells
    (fun pub words => continuationCells pub words coins)

theorem cells_paid (run : RunView) (coins : Coins) : cells run coins ≤ (eval run coins).2 := by
  change cells run coins ≤ (ActualExtractorStagesM10.eval run coins).2
  unfold cells ActualExtractorStagesM10.eval
  exact AllocationPublicStageM10.paid _ _ _ _ _ _
    (PublicRecoveryAllocationM10.stage_work run)
    (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).paid
    (fun pub words => continuation_paid pub words coins)

def width : Nat := 1024

/-- Fixed constructor composition; callers cannot choose the allocation bill. -/
def actualHeap : CostedAllocationLedgerM10.HeapMirror := ⟨fun run coins => width*cells run coins⟩

theorem allocation_work (run : RunView) (coins : Coins) :
    actualHeap.bits run coins ≤ width*(eval run coins).2 :=
  Nat.mul_le_mul_left width (cells_paid run coins)

theorem allocation_linear (run : RunView) (coins : Coins) :
    actualHeap.bits run coins ≤ (width*CostedAllocationLedgerM10.workConstant)*(run.1.length+1) := by
  have hw := work run coins
  exact (allocation_work run coins).trans (by
    simpa only [Nat.mul_assoc] using Nat.mul_le_mul_left width hw)

theorem width_le_footprint : width ≤ CostedAllocationLedgerM10.footprint := by
  unfold width CostedAllocationLedgerM10.footprint
  omega

theorem allocation_bound (Q : Nat) : CostedAllocationLedgerM10.AllocationBound actualHeap Q := by
  intro run coins _ _
  exact (allocation_linear run coins).trans
    (Nat.mul_le_mul_right _ (Nat.mul_le_mul_right _ width_le_footprint))

theorem counted_allocations (run : RunView) (coins : Coins) :
    actualHeap.bits run coins = 1024*(PublicRecoveryAllocationM10.stageCells run+
      (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).cells+
      (match (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run.1 with
      | none => 0
      | some pub => continuationCells pub (BinPublicRecoverySharedStageM10.stage run).1 coins)+1+32) := rfl

theorem positive (run : RunView) (coins : Coins) : 0 < actualHeap.bits run coins := by
  rw [counted_allocations]
  have h : 0 < PublicRecoveryAllocationM10.stageCells run+
      (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).cells+
      (match (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run.1 with
      | none => 0 | some pub => continuationCells pub (BinPublicRecoverySharedStageM10.stage run).1 coins)+1+32 := by omega
  exact Nat.mul_pos (by omega) h

/-- This is an accepted-flag RunView, precisely the domain of AllocationBound;
it does not assert that a recorded protocol experiment produces this view. -/
def acceptedRun (pub : RawCutProverCompiler.Public) (proof : RawCutProverCompiler.Proof) : RunView :=
  ([],((pub,proof),true))

theorem accepted_nonvacuous (pub : RawCutProverCompiler.Public) (proof : RawCutProverCompiler.Proof)
    (coins : Coins) :
    (acceptedRun pub proof).2.2 = true ∧
    actualHeap.bits (acceptedRun pub proof) coins = width*cells (acceptedRun pub proof) coins ∧
    0 < actualHeap.bits (acceptedRun pub proof) coins := ⟨rfl,rfl,positive _ _⟩
