@[never_extract,noinline,irreducible] def continuation (pub : R0P.Public M31Exact) (words : Words) (coins : Coins) : Option Wspend × Nat :=
  (AllocatedContinuationM10.spec pub (CostedExtractorStagesM10.received words) coins).run
    (CostedExtractorStagesM10.received words,fun _ => words.final)

def continuationCells (pub : R0P.Public M31Exact) (words : Words) (coins : Coins) : Nat :=
  ((AllocatedContinuationM10.spec pub (CostedExtractorStagesM10.received words) coins).allocation
    (CostedExtractorStagesM10.received words,fun _ => words.final)).elements

theorem continuation_value (pub : R0P.Public M31Exact) (words : Words) (coins : Coins) :
    (continuation pub words coins).1 = RawRecordedEagerExtractorReadsBin.continueWithWords pub words coins := by
  unfold continuation
  have h := AllocatedContinuationM10.value pub (CostedExtractorStagesM10.received words)
    (fun _ => words.final) coins
  simpa only [CostedExtractorStagesM10.received,RawRecordedEagerExtractorReadsBin.continueWithWords] using h

theorem continuation_work (pub : R0P.Public M31Exact) (hp : pub.nextPairIndex < 2^64)
    (words : Words) (coins : Coins) :
    (continuation pub words coins).2 ≤ ComposedContinuationSpecM10.budget InitialPublicImageWorkM10.budget := by
  unfold continuation
  exact AllocatedContinuationM10.work pub hp _ _ coins

theorem continuation_paid (pub : R0P.Public M31Exact) (words : Words) (coins : Coins) :
    continuationCells pub words coins ≤ (continuation pub words coins).2 :=
  ((AllocatedContinuationM10.spec pub (CostedExtractorStagesM10.received words) coins).allocation
    (CostedExtractorStagesM10.received words,fun _ => words.final)).creation_work

@[irreducible] def eval (run : RunView) (coins : Coins) : Option Wspend × Nat :=
  ExtractorStagesM10.publicStage (BinPublicRecoverySharedStageM10.stage run)
    (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run
    (fun pub words => continuation pub words coins)

theorem value (run : RunView) (coins : Coins) :
    (eval run coins).1 = RawRecordedPublicSpendExtractorBin.extract run coins := by
  have hp : (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run.1 =
      RawCutWordsFromRunPublicValidBin.runDecodedPublic run := by
    rw [PublicParserAllocationM10.erase]
    unfold RawCutWordsFromRunPublicValidBin.runDecodedPublic
    exact PublicParserWork.parsePublic_value _
  have hh := ExtractorStagesM10.public_refines (BinPublicRecoverySharedStageM10.stage run)
    (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run
    (fun pub words => continuation pub words coins)
    (BinEagerWordsReads.words run).1 (RawCutWordsFromRunPublicValidBin.runDecodedPublic run)
    (fun pub words => RawRecordedEagerExtractorReadsBin.continueWithWords pub words coins)
    (BinPublicRecoverySharedStageM10.stage_value run) hp
    (fun pub words => continuation_value pub words coins)
  have he := RawRecordedEagerExtractorReadsBin.optionReads_value
    (RawCutWordsFromRunPublicValidBin.runDecodedPublic run)
    (fun pub => (RawRecordedEagerExtractorReadsBin.continueWithWords pub
      (BinEagerWordsReads.words run).1 coins,(BinEagerWordsReads.words run).2))
  unfold eval
  exact (hh.trans he.symm).trans (RawRecordedEagerExtractorReadsBin.value run coins)

theorem work (run : RunView) (coins : Coins) :
    (eval run coins).2 ≤ CostedAllocationLedgerM10.workConstant*(run.1.length+1) := by
  have hp : (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run.2 ≤ 110260 := by
    rw [PublicParserAllocationM10.erase]
    exact PublicParserWork.parsePublic_work _
  have hh := ExtractorStagesM10.public_work (BinPublicRecoverySharedStageM10.stage run)
    (PublicParserAllocationM10.parsePublic run.2.1.1.pubBytes).run
    (fun pub words => continuation pub words coins)
    (4000000000000*(run.1.length+1)) 110260 (ComposedContinuationSpecM10.budget InitialPublicImageWorkM10.budget)
    (BinPublicRecoverySharedStageM10.stage_work run) hp
    (fun pub hpub => continuation_work pub (PublicParsedIndexM10.bound _ pub
      (by simpa only [PublicParserAllocationM10.erase] using hpub)) _ coins)
  have hc : 110260+ComposedContinuationSpecM10.budget InitialPublicImageWorkM10.budget+1+1000 ≤
      (110260+ComposedContinuationSpecM10.budget InitialPublicImageWorkM10.budget+1+1000)*(run.1.length+1) :=
    Nat.le_mul_of_pos_right _ (by omega)
  unfold eval CostedAllocationLedgerM10.workConstant CostedExtractorStagesM10.workConstant
  nlinarith
