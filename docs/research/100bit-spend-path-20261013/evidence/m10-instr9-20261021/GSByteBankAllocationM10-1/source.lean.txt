  have hs := GSPointSamplerAllocationM10.sample_paid 21377328 coins
  have hr : reindexCells ≤ GSByteBankWorkM10.reindexBudget := by
    unfold reindexCells GSByteBankWorkM10.reindexBudget
    omega
  cases h : (GSByteSamplerWorkM10.sample 21377328 coins).1 with
  | none => simp only [cells,GSByteBankWorkM10.run,PublicParserWork.bind,h]; omega
  | some H =>
    have hb := GSBankAllocationM10.bank_paid (P31^8) 149 inputs
      (flatBank WideExact (1025*112) (256*112) 149 H)
    simp only [cells,GSByteBankWorkM10.run,PublicParserWork.bind,h]
    omega
