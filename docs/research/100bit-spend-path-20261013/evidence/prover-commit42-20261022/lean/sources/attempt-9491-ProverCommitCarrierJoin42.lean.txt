theorem uVec_normalize {N : Nat} (nonce : Fin N → NZ) (answers : Fin N → Fin 2 → OracleAnswer)
    (s : Fin N → AttS (F := F) W Pr) :
    uVec W Pr (ProverCommitNormalCarrier42.normalize W Pr nonce answers s) =
      ProverCommitGlobal42.normalized W Pr answers (uVec W Pr s) := by
  funext g
  simp only [uVec,ProverCommitNormalCarrier42.normalize,block,vec,ProverCommitGlobal42.normalized,
    ProverCommitPublicCoins41.normalize,ProverCommitGlobal42.positions,
    ProverCommitGlobal42.values,Finset.mem_filter,Finset.mem_univ,true_and,posE,Equiv.trans_apply,Equiv.prodCongr_apply,Equiv.refl_apply]
  cases hp : ProverCommitGlobal42.publicIndex W Pr
      (Pr.ord (finProdFinEquiv.symm g).2) with
  | none => rfl
  | some b => fin_cases b <;> rfl

theorem public_nonce_fixed {N : Nat} (nonce : Fin N → NZ)
    (answers : Fin N → Fin 2 → OracleAnswer) (s : Fin N → AttS (F := F) W Pr) (k : Fin N) :
    (ProverCommitNormalCarrier42.normalize W Pr nonce answers s k).1 0 = nonce k := by simp [ProverCommitNormalCarrier42.normalize,block]

theorem private_seeds_retained {N : Nat} (nonce : Fin N → NZ)
    (answers : Fin N → Fin 2 → OracleAnswer) (s : Fin N → AttS (F := F) W Pr)
    (k : Fin N) (i : Fin 3) (hi : i ≠ 0) :
    (ProverCommitNormalCarrier42.normalize W Pr nonce answers s k).1 i = (s k).1 i := by
  simp only [ProverCommitNormalCarrier42.normalize,block,Function.update_of_ne hi]

theorem charged_hit_mass (hb : RCommitBytes W Pr) (hop : ROpen Pr) (hcom : RComLog W Pr eCm h B outputs x)
    (nonce : Fin N → NZ) (answers : Fin N → Fin 2 → OracleAnswer)
    (k : Fin N) (c : CPos Pr.E₁ Pr.E₂) (hc : ProverCommitPrivateBounds42.privatePos W Pr c)
    (i : Fin W.qmax) :
    mass (fun s => chargedQ W Pr eCm N P h V w (ProverCommitNormalCarrier42.normalize W Pr nonce answers s) (k,.inr c) ∧
      qryQ W Pr eCm N hN P h V B outputs x w adv i (ProverCommitNormalCarrier42.normalize W Pr nonce answers s) =
      addrQ W Pr R N hN h B outputs x w (ProverCommitNormalCarrier42.normalize W Pr nonce answers s) (k,.inr c)) ≤
        1 / (2 : ℚ)^255 := by
  simpa only [Nat.cast_pow,Nat.cast_ofNat] using hitFresh_mass _ _ _ (2^255) (by positivity)
    (ProverCommitCharged42.hit_private W Pr R eCm N hN P h V B outputs x w adv hb hop hcom nonce answers k c hc i)
