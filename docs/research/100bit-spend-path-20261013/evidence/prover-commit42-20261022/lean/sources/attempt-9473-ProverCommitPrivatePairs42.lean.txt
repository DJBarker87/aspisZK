theorem publicSec_update (nonce : NZ) (seeds : ProverCommitPublicBinding40.Private) (i : Fin 2) (z : NZ) :
    ProverCommitPublicBinding40.publicSec nonce (Function.update seeds i z) =
      Function.update (ProverCommitPublicBinding40.publicSec nonce seeds) i.succ z := by
  funext j
  refine Fin.cases ?_ (fun j => ?_) j
  · simp [ProverCommitPublicBinding40.publicSec]
  · simp [ProverCommitPublicBinding40.publicSec]

theorem vector_update (first second : OracleAnswer) (q : RIdx)
    (hq : ∀ b : Fin 2, q ≠ .inl b) (a : Carrier W Pr) (u : OracleAnswer) :
    vector W Pr first second ((answerLens W Pr q).set a u) =
      Function.update (vector W Pr first second a) (.inr (.inr (.inr q))) u := by
  funext p
  by_cases hp : p = .inr (.inr (.inr q))
  · subst p
    cases q with
    | inl b => exact False.elim (hq b rfl)
    | inr q => simp [vector,ProverCommitGlobal42.publicIndex,answerLens]
  · simp [vector,answerLens,Function.update_of_ne hp]

theorem answer_parts (first second : OracleAnswer) (q : RIdx)
    (hq : ∀ b : Fin 2, q ≠ .inl b) (a : Carrier W Pr) (u : OracleAnswer) :
    tapeR W Pr (vector W Pr first second ((answerLens W Pr q).set a u)) = tapeR W Pr (vector W Pr first second a) ∧
    trOf (vector W Pr first second ((answerLens W Pr q).set a u)) = trOf (vector W Pr first second a) ∧
    restOf (vector W Pr first second ((answerLens W Pr q).set a u)) =
      Function.update (restOf (vector W Pr first second a)) q u := by
  rw [vector_update W Pr first second q hq]
  constructor
  · simp only [tapeR,e₁Of,e₂Of]; congr 3 <;> funext c <;> simp
  · constructor
    · funext j; simp [trOf]
    · funext q'; by_cases eq : q'=q <;> simp [restOf,eq,Function.update_of_ne]

theorem seed_inv (hb : RCommitBytes W Pr) (nonce : NZ) (digest first second : OracleAnswer)
    (i : Fin 2) (a : Carrier W Pr) (z : NZ) (c : CPos Pr.E₁ Pr.E₂)
    (independent : ¬ depS c (.inl i.succ)) :
    address W Pr h B outputs x w nonce digest first second ((seedLens W Pr i).set a z) c =
      address W Pr h B outputs x w nonce digest first second a c := by
  simp only [address,attAddr,hb,seedLens]
  rw [publicSec_update]
  exact gram_sec_update _ _ _ _ _ _ _ _ c independent

theorem answer_inv (hb : RCommitBytes W Pr) (nonce : NZ) (digest first second : OracleAnswer)
    (q : RIdx) (hq : ∀ b : Fin 2, q ≠ .inl b) (a : Carrier W Pr) (u : OracleAnswer)
    (c : CPos Pr.E₁ Pr.E₂) (independent : ¬ depS c (.inr q)) :
    address W Pr h B outputs x w nonce digest first second ((answerLens W Pr q).set a u) c =
      address W Pr h B outputs x w nonce digest first second a c := by
  have parts := answer_parts W Pr first second q hq a u
  simp only [address,attAddr,hb]
  rw [parts.1,parts.2.1,parts.2.2]
  exact gram_rest_update _ _ _ _ _ _ _ _ c independent

/-- Both private positions, with fixed public nonce and memo bindings. No
public coordinate is admitted as the source of the private-pair bound. -/
theorem pair_all (hb : RCommitBytes W Pr) (hE₁ : Pr.E₁ ≤ 2^64) (hE₂ : Pr.E₂ ≤ 2^64)
    (nonce : NZ) (digest first second : OracleAnswer)
    (c c' : CPos Pr.E₁ Pr.E₂) (hc : privatePos W Pr c) (hc' : privatePos W Pr c') (hne : c ≠ c') :
    PairFresh (fun a => address W Pr h B outputs x w nonce digest first second a c)
      (fun a => address W Pr h B outputs x w nonce digest first second a c') (2^255) ∨
    PairFresh (fun a => address W Pr h B outputs x w nonce digest first second a c')
      (fun a => address W Pr h B outputs x w nonce digest first second a c) (2^255) := by
  rcases fresh_or_sep hE₁ hE₂ c c' hne with left | right | separate
  · refine Or.inr (private_pair W Pr h B outputs x w hb nonce digest first second c hc _ ?_ ?_)
    · intro i hi a z
      exact seed_inv W Pr h B outputs x w hb nonce digest first second i a z c' (hi ▸ left)
    · intro q hq a u
      exact answer_inv W Pr h B outputs x w hb nonce digest first second q
        ((src_private W Pr c hc).2 q hq) a u c' (hq ▸ left)
  · refine Or.inl (private_pair W Pr h B outputs x w hb nonce digest first second c' hc' _ ?_ ?_)
    · intro i hi a z
      exact seed_inv W Pr h B outputs x w hb nonce digest first second i a z c (hi ▸ right)
    · intro q hq a u
      exact answer_inv W Pr h B outputs x w hb nonce digest first second q
        ((src_private W Pr c' hc').2 q hq) a u c (hq ▸ right)
  · refine Or.inl (pairFresh_of_ne _ _ _ (fun a => ?_))
    simp only [address,attAddr,hb]
    exact separate _ _ _ _ _ _

theorem pair_mass (hb : RCommitBytes W Pr) (hE₁ : Pr.E₁ ≤ 2^64) (hE₂ : Pr.E₂ ≤ 2^64)
    (nonce : NZ) (digest first second : OracleAnswer)
    (c c' : CPos Pr.E₁ Pr.E₂) (hc : privatePos W Pr c) (hc' : privatePos W Pr c') (hne : c ≠ c') :
    mass (fun a : Carrier W Pr => address W Pr h B outputs x w nonce digest first second a c =
      address W Pr h B outputs x w nonce digest first second a c') ≤ 1/(2 : ℚ)^255 := by
  simpa only [Nat.cast_pow,Nat.cast_ofNat] using
    pairFresh_mass _ _ (2^255) (by positivity)
      (pair_all W Pr h B outputs x w hb hE₁ hE₂ nonce digest first second c c' hc hc' hne)
