//! Split successor to timed-out 16-by-64 symbolic tape harnesses.
//! Every fill chooses a new arbitrary reply; this is support coverage, not E1.
use super::*;
static mut CALLS: usize = 0;
static mut TERMINAL: usize = 0;
static mut FAIL: bool = false;
static mut LAST: [u8; 64] = [0;64];
static mut CODE: i32 = 1;
fn error(code:i32)->getrandom::Error {
    let raw=core::num::NonZeroI32::new(code).unwrap_or(core::num::NonZeroI32::new(1).unwrap());
    unsafe { core::mem::transmute(raw) }
}
fn nonzero(bytes:&[u8;32])->bool {
        let a=u64::from_le_bytes([bytes[0],bytes[1],bytes[2],bytes[3],bytes[4],bytes[5],bytes[6],bytes[7]]);
        let b=u64::from_le_bytes([bytes[8],bytes[9],bytes[10],bytes[11],bytes[12],bytes[13],bytes[14],bytes[15]]);
        let c=u64::from_le_bytes([bytes[16],bytes[17],bytes[18],bytes[19],bytes[20],bytes[21],bytes[22],bytes[23]]);
        let d=u64::from_le_bytes([bytes[24],bytes[25],bytes[26],bytes[27],bytes[28],bytes[29],bytes[30],bytes[31]]);
        (a|b|c|d)!=0
    }
fn stream(destination:&mut[u8])->Result<(),getrandom::Error> {
    assert_eq!(destination.len(),64);
    assert!(destination.iter().all(|byte|*byte==0));
    let call=unsafe{CALLS};assert!(call<16);
    unsafe{CALLS=call+1;}
    let mut reply:[u8;64]=kani::any();
    if call<unsafe{TERMINAL} {
        if kani::any::<bool>() {reply[..32].fill(0);} else {reply[32..].fill(0);}
    } else {
        assert_eq!(call,unsafe{TERMINAL});
        if !unsafe{FAIL} {
            kani::assume(reply[..32].iter().any(|byte|*byte!=0));
            kani::assume(reply[32..].iter().any(|byte|*byte!=0));
        }
        unsafe{LAST=reply;}
    }
    destination.copy_from_slice(&reply);
    if call==unsafe{TERMINAL} && unsafe{FAIL} {Err(error(unsafe{CODE}))} else {Ok(())}
}
fn terminal<const N:usize,const IS_ERROR:bool>() {
    let nonce:[u8;32]=kani::any();kani::assume(nonce!=[0;32]);
    unsafe{CALLS=0;TERMINAL=N;FAIL=IS_ERROR;CODE=kani::any();}
    let byte=usize::from(kani::any::<u8>() & 31);
    match StateOnlyAttemptSecrets::generate_for_mask_nonce(nonce) {
        Ok(secret)=>{
            assert!(!IS_ERROR);assert_eq!(secret.mask_nonce[byte],nonce[byte]);
            assert_eq!(secret.field_mask_entropy[byte],unsafe{LAST}[byte]);
            assert_eq!(secret.leaf_salt_seed[byte],unsafe{LAST}[32+byte]);
        }
        Err(StateOnlyAttemptEntropyError::Os(value))=>{
            assert!(IS_ERROR);assert_eq!(value,error(unsafe{CODE}));
        }
        Err(_)=>assert!(false),
    }
    assert_eq!(unsafe{CALLS},N+1);
}

// Control-only split: payload/ABI/Drop contracts remain independently retained.
fn control_stream(destination:&mut[u8])->Result<(),getrandom::Error> {
    assert_eq!(destination.len(),64);
    let call=unsafe{CALLS};unsafe{CALLS=call+1;}
    let mut reply:[u8;64]=kani::any();
    if call<unsafe{TERMINAL} {
        if kani::any::<bool>() {reply[..32].fill(0);} else {reply[32..].fill(0);}
    } else {
        assert_eq!(call,unsafe{TERMINAL});
        if !unsafe{FAIL} {
            kani::assume(reply[..32].iter().any(|byte|*byte!=0));
            kani::assume(reply[32..].iter().any(|byte|*byte!=0));
        }
    }
    destination.copy_from_slice(&reply);
    if call==unsafe{TERMINAL} && unsafe{FAIL} {Err(error(unsafe{CODE}))} else {Ok(())}
}
fn control<const N:usize,const IS_ERROR:bool>() {
    let nonce:[u8;32]=kani::any();kani::assume(nonce!=[0;32]);
    unsafe{CALLS=0;TERMINAL=N;FAIL=IS_ERROR;CODE=kani::any();}
    match StateOnlyAttemptSecrets::generate_for_mask_nonce(nonce) {
        Ok(_)=>assert!(!IS_ERROR),
        Err(StateOnlyAttemptEntropyError::Os(_))=>assert!(IS_ERROR),
        Err(_)=>assert!(false),
    }
    assert_eq!(unsafe{CALLS},N+1);
}

fn error_control_range(low:usize,high:usize) {
    let index=usize::from(kani::any::<u8>() & 15);kani::assume(low<=index && index<high);
    let nonce:[u8;32]=kani::any();kani::assume(nonce!=[0;32]);
    unsafe{CALLS=0;TERMINAL=index;FAIL=true;CODE=kani::any();}
    match StateOnlyAttemptSecrets::generate_for_mask_nonce(nonce) {
        Err(StateOnlyAttemptEntropyError::Os(value))=>assert_eq!(value,error(unsafe{CODE})),
        _=>assert!(false),
    }
    assert_eq!(unsafe{CALLS},index+1);
}
