import R1Z.PV_CurrentScaleInstall
set_option autoImplicit false
namespace R1Z.PV_WordZero
open Aeneas Aeneas.Std

theorem or_zero (a b : U32) : (a ||| b)=0#u32 ↔ a=0#u32 ∧ b=0#u32 := by
  simp only [U32.eq_equiv_bv_eq,UScalar.bv_or]
  exact BitVec.or_eq_zero_iff

theorem xor_zero (a b : U32) : (a ^^^ b)=0#u32 ↔ a=b := by
  simp only [U32.eq_equiv_bv_eq,UScalar.bv_xor]
  exact BitVec.xor_eq_zero_iff
