    ∃ out, (selected ops).op110 x = ok out := by
  change ∃ out, ProverRwDictionaryEmbedding.embed x = ok out
  obtain ⟨r,hr,_⟩ := WP.spec_imp_exists
    (UScalar.rem_spec x (y := AspisR1ProducerRelease47.field.P) (by decide))
  simp only [ProverRwDictionaryEmbedding.embed,ProverRwDictionaryScalar.qFrom,
    ProverRwDictionaryScalar.mFrom,hr,bind_tc_ok,
    AspisR1ProducerRelease47.field.CM31.from_m31,
    AspisR1ProducerRelease47.field.QM31.from_cm31,
    AspisR1ProducerRelease47.field.wide_exact.WideExact.from_qm31,mapResult]
  exact ⟨_,rfl⟩
