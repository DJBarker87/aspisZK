
fn native_channel49(mut sums:[[u64;3];3],a:[[u32;3];3],b:[[u32;3];3],c:usize,h:usize)->[[u64;3];3] {
 sums[c][h]+=a[c][h] as u64*b[c][h] as u64;sums
}
fn owner_channel49(mut sums:[[u64;3];3],a:[[u32;3];3],b:[[u32;3];3],c:usize,h:usize)->[[u64;3];3] {
 let product=(a[c][h] as u64).wrapping_mul(b[c][h] as u64);
 sums[c][h]=sums[c][h].wrapping_add(product);sums
}
#[cfg(kani)] mod channel_proofs {
 use super::*;
 #[kani::proof] #[kani::unwind(12)] fn integer_channel_iteration() {
  let sums:[[u64;3];3]=kani::any();let a:[[u32;3];3]=kani::any();let b:[[u32;3];3]=kani::any();
  let c:usize=kani::any();let h:usize=kani::any();kani::assume(c<3 && h<3);
  kani::assume(a[c][h]<2147483647 && b[c][h]<2147483647);
  kani::assume(sums[c][h]<=2*2147483646u64*2147483646u64);
  assert_eq!(native_channel49(sums,a,b,c,h),owner_channel49(sums,a,b,c,h));
 }
}
