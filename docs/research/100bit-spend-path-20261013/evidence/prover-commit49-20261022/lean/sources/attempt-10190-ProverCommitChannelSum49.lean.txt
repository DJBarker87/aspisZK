abbrev P : Nat := AspisCircleGroupOrder.P

def sum (a b : Nat → U32) : Nat → Nat → Nat
 | 0,_ => 0
 | n+1,j => (a j).val*(b j).val+sum a b n (j+1)

theorem product_bound (a b : U32) (ha : a.val<P) (hb : b.val<P) :
    a.val*b.val≤(P-1)^2 := by
  have h := Nat.mul_le_mul (show a.val≤P-1 by omega) (show b.val≤P-1 by omega)
  simpa only [pow_two] using h

theorem sum_bound (a b : Nat → U32) (ha : ∀ i, i<3 → (a i).val<P)
    (hb : ∀ i, i<3 → (b i).val<P) (n j : Nat) (bound : j+n≤3) :
    sum a b n j≤n*(P-1)^2 := by
  induction n generalizing j with
  | zero => simp only [sum,Nat.zero_mul,Nat.le_refl]
  | succ n ih =>
    rw [sum,Nat.succ_mul]
    have hp := product_bound (a j) (b j) (ha j (by omega)) (hb j (by omega))
    have ht := ih (j+1) (by omega)
    omega

attribute [local irreducible] ProverCommitM31Array48.array

theorem sum_value (a b : Nat → U32) (n : Nat) (i : Usize) (left : U8)
    (hleft : left.val=n) (bound : i.val+n≤3) (total : U64)
    (safe : total.val+sum a b n i.val<2^64) :
    ∃ z : U64, channels.sum (ProverCommitM31Array48.array 3#usize a)
      (ProverCommitM31Array48.array 3#usize b) i left total=ok z ∧
      z.val=total.val+sum a b n i.val := by
  induction n generalizing i left total with
  | zero =>
    have hz : left=0#u8 := by scalar_tac
    refine ⟨total,?_,?_⟩
    · rw [channels.sum,hz];rfl
    · rw [sum,Nat.add_zero]
  | succ n ih =>
    have hn : left≠0#u8 := by scalar_tac
    have hdec : (U8.wrapping_sub left 1#u8).val=n := by
      rw [ProverCommitBasisCounter47.sub_one left (by omega)];omega
    have hinc : (Usize.wrapping_add i 1#usize).val=i.val+1 := by
      simpa only [show (1#usize).val=1 from rfl] using
        ProverCommitBasisFillValue47.add_val i 1#usize (by change i.val+1<2048;omega)
    let product := core.num.U64.wrapping_mul (core.convert.num.FromU64U32.from (a i.val))
      (core.convert.num.FromU64U32.from (b i.val))
    have hp : product.val=(a i.val).val*(b i.val).val := by
      have ha : (a i.val).val<2^32 := (a i.val).bv.isLt
      have hb : (b i.val).val<2^32 := (b i.val).bv.isLt
      have hs : (a i.val).val*(b i.val).val<2^64 := by
        calc
          _ < 2^32*2^32 := Nat.mul_lt_mul_of_lt_of_lt ha hb
          _ = _ := by decide
      exact (ProverCommitNativeSquare49.mul64 _ _ (by rw [ProverCommitNativeSquare49.widen,ProverCommitNativeSquare49.widen];exact hs)).trans
        (by rw [ProverCommitNativeSquare49.widen,ProverCommitNativeSquare49.widen])
    let next := core.num.U64.wrapping_add total product
    have hnext : next.val=total.val+(a i.val).val*(b i.val).val := by
      exact (ProverCommitNativeSquare49.add64 total product (by rw [hp];rw [sum] at safe;omega)).trans (by rw [hp])
    obtain ⟨z,hz,hzv⟩ := ih _ _ hdec (by rw [hinc];omega) next
      (by rw [hnext,hinc];rw [sum] at safe;omega)
    refine ⟨z,?_,?_⟩
    · rw [channels.sum]
      simp only [hn,if_false,lift,bind_tc_ok]
      rw [ProverCommitM31Array48.index 3#usize a i (by change i.val<3;omega)]
      simp only [bind_tc_ok]
      rw [ProverCommitM31Array48.index 3#usize b i (by change i.val<3;omega)]
      simp only [lift,bind_tc_ok]
      exact hz
    · rw [hzv,hnext,hinc,sum];omega

theorem dot3_value (a b : Nat → U32) (ha : ∀ i, i<3 → (a i).val<P) (hb : ∀ i, i<3 → (b i).val<P) :
    ∃ z : U32, channels.dot3 (ProverCommitM31Array48.array 3#usize a) (ProverCommitM31Array48.array 3#usize b)=ok z ∧
      z.val<P ∧ (z.val : M31Exact)=(sum a b 3 0 : M31Exact) := by
  have hbound := sum_bound a b ha hb 3 0 (by decide)
  obtain ⟨total,ht,htv⟩ := sum_value a b 3 0#usize 3#u8 rfl (by decide) 0#u64
    (by change 0+sum a b 3 0<2^64;change sum a b 3 0≤3*(2147483647-1)^2 at hbound;omega)
  obtain ⟨z,hz,hcz,hvz⟩ := ProverCommitNativeSquare49.reduce_exact total
  refine ⟨z,?_,hcz,?_⟩
  · simp only [channels.dot3,ht,hz,bind_tc_ok]
  · rw [hvz,htv]
    rfl
