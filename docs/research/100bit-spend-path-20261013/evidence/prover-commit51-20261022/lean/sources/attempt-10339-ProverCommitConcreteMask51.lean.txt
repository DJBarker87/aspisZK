    (t : R0P.Trace QM31Exact) (ht : ∀ l : Fin 29, ∀ j : Fin 1024, decode (c l.val j.val)=t l j) :
    ∃ z : Old.Q, ProverCommitMaskRead51.run (ProverCommitQArray48.array 10#usize p) (ProverCommitMaskColumns51.columns c)
      (ProverCommitQArray48.array 1024#usize (c 27))=ok (ProverCommitNativeQ48.source z) ∧

        (honestClaims t (fun j : Fin 10 => decode (p j.val)) 0 27) (fun i : Fin 10 => decode (p i.val)) := by

    ProverCommitQBasis51.basis_i ProverCommitQBasis51.basis_u p hp c hc

  have h1 : (fun i : Fin 16 => ProverCommitMaskColumns51.value p c i.val)=
      (fun i : Fin 16 => honestClaims t (fun j : Fin 10 => decode (p j.val)) 0 (Fin.castLE (by omega) i)) := by

      (fun i : Fin 10 => honestClaims t (fun j : Fin 10 => decode (p j.val)) 0 ⟨16+i.val,by omega⟩) := by
    funext i;exact ProverCommitMaskRead51.claims_value p c t ht ⟨16+i.val,by omega⟩
  rw [h1,h2,ProverCommitMaskRead51.claims_value p c t ht 27] at hdz
  exact hdz
