fn parse_records_into<'a>(
    bytes: &'a [u8], records: &mut [&'a [u8]; 27],
) -> Result<(), Error> {
    let mut offset = 32;
    let mut row = 0u8;
    while row < 27 {
        let length = 5 + row_shape(row).unwrap().1;
        let framed = bytes
            .get(offset..offset + length)
            .ok_or(Error::Length { row })?;
        parse_record(row, framed)?;
        records[row as usize] = framed;
        offset += length;
        row += 1;
    }
    if offset != bytes.len() {
        return Err(Error::TrailingBytes);
    }
    Ok(())
}
