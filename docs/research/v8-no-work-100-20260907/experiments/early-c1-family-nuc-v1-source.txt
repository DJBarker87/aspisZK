def qualifies (c1 : C1InitialWords) (p : C1InitialMessages) : Prop :=
  38228 ≤ (c1JointAgreementSet c1 p).card

/-- No per-lane decoder-membership filter appears in this bound. -/
theorem finite_subfamily_card (c1 : C1InitialWords)
    (candidates : Finset C1InitialMessages)
    (large : ∀ p∈candidates, qualifies c1 p) : candidates.card≤100 := by
  classical
  let Candidate := {p // p∈candidates}
  let agreement : Candidate → Finset (Fin 1048576) := fun p => c1JointAgreementSet c1 p.1
  have size : ∀ p : Candidate, 38228≤(agreement p).card := fun p => large p.1 p.2
  have overlap : ∀ p r : Candidate, p≠r → (agreement p ∩ agreement r).card≤1024 := by
    intro p r different
    exact distinct_c1_tuple_joint_agreement_overlap_le_1024 c1 p.1 r.1
      (fun same => different (Subtype.ext same))
  have forbidden := list_card_lt_of_johnson_parameters agreement 1048576 38228 1024 101
    (by simp) size overlap (by norm_num) (by norm_num) (by norm_num)
  rw [Fintype.card_coe] at forbidden
  omega

/-- Generic finite construction prevents reduction of the concrete message
universe when establishing exact membership. It is not an efficient algorithm. -/
def filterFamily {M : Type*} [Fintype M] (predicate : M → Prop) : Finset M := by
  classical
  exact Finset.univ.filter predicate

theorem mem_filterFamily {M : Type*} [Fintype M] (predicate : M → Prop) (m : M) :
    m∈filterFamily predicate ↔ predicate m := by
  classical
  simp only [filterFamily,Finset.mem_filter,Finset.mem_univ,true_and]

def family (c1 : C1InitialWords) : Finset C1InitialMessages := filterFamily (qualifies c1)

theorem mem_family (c1 : C1InitialWords) (p : C1InitialMessages) :
    p∈family c1 ↔ qualifies c1 p := mem_filterFamily (qualifies c1) p

theorem family_card_le_100 (c1 : C1InitialWords) : (family c1).card≤100 := by
  apply finite_subfamily_card c1
  intro p member
  exact (mem_family c1 p).mp member

/-- Literal 26+3 source assembly, not a freely supplied correspondence
between a late tuple and another convenient early committed word. -/
theorem actual_joint_subset (c1 : C1InitialWords) (c2 : C2Received)
    (p : Fin 29 → EarlyC1Projection.Message) :
    width29JointAgreementSet exactInitialEncoder (received29 c1 c2) p ⊆
      c1JointAgreementSet c1 (c1Projection p) := by
  classical
  intro index member
  have allLanes := (Finset.mem_filter.mp member).2
  apply Finset.mem_filter.mpr
  refine ⟨Finset.mem_univ _,?_⟩
  intro column
  have lane := allLanes (c1LaneIndex column)
  rw [received29_c1] at lane
  exact lane

theorem actual_late_projection_member (c1 : C1InitialWords) (c2 : C2Received)
    (p : Fin 29 → EarlyC1Projection.Message)
    (large : 38228≤(width29JointAgreementSet exactInitialEncoder (received29 c1 c2) p).card) :
    c1Projection p∈family c1 := by
  apply (mem_family c1 (c1Projection p)).mpr
  exact large.trans (Finset.card_le_card (actual_joint_subset c1 c2 p))

/-- Every member has base-field coefficients when the fixed committed C1
values are base-field values. Uses the member's OWN support, with no
decoder-instantiation or same-selected-support recovery premise. -/
theorem member_is_base (c1 : C1InitialWords) (p : C1InitialMessages)
    (member : p∈family c1)
    (fixed : ∀ column index, projectBase (c1 column index)=c1 column index) :
    ∀ column, projectMessage (p column)=p column := by
  classical
  intro column
  by_contra different
  have subset : c1JointAgreementSet c1 p ⊆ Finset.univ.filter (fun index =>
      exactInitialEncoder (projectMessage (p column)) index=exactInitialEncoder (p column) index) := by
    intro index good
    have same := (Finset.mem_filter.mp good).2 column
    apply Finset.mem_filter.mpr
    refine ⟨Finset.mem_univ _,?_⟩
    rw [congrFun (exactInitialEncoder_commutes (p column)) index,← same,fixed]
  have upper := exactInitialEncoder_overlap_cap (projectMessage (p column)) (p column) different
  have lower := (mem_family c1 p).mp member
  have counts := Finset.card_le_card subset
  change (Finset.univ.filter (fun index =>
    exactInitialEncoder (projectMessage (p column)) index=exactInitialEncoder (p column) index)).card≤1024 at upper
  unfold qualifies at lower
  omega
