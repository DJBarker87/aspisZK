/-! The exact original-message GRS polynomial evaluated at either actual
OOD circle point. This is a pointwise algebraic identity, not an inference
from agreement only on the stored domain. The multiplier is (1+t^2)^512,
fixed by the OOD point before gamma, with no component recovery premise. -/
set_option autoImplicit false
set_option Elab.async false
set_option maxRecDepth 200
set_option maxHeartbeats 100000
namespace AspisV8.CoveredOODGRSAlgebra
open Polynomial
open AspisV8.ComponentOODBinding
open AspisV5FriCircleEncoderDistance AspisV5FriInitialCircleEncoderIdentity
variable {F : Type*} [Field F] [NeZero (2 : F)] [DecidableEq F]

theorem message_numerator_eval (message : Fin 1024 → F) (x y : F)
    (circle : x^2+y^2=1) (west : x≠-1) :
    (circleNumerator (initialP0 message) (initialP1 message)).eval (y/(1+x))=
      (1+(y/(1+x))^2)^512*circleFunctional x y message := by
  rw [circleFunctional_apply]
  exact circleNumerator_eval_stereo (initialP0 message) (initialP1 message)
    (initialP0_degree_lt message) (initialP1_degree_lt message) x y circle west

theorem factor_nonzero (x y : F) (circle : x^2+y^2=1) (west : x≠-1) :
    (1+(y/(1+x))^2)^512≠0 :=
  pow_ne_zero 512 (stereo_identities x y circle west).1

/-- Keep the parameter and cleared denominator opaque during concrete
specialization; their defining equations contain only degree-two arithmetic. -/
theorem message_numerator_at (message : Fin 1024 → F) (x y t denominator : F)
    (parameter : y/(1+x)=t) (denom : 1+t^2=denominator)
    (circle : x^2+y^2=1) (west : x≠-1) :
    (circleNumerator (initialP0 message) (initialP1 message)).eval t=
      denominator^512*circleFunctional x y message := by
  rw [←denom,←parameter]
  exact message_numerator_eval message x y circle west

theorem denominator_nonzero (x y t denominator : F)
    (parameter : y/(1+x)=t) (denom : 1+t^2=denominator)
    (circle : x^2+y^2=1) (west : x≠-1) : denominator≠0 := by
  rw [←denom,←parameter]
  exact (stereo_identities x y circle west).1
end AspisV8.CoveredOODGRSAlgebra

noncomputable section
local instance : NeZero (2 : K) := SelectedReceivedOracle.twoNonzero

/-- V7's exact GRS message map is a cleared stereographic numerator,
not the natural message itself and not an inverse-FFT evaluation table. -/
theorem grs_at_circle (message : Fin 1024 → K) (x y : K)
    (circle : x^2+y^2=1) (west : x≠-1) :
    (exactCircleGRSPolynomial message).eval (y/(1+x))=
      (1+(y/(1+x))^2)^512*circleFunctional x y message := by
  generalize parameter : y/(1+x)=t
  generalize denom : 1+t^2=denominator
  rw [exactCircleGRSPolynomial_eq_released]
  exact CoveredOODGRSAlgebra.message_numerator_at message x y t denominator
    parameter denom circle west

theorem stereographic_factor_nonzero (x y : K)
    (circle : x^2+y^2=1) (west : x≠-1) :
    (1+(y/(1+x))^2)^512≠0 := by
  generalize parameter : y/(1+x)=t
  generalize denom : 1+t^2=denominator
  exact pow_ne_zero 512 (CoveredOODGRSAlgebra.denominator_nonzero x y t denominator
    parameter denom circle west)

/-- An image-valid quotient reconstructs a message whose literal two OOD
values equal the two gamma-batched answers. Transport to the GRS polynomial
multiplies those answers by the fixed nonzero stereographic factor. -/
theorem original_grs_at_ood (d : Data (K := K)) (checked : d.Checked)
    (Q : Fin 1024 → K) (image : Q 1023=0 ∧ d.b*Q 1022-d.c*Q 1021=0)
    (circles : d.x0^2+d.y0^2=1 ∧ d.x1^2+d.y1^2=1)
    (r : Fin 2) (west : pointX d r≠-1) :
    (exactCircleGRSPolynomial (d.original Q)).eval (pointY d r/(1+pointX d r))=
      (1+(pointY d r/(1+pointX d r))^2)^512*d.batch r := by
  have circle : (pointX d r)^2+(pointY d r)^2=1 := by
    fin_cases r
    · exact circles.1
    · exact circles.2
  rw [grs_at_circle (d.original Q) (pointX d r) (pointY d r) circle west]
  rw [original_point_values d checked Q image circles r]
