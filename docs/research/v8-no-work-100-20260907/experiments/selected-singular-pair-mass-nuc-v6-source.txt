local instance decision (p : Prop) : Decidable p := Classical.propDecidable p

/-- The same set is fixed by C1/C2 before either OOD sample.  Filtering by
`Admissible` matches the exact accepted parameter domain and does not
condition away ordinary decode or retry failure. -/
def admissibleDomain : Finset K := Finset.univ.filter Admissible

def admissibleRoots (c1 : C1Received) (c2 : C2Received) : Finset K :=
  rootsIn (obstruction c1 c2) admissibleDomain

theorem mem_admissibleRoots_iff (c1 : C1Received) (c2 : C2Received) (t : K) :
    t ∈ admissibleRoots c1 c2 ↔
      Admissible t ∧ (obstruction c1 c2).eval t = 0 := by
  have nonzero := (SelectedSingularOODFamily.fixed_cover c1 c2).1
  simp only [admissibleRoots, mem_rootsIn, admissibleDomain, Finset.mem_filter,
    Finset.mem_univ, true_and, nonzero]

theorem admissibleRoots_card_le (c1 : C1Received) (c2 : C2Received) :
    (admissibleRoots c1 c2).card ≤ 25345827 := by
  exact (rootsIn_card (obstruction c1 c2) admissibleDomain).trans
    (SelectedSingularOODFamily.fixed_cover c1 c2).2.1

/-- The literal two parameters belong to the ordered off-diagonal target
exactly when they are admissible, distinct and both roots.  The theorem says
nothing about how source bytes or transcript callbacks produced them. -/
theorem pair_mem_offDiag (c1 : C1Received) (c2 : C2Received) (first second : K)
    (firstAdmissible : Admissible first) (secondAdmissible : Admissible second)
    (different : first ≠ second)
    (firstRoot : (obstruction c1 c2).eval first = 0)
    (secondRoot : (obstruction c1 c2).eval second = 0) :
    (first, second) ∈ (admissibleRoots c1 c2).offDiag := by
  refine Finset.mem_offDiag.mpr ⟨?_, ?_, different⟩
  · exact (mem_admissibleRoots_iff c1 c2 first).mpr ⟨firstAdmissible, firstRoot⟩
  · exact (mem_admissibleRoots_iff c1 c2 second).mpr ⟨secondAdmissible, secondRoot⟩

variable {O H : Type*}

/-- Abort-preserving ideal mass of the complete selected singular pair-root
alternative.  The only probabilistic premise is the existing history-uniform
ordinary decoder law, stated for every history. -/
theorem selected_pair_root_mass_le (coins : Finset O)
    (draw : H → O → Option (K × H)) (between : K → H → H)
    (law : NestedCircleMass.Generic.Uniform coins draw
      (ordinaryAtomMass (FiniteOptionMass.successMass OrdinaryPrefixMass.value)))
    (c1 : C1Received) (c2 : C2Received) (h : H) :
    targetMass coins draw between (admissibleRoots c1 c2) h ≤
      ((admissibleRoots c1 c2).card : ℚ) *
        ((admissibleRoots c1 c2).card - 1 : ℚ) /
          ((domainSize : ℚ) * ((domainSize : ℚ) - 1)) := by
  apply actual_one_call_target_bound coins draw between law
  intro t member
  exact ((mem_admissibleRoots_iff c1 c2 t).mp member).1

/-- Concrete selected numerator.  No finite field is enumerated: the root
cardinality is bounded by the polynomial degree and the denominator is the
checked exact admissible-domain size. -/
theorem selected_pair_root_mass_numeric_le (coins : Finset O)
    (draw : H → O → Option (K × H)) (between : K → H → H)
    (law : NestedCircleMass.Generic.Uniform coins draw
      (ordinaryAtomMass (FiniteOptionMass.successMass OrdinaryPrefixMass.value)))
    (c1 : C1Received) (c2 : C2Received) (h : H) :
    targetMass coins draw between (admissibleRoots c1 c2) h ≤
      (25345827 : ℚ) * 25345826 /
        ((domainSize : ℚ) * ((domainSize : ℚ) - 1)) := by
  refine (selected_pair_root_mass_le coins draw between law c1 c2 h).trans ?_
  have cardBound : ((admissibleRoots c1 c2).card : ℚ) ≤ 25345827 := by
    exact_mod_cast admissibleRoots_card_le c1 c2
  have cardNonnegative : (0 : ℚ) ≤ (admissibleRoots c1 c2).card :=
    Nat.cast_nonneg _
  have denominatorPositive :
      (0 : ℚ) < (domainSize : ℚ) * ((domainSize : ℚ) - 1) := by
    have domainTwo : (2 : ℚ) ≤ domainSize := by
      exact_mod_cast domain_size_bounds.2
    have domainPositive : (0 : ℚ) < domainSize := lt_of_lt_of_le (by norm_num) domainTwo
    have domainMinusOnePositive : (0 : ℚ) < (domainSize : ℚ) - 1 := by linarith
    exact mul_pos domainPositive domainMinusOnePositive
  apply div_le_div_of_nonneg_right _ (le_of_lt denominatorPositive)
  nlinarith
