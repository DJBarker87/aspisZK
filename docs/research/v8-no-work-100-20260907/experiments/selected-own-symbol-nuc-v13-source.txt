def chargedSet (U C : Finset I) (G : Finset K) (R : I → K → Prop) : Finset K := by
  classical
  exact (U \ C).biUnion fun i => G.filter (R i)

/-- Field-free finite transport: opaque sets/predicates avoid comparing
concrete QM31 dictionaries while selecting the outside symbol. -/
theorem excess_charged (U C M : Finset I) (G : Finset K) (R : I → K → Prop)
    (gamma : K) (inGamma : gamma∈G) (inside : M⊆U)
    (roots : ∀i∈M, R i gamma) (excess : C.card<M.card) :
    gamma∈chargedSet U C G R := by
  classical
  have missing : ¬M⊆C := fun subset => Nat.not_lt_of_ge (Finset.card_le_card subset) excess
  obtain ⟨i, member, outside⟩ := Finset.not_subset.mp missing
  apply Finset.mem_biUnion.mpr
  exact ⟨i,Finset.mem_sdiff.mpr ⟨inside member,outside⟩,
    Finset.mem_filter.mpr ⟨inGamma,roots i member⟩⟩
end
end Generic

theorem matching_eq_symbols (c1 : C1Received) (c2 : C2Received)
    (p : Fin 29 → Fin 1024 → K) (gamma : K) :
    OwnSymbolCollision.matching univ (received29 c1 c2) (expected p) gamma=
      CoveredOriginalSymbols.symbolSupport
        (exactInitialEncoder (ClaimTransport.batch gamma p))
        (NearGammaSelectedC1.rawBatch c1 c2 gamma) := by
  ext i
  simp only [OwnSymbolCollision.matching, CoveredOriginalSymbols.symbolSupport,
    Finset.mem_filter, Finset.mem_univ, true_and]
  have value := symbolError_eval c1 c2 p gamma i
  dsimp only [symbolError] at value
  rw [value, sub_eq_zero]
  exact eq_comm

/-- Q and its final may be adaptive; a large literal quotient support plus
its actual image reconstruction forces a root of a pre-gamma polynomial
whenever this represented tuple lacks the source's own-support threshold. -/
theorem covered_forces_exception (c1 : C1Received) (c2 : C2Received)
    (p : Fin 29 → Fin 1024 → K) (data : Data (K := K)) (checked : data.Checked)
    (G : Finset K) (gamma : K) (inGamma : gamma∈G) (Q : Fin 1024 → K)
    (image : Q 1023=0 ∧ (atGamma data gamma).b*Q 1022-(atGamma data gamma).c*Q 1021=0)
    (member : Q∈literalFamily (SelectedComponentGame.received c1 c2 data gamma))
    (represented : (atGamma data gamma).original Q=ClaimTransport.batch gamma p)
    (small : (own c1 c2 p).card<38228) : gamma∈badGamma c1 c2 p G := by
  letI : DecidableEq K := Classical.decEq K
  have large := CoveredOriginalSymbols.family_member_original_38230 (atGamma data gamma)
    (SelectedComponentGame.checked_atGamma data checked gamma) Q image
    (NearGammaSelectedC1.rawBatch c1 c2 gamma) member
  rw [represented] at large
  have enough : 38230≤(OwnSymbolCollision.matching univ
      (received29 c1 c2) (expected p) gamma).card := by
    rw [matching_eq_symbols]
    exact large
  have excess : (OwnSymbolCollision.own univ (received29 c1 c2) (expected p)).card<
      (OwnSymbolCollision.matching univ (received29 c1 c2) (expected p) gamma).card := by
    dsimp only [own] at small
    omega
  have inside : OwnSymbolCollision.matching univ (received29 c1 c2) (expected p) gamma ⊆
      (Finset.univ : Finset (Fin 1048576)) := Finset.subset_univ _
  have roots : ∀i∈OwnSymbolCollision.matching univ (received29 c1 c2) (expected p) gamma,
      (symbolError c1 c2 p i).eval gamma=0 := by
    intro i member
    simpa only [OwnSymbolCollision.matching, Finset.mem_filter, Finset.mem_univ,
      true_and, symbolError] using member
  let U : Finset (Fin 1048576) := univ
  let C := OwnSymbolCollision.own U (received29 c1 c2) (expected p)
  let M := OwnSymbolCollision.matching U (received29 c1 c2) (expected p) gamma
  let R : Fin 1048576 → K → Prop := fun i g => (symbolError c1 c2 p i).eval g=0
  have step := Generic.excess_charged (I := Fin 1048576) (K := K) U C M G R gamma
  have forced := step inGamma inside roots excess
  simpa only [badGamma, OwnSymbolCollision.exception, Generic.chargedSet, symbolError,
    U, C, M, R] using forced

/-- The complementary own-supported branch really enters the unfiltered
early-C1 mathematical family. This does not make that family an algorithm. -/
theorem own_supported_early_member (c1 : C1Received) (c2 : C2Received)
    (p : Fin 29 → Fin 1024 → K) (large : 38228≤(own c1 c2 p).card) :
    c1Projection p∈EarlyC1Family.family c1 := by
  apply EarlyC1Family.actual_late_projection_member c1 c2 p
  rw [←own_eq_joint]
  exact large
