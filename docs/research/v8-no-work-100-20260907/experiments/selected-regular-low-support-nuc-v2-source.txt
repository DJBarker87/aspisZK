def fibreCount (received : Fin 1048576 → K) (Q : Fin 1024 → K) : Nat :=
  (fullSupport univ received Q).card

theorem full_bad_card (received : Fin 1048576 → K) (Q : Fin 1024 → K) :
    fibreCount received Q+
      (fibreBad (m := 262144) (exactInitialEncoder Q) received).card ≤ 262144 := by
  have disjoint : Disjoint (fullSupport univ received Q)
      (fibreBad (m := 262144) (exactInitialEncoder Q) received) := by
    apply Finset.disjoint_left.mpr
    intro i full bad
    obtain ⟨slot, different⟩ := (Finset.mem_filter.mp bad).2
    exact different ((Finset.mem_filter.mp full).2 slot)
  have size := Finset.card_le_univ
    (fullSupport univ received Q ∪ fibreBad (m := 262144) (exactInitialEncoder Q) received)
  rw [Finset.card_union_of_disjoint disjoint] at size
  simpa only [fibreCount, Fintype.card_fin] using size

/-- Literal field query support is the image of these same indexed fibres. -/
theorem full_query_mass (received : Fin 1048576 → K) (Q : Fin 1024 → K) (q : Nat) :
    fullQueryMass received Q q=
      ((fibreCount received Q).choose q : ℚ)/(262144).choose q := by
  have sets : domain.filter (fun t => FullSlots received Q (index t))=
      (fullSupport univ received Q).image (storedPoint (K := K)) := by
    ext t
    constructor
    · intro member
      obtain ⟨inside, full⟩ := Finset.mem_filter.mp member
      refine Finset.mem_image.mpr ⟨index t, ?_, point_index t inside⟩
      exact Finset.mem_filter.mpr ⟨Finset.mem_univ _, full⟩
    · intro member
      obtain ⟨i, full, rfl⟩ := Finset.mem_image.mp member
      refine Finset.mem_filter.mpr
        ⟨Finset.mem_image.mpr ⟨i, Finset.mem_univ _, rfl⟩, ?_⟩
      rw [index_point]
      exact (Finset.mem_filter.mp full).2
  unfold fullQueryMass
  rw [sets, Finset.card_image_of_injective _ storedPoint_injective, domain_card]

/-- Parameterized sharp TWO-SYMBOL loss, before choosing a tail threshold. -/
theorem fibre_to_original_support (c1 : EarlyC1Projection.C1Received)
    (c2 : EarlyC1LateProjection.C2Received) (d : Data (K := K)) (checked : d.Checked)
    (gamma : K) (Q : Fin 1024 → K)
    (image : Q 1023=0 ∧ d.b*Q 1022-d.c*Q 1021=0) :
    4*fibreCount (SelectedComponentGame.received c1 c2 d gamma) Q ≤
      (originalSupport c1 c2 d gamma Q).card+2 := by
  have slots := complete_fibres_supply_symbols
    (SelectedComponentGame.received c1 c2 d gamma) Q
  have loss := original_symbol_loss (atGamma d gamma)
    (SelectedComponentGame.checked_atGamma d checked gamma) Q image
    (NearGammaSelectedC1.rawBatch c1 c2 gamma)
  exact slots.trans loss

def lowRegularPrefix {q : Nat} (e : Execution q) (F : TrivariatePolynomial K)
    (r : Fin 2) (gamma kappa tau alpha : K) : Prop :=
  LowPrefix e gamma kappa tau alpha ∧ fixedRegularPrefix e F r gamma kappa tau alpha

/-- The chosen actual regular witness is low, not a separately assumed Q.
No derivative uniqueness or pre-alpha final selection is needed here. -/
theorem low_regular_witness {q : Nat} (e : Execution q)
    (F : TrivariatePolynomial K) (factor : F ∈ curvePrimeFactors (parent e.c1 e.c2))
    (retained : Retained (point e.data)
      (fun r => CurveOODGate.answerCurve (answers e.data r)) F)
    (higher : 3 ≤ F.natDegree) (r : Fin 2) (gamma kappa tau alpha : K)
    (selected : lowRegularPrefix e F r gamma kappa tau alpha) :
    ∃ Q, Qualified e.c1 e.c2 e.data F gamma Q ∧
      (e.strategy gamma kappa).final tau alpha=coefficientFoldLayer 256 alpha Q ∧
      9558 ≤ fibreCount (e.raw gamma) Q ∧ fibreCount (e.raw gamma) Q ≤ 200807 := by
  obtain ⟨low, Q, qualified, notBad, final, _⟩ := selected
  have witness : Witness e gamma kappa tau alpha Q :=
    ⟨qualified.1, notBad, final, F, factor, retained, qualified.2.2, higher⟩
  have bad := low_witness_bad e gamma kappa tau alpha low Q witness
  have total := full_bad_card (e.raw gamma) Q
  have lower : 9558 ≤ fibreCount (e.raw gamma) Q :=
    (mem_literalFamily (e.raw gamma) Q).mp qualified.1
  exact ⟨Q, qualified, final, lower, by omega⟩

/-- A same-Q fibre threshold enters the actual existing support-tail set. -/
theorem regular_fibre_tail_mem {q : Nat} (e : Execution q)
    (checked : e.data.Checked) (F : TrivariatePolynomial K) (r : Fin 2)
    (Gamma : Finset K) (gamma : K) (inside : gamma ∈ Gamma)
    (Q : Fin 1024 → K) (qualified : Qualified e.c1 e.c2 e.data F gamma Q)
    (regular : (FactorCoherence.derivativeCurve F (point e.data r)
      (CurveOODGate.answerCurve (answers e.data r))).eval gamma ≠ 0)
    (m : Nat) (many : m ≤ fibreCount (e.raw gamma) Q) :
    gamma ∈ supportGammas e.c1 e.c2 e.data F r Gamma (4*m-2) := by
  have support := fibre_to_original_support e.c1 e.c2 e.data checked gamma Q qualified.2.1
  change 4*fibreCount (e.raw gamma) Q ≤
    (originalSupport e.c1 e.c2 e.data gamma Q).card+2 at support
  apply (mem_supportGammas e.c1 e.c2 e.data F r Gamma (4*m-2) gamma).mpr
  exact ⟨inside, regular, Q, qualified, by omega⟩

/-- Restrict the already checked fixed-factor adaptive matching moment.
This has NO suffix charge and is therefore safe to sum over actual factors. -/
theorem low_regular_moment {q : Nat} (e : Execution q)
    (checked : e.data.Checked)
    (circles : e.data.x0^2+e.data.y0^2=1 ∧ e.data.x1^2+e.data.y1^2=1)
    (west : ∀ r, pointX e.data r ≠ -1) (F : TrivariatePolynomial K)
    (r : Fin 2) (gamma : K) (Q : Fin 1024 → K)
    (qualified : Qualified e.c1 e.c2 e.data F gamma Q) (kappa tau : K)
    (A : Finset K) (nonempty : A.Nonempty) (count : q ≤ 262144) :
    avg A (fun alpha => if lowRegularPrefix e F r gamma kappa tau alpha then
      compatibleMoment (atFold ((e.rows gamma).before kappa)
        (e.strategy gamma kappa) tau alpha) ((e.strategy gamma kappa).final tau alpha)
        ((oracle 0 (e.raw gamma)).folded alpha) domain q else 0) ≤
      fullQueryMass (e.raw gamma) Q q+
        (1-fullQueryMass (e.raw gamma) Q q)*min 1 ((3 : ℚ)/A.card) := by
  apply (avg_mono A _ _ ?_).trans
    (adaptive_regular_moment e checked circles west F r gamma Q qualified
      kappa tau A nonempty count)
  intro alpha _
  have nonnegative : 0 ≤ compatibleMoment
      (atFold ((e.rows gamma).before kappa) (e.strategy gamma kappa) tau alpha)
      ((e.strategy gamma kappa).final tau alpha)
      ((oracle 0 (e.raw gamma)).folded alpha) domain q := by
    unfold compatibleMoment matchingRatio
    split_ifs <;> positivity
  by_cases low : lowRegularPrefix e F r gamma kappa tau alpha
  · simp only [if_pos low, if_pos low.2, le_refl]
  · rw [if_neg low]
    split_ifs
    · exact nonnegative
    · exact le_refl _

def regularLowUnion {q : Nat} (e : Execution q) (r : Fin 2)
    (gamma kappa tau alpha : K) : Prop :=
  ∃ F ∈ SingularOODFamily.higherFactors (parent e.c1 e.c2),
    Retained (point e.data) (fun row => CurveOODGate.answerCurve (answers e.data row)) F ∧
      lowRegularPrefix e F r gamma kappa tau alpha

/-- Reduce the UNION event only once. It still contains the actual shared
strategy and the exact finite matching moment, before any support integration. -/
theorem shared_suffix {q : Nat} (e : Execution q) (r : Fin 2)
    (gamma kappa : K) (A G : Finset K) (ha : A.Nonempty) (hg : G.Nonempty)
    (positive : 0<q) (count : q≤262144) :
    avg G (fun tau => avg A (fun alpha =>
      if regularLowUnion e r gamma kappa tau alpha then suffix e gamma kappa tau alpha A G
      else 0)) ≤
    avg G (fun tau => avg A (fun alpha =>
      if regularLowUnion e r gamma kappa tau alpha then
        compatibleMoment (atFold ((e.rows gamma).before kappa)
          (e.strategy gamma kappa) tau alpha) ((e.strategy gamma kappa).final tau alpha)
          ((oracle 0 (e.raw gamma)).folded alpha) domain q else 0))+
      (q : ℚ)/G.card+18/A.card :=
  causal_supported_bound ((e.rows gamma).before kappa) e.quarterChecked
    (oracle 0 (e.raw gamma)) (e.strategy gamma kappa)
    (regularLowUnion e r gamma kappa) A G ha hg positive (by rwa [domain_card])
