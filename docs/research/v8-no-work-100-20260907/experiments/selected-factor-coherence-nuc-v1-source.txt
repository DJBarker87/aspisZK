theorem pointSubstitution_eq_gate {K : Type*} [Field K]
    {maximumDegree curveDegree xBound yRows zBound : Nat}
    (coefficients : CurveMonomialIndex maximumDegree curveDegree xBound yRows zBound → K)
    (x : K) (answers : Fin (curveDegree+1) → K) :
    pointSubstitution x (answerCurve answers) (curveTrivariatePolynomial coefficients) =
      curveConstraintPolynomial (fun _ : Fin 1 => x)
        (fun lane (_ : Fin 1) => answers lane) coefficients 0 0 := by
  classical
  simp only [pointSubstitution,RingHom.coe_comp,Function.comp_apply,
    specializeEvaluationPoint,evaluateInnerVariable,curveTrivariatePolynomial,
    Polynomial.coe_mapRingHom,Polynomial.map_sum,Polynomial.map_monomial,
    Polynomial.coe_evalRingHom,Polynomial.eval_finsetSum,Polynomial.eval_monomial,
    curveConstraintPolynomial,hasseXOrder_zero,hasseYOrder_zero,
    Nat.choose_zero_right,Nat.cast_one,mul_one,Nat.sub_zero,answerCurve]
  apply Finset.sum_congr rfl
  intro m _
  simp only [Polynomial.C_mul,Polynomial.C_pow]
  ring

theorem derivativeCurve_eq_gate {K : Type*} [Field K]
    {maximumDegree curveDegree xBound yRows zBound : Nat}
    (coefficients : CurveMonomialIndex maximumDegree curveDegree xBound yRows zBound → K)
    (x : K) (answers : Fin (curveDegree+1) → K) :
    derivativeCurve (curveTrivariatePolynomial coefficients) x (answerCurve answers) =
      CurveOODDerivative.atPoint coefficients x answers := by
  classical
  rw [CurveOODDerivative.atPoint_sum]
  simp only [derivativeCurve,pointSubstitution,RingHom.coe_comp,Function.comp_apply,
    specializeEvaluationPoint,evaluateInnerVariable,curveTrivariatePolynomial,
    Polynomial.derivative_sum,Polynomial.derivative_monomial,
    Polynomial.coe_mapRingHom,Polynomial.map_sum,Polynomial.map_monomial,
    Polynomial.coe_evalRingHom,Polynomial.eval_finsetSum,Polynomial.eval_monomial,
    Polynomial.map_mul,map_natCast,Polynomial.eval_mul,Polynomial.eval_natCast]
  apply Finset.sum_congr rfl
  intro m _
  simp only [Polynomial.C_mul,Polynomial.C_natCast]
  ring

def parent (c1 : C1Received) (c2 : C2Received) : TrivariatePolynomial K :=
  curveTrivariatePolynomial (fixedInterpolant c1 c2)

def derivativeGate (c1 : C1Received) (c2 : C2Received)
    (d : Data (K := K)) (r : Fin 2) : K[X] :=
  CurveOODDerivative.atPoint (fixedInterpolant c1 c2) (point d r) (answers d r)

/-- Fixed before gamma, selected from the whole actual OOD identity. In
the regular branch EVERY later covered image-valid quotient follows F away
from a single explicit root set. No candidate or factor union is charged. -/
theorem selected_fixed_factor
    (c1 : C1Received) (c2 : C2Received) (d : Data (K := K)) (r : Fin 2)
    (identity : gate c1 c2 d r=0) :
    derivativeGate c1 c2 d r=0 ∨
      ∃ F ∈ curvePrimeFactors (parent c1 c2), 0<F.natDegree ∧
        pointSubstitution (point d r) (answerCurve (answers d r)) F=0 ∧
        ∃ B : Finset K, B.card≤117049 ∧
          ∀ (checked : d.Checked)
            (circles : d.x0^2+d.y0^2=1 ∧ d.x1^2+d.y1^2=1)
            (west : ∀ j, pointX d j≠-1) (gamma : K) (Q : Fin 1024 → K),
            gamma∉B →
            Q∈literalFamily (SelectedComponentGame.received c1 c2 d gamma) →
            (Q 1023=0 ∧ d.b*Q 1022-d.c*Q 1021=0) →
            challengeCandidateHom gamma
              (exactCircleGRSPolynomial ((atGamma d gamma).original Q)) F=0 := by
  have equal : derivativeCurve (parent c1 c2) (point d r)
      (answerCurve (answers d r))=derivativeGate c1 c2 d r :=
    derivativeCurve_eq_gate _ _ _
  have actualIdentity : pointSubstitution (point d r) (answerCurve (answers d r))
      (parent c1 c2)=0 := by
    rw [parent,pointSubstitution_eq_gate]
    exact identity
  have nonzero : parent c1 c2≠0 :=
    curveTrivariatePolynomial_ne_zero _ (fixedInterpolant_nonzero c1 c2)
  by_cases singular : derivativeGate c1 c2 d r=0
  · exact Or.inl singular
  · right
    obtain singular' | ⟨F,member,positive,factorIdentity,coherent⟩ :=
      fixed_factor_or_singular (parent c1 c2) nonzero (point d r)
        (answerCurve (answers d r)) actualIdentity
    · exact (singular (equal.symm.trans singular')).elim
    · let B := (derivativeGate c1 c2 d r).roots.toFinset
      refine ⟨F,member,positive,factorIdentity,B,?_,?_⟩
      · exact (Multiset.toFinset_card_le _).trans
          ((Polynomial.card_roots' _).trans (CurveOODDerivative.selected_degree _ _ _))
      · intro checked circles west gamma Q notBad qmember image
        have compatible := covered_compatible c1 c2 d checked circles west gamma Q image qmember
        -- Use the literal candidate from the bridge, not an arbitrary
        -- existential replacement with possibly different OOD values.
        have valid := CoveredOriginalSymbols.selected_width29_valid c1 c2 d checked
          (fun _ => Q) gamma image qmember
        have root := AspisK1.V7ExactCorrelatedAgreement.exactInitialValidCandidate_substitute_eq_zero
          (received29 c1 c2) (CoveredOriginalSymbols.originalStrategy c1 c2 d (fun _ => Q))
          (fixedInterpolant c1 c2) (fixedInterpolant_kernel c1 c2) gamma valid
        apply coherent gamma _
        · simpa only [parent,challengeCandidateHom,RingHom.coe_comp,Function.comp_apply,
            specializeChallenge_curveTrivariatePolynomial lastRow,
            substituteCandidate_weightedBivariatePolynomial] using root
        · rw [answers_eval]
          simpa only [point,factor,pointX,pointY,atGamma] using
            CoveredOODGRS.original_grs_at_ood (atGamma d gamma) checked Q image circles r (west r)
        · rw [equal]
          intro rootZero
          exact notBad (Multiset.mem_toFinset.mpr ((Polynomial.mem_roots singular).mpr rootZero))
