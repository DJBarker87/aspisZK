/-- Both candidate membership and final coefficient identity are derived from
the actual interleaved basis and canonical inverse tables. -/
theorem actual_covered_iff (received : Fin 1048576 → K)
    (final : K → Fin 256 → K) (a : Nat) (alpha : K) :
    MaskedCurveTail.CoveredAt exactFinalLinear (lanes received) a
      (strategy received final) alpha ↔
      OffFamilyIntersection.CoveredAt Finset.univ received a alpha (final alpha) := by
  have inverses := canonical_one_fold_schedule_exact 0
  have generic := MaskedCurveRepresentation.covered_iff exactFinalLinear
    exactFinalEncoder_injective exactCircleX exactCircleY
    (canonicalOneFoldSchedule 0).circleInv2x (canonicalOneFoldSchedule 0).circleInv2y
    inverses.1 inverses.2 received a (strategy received final) alpha
  constructor
  · intro covered
    obtain ⟨Q,own,represented⟩ := generic.mp covered
    rw [actual_support] at own
    exact ⟨Q,own,represented⟩
  · rintro ⟨Q,own,represented⟩
    apply generic.mpr
    refine ⟨Q,?_,represented⟩
    rwa [actual_support]

/-- Full-field alpha tail for the actual selected candidate, with no assumed
family membership or decoded-word/quotient correspondence. The borrowed

theorem actual_tail_card_le_on (received : Fin 1048576 → K)
    (final : K → Fin 256 → K) (a : Nat) (floor : a≤9558) (challenges : Finset K) :
    (OffFamilyIntersection.tailSet Finset.univ received a 9558 challenges final).card≤
      foldChallengeCap := by
  classical
  have masked := MaskedCurveTail.tail_card_le exactFinalLinear (lanes received)
    a 9558 foldChallengeCap (by omega) floor
    exactV7FinalDegreeThreeCurveDecodable (strategy received final)
  have inclusion : OffFamilyIntersection.tailSet Finset.univ received a 9558 challenges final⊆
      MaskedCurveTail.tailSet exactFinalLinear (lanes received) a 9558 (strategy received final) := by
    intro alpha member
    obtain ⟨_,size,outside⟩ :=
      (actual_tail_mem_iff Finset.univ received a 9558 challenges final alpha).mp member
    apply Finset.mem_filter.mpr
    refine ⟨?_,?_⟩
    · apply (mem_goodChallenges_iff _ _ _ _ _).mpr
      apply (actual_valid_iff received final (9558-1) alpha).mpr
      omega
    · intro covered
      exact outside ((actual_covered_iff received final a alpha).mp covered)
  exact (Finset.card_le_card inclusion).trans masked

theorem actual_tail_card_le (received : Fin 1048576 → K)
    (final : K → Fin 256 → K) (a : Nat) (floor : a≤9558) :
    (OffFamilyIntersection.tailSet Finset.univ received a 9558 Finset.univ final).card≤
