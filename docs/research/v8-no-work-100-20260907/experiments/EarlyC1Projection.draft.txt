/-- Concrete V7 base projection preserves every correctly supported C1 fibre.
This works on the tuple's own support and needs no decoder/provider success. -/
theorem projected_support (received : C1Received) (p : C1Messages)
    (base : ∀ lane index, projectBase (received lane index) = received lane index) :
    support fibreEncode (receivedFibres received) p ⊆
      support fibreEncode (receivedFibres received) (fun lane => projectMessage (p lane)) := by
  exact projection_preserves_support fibreEncode (receivedFibres received) p projectMessage
    (fun value slot => projectBase (value slot))
    (fun message f => funext fun slot => congrFun (exactInitialEncoder_commutes message) _)
    (fun lane f => funext fun slot => base lane _)

theorem supported_C1_is_base (received : C1Received) (p : C1Messages)
    (base : ∀ lane index, projectBase (received lane index) = received lane index)
    (own : 245609 ≤ (support fibreEncode (receivedFibres received) p).card) :
    ∀ lane, projectMessage (p lane) = p lane := by
  have projected := own.trans (card_le_card (projected_support received p base))
  have left := earlyC1_identifies_own_support received p own
  have right := earlyC1_identifies_own_support received
    (fun lane => projectMessage (p lane)) projected
  have same := Option.some.inj (right.symm.trans left)
  exact congrFun same

theorem earlyC1_value_is_embedded (received : C1Received) (p : C1Messages)
    (base : ∀ lane index, projectBase (received lane index) = received lane index)
    (found : earlyC1 received = some p) (row : Fin 1024) (lane : Fin 16) :
    embedM31Exact ((p (semanticColumnIndex lane) row).re.re) =
      p (semanticColumnIndex lane) row := by
  have own := some_early_has_support fibreEncode (receivedFibres received) 245609 p found
  have fixed := congrFun (supported_C1_is_base received p base own
    (semanticColumnIndex lane)) row
  exact fixed

noncomputable def jointOwnSupport (received : Width29Received) (p : Width29Messages) :
    Finset (Fin 262144) :=
  support fibreEncode (fun lane f slot => received lane (fibreEmbed (f,slot))) p

/-- Consumes the dense near-gamma branch's joint own-support conclusion.
The full tuple can depend on C2/earlier challenges; its C1 projection cannot
vary once the fixed received C1 is fixed. The small-Good/no-tuple branch is
not asserted to satisfy this premise. -/
theorem late_width29_projection_is_early
    (receivedC1 : C1Received) (received : Width29Received) (p : Width29Messages)
    (c1fixed : ∀ column index,
      received (c1LaneIndex column) index = receivedC1 column index)
    (own : 245609 ≤ (jointOwnSupport received p).card) :
    earlyC1 receivedC1 = some (fun column => p (c1LaneIndex column)) := by
  have projected := support_restrict fibreEncode
    (fun lane f slot => received lane (fibreEmbed (f,slot))) p c1LaneIndex
    (receivedFibres receivedC1) (fun column f => funext fun slot => c1fixed column _)
  exact earlyC1_identifies_own_support receivedC1 (fun column => p (c1LaneIndex column))
    (own.trans (card_le_card projected))

theorem no_early_excludes_late_own_support
    (receivedC1 : C1Received) (received : Width29Received) (p : Width29Messages)
    (c1fixed : ∀ column index,
      received (c1LaneIndex column) index = receivedC1 column index)
    (none : earlyC1 receivedC1 = Option.none) :
    ¬ 245609 ≤ (jointOwnSupport received p).card := by
  intro own
  have found := late_width29_projection_is_early receivedC1 received p c1fixed own
  rw [none] at found
  contradiction

theorem adaptive_C2_cannot_change_C1
    (receivedC1 : C1Received) (leftWord rightWord : Width29Received)
    (left right : Width29Messages)
    (leftFixed : ∀ column index,
      leftWord (c1LaneIndex column) index = receivedC1 column index)
    (rightFixed : ∀ column index,
      rightWord (c1LaneIndex column) index = receivedC1 column index)
    (leftOwn : 245609 ≤ (jointOwnSupport leftWord left).card)
    (rightOwn : 245609 ≤ (jointOwnSupport rightWord right).card) :
    (fun column => left (c1LaneIndex column)) =
      (fun column => right (c1LaneIndex column)) := by
  have hl := late_width29_projection_is_early receivedC1 leftWord left leftFixed leftOwn
  have hr := late_width29_projection_is_early receivedC1 rightWord right rightFixed rightOwn
  exact Option.some.inj (hl.symm.trans hr)

noncomputable def earlySemanticTrace (received : C1Received) :
    Option (Fin 1024 → Fin 16 → M31Exact) :=
  (earlyC1 received).map fun p row lane => (p (semanticColumnIndex lane) row).re.re

/-- The selected sixteen-column M31 trace projection is the old V7 projection,
now attached to an early object rather than to a post-C2 tuple by assertion. -/
theorem late_semantic_trace_is_early
    (receivedC1 : C1Received) (received : Width29Received) (p : Width29Messages)
    (c1fixed : ∀ column index,
      received (c1LaneIndex column) index = receivedC1 column index)
    (own : 245609 ≤ (jointOwnSupport received p).card) :
    earlySemanticTrace receivedC1 = some (semanticTrace p) := by
  unfold earlySemanticTrace
  rw [late_width29_projection_is_early receivedC1 received p c1fixed own]
  rfl
