#!/usr/bin/env python3
"""Generator for lean/R0Z/PoseidonConstLift.lean (DESIGN-BUILDER-MODEL §11.5 (1)).

Reads the Poseidon round-constant tables from four sources and aborts on any
mismatch:
  P  production   crates/aspis-statement/src/poseidon2.rs
  C  copy         docs/research/v8-r0-zk-20261009/refine/poseidon/src/lib.rs
  X  extracted    docs/research/v8-r0-zk-20261009/lean/R0Z/PoseidonExtracted.lean
  M  R0P model    docs/research/v8-r0-sem-proof-20261008/lean/R0P/PoseidonConstants.lean
Tables: EXTERNAL_INITIAL [4][16], EXTERNAL_FINAL [4][16], INTERNAL [14]
(R0P names poseidonExternalInitial, poseidonExternalFinal,
poseidonInternalConstants).  Every value must be >= 2 (Nat.AtLeastTwo) and < p.

Emitted proof shape, per lane, a named application only:
  lane_lift <n> (by unfold <TABLE>; exact UScalar.ofNatCore_val_eq _) rfl
where lane_lift unfolds liftU and applies Nat.cast_ofNat; the u32 side is
read by UScalar.ofNatCore_val_eq after unfolding the one table definition
(list access only), and the R0P side `rfl` is a vector access.  No Nat.cast
is unfolded, no literal is decomposed, no decide/norm_num/simp is emitted.
Then one row lemma per row (match over Fin 16) and one dispatch lemma per
table (match over the row index).

Usage (from the repository root):
  gen-const-lift.py.txt            write the Lean file
  gen-const-lift.py.txt --check    regenerate in memory; exit 1 if the
                                   committed file differs or any source
                                   disagrees (nothing is written)
"""
import re, sys, os

P = 2147483647
ROOT = os.getcwd()
R = "docs/research/v8-r0-zk-20261009"
SRC = {
    "P": "crates/aspis-statement/src/poseidon2.rs",
    "C": f"{R}/refine/poseidon/src/lib.rs",
    "X": f"{R}/lean/R0Z/PoseidonExtracted.lean",
    "M": "docs/research/v8-r0-sem-proof-20261008/lean/R0P/PoseidonConstants.lean",
}
OUT = f"{R}/lean/R0Z/PoseidonConstLift.lean"
TABLES = [("EXTERNAL_INITIAL", "poseidonExternalInitial", 4, 16, "ei"),
          ("EXTERNAL_FINAL", "poseidonExternalFinal", 4, 16, "ef"),
          ("INTERNAL", "poseidonInternalConstants", 1, 14, "ic")]

def die(msg):
    sys.stderr.write("gen-const-lift: ABORT: " + msg + "\n"); sys.exit(2)

def read(k):
    return open(os.path.join(ROOT, SRC[k])).read()

def rust_table(src, name):
    m = re.search(r"const " + name + r"\s*:[^=]*=\s*\[", src)
    if not m: die(f"{name} not found in Rust source")
    depth, i = 1, m.end()
    while depth:
        c = src[i]; depth += (c == "[") - (c == "]"); i += 1
    body = re.sub(r"//[^\n]*", "", src[m.end():i - 1])
    return [int(t.replace("_", ""), 0) for t in re.findall(r"0x[0-9a-fA-F_]+|\b\d[\d_]*\b", body)]

def lean_x_table(src, name):
    m = re.search(r"def " + name + r" :[^\n]*\n?[^\n]*:=", src)
    if not m: die(f"{name} not found in extracted Lean")
    end = src.find("/--", m.end()); end = len(src) if end < 0 else end
    body = src[m.end():end]
    return [int(t) for t in re.findall(r"(\d+)#u32", body)]

def lean_m_table(src, name):
    m = re.search(r"def " + name + r" :[^\n]*:=", src)
    if not m: die(f"{name} not found in R0P constants")
    end = src.find("/--", m.end()); end = len(src) if end < 0 else end
    end2 = src.find("end R0P", m.end()); end = min(end, end2) if end2 >= 0 else end
    return [int(t, 16) for t in re.findall(r"0x([0-9a-fA-F]+)", src[m.end():end])]

def tables():
    srcs = {k: read(k) for k in SRC}
    out = {}
    for rname, mname, rows, cols, _ in TABLES:
        vals = {"P": rust_table(srcs["P"], rname), "C": rust_table(srcs["C"], rname),
                "X": lean_x_table(srcs["X"], rname), "M": lean_m_table(srcs["M"], mname)}
        for k, v in vals.items():
            if len(v) != rows * cols: die(f"{rname} in {k}: {len(v)} values, expected {rows * cols}")
        for k in "CXM":
            if vals[k] != vals["P"]:
                bad = [j for j in range(rows * cols) if vals[k][j] != vals["P"][j]]
                die(f"{rname}: source {k} differs from production at flat index {bad[:5]}")
        for j, x in enumerate(vals["P"]):
            if not (2 <= x < P): die(f"{rname}[{j}] = {x} outside [2, p)")
        out[rname] = vals["P"]
    return out

def emit(tab):
    L = []
    w = L.append
    w("-- GENERATED by refine/extract/gen-const-lift.py.txt; do not edit (check with --check).")
    w("import R0Z.PoseidonTables")
    w("")
    w("/-! Round-constant lifts (DESIGN-BUILDER-MODEL §11.5 (1)).  The copy's u32 tables")
    w("EXTERNAL_INITIAL, EXTERNAL_FINAL and INTERNAL lift lane by lane to R0P's")
    w("poseidonExternalInitial, poseidonExternalFinal and poseidonInternalConstants.")
    w("The generator checked production poseidon2.rs, the copy, the extracted Lean and")
    w("R0P PoseidonConstants.lean agree on every value.  Each lane is one application of")
    w("`lane_lift` (Nat.cast_ofNat); no Nat.cast or literal is unfolded. -/")
    w("set_option autoImplicit false")
    w("")
    w("namespace R0Z.PoseidonRefine")
    w("open Aeneas Aeneas.Std R0PoseidonCopy R0Z.WriterRefine")
    w("")
    w("variable {K : Type} [Field K]")
    w("")
    w("/-- One lane: a u32 whose value is the numeral `n` lifts to the K-numeral `n`. -/")
    w("theorem lane_lift {x : U32} {c : K} (n : Nat) [Nat.AtLeastTwo n]")
    w("    (hx : x.val = (OfNat.ofNat n : Nat)) (hc : c = (OfNat.ofNat n : K)) : liftU x = c := by")
    w("  unfold liftU")
    w("  rw [hx, hc]")
    w("  exact Nat.cast_ofNat")
    w("")
    for rname, mname, rows, cols, tag in TABLES:
        vals = tab[rname]
        for r in range(rows):
            for i in range(cols):
                n = vals[r * cols + i]
                if rows > 1:
                    lhs = f"(({rname}.val[{r}]!).val[{i}]!)"
                    rhs = f"R0P.{mname} {r} {i}"
                    nm = f"{tag}_{r}_{i}"
                else:
                    lhs = f"({rname}.val[{i}]!)"
                    rhs = f"R0P.{mname} {i}"
                    nm = f"{tag}_{i}"
                w(f"theorem {nm} : liftU (K := K) {lhs} = {rhs} :=")
                w(f"  lane_lift {n} (by unfold {rname}; exact UScalar.ofNatCore_val_eq _) rfl")
            w("")
        if rows > 1:
            for r in range(rows):
                w(f"theorem {tag}_row_{r} : ∀ i : Fin {cols},")
                w(f"    liftU (K := K) (({rname}.val[{r}]!).val[i.val]!) = R0P.{mname} {r} i")
                for i in range(cols):
                    w(f"  | ⟨{i}, _⟩ => {tag}_{r}_{i}")
                w(f"  | ⟨_ + {cols}, h⟩ => absurd h (by omega)")
                w("")
            w(f"theorem {tag}_rows : ∀ (r : Fin {rows}) (i : Fin {cols}),")
            w(f"    liftU (K := K) (({rname}.val[r.val]!).val[i.val]!) = R0P.{mname} r i")
            for r in range(rows):
                w(f"  | ⟨{r}, _⟩ => {tag}_row_{r}")
            w(f"  | ⟨_ + {rows}, h⟩ => absurd h (by omega)")
            w("")
        else:
            w(f"theorem {tag}_all : ∀ j : Fin {cols}, liftU (K := K) ({rname}.val[j.val]!) = R0P.{mname} j")
            for i in range(cols):
                w(f"  | ⟨{i}, _⟩ => {tag}_{i}")
            w(f"  | ⟨_ + {cols}, h⟩ => absurd h (by omega)")
            w("")
    w("end R0Z.PoseidonRefine")
    w("")
    for nm in ["lane_lift", "ei_rows", "ef_rows", "ic_all"]:
        w(f"#print axioms R0Z.PoseidonRefine.{nm}")
    return "\n".join(L) + "\n"

def main():
    text = emit(tables())
    if "--check" in sys.argv[1:]:
        cur = open(os.path.join(ROOT, OUT)).read() if os.path.exists(os.path.join(ROOT, OUT)) else ""
        if cur != text:
            sys.stderr.write("gen-const-lift: CHECK FAILED: committed file differs from regeneration\n"); sys.exit(1)
        print("gen-const-lift: check ok (4 sources agree; 142 lanes; file up to date)")
        return
    for bad in ("decide", "norm_num", "simp", "native_decide", "sorry", "maxHeartbeats", "maxRecDepth"):
        if re.search(r"\b" + bad + r"\b", text): die(f"emitted text contains {bad}")
    open(os.path.join(ROOT, OUT), "w").write(text)
    print(f"gen-const-lift: wrote {OUT}")

main()
