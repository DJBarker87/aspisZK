#!/usr/bin/env python3
"""D64 hRel: split the Aeneas output of c3/build (tag br, which only APPENDS builder 4 to
lib.rs) into the unchanged R1Z/CertBuildExtracted and the new R1Z/CertRelExtracted.
After the two substitute.py.txt edits, the new output must start with the committed
R1Z/CertBuildExtracted.lean minus its closing `end R1CertBuild` (asserted byte for byte),
so no compiled builder module changes; the remainder (the builder-4 definitions, verbatim)
becomes R1Z/CertRelExtracted, which imports R1Z.CertBuildExtracted and reopens the same
namespace with the same opens and options.
Usage: split.py.txt <substituted Build.lean> <CertBuildExtracted.lean> <out CertRelExtracted.lean>"""
import sys
new = open(sys.argv[1]).read()
old = open(sys.argv[2]).read()
tail = "end R1CertBuild\n"
assert old.endswith(tail) and new.endswith(tail)
prefix = old[: -len(tail)]
assert new.startswith(prefix), "builder 1-3 definitions changed"
rest = new[len(prefix):]
head = old[: old.index("namespace R1CertBuild\n")]
hdr = head.replace("import Aeneas.Std\nimport Aeneas.Tactic.RustAttributes\n", "import R1Z.CertBuildExtracted\n", 1)
assert hdr != head
out = hdr + "namespace R1CertBuild\n" + rest
open(sys.argv[3], "w").write(out)
print("prefix_bytes", len(prefix), "new_bytes", len(rest))
