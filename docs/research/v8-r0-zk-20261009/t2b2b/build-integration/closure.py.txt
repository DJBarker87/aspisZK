# closure.py <lean-path (colon list, first wins)> <src-roots (colon list)> <Module>...
# Resolve the user-module import closure of the given modules: each module is resolved to the
# first .olean on the LEAN_PATH (as lean does); its imports are read from the first matching
# source under src-roots.  Library modules (Mathlib, Batteries, Aesop, Qq, Lean, Init, Std,
# Plausible, ProofWidgets, ImportGraph, LeanSearchClient, Lake, Cli) are leaves.  Prints JSON:
# module -> {olean, olean_sha256, source, source_sha256}.
import sys, os, re, json, hashlib
LIB = ('Mathlib','Batteries','Aesop','Qq','Lean','Init','Std','Plausible','ProofWidgets',
       'ImportGraph','LeanSearchClient','Lake','Cli')
lp = sys.argv[1].split(':'); roots = sys.argv[2].split(':'); todo = list(sys.argv[3:])
def sha(p):
    return hashlib.sha256(open(p,'rb').read()).hexdigest() if p and os.path.exists(p) else None
def find(dirs, rel):
    for d in dirs:
        p = os.path.join(d, rel)
        if os.path.exists(p): return p
    return None
out = {}
while todo:
    m = todo.pop()
    if m in out or m.split('.')[0] in LIB: continue
    rel = m.replace('.', '/')
    ol = find(lp, rel + '.olean'); src = find(roots, rel + '.lean')
    out[m] = {'olean': ol and os.path.realpath(ol), 'olean_sha256': sha(ol),
              'olean_mtime': ol and int(os.path.getmtime(ol)),
              'source': src and os.path.realpath(src), 'source_sha256': sha(src)}
    out[m]['imports'] = []
    if src:
        for line in open(src, encoding='utf-8'):
            mm = re.match(r'^(?:public\s+)?import\s+(.+)$', line.rstrip())
            if mm:
                xs = [x for x in mm.group(1).split() if re.match(r'^[A-Z][A-Za-z0-9_.]*$', x)]
                out[m]['imports'].extend(xs); todo.extend(xs)
print(json.dumps(out, indent=1, sort_keys=True))
