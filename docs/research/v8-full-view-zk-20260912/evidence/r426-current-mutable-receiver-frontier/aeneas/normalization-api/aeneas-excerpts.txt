### string literal borrow temp insertion
  2582	(** Whenever we write a string literal in Rust, rustc actually introduces a
  2583	    constant of type [&str]. Generally speaking, because [str] is unsized, it
  2584	    doesn't make sense to manipulate values of type [str] directly. But in the
  2585	    context of Aeneas, it is reasonable to decompose those literals into: a
  2586	    string stored in a local variable, then a borrow of this variable.
  2587	
  2588	    Remark: the new statements all have the id 0: this pass requires to refresh
  2589	    the ids later. *)
  2590	let decompose_str_borrows (_ : crate) (f : fun_decl) : fun_decl =
  2591	  (* Map  *)
  2592	  let body =
  2593	    match f.body with
  2594	    | StructuredBody body ->
  2595	        let new_locals = ref [] in
  2596	        let _, gen =
  2597	          LocalId.mk_stateful_generator_starting_at_id
  2598	            (LocalId.of_int (List.length body.locals.locals))
  2599	        in
  2600	        let fresh_local ty =
  2601	          let local =
  2602	            {
  2603	              index = gen ();
  2604	              local_ty = ty;
  2605	              name = None;
  2606	              span = f.item_meta.span;
  2607	            }
  2608	          in
  2609	          new_locals := local :: !new_locals;
  2610	          local.index
  2611	        in
  2612	
  2613	        (* Function to decompose a constant literal *)
  2614	        let decompose_rvalue (span : Meta.span) (lv : place) (rv : rvalue) :
  2615	            statement list =
  2616	          let new_statements = ref [] in
  2617	
  2618	          (* Visit the rvalue *)
  2619	          let visitor =
  2620	            object
  2621	              inherit [_] map_statement as super
  2622	
  2623	              (* We have to visit all the constant operands.
  2624	                 As we might need to replace them with borrows, while borrows
  2625	                 are rvalues (i.e., not operands) we have to introduce two
  2626	                 intermediate statements: the string initialization, then
  2627	                 the borrow, that we can finally move.
  2628	              *)
  2629	              method! visit_Constant env (cv : constant_expr) =
  2630	                match (cv.kind, cv.ty) with
  2631	                | ( CLiteral (VStr str),
  2632	                    TRef
  2633	                      (_, (TAdt { id = TBuiltin TStr; _ } as str_ty), ref_kind)
  2634	                  ) ->
  2635	                    (* We need to introduce intermediate assignments *)
  2636	                    (* First the string initialization *)
  2637	                    let local_id =
  2638	                      let local_id = fresh_local str_ty in
  2639	                      let new_cv : constant_expr =
  2640	                        { kind = CLiteral (VStr str); ty = str_ty }
  2641	                      in
  2642	                      let st =
  2643	                        {
  2644	                          span;
  2645	                          statement_id = StatementId.zero;
  2646	                          kind =
  2647	                            Assign
  2648	                              ( { kind = PlaceLocal local_id; ty = str_ty },
  2649	                                Use (Constant new_cv, NoRetag) );
  2650	                          comments_before = [];
  2651	                        }
  2652	                      in
  2653	                      new_statements := st :: !new_statements;
  2654	                      local_id
  2655	                    in
  2656	                    let str_len =
  2657	                      Constant
  2658	                        {
  2659	                          kind =
  2660	                            CLiteral
  2661	                              (VScalar
  2662	                                 (UnsignedScalar
  2663	                                    (Usize, Z.of_int (String.length str))));
  2664	                          ty = TLiteral (TUInt Usize);
  2665	                        }
  2666	                    in
  2667	                    (* Then the borrow *)
  2668	                    let local_id =
  2669	                      let nlocal_id = fresh_local cv.ty in
  2670	                      let bkind =
  2671	                        match ref_kind with
  2672	                        | RMut -> BMut
  2673	                        | RShared -> BShared
  2674	                      in
  2675	                      let rv =
  2676	                        RvRef
  2677	                          ( { kind = PlaceLocal local_id; ty = str_ty },
  2678	                            bkind,
  2679	                            str_len )
  2680	                      in
  2681	                      let lv = { kind = PlaceLocal nlocal_id; ty = cv.ty } in
  2682	                      let st =
  2683	                        {
  2684	                          span;
  2685	                          statement_id = StatementId.zero;
  2686	                          kind = Assign (lv, rv);
  2687	                          comments_before = [];
  2688	                        }
  2689	                      in
  2690	                      new_statements := st :: !new_statements;
  2691	                      nlocal_id
  2692	                    in
  2693	                    (* Finally we can move the value *)
  2694	                    Move { kind = PlaceLocal local_id; ty = cv.ty }
  2695	                | _ -> super#visit_Constant env cv
  2696	            end
  2697	          in
  2698	
  2699	          let rv = visitor#visit_rvalue () rv in
  2700	
  2701	          (* Construct the sequence *)
  2702	          let assign =
  2703	            {
  2704	              span;
  2705	              statement_id = StatementId.zero;
  2706	              kind = Assign (lv, rv);
  2707	              comments_before = [];
  2708	            }
  2709	          in
  2710	          (* Note that the new statements are in reverse order *)
  2711	          let statements = assign :: !new_statements in
  2712	          List.rev statements
### global reference and deref insertion plus call visitor
  2835	(** This micro-pass introduces intermediate assignments to access the global
  2836	    values in order to simplify the semantics.
  2837	
  2838	    Whenever we access a constant, we introduce a shared borrow and a
  2839	    dereference. Ex.:
  2840	
  2841	    {[
  2842	      let x = copy C;
  2843	
  2844	        ~~>
  2845	
  2846	      let tmp = &C;
  2847	      let x = copy *C;
  2848	    ]}
  2849	
  2850	    Remark: we use the crate to lookup the type of the globals.
  2851	
  2852	    TODO: generalize the evaluation of globals in the symbolic interpreter. *)
  2853	let decompose_global_accesses (crate : crate) (f : fun_decl) : fun_decl =
  2854	  (* Map  *)
  2855	  let body =
  2856	    match f.body with
  2857	    | StructuredBody body -> (
  2858	        let new_locals = ref [] in
  2859	        let _, gen =
  2860	          LocalId.mk_stateful_generator_starting_at_id
  2861	            (LocalId.of_int (List.length body.locals.locals))
  2862	        in
  2863	        let fresh_local ty =
  2864	          let local =
  2865	            {
  2866	              index = gen ();
  2867	              local_ty = ty;
  2868	              name = None;
  2869	              span = f.item_meta.span;
  2870	            }
  2871	          in
  2872	          new_locals := local :: !new_locals;
  2873	          local.index
  2874	        in
  2875	
  2876	        (* Function to decompose the operands in a statement *)
  2877	        let decompose_in_statement (st : statement) : statement list =
  2878	          let span = st.span in
  2879	          let new_statements = ref [] in
  2880	
  2881	          (* Visit the rvalue *)
  2882	          let visitor =
  2883	            object
  2884	              inherit [_] map_statement as super
  2885	              method! visit_place _ p = super#visit_place p.ty p
  2886	
  2887	              method! visit_PlaceGlobal ty gref =
  2888	                (* Compute the type of the reference *)
  2889	                let ref_ty = TRef (RErased, ty, RShared) in
  2890	
  2891	                (* Introduce the intermediate reference *)
  2892	                let local_id =
  2893	                  let local_id = fresh_local ref_ty in
  2894	                  let metadata = Constant mk_unit_const in
  2895	                  let st =
  2896	                    {
  2897	                      span;
  2898	                      statement_id = StatementId.zero;
  2899	                      kind =
  2900	                        Assign
  2901	                          ( { kind = PlaceLocal local_id; ty = ref_ty },
  2902	                            RvRef
  2903	                              ( { kind = PlaceGlobal gref; ty },
  2904	                                BShared,
  2905	                                metadata ) );
  2906	                      comments_before = [];
  2907	                    }
  2908	                  in
  2909	                  new_statements := st :: !new_statements;
  2910	                  local_id
  2911	                in
  2912	
  2913	                (* Finally we can update the place *)
  2914	                PlaceProjection
  2915	                  ({ kind = PlaceLocal local_id; ty = ref_ty }, Deref)
  2916	            end
  2917	          in
  2918	
  2919	          let kind =
  2920	            match st.kind with
  2921	            | Assign (lv, rv) -> Assign (lv, visitor#visit_rvalue mk_unit_ty rv)
  2922	            | Assert ({ cond; expected; check_kind }, on_failure, on_unwind) ->
  2923	                let cond = visitor#visit_operand mk_unit_ty cond in
  2924	                Assert ({ cond; expected; check_kind }, on_failure, on_unwind)
  2925	            | Call ({ func; args; dest }, on_unwind) ->
  2926	                let func = visitor#visit_fn_operand mk_unit_ty func in
  2927	                let args = List.map (visitor#visit_operand mk_unit_ty) args in
  2928	                Call ({ func; args; dest }, on_unwind)
  2929	            | Switch (If (cond, yes, no)) ->
  2930	                Switch (If (visitor#visit_operand mk_unit_ty cond, yes, no))
  2931	            | Switch (SwitchInt (op, ty, branches, otherwise)) ->
  2932	                Switch (SwitchInt
  2933	                  (visitor#visit_operand mk_unit_ty op, ty, branches, otherwise))
  2934	            | Switch (Match (scrut, branches, otherwise)) ->
  2935	                Switch (Match
  2936	                  (visitor#visit_place mk_unit_ty scrut, branches, otherwise))
  2937	            | SetDiscriminant _ | StorageLive _ | StorageDead _ | PlaceMention _
  2938	            | Drop (_, _, _, _)
  2939	            | Abort _
  2940	            | Return
  2941	            | UnwindResume
  2942	            | Break _
  2943	            | Continue _
  2944	            | Nop
  2945	            | Loop _
  2946	            | Error _ -> st.kind
  2947	            | _ ->
  2948	                [%craise] st.span
  2949	                  ("unsupported statement: " ^ show_statement_kind st.kind)
  2950	          in
  2951	          let st = { st with kind } in
  2952	
  2953	          List.rev (st :: !new_statements)
  2954	        in
  2955	
  2956	        (* Visit all the statements and decompose the operands *)
  2957	        try
  2958	          let body_body = map_statement decompose_in_statement body.body in
  2959	          StructuredBody
  2960	            {
  2961	              body with
  2962	              body = body_body;
  2963	              locals =
  2964	                {
  2965	                  body.locals with
  2966	                  locals = body.locals.locals @ List.rev !new_locals;
  2967	                };
  2968	            }

### ref evaluator ignores ptr_metadata
  1222	(** Evaluate an rvalue which creates a reference (i.e., an rvalue which is `&p`
  1223	    or `&mut p` or `&two-phase p`) *)
  1224	let eval_rvalue_ref (config : config) (span : Meta.span) (p : place)
  1225	    (bkind : borrow_kind) (ctx : eval_ctx) :
  1226	    tvalue * eval_ctx * (SymbolicAst.expr -> SymbolicAst.expr) =
  1227	  match bkind with
  1228	  | BShared | BTwoPhaseMut | BShallow ->
  1229	      (* **REMARK**: we initially treated shallow borrows like shared borrows.
  1230	         In practice this restricted the behaviour too much, so for now we
  1231	         forbid them and remove them in the prepasses (see the comments there
  1232	         as to why this is sound).
  1233	      *)
  1234	      [%sanity_check] span (bkind <> BShallow);
  1235	
  1236	      (* Access the value *)
  1237	      let access =
  1238	        match bkind with
  1239	        | BShared | BShallow -> Read
  1240	        | BTwoPhaseMut -> Write
  1241	        | _ -> [%craise] span "Unreachable"
  1242	      in
  1243	
  1244	      let greedy_expand = false in
  1245	      let lid, v, ctx, cc =
  1246	        access_rplace_reorganize_and_read config span greedy_expand access p ctx
  1247	      in
  1248	      (* Generate the fresh shared borrow id *)
  1249	      let sid = ctx.fresh_shared_borrow_id () in
  1250	      (* Compute the loan value, with which to replace the value at place p *)
  1251	      let bid, nv =
  1252	        match (lid, v.value) with
  1253	        | Some lid, _ | None, VLoan (VSharedLoan (lid, _)) ->
  1254	            (* The value is (directly inside) a shared loan: we do not need to do anything *)
  1255	            (lid, v)
  1256	        | _ ->
  1257	            (* Not a shared loan: add a wrapper *)
  1258	            let bid = ctx.fresh_borrow_id () in
  1259	            let v' = VLoan (VSharedLoan (bid, v)) in
  1260	            (bid, { v with value = v' })
  1261	      in
  1262	      (* Update the value in the context to replace it with the loan *)
  1263	      let ctx = write_place span access p nv ctx in
  1264	      (* Compute the rvalue - simply a shared borrow with the fresh id.
  1265	       * Note that the reference is *mutable* if we do a two-phase borrow *)
  1266	      let ref_kind =
  1267	        match bkind with
  1268	        | BShared | BShallow -> RShared
  1269	        | BTwoPhaseMut -> RMut
  1270	        | _ -> [%craise] span "Unreachable"
  1271	      in
  1272	      let rv_ty = TRef (RErased, v.ty, ref_kind) in
  1273	      let bc =
  1274	        match bkind with
  1275	        | BShared | BShallow ->
  1276	            (* See the remark at the beginning of the match branch: we
  1277	               handle shallow borrows like shared borrows *)
  1278	            VSharedBorrow (bid, sid)
  1279	        | BTwoPhaseMut -> VReservedMutBorrow (bid, sid)
  1280	        | _ -> [%craise] span "Unreachable"
  1281	      in
  1282	      let rv : tvalue = { value = VBorrow bc; ty = rv_ty } in
  1283	      (* Return *)
  1284	      (rv, ctx, cc)
  1285	  | BMut | BUniqueImmutable ->
  1286	      (* We treat unique immutable borrows or mutable borrows: the difference is not
  1287	         that important in our case *)
  1288	      (* Access the value *)
  1289	      let access = Write in
  1290	      let greedy_expand = false in
  1291	      let _, v, ctx, cc =
  1292	        access_rplace_reorganize_and_read config span greedy_expand access p ctx
  1293	      in
  1294	      (* Compute the rvalue - wrap the value in a mutable borrow with a fresh id *)
  1295	      let bid = ctx.fresh_borrow_id () in
  1296	      let rv_ty = TRef (RErased, v.ty, RMut) in
  1297	      let rv : tvalue = { value = VBorrow (VMutBorrow (bid, v)); ty = rv_ty } in
  1298	      (* Compute the loan value with which to replace the value at place p *)
  1299	      let nv = { v with value = VLoan (VMutLoan bid) } in
  1300	      (* Update the value in the context to replace it with the loan *)
  1301	      let ctx = write_place span access p nv ctx in
  1302	      (* Return *)
  1303	      (rv, ctx, cc)

  1490	    (rvalue : rvalue) (ctx : eval_ctx) :

  1502	  | UnaryOp (unop, op) -> eval_unary_op config span unop op ctx
