   303	                    for pass in passes.iter() {
   304	                        if pass.should_run(&ctx.options) {
   305	                            trace!("# Starting pass {}", pass.name());
   306	                            pass.transform_item(ctx, item.reborrow());
   307	                        }
   308	                    }
   309	                });
   310	                for pass in passes.iter() {
   311	                    pass.finalize(self);
   312	                }

   315	                self.for_each_fun_decl(|ctx, decl| {
