   278	    pub fn enqueue_module_item(&mut self, def_id: &hax::DefId) {
   279	        if let Some(trait_def_id) = self.is_method_decl_without_default(def_id) {
   280	            // Don't translate the method itself as it doesn't correspond to an item, translate the
   281	            // trait instead.
   282	            self.enqueue_module_item(&trait_def_id);
   283	            return;
   284	        }
   285	        let Some(kind) = self.base_kind_for_item(def_id) else {
   286	            return;
   287	        };
   288	        let item_src = if self.options.monomorphize_with_hax {
   289	            if let Ok(def) = self.poly_hax_def(def_id)
   290	                && !def.has_any_generics()
   291	            {
   292	                // Monomorphize this item and the items it depends on.
   293	                TransItemSource::monomorphic(def.this(), kind)
   294	            } else {
   295	                // Skip polymorphic items and items that cause errors.
   296	                return;
   297	            }
   298	        } else {
   299	            TransItemSource::polymorphic(def_id, kind)
   300	        };
   301	        let _: Option<ItemId> = self.register_and_enqueue(&None, item_src);
   302	    }

   303	fn compute_declarations_graph(ctx: &TransformCtx) -> DiGraphMap<ItemId, ()> {
   304	    let mut deps = Deps::default();
   305	    // Start from the items included in `start_from`. We've mostly only translated items accessible
   306	    // from that, but some passes render items inaccessible again, which we filter out here.
   307	    deps.unprocessed = ctx
   308	        .translated
   309	        .all_items()
   310	        .filter(|item| {
   311	            ctx.options
   312	                .start_from
   313	                .iter()
   314	                .any(|pat| pat.matches(&ctx.translated, item.item_meta()))
   315	        })
   316	        .map(|item| item.id())
   317	        .collect();
   318	
   319	    // Explore reachable items.
   320	    while let Some(id) = deps.unprocessed.pop() {
   321	        if !deps.visited.insert(id) {
   322	            continue;
   323	        }
   324	        let Some(item) = ctx.translated.get_item(id) else {
   325	            continue;
   326	        };
   327	        let mut visitor = deps.visitor_for_item(ctx, item);
   328	        match item {
   329	            ItemRef::Type(..) | ItemRef::TraitImpl(..) | ItemRef::Global(..) => {
   330	                let _ = item.drive(&mut visitor);
   331	            }
   332	            ItemRef::Fun(d) => {
   333	                let FunDecl {
   334	                    def_id,
   335	                    item_meta: _,
   336	                    generics,
   337	                    signature,
   338	                    src,
   339	                    is_global_initializer: _,
   340	                    body,
   341	                } = d;
   342	                let _ = def_id.drive(&mut visitor); // For `seen_current_id`
   343	                // Skip `d.is_global_initializer` to avoid incorrect mutual dependencies.
   344	                // TODO: add `is_global_initializer` to `ItemSource`.
   345	                let _ = generics.drive(&mut visitor);
   346	                let _ = signature.drive(&mut visitor);
   347	                let _ = body.drive(&mut visitor);
   348	                if let ItemSource::TraitDecl { trait_ref, .. } = src {
   349	                    visitor.insert_edge(trait_ref.id);
   350	                }
   351	            }
   352	            ItemRef::TraitDecl(d) => {
   353	                let TraitDecl {
   354	                    def_id,
   355	                    item_meta: _,
   356	                    generics,
   357	                    implied_clauses: parent_clauses,
   358	                    consts,
   359	                    types,
   360	                    methods,
   361	                    vtable,
   362	                } = d;
   363	                let _ = def_id.drive(&mut visitor); // For `seen_current_id`
   364	                // Visit the traits referenced in the generics
   365	                let _ = generics.drive(&mut visitor);
   366	
   367	                // Visit the parent clauses
   368	                let _ = parent_clauses.drive(&mut visitor);
   369	
   370	                // Visit the items
   371	                let _ = types.drive(&mut visitor);
   372	                let _ = vtable.drive(&mut visitor);
   373	
   374	                // We consider that a trait decl only contains the function/constant signatures.
   375	                // Therefore we don't explore the default const/method ids.
   376	                for assoc_const in consts {
   377	                    let TraitAssocConst {
   378	                        name: _,
   379	                        attr_info: _,
   380	                        ty,
   381	                        default,
   382	                    } = assoc_const;
   383	                    let _ = ty.drive(&mut visitor);
   384	                    if let Some(gref) = default {
   385	                        visitor.insert_node(gref.id); // Still count the item as reachable.
   386	                        let _ = gref.generics.drive(&mut visitor);
   387	                    }
   388	                }
   389	                for bound_method in methods {
   390	                    let _ = bound_method.params.drive(&mut visitor);
   391	                    let _ = bound_method.skip_binder.signature.drive(&mut visitor);
   392	                    if let Some(funref) = &bound_method.skip_binder.default {
   393	                        visitor.insert_node(funref.id); // Still count the item as reachable.
   394	                        let _ = funref.generics.drive(&mut visitor);
   395	                    }
   396	                }
   397	            }
   398	        }
