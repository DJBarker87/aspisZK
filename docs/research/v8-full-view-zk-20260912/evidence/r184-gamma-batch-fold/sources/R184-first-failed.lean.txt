import AspisV8R19.R183DefaultFoldExecution

set_option autoImplicit false
namespace AspisV8R19.R184GammaBatchFold

open Aeneas Aeneas.Std Result
open AspisR156FullFreeze.aspis_core
open AspisV8R15.ExactTowerBase
noncomputable section

def closedGammaCallMut
    (c : field.QM31 × field.QM31)
    (args : field.QM31 × field.QM31) :
    Result (field.QM31 × (field.QM31 × field.QM31)) :=
  AspisV8R17.MaskClosureWriteback.finishCall
    (R174GammaBatchStep.rawCallMut c args)

def closedGammaCallOnce
    (c : field.QM31 × field.QM31)
    (args : field.QM31 × field.QM31) : Result field.QM31 := do
  let (value, _) ← closedGammaCallMut c args
  ok value

def closedGammaFnOnce :
    core.ops.function.FnOnce (field.QM31 × field.QM31)
      (field.QM31 × field.QM31) field.QM31 := {
  call_once := closedGammaCallOnce
}

def closedGamma :
    core.ops.function.FnMut (field.QM31 × field.QM31)
      (field.QM31 × field.QM31) field.QM31 := {
  FnOnceInst := closedGammaFnOnce
  call_mut := closedGammaCallMut
}

def gammaFold : List QM31Exact → QM31Exact → QM31Exact → QM31Exact →
    QM31Exact × QM31Exact
  | [], power, _gamma, acc => (acc, power)
  | value :: values, power, gamma, acc =>
      gammaFold values (power * gamma) gamma (acc + power * value)

theorem closedGamma_foldMutList (values : List QM31Exact)
    (power gamma acc : QM31Exact) :
    R183DefaultFoldExecution.foldMutList closedGamma
        (values.map R164ProductExecution.encode)
        (R164ProductExecution.encode power, R164ProductExecution.encode gamma)
        (R164ProductExecution.encode acc) =
      .ok (R164ProductExecution.encode (gammaFold values power gamma acc).1,
        (R164ProductExecution.encode (gammaFold values power gamma acc).2,
          R164ProductExecution.encode gamma)) := by
  induction values generalizing power acc with
  | nil =>
      simp [R183DefaultFoldExecution.foldMutList, gammaFold]
  | cons value values ih =>
      simp only [List.map_cons, R183DefaultFoldExecution.foldMutList,
        gammaFold, closedGamma, closedGammaCallMut]
      rw [R174GammaBatchStep.encoded_closed]
      simp only [bind_tc_ok]
      exact ih (power * gamma) (acc + power * value)

#print axioms closedGamma_foldMutList

end
end AspisV8R19.R184GammaBatchFold
