struct FinishLog{events:[u8;7],count:usize,round_reply:[Result<E,Error>;2],round_ptr:[*const u8;2],round_len:[usize;2],round_alpha:[E;2],round_claim:[E;2],new:NewPort,pair:PairPort,dot_reply:Result<E,Error>,dot_f:*const u8,dot_f_len:usize,dot_weights:*const K,dot_len:usize}
fn finish_event(l:&mut FinishLog,tag:u8){l.events[l.count]=tag;l.count+=1;}
fn round_packet(poly:&[u8],alpha:E,claim:E,round:usize,l:&mut FinishLog)->Result<E,Error>{l.round_ptr[round]=poly.as_ptr();l.round_len[round]=poly.len();l.round_alpha[round]=alpha;l.round_claim[round]=claim;finish_event(l,round as u8);l.round_reply[round]}
fn dot_packet(f:&[u8],weights:&[K],l:&mut FinishLog)->Result<E,Error>{l.dot_f=f.as_ptr();l.dot_f_len=f.len();l.dot_weights=weights.as_ptr();l.dot_len=weights.len();finish_event(l,5);l.dot_reply}
fn finish_cut(v:&View<'_>,p:&Prepared,a:&Accumulator,l:&mut FinishLog)->Result<(),Error>{
 let mut claim=p.claim;claim=round_packet(v.round0,p.alpha[0],claim,0,l)?;
 if p.variant.folds()==2{claim=round_packet(v.round1,p.alpha[1],claim,1,l)?;}
 finish_event(l,2);let weights=new_packet(p.variant,&p.semantic_alpha,&p.z,&mut l.new)?;
 finish_event(l,3);let pair=pair_packet(&weights,&p.alpha[..p.variant.folds()],v.final_message,p.kappa,&mut l.pair)?;
 if pair!=claim{return Err(Error::Relation)}
 finish_event(l,4);let actual=dot_packet(v.final_message,&a.weights,l)?;
 if a.next!=22||actual!=a.rhs{return Err(Error::Query)}
 finish_event(l,6);Ok(())
}
fn finish_expected(a:&Accumulator,l:&FinishLog)->(Result<(),u8>,usize){
 if let Err(e)=l.round_reply[0]{return(Err(error_code(e)),1)}
 let claim=match l.round_reply[1]{Ok(v)=>v,Err(e)=>return(Err(error_code(e)),2)};
 if let Some(e)=l.new.error{return(Err(error_code(e)),3)}
 let pair=match l.pair.reply{Ok(v)=>v,Err(e)=>return(Err(error_code(e)),4)};
 if pair.to_limbs()!=claim.to_limbs(){return(Err(3),4)}
 let actual=match l.dot_reply{Ok(v)=>v,Err(e)=>return(Err(error_code(e)),6)};
 if a.next!=22||actual.to_limbs()!=a.rhs.to_limbs(){return(Err(5),6)}
 (Ok(()),7)
}
