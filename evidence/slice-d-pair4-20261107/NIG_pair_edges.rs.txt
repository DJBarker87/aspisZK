struct PairEdges{who:*const Weights,shared:bool,folds:usize,n:usize,alpha:*const E,alpha_len:usize,message:*const u8,message_len:usize,scratch:*const K,scratch_len:usize,parts:*const E,p0:E,p1:E,regular_post:[E;7],common:*const K,common_value:K,partial_message:*const u8,partial_message_len:usize,partial_reply:[E;4],high_who:*const Weights,high_partial:E,high_p0:E,high_p1:E,high_parts:*const E}
fn edge_regular(w:&Weights,shared:bool,alpha:&[E],f:&[u8],folds:usize,n:usize,ap0:&[E;4],ap1:&[E;4],scratch:&mut [K],parts:&mut [E;7],power:usize,l:&mut PairEdges){
 l.who=w;l.shared=shared;l.folds=folds;l.n=n;l.alpha=alpha.as_ptr();l.alpha_len=alpha.len();l.message=f.as_ptr();l.message_len=f.len();l.scratch=scratch.as_ptr();l.scratch_len=scratch.len();l.parts=parts.as_ptr();l.p0=ap0[power];l.p1=ap1[power];
 parts[2]=l.regular_post[2];parts[5]=l.regular_post[5];parts[6]=l.regular_post[6];if !shared{parts[1]=l.regular_post[1];parts[3]=l.regular_post[3];}
}
fn edge_partial(f:&[u8],common:&[K;16],i:usize,l:&mut PairEdges)->[E;4]{l.common=common.as_ptr();l.common_value=common[i];l.partial_message=f.as_ptr();l.partial_message_len=f.len();l.partial_reply}
fn edge_high(partials:&[E;4],w:&Weights,ap0:&[E;4],ap1:&[E;4],parts:&mut [E;7],power:usize,l:&mut PairEdges){l.high_who=w;l.high_partial=partials[power];l.high_p0=ap0[power];l.high_p1=ap1[power];l.high_parts=parts.as_ptr();}

 let alpha:[E;2]=[anye(),anye()];let f:[u8;2048]=kani::any();let ap0=core::array::from_fn(|_|anye());let ap1=core::array::from_fn(|_|anye());let mut scratch:[K;64]=core::array::from_fn(|_|anyq());let mut parts:[E;7]=core::array::from_fn(|_|anye());let common:[K;16]=core::array::from_fn(|_|anyq());
 let shared:bool=kani::any();let power:usize=kani::any();kani::assume(power<4);let i:usize=kani::any();kani::assume(i<16);let limb:usize=kani::any();kani::assume(limb<8);

}
