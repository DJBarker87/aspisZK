
fn error(i:u8)->Error{match i%8{0=>Error::Parse,1=>Error::Semantic,2=>Error::Transcript,3=>Error::Relation,4=>Error::Merkle,5=>Error::Query,6=>Error::Checkpoint,_=>Error::Allocation}}
fn error_code(e:Error)->u8{match e{Error::Parse=>0,Error::Semantic=>1,Error::Transcript=>2,Error::Relation=>3,Error::Merkle=>4,Error::Query=>5,Error::Checkpoint=>6,Error::Allocation=>7}}
fn unit_reply()->Result<(),Error>{if kani::any(){Ok(())}else{Err(error(kani::any()))}}
struct GuardLog{events:[u8;9],count:usize,p0:E,p1:E,ap0:[E;4],ap1:[E;4],requested:usize,value:K,fill_ok:bool,ptr:*mut K,who:*const Weights,folds:usize,guard:bool,forwarded:bool,regular:Result<(),Error>,partial:Result<(),Error>,high:Result<(),Error>,regular_calls:usize,common_calls:usize,init:E,reg_ptr:*const K,reg_len:usize,reg_f:*const u8,reg_f_len:usize,reg_alpha:*const E,reg_alpha_len:usize,reg_who:*const Weights,reg_folds:usize,reg_n:usize,reg_ap0:E,reg_ap1:E,common_reply:[K;16],common_who:*const Weights,common_initial:K,partial_reply:[E;4],partial_ptr:*const u8,partial_len:usize,partial_common:K,high_who:*const Weights,high_partial:E,high_ap0:E,high_ap1:E,regular_parts:[E;7],high_parts:[E;7],high_before:E}
fn event(l:&mut GuardLog,tag:u8){l.events[l.count]=tag;l.count+=1;}
fn p0(x:E,l:&mut GuardLog)->[E;4]{l.p0=x;event(l,0);l.ap0}
fn p1(x:E,l:&mut GuardLog)->[E;4]{l.p1=x;event(l,1);l.ap1}
fn fill_call(n:usize,value:K,l:&mut GuardLog)->Result<Vec<K>,()>{l.requested=n;l.value=value;event(l,2);if !l.fill_ok{return Err(())}
 let mut out=Vec::<K>::with_capacity(n);unsafe{core::ptr::write_bytes(out.as_mut_ptr(),0,n);out.set_len(n);}l.ptr=out.as_mut_ptr();Ok(out)}
fn guard_call(w:&Weights,folds:usize,l:&mut GuardLog)->bool{l.who=w;l.folds=folds;event(l,3);l.guard}
fn regular_call(w:&Weights,shared:bool,folds:usize,n:usize,ap0:&[E;4],ap1:&[E;4],power_index:usize,alpha:&[E],f:&[u8],scratch:&mut [K],parts:&mut [E;7],slot:usize,l:&mut GuardLog)->Result<(),Error>{l.reg_who=w;l.reg_folds=folds;l.reg_n=n;l.reg_ap0=ap0[power_index];l.reg_ap1=ap1[power_index];l.forwarded=shared;l.regular_calls+=1;l.init=parts[slot];l.reg_ptr=scratch.as_ptr();l.reg_len=scratch.len();l.reg_f=f.as_ptr();l.reg_f_len=f.len();l.reg_alpha=alpha.as_ptr();l.reg_alpha_len=alpha.len();event(l,4);parts[2]=l.regular_parts[2];parts[5]=l.regular_parts[5];parts[6]=l.regular_parts[6];if !shared{parts[1]=l.regular_parts[1];parts[3]=l.regular_parts[3];}l.regular}
fn common_call(w:&Weights,qindex:usize,l:&mut GuardLog)->[K;16]{l.common_who=w;l.common_initial=[K::ZERO;16][qindex];l.common_calls+=1;event(l,5);l.common_reply}
fn partial_call(f:&[u8],common:&[K;16],qindex:usize,l:&mut GuardLog)->Result<[E;4],Error>{l.partial_ptr=f.as_ptr();l.partial_len=f.len();l.partial_common=common[qindex];event(l,6);l.partial.map(|_|l.partial_reply)}
fn high_call(partials:&[E;4],w:&Weights,ap0:&[E;4],ap1:&[E;4],power_index:usize,parts:&mut [E;7],slot:usize,l:&mut GuardLog)->Result<(),Error>{l.high_who=w;l.high_partial=partials[power_index];l.high_ap0=ap0[power_index];l.high_ap1=ap1[power_index];l.high_before=parts[slot];event(l,7);parts[1]=l.high_parts[1];parts[3]=l.high_parts[3];l.high}
fn guard_cut(w:&Weights,alpha:&[E],f:&[u8],slot:usize,power_index:usize,qindex:usize,l:&mut GuardLog)->Result<(),Error>{
 let n=f.len()/32;let folds=alpha.len();let mut expected_n=1024usize;for _ in 0..folds{expected_n>>=2u32;}if n!=expected_n{return Err(Error::Parse)}
 let ap0=p0(alpha[0],l);let ap1=if folds==2{p1(alpha[1],l)}else{[E::ONE;4]};
 let mut scratch=match fill_call(n,K::ZERO,l){Ok(v)=>v,Err(_)=>return Err(Error::Allocation)};
 let mut parts=[E::ZERO;7];let shared=guard_call(w,folds,l);
 regular_call(w,shared,folds,n,&ap0,&ap1,power_index,alpha,f,&mut scratch,&mut parts,slot,l)?;
 if shared{let common=common_call(w,qindex,l);let partials=partial_call(f,&common,qindex,l)?;high_call(&partials,w,&ap0,&ap1,power_index,&mut parts,slot,l)?;}
 event(l,8);Ok(())
}
fn guard_check(g:usize){
 let w=Weights{variant:Variant::S2,factors:core::array::from_fn(|_|core::array::from_fn(|_|core::array::from_fn(|_|anyq()))),points:core::array::from_fn(|_|core::array::from_fn(|_|anyq())),extra:[anyq(),anyq()]};
 let f:[u8;2048]=kani::any();let alpha:[E;2]=[anye(),anye()];let slot:usize=kani::any();kani::assume(slot<7);let limb:usize=kani::any();kani::assume(limb<8);let power_index:usize=kani::any();kani::assume(power_index<4);let qindex:usize=kani::any();kani::assume(qindex<16);
 let mut l=GuardLog{events:[99;9],count:0,p0:E::ZERO,p1:E::ZERO,ap0:core::array::from_fn(|_|anye()),ap1:core::array::from_fn(|_|anye()),requested:0,value:K::ONE,fill_ok:kani::any(),ptr:core::ptr::null_mut(),who:core::ptr::null(),folds:0,guard:kani::any(),forwarded:false,regular:unit_reply(),partial:unit_reply(),high:unit_reply(),regular_calls:0,common_calls:0,init:E::ONE,reg_ptr:core::ptr::null(),reg_len:0,reg_f:core::ptr::null(),reg_f_len:0,reg_alpha:core::ptr::null(),reg_alpha_len:0,reg_who:core::ptr::null(),reg_folds:0,reg_n:0,reg_ap0:E::ZERO,reg_ap1:E::ZERO,common_reply:core::array::from_fn(|_|anyq()),common_who:core::ptr::null(),common_initial:K::ONE,partial_reply:core::array::from_fn(|_|anye()),partial_ptr:core::ptr::null(),partial_len:0,partial_common:K::ZERO,high_who:core::ptr::null(),high_partial:E::ZERO,high_ap0:E::ZERO,high_ap1:E::ZERO,regular_parts:core::array::from_fn(|_|anye()),high_parts:core::array::from_fn(|_|anye()),high_before:E::ZERO};
 let r=guard_cut(&w,&alpha,&f,slot,power_index,qindex,&mut l);
 let expected_reply=if !l.fill_ok{Err(7)}else if let Err(e)=l.regular{Err(error_code(e))}else if l.guard{if let Err(e)=l.partial{Err(error_code(e))}else if let Err(e)=l.high{Err(error_code(e))}else{Ok(())}}else{Ok(())};
 let expected_count=if !l.fill_ok{3}else if l.regular.is_err(){5}else if !l.guard{6}else if l.partial.is_err(){7}else if l.high.is_err(){8}else{9};
 let expected_common=l.fill_ok&&l.regular.is_ok()&&l.guard;
 if cfg!(NIG_corrupt){match g{0=>l.requested=63,1=>l.forwarded=!l.guard,2=>l.events[0]=99,3=>l.count+=1,4=>l.init=E::ONE,_=>l.reg_len=63}}
 let properties=[
 l.requested==64&&l.value==K::ZERO&&l.p0.to_limbs()[limb]==alpha[0].to_limbs()[limb]&&l.p1.to_limbs()[limb]==alpha[1].to_limbs()[limb],
 !l.fill_ok||(l.who==&w as *const Weights&&l.folds==2&&(l.regular_calls==0||l.forwarded==l.guard)),
 l.events[0]==0&&l.events[1]==1&&l.events[2]==2&&(!l.fill_ok||(l.events[3]==3&&l.events[4]==4))&&(!expected_common||(l.events[5]==5&&l.events[6]==6))&&(!(expected_common&&l.partial.is_ok())||l.events[7]==7)&&(r.is_err()||l.events[expected_count-1]==8),
 r.map_err(error_code)==expected_reply&&l.count==expected_count&&l.common_calls==usize::from(expected_common),
 (l.regular_calls==0||l.init.to_limbs()[limb]==0)&&(!expected_common||l.common_initial==K::ZERO)&&(!(expected_common&&l.partial.is_ok())||l.high_before.to_limbs()[limb]==if slot==2||slot==5||slot==6{l.regular_parts[slot]}else{E::ZERO}.to_limbs()[limb]),
 l.regular_calls==0||(l.reg_ptr==l.ptr&&l.reg_len==64&&l.reg_f==f.as_ptr()&&l.reg_f_len==2048&&l.reg_alpha==alpha.as_ptr()&&l.reg_alpha_len==2&&l.reg_who==&w as *const Weights&&l.reg_folds==2&&l.reg_n==64&&l.reg_ap0.to_limbs()[limb]==l.ap0[power_index].to_limbs()[limb]&&l.reg_ap1.to_limbs()[limb]==l.ap1[power_index].to_limbs()[limb])&&(!expected_common||(l.common_who==&w as *const Weights&&l.partial_ptr==f.as_ptr()&&l.partial_len==2048&&qw(l.partial_common)[limb&3]==qw(l.common_reply[qindex])[limb&3]))&&(!(expected_common&&l.partial.is_ok())||(l.high_who==&w as *const Weights&&l.high_partial.to_limbs()[limb]==l.partial_reply[power_index].to_limbs()[limb]&&l.high_ap0.to_limbs()[limb]==l.ap0[power_index].to_limbs()[limb]&&l.high_ap1.to_limbs()[limb]==l.ap1[power_index].to_limbs()[limb]))
 ];
 if g==6{assert!(properties.into_iter().all(|p|p));}else{assert!(properties[g]);}
}
macro_rules! guardgate{($name:ident,$g:expr)=>{#[kani::proof]#[kani::unwind(2200)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
#[kani::stub(<[u8]>::copy_from_slice,KMX_copy_from_slice)]
fn $name(){guard_check($g)}};}
