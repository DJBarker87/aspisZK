#[cfg(kani)]
pub mod NIX_sampler {
 use super::*;use aspis_core::{field::{QM31,CM31},circle::{SecureCirclePoint,CirclePointError}};
 fn same(a:QM31,b:QM31)->bool {a.c0.a.0==b.c0.a.0 && a.c0.b.0==b.c0.b.0 && a.c1.a.0==b.c1.a.0 && a.c1.b.0==b.c1.b.0}
 fn point()->SecureCirclePoint {SecureCirclePoint{x:kani::any(),y:kani::any()}}
 fn canon(q:QM31)->bool{q.c0.a.0<aspis_core::field::P&&q.c0.b.0<aspis_core::field::P&&q.c1.a.0<aspis_core::field::P&&q.c1.b.0<aspis_core::field::P}
 struct Log{block:*const u8,parameter:QM31,index:usize,count:u8,at:[u8;3],frame:u64}
 fn sample(block:&[u8;32],q:QM31,log:&mut Log)->QM31{log.block=block.as_ptr();log.at[0]=log.count;log.count+=1;q}
 fn map(q:QM31,reply:Result<SecureCirclePoint,CirclePointError>,log:&mut Log)->Result<SecureCirclePoint,CirclePointError>{log.parameter=q;log.at[1]=log.count;log.count+=1;reply}
 fn fallback(index:usize,p:SecureCirclePoint,log:&mut Log)->SecureCirclePoint{log.index=index;log.at[2]=log.count;log.count+=1;p}
 fn step(block:&[u8;32],row:CircleRow,q:QM31,mapped:Result<SecureCirclePoint,CirclePointError>,fb:SecureCirclePoint,log:&mut Log)->SecureCirclePoint{
  map(sample(block,q,log),mapped,log).unwrap_or_else(|_| {
   fallback(match row {CircleRow::First=>0,CircleRow::Second=>1},fb,log)
  })
 }
 #[kani::proof] #[kani::unwind(34)]
 #[kani::stub(core::ptr::copy_nonoverlapping,crate::NIX_kmx::KMX_copy_nonoverlapping)]
 pub fn caller(){
  let block:[u8;32]=kani::any();let row=if kani::any(){CircleRow::First}else{CircleRow::Second};
  let q:QM31=kani::any();kani::assume(canon(q));let p=point();let fb=point();let tag:u8=kani::any();kani::assume(tag<3);
  let mapped=match tag{0=>Ok(p),1=>Err(CirclePointError::SingularParameter),_=>Err(CirclePointError::ParameterInCm31Subfield)};
  let count:u8=kani::any();kani::assume(count<253);let old_index:usize=kani::any();let at:[u8;3]=kani::any();let frame:u64=kani::any();
  let mut log=Log{block:core::ptr::null(),parameter:kani::any(),index:old_index,count,at,frame};
  let mut result=step(&block,row,q,mapped,fb,&mut log);
  #[cfg(all(NIX_negative,NIX_case_args))] {log.parameter.c0.a.0^=1;}
  #[cfg(all(NIX_negative,NIX_case_order))] {log.at[0]=log.at[0].wrapping_add(1);}
  #[cfg(all(NIX_negative,NIX_case_frame))] {log.frame^=1;}
  #[cfg(all(NIX_negative,NIX_case_reply))] {result.x.c0.a.0^=1;}
  #[cfg(NIX_case_args)] {assert_eq!(log.block,block.as_ptr());assert!(same(log.parameter,q));if tag!=0 {assert_eq!(log.index,match row{CircleRow::First=>0,CircleRow::Second=>1});}else{assert_eq!(log.index,old_index);}}
  #[cfg(NIX_case_order)] {assert_eq!(log.at[0],count);assert_eq!(log.at[1],count+1);if tag!=0{assert_eq!(log.at[2],count+2);assert_eq!(log.count,count+3);}else{assert_eq!(log.at[2],at[2]);assert_eq!(log.count,count+2);}}
  #[cfg(NIX_case_frame)] {assert_eq!(log.frame,frame);}
  #[cfg(NIX_case_reply)] {let expected=if tag==0{p}else{fb};assert!(same(result.x,expected.x)&&same(result.y,expected.y));}
 }
}
