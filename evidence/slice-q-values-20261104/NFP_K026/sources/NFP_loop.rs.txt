type R=Result<(Accumulator,BatchedWeights),Error>;
enum Step{Done(R),Continue(R)}
struct Call{called:bool,error:bool,seen:[usize;9],rhs:E,power:K,next:usize,terms:usize,raw:[[u64;4];64]}
fn child(proof:&[u8],p:&Prepared,mut a:Accumulator,g:&[E;29],_:Observer,geometry:&QueryGeometry,mut b:BatchedWeights,c:&mut Call)->R{
 assert!(!c.called);c.called=true;c.seen=[proof.as_ptr() as usize,proof.len(),p as *const _ as usize,g.as_ptr() as usize,geometry as *const _ as usize,a.next,a.weights.as_ptr() as usize,b.raw.as_ptr() as usize,b.terms];

}
fn NFP_step(proof:&[u8],p:&Prepared,g:&[E;29],obs:Observer,geometry:&QueryGeometry,end:usize,state:R,c:&mut Call)->Step{
 match state{
  Err(error)=>Step::Done(Err(error)),
  Ok((a,pending))=>{
   if a.next>=end{return Step::Done(Ok((a,pending)));}
   let state=child(proof,p,a,g,obs,geometry,pending,c);
   Step::Continue(state)
  }
 }
}
