fn inverse<const N:usize>(start:usize,mode:u8){
 let raw:[[u32;5];N]=kani::any();let data=raw.map(|a|a.map(M31));
 let geometry=QueryGeometry{start,inverses:Vec::from(data),line_z:Vec::new()};
 let i:usize=kani::any();let lane:usize=kani::any();kani::assume(start<=i && i<start+N && lane<5);
 let inverse=*geometry.for_query(i);let expected=raw[i-start][lane];

}
fn line<const N:usize>(start:usize,mode:u8){
 let raw:[u32;N]=kani::any();let data=raw.map(M31);
 let geometry=QueryGeometry{start,inverses:Vec::new(),line_z:Vec::from(data)};
 let i:usize=kani::any();kani::assume(start<=i && i<start+N);
 let z=geometry.line_z[i-geometry.start];let expected=raw[i-start];
 assert_eq!(z.0,if mode==1{expected^1}else{expected});assert_eq!(geometry.line_z.len(),N);
}
