#[kani::proof] #[kani::unwind(34)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
#[kani::stub(<[u8]>::copy_from_slice,KMX_copy_from_slice)]
fn NET_wide832(){
 let mut raw:[u8;832]=kani::any();let i:usize=kani::any();kani::assume(i<4);
 let j:usize=kani::any();kani::assume(j<832);let before=raw[j];let start=672+i*32;
 let selected=&raw[start..start+32];let ptr=selected.as_ptr();
 let actual=WideExact::from_le_bytes(selected).map(|x|x.to_limbs());let expected=NET_reference::wide(&raw[start..start+32]);
 #[cfg(NET_bad_reply)] assert_ne!(actual,expected);
 #[cfg(not(NET_bad_reply))] assert_eq!(actual,expected);
 assert_eq!(selected.len(),32);assert_eq!(ptr,unsafe{raw.as_ptr().add(start)});
 #[cfg(NET_bad_frame)] {raw[j]^=1;}
 assert_eq!(raw[j],before);
}
