 let j:usize=kani::any();kani::assume(j<608);let before=raw[j];
 let native=&raw[start..start+16];let native_ptr=native.as_ptr();
 let actual=QM31::from_le_bytes(native).map(q_words);let expected=NET_reference::quadratic(&raw[start..start+16]);
