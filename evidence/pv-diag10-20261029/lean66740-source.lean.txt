import PV_DIAG10_FILE
import PV_DIAG10_JSON_STDOUT
import PV_DIAG10_ERROR_STDERR
set_option autoImplicit false
namespace PV_DIAG10
open PV_RJ4IDB
/-! The source-model observer is the EXISTING decoder on translated events.
Its local links are proved here. Readout/EventOrder for deployed execution are
not constructed by defining a model observer. All abnormal prefixes remain. -/
def sourceFile {I : Type} (_ : I) (es : List NativeEvent) := files (es.map translateEvent)
def sourceJSON {I : Type} (_ : I) (es : List NativeEvent) :=
 (json (es.map translateEvent),stdout (es.map translateEvent))
def sourceError {I : Type} (_ : I) (es : List NativeEvent) :=
 (diagnostics2 (es.map translateEvent),diagnostics3 (es.map translateEvent))
theorem file_source_agrees {I : Type} (i : I) (es : List NativeEvent) :
 sourceFile i es=nativeFiles es := by
 simp only [sourceFile,nativeFiles_translation]
#print axioms file_source_agrees
theorem file_init_link {I : Type} :
 PV_DIAG10_Link_Init (@sourceFile I) (nativeFiles []) := by
 intro i
 exact file_source_agrees i []
#print axioms file_init_link
theorem file_step_link {I : Type} :
 PV_DIAG10_Link_Step (@sourceFile I) List.append (fun e => nativeFiles [e]) := by
 intro i es e
 rw [file_source_agrees,file_source_agrees]
 simp only [nativeFiles,List.filterMap_append] <;> rfl
#print axioms file_step_link
theorem json_source_agrees {I : Type} (i : I) (es : List NativeEvent) :
 sourceJSON i es=(fun es => (nativeJSON es,nativeStdout es)) es := by
 simp only [sourceJSON,nativeJSON_translation,nativeStdout_translation]
#print axioms json_source_agrees
theorem json_init_link {I : Type} :
 PV_DIAG10_Link_Init (@sourceJSON I) ((fun es => (nativeJSON es,nativeStdout es)) []) := by
 intro i
 exact json_source_agrees i []
#print axioms json_init_link
theorem json_step_link {I : Type} :
 PV_DIAG10_Link_Step (@sourceJSON I) (fun a b => (a.1++b.1,a.2++b.2)) (fun e => (fun es => (nativeJSON es,nativeStdout es)) [e]) := by
 intro i es e
 rw [json_source_agrees,json_source_agrees]
 simp only [nativeJSON,nativeStdout,List.filterMap_append] <;> rfl
#print axioms json_step_link
theorem error_source_agrees {I : Type} (i : I) (es : List NativeEvent) :

 simp only [sourceError,nativeDiag2_translation,nativeDiag3_translation]
#print axioms error_source_agrees

#print axioms error_init_link
theorem error_step_link {I : Type} :
 PV_DIAG10_Link_Step (@sourceError I) (fun a b => (a.1++b.1,a.2++b.2)) (fun e => (fun es => (nativeDiag2 es,nativeDiag3 es)) [e]) := by

#print axioms error_step_link
theorem file_acquisition_from_source_links {I A : Type}
 (actual : I → List FileOp) (native : I → NativeTrace A) (raw : I → List NativeEvent)
 (readout : PV_DIAG10_Link_Readout actual sourceFile raw)
 (order : PV_DIAG10_Link_EventOrder raw native) :
 OpenInputs.DIAG_FILE_TRACE actual native :=
 DIAG_FILE_TRACE actual native raw sourceFile file_init_link file_step_link readout order
#print axioms file_acquisition_from_source_links
theorem json_acquisition_from_source_links {I A : Type}
 (actual : I → List Bytes × List (Bytes × Bytes)) (native : I → NativeTrace A) (raw : I → List NativeEvent)

theorem error_acquisition_from_source_links {I A : Type}
 (actual : I → List PV_DIAG2.Effect × List PV_DIAG3.Event) (native : I → NativeTrace A) (raw : I → List NativeEvent)
 (readout : PV_DIAG10_Link_Readout actual sourceError raw)
