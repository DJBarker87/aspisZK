def step 
  (lst : Slice Std.U32) (tag : Slice Std.U32) (n : Std.Usize) (ok1 : Bool) (c : Usize) : Result (Bool × Usize) := do
  let i ← Slice.index_usize tag c
  let i1 ← lift (UScalar.cast .Usize i)
  let i2 ← Slice.index_usize lst i1
  let i3 ← lift (UScalar.cast .U64 i2)
  let i4 ← lift (UScalar.cast .U64 c)
  let c1 ← c + 1#usize
  ok (ok1 && (i3 = i4),c1)

  let i3 := UScalar.cast .U64 i2

  let c1 ← operation (.add c.val (1#usize).val) (c + 1#usize)
