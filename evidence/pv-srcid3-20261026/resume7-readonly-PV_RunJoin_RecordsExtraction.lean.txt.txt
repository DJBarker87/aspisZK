/-! Extraction adapter. The shipped/reference association is an external P6
certificate, not a Lean theorem. Do not install until that gate is accepted. -/
set_option autoImplicit false
namespace R1Z.PV_RunJoin_RecordsExtraction
open Aeneas Aeneas.Std Result
open BuilderForestAdmissionPort BuilderForestAdmissionPort.pool_v1

abbrev E_shipped : ExternalLinear := PV_Src_ExternalReference.dictionary

def R_reference : Records := ⟨fun input out rec chg snapshot asset => do
  let empty ← @pair_trace.pair_empty_roots E_shipped
  let (w,p) ← AspisBuilderRecordsReference.pool_v1.pair_forest_copy_builder.records_reference.records_reference_with_roots
    (PV_Src_NativeDirections.reference input) out (rec.map PV_Src_RecordReference.note)
    (PV_Src_RecordReference.note chg) (PV_Src_RecordReference.snapshot snapshot) asset empty
  ok (PV_Src_RecordReference.witness w, PV_Src_RecordReference.publicRecord p)⟩

theorem reference_identification
    (w : pair_forest_trace.PoolV1PairForestInputNoteWitnessV1)
    (out : U32) (rec : Option payment_relation.PoolV1OutputNoteWitnessV1)
    (change : payment_relation.PoolV1OutputNoteWitnessV1)
    (snapshot : pair_tree_profile.PoolV1PairLiveSnapshotV1) (asset : U32) :
    Records.call R_reference w out rec change snapshot asset =
      @PV_Src_RecordDynamic.call E_shipped w out rec change snapshot asset := rfl

#print axioms reference_identification
end R1Z.PV_RunJoin_RecordsExtraction
