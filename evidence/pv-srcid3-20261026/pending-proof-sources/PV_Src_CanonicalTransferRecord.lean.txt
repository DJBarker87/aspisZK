import R1Z.PV_Src_RecordAdmission
import R1Z.PV_Src_CanonicalRequest

/-! Canonical transfer requests retain their original public fields, rather
than replacing them to force the output condition. All private record fields
are identified for their path-packed source encoding. -/
set_option autoImplicit false
noncomputable section
namespace R1Z.PV_Src_CanonicalTransferRecord
open Aeneas Aeneas.Std Result R1J.SpendWitnessMap
open BuilderForestAdmissionPort.pool_v1
variable {K : Type} [Field K] [Fintype K] [DecidableEq K]
  [Algebra (ZMod AspisCircleGroupOrder.P) K] [CharP K 2147483647]

theorem witness (c : BuilderCanonical.Instance (K := K))
    (hv : c.val.1.variant = .privateTransfer)
    (roots : BuilderRecordsReference.Roots)
    (pool domain destination : PV_Src_PublicRecords.Context)
    (snapshot : pair_tree_profile.PoolV1PairLiveSnapshotV1)
    (compiled : pair_forest_trace.PoolV1PairForestMergedC1CompilationV1)
    (anchor : Array U32 8#usize) :
    PV_Src_Forest_PrimitiveArguments.wordWitness
      (PV_Src_RecordReference.returned roots
        (PV_Src_CheckedEncoding.input c.val.2
          (PV_Src_CanonicalRequest.request c pool domain destination)
          (PV_Src_PathIndex.index c.val.2) snapshot compiled anchor)).w =
      BuilderCanonical.rustWitness c.val.2 := by
  simpa only [PV_Src_CanonicalRequest.request,hv,PV_Src_CheckedEncoding.transferRequest] using
    PV_Src_ReferenceTransfer.witness roots c.val.2
      (PV_Src_PublicRecords.transfer c.val.1 pool domain) snapshot compiled anchor
      (PV_Src_CanonicalSibling.sibling c)

#print axioms witness
end R1Z.PV_Src_CanonicalTransferRecord
