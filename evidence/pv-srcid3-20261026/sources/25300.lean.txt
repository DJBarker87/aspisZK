import R1Z.PV_Src_RecordReference
import R1Z.BuilderAdmissionRecords

/-! Primitive admission at the actual extracted common validator. The output
and forest guards are named separately because that extraction contains only
validate_common, not the two public entry wrappers. -/
set_option autoImplicit false
namespace R1Z.PV_Src_CheckedValidation
open Aeneas Aeneas.Std Result R0Z.WriterRefine
open BuilderForestAdmissionPort.pool_v1 PV_Src_NativeInput

def sourcePair (i : Input) :=
  BuilderAdmissionRecords.sourcePair (PV_Src_NativeDirections.reference i.witness)

theorem checked_of_common_success (i : Input)
    (pool domain : Array U8 32#usize) (seq : U64)
    (anchor nullifier : Array U32 8#usize) (asset : U32)
    (context : AspisBuilderAdmissionSource.pool_v1.payment_relation.PoolV1PaymentRelationContextV1)
    (success : AspisBuilderAdmissionSource.pool_v1.pair_trace.validate_common
      pool domain seq anchor nullifier asset (sourcePair i) context = ok (.Ok ()))
    (outputBound : (outputValue i.request).val < 2^30)
    (changeBound : i.change.value.val < 2^30)
    (superRoot : ∀ l, l < 3 → Canon8 i.witness.super_root_siblings.val[l]!) :
    PV_Src_RecordReference.Checked i := by
  have h := BuilderAdmissionCommon.common_facts
    pool domain seq anchor nullifier asset (sourcePair i) context success
  exact ⟨⟨h.value_bound,outputBound,changeBound⟩,
    h.pair_first,h.pair_second,h.membership,superRoot⟩

theorem excludes_boundary (i : Input) (h : PV_Src_RecordReference.Checked i) :
    (outputValue i.request).val ≠ 2^30 := by
  have hb := h.values.2.1
  omega

theorem transfer_guards (out change : U32)
    (guard : ¬(out = 0#u32 ∨ change = 0#u32 ∨
      out ≥ 1073741824#u32 ∨ change ≥ 1073741824#u32)) :
    out.val < 2^30 ∧ change.val < 2^30 := by
  have ho : ¬out ≥ 1073741824#u32 := fun h => guard (Or.inr (Or.inr (Or.inl h)))
  have hc : ¬change ≥ 1073741824#u32 := fun h => guard (Or.inr (Or.inr (Or.inr h)))
  simp only [UScalar.le_equiv,UScalar.ofNatCore_val_eq] at ho hc
  exact ⟨Nat.lt_of_not_ge ho,Nat.lt_of_not_ge hc⟩

/-- Successful checked addition cannot admit an output beyond a bounded input.
The equality here is the Nat interpretation of the successful checked_add,
not a wrapping-add equality. -/
theorem withdrawal_conservation (input out change : U32)
    (inputBound : input.val < 2^30)
    (conservation : out.val + change.val = input.val) : out.val < 2^30 := by omega

end R1Z.PV_Src_CheckedValidation
