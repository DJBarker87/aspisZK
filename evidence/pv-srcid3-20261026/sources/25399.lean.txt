      (BuilderCanonical.liftWitness (BuilderRecordCanonical.witness
        (PV_Src_Forest_PrimitiveArguments.wordWitness records.w))) := by
  rw [private,PV_Src_CanonicalDecode.roundtrip,public]
  exact canonical.property
