use super::*;
// Recheck inherited fixed stores with an explicit bound covering the largest copy.
#[kani::proof]
#[kani::unwind(801)]
fn NBN_after_00() {
  let mut out:[u8;688]=kani::any(); let old=out;
  let bytes:[u8;32]=kani::any();
  let j:usize=kani::any();kani::assume(j<688);
  out[48..80].copy_from_slice(&bytes);
  assert_eq!(out[j],if j>=48 && j<80 {bytes[j-48]} else {old[j]});
}
