/-- Complete chronological source projection of the owned replay. Every233
clause is obtained from its FULL named packet at SAME arguments. This remains
conditional source interpretation; independent native stops and canonical
opaque-child realization are not silently discharged by this theorem. -/
theorem erases {Ambient CoreError : Type}
 (missing : AspisR1.IDa_Deployed.Uninstalled CoreError) (indexError : CoreError)
 (reservation : AspisR1.NP6_StdModel.Reservation)
 (owners : NMS_RowSpec233.Args → List Nat) (native : NMS_RowSpec233.Native Ambient)
 (packet : NMS_RowSpec233.RowSpec_233 missing indexError reservation owners native)
 (before : World Ambient) (inputs : List Nat) (program : solana_pubkey.Pubkey)
 (accounts : Slice r1_pool_adapter.owned.OwnedAccount) (data : Slice U8)
 (slot : core.result.Result U64 PE) :
 NMS_LogMap.run Event.erase (NMS_LogMap.rust Validated.erase)
  (NMS_Owned242.G (NLM_DeployedOps.ops missing indexError reservation) native before inputs
   program accounts data slot) =
 NLM_Validate242.G (NLM_DeployedOps.ops missing indexError reservation) program accounts data slot := by
 rw [NMS_242Prefix.factor_owned,NMS_242Prefix.factor_raw]
 unfold NMS_242Prefix.owned NMS_242Prefix.raw
 repeat' first
 | (solve | rfl)
 | (solve | exact NMS_242Auth.erases missing indexError reservation owners native packet before inputs _ _ _ _ _ _)
 | (solve | apply NMS_PlainMap.record; intro; trivial)
 | (solve | apply NMS_PlainMap.residual)
 | (solve | rw [NMS_TryFold.lifted,NMS_LogMap.result_id])
 | (rw [NMS_LogMap.ite_map]; split)
 | (rw [NMS_LogMap.flow]; split)
 | (apply NMS_LogMap.bind Event.erase id (NMS_LogMap.rust Validated.erase))
 | intro
#print axioms erases
end NMS_242PrefixMap
