#[cfg(ncp_for)]
#[kani::proof]
fn NCP_for_invariant(){let a:[u8;4]=kani::any();let mut count:usize=0;
 #[kani::loop_invariant(count == kani::index && count <= 4)]
 for _x in a.iter().enumerate(){count+=1;}
 assert_eq!(count,4);
}
#[cfg(ncp_for)]
#[kani::proof]
fn NCP_for_invariant_negative(){let a:[u8;4]=kani::any();let mut count:usize=0;
 #[kani::loop_invariant(count == kani::index && count <= 4)]
 for _x in a.iter().enumerate(){count+=1;}
 assert_eq!(count,3);
}
#[cfg(ncp_while)]
#[kani::proof]
fn NCP_while_invariant(){let mut i=0usize;
 #[kani::loop_invariant(i<=4)]

 assert_eq!(i,4);
}
