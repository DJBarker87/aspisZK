#[kani::requires(input.len() == 1)]
#[kani::ensures(|r: &Result<u8,u8>| *r != Ok(input[0]))]
fn NCP_slice_inverted(input:&[u8])->Result<u8,u8>{Ok(input[0])}
#[kani::requires(input.len() == 1)]
#[kani::ensures(|_: &Result<u8,u8>| true)]
fn NCP_slice_weak(input:&[u8])->Result<u8,u8>{Ok(input[0])}
#[kani::requires(a.len() == 1 && b.len() == 1)]
#[kani::modifies(a)]
#[kani::ensures(|_: &Result<(),u8>| true)]
fn NCP_slice_frame(a:&mut [u8],b:&mut [u8])->Result<(),u8>{a[0]=1;b[0]=2;Ok(())}
