fn q()->QM31 {let a:[u32;4]=kani::any();QM31{c0:CM31{a:M31(a[0]),b:M31(a[1])},c1:CM31{a:M31(a[2]),b:M31(a[3])}}}
fn byte(v:u8)->QM31{QM31{c0:CM31{a:M31(v as u32),b:M31(0)},c1:CM31{a:M31(0),b:M31(0)}}}
fn select_ch(c:&Challenges,i:usize)->QM31{match i%15{0=>c.lambda,1=>c.chi,2=>c.theta,3..=12=>c.zc[i%15-3],13=>c.mu,_=>c.eta}}
fn challenges()->Challenges { Challenges{lambda:q(),chi:q(),theta:q(),zc:core::array::from_fn(|_|q()),mu:q(),eta:q()} }
fn source_construct<H:Digest32>(c1_root:&[u8;32],root:[u8;32],challenges:Challenges,alpha:[QM31;10],claims:Box<[[QM31;29];3]>,extra:[QM31;29],before_z1:[QM31;29],state_before_z0:[u8;32],z0:SecureCirclePoint,z1:SecureCirclePoint,transcript:SemanticTranscript<H>)->Verified<H>{
    Verified {
        roots: [*c1_root, root],
        challenges,
        alpha,
        claims,
        extra,
        before_z1,
        state_before_z0,
        z: [z0, z1],
        transcript,
    }
}
fn check_construct(bad:bool){
 let c1:[u8;32]=kani::any();let root:[u8;32]=kani::any();let ch=challenges();let alpha=core::array::from_fn(|_|q());

 let expected=match mode{0=>Some(byte(c1[j])),1=>Some(select_ch(&ch,j)),2=>Some(alpha[j%10]),3=>Some(claims[row][col]),4=>Some(extra[col]),5=>Some(before[col]),6=>Some(byte(state[j])),7=>Some(match j%4{0=>z0.x,1=>z0.y,2=>z1.x,_=>z1.y}),_=>None};

fn core_error()->aspis_core::state_only_prefix::r0::Error{
