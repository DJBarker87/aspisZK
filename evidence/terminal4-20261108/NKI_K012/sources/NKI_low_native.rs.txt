struct NKI_LowLog {seek:usize,calls:usize,kind:u8,left:QM31,right:QM31,cache:[[u32;3];3],new_reply:PreparedQm31Multiplier,reply:[QM31;5]}
fn NKI_low_enter(k:u8,a:QM31,b:QM31,l:&mut NKI_LowLog)->QM31{let i=l.calls;if i==l.seek{l.kind=k;l.left=a;l.right=b;}l.calls+=1;l.reply[i]}
fn NKI_low_new(a:QM31,l:&mut NKI_LowLog)->PreparedQm31Multiplier{NKI_low_enter(3,a,QM31::ZERO,l);l.new_reply}
fn NKI_low_mul(p:PreparedQm31Multiplier,a:QM31,l:&mut NKI_LowLog)->QM31{if l.calls==l.seek{l.cache=NKI_cache(p);}NKI_low_enter(4,QM31::ZERO,a,l)}
fn NKI_low_add(a:QM31,b:QM31,l:&mut NKI_LowLog)->QM31{NKI_low_enter(1,a,b,l)}
fn NKI_low_mix(mut output:[QM31;2],low:QM31,local:[QM31;2],l:&mut NKI_LowLog)->[QM31;2]{
 let prepared=NKI_low_new(low,l);
 output[0]=NKI_low_add(output[0],NKI_low_mul(prepared,local[0],l),l);
 output[1]=NKI_low_add(output[1],NKI_low_mul(prepared,local[1],l),l);
 output
}
