fn NKI_words(q:QM31)->[u32;4]{[q.c0.a.0,q.c0.b.0,q.c1.a.0,q.c1.b.0]}
fn NKI_q(w:[u32;4])->QM31{QM31{c0:CM31{a:M31(w[0]),b:M31(w[1])},c1:CM31{a:M31(w[2]),b:M31(w[3])}}}
fn NKI_cache(p:PreparedQm31Multiplier)->[[u32;3];3]{unsafe{core::mem::transmute(p)}}
struct NKI_Log {seek:usize,calls:usize,kind:u8,left:[u32;4],right:[u32;4],pack:[[u32;4];4],cache:[[u32;3];3],new_reply:PreparedQm31Multiplier,reply:[QM31;23],empty:[M31;8],level:usize}
impl NKI_Log {fn enter(&mut self,kind:u8,a:QM31,b:QM31)->QM31{let k=self.calls;if k==self.seek{self.kind=kind;self.left=NKI_words(a);self.right=NKI_words(b);}self.calls+=1;self.reply[k]}}
fn NKI_new(h:QM31,l:&mut NKI_Log)->PreparedQm31Multiplier{l.enter(3,h,QM31::ZERO);l.new_reply}
fn NKI_empty(level:usize,l:&mut NKI_Log)->Digest{l.level=level;l.enter(7,QM31::ZERO,QM31::ZERO);l.empty}
fn NKI_tweak(a:M31,b:M31,l:&mut NKI_Log)->M31{let r=l.enter(8,NKI_q([a.0,0,0,0]),NKI_q([b.0,0,0,0]));M31(r.c0.a.0)}
fn NKI_lift(a:M31,l:&mut NKI_Log)->QM31{l.enter(5,NKI_q([a.0,0,0,0]),QM31::ZERO)}
fn NKI_sub(a:QM31,b:QM31,l:&mut NKI_Log)->QM31{l.enter(2,a,b)}
fn NKI_pack(a:&[QM31;4],l:&mut NKI_Log)->QM31{if l.calls==l.seek{l.pack=a.map(NKI_words);}l.enter(6,QM31::ZERO,QM31::ZERO)}
fn NKI_mul(p:PreparedQm31Multiplier,a:QM31,l:&mut NKI_Log)->QM31{if l.calls==l.seek{l.cache=NKI_cache(p);}l.enter(4,QM31::ZERO,a)}
fn NKI_append(next_pair_index:u64,level:usize,frontier:Digest,opened:[QM31;16],high:QM31,l:&mut NKI_Log)->([QM31;2],[QM31;2]){
    let prepared_high = NKI_new(high,l);
    if ((next_pair_index >> level) & 1) == 0 {
        let expected = NKI_empty(level,l);
        let residual0 = [
            NKI_sub(opened[RATE],NKI_lift(expected[0],l),l),
            NKI_sub(opened[RATE + 1],NKI_lift(expected[1],l),l),
            NKI_sub(opened[RATE + 2],NKI_lift(expected[2],l),l),
            NKI_sub(opened[RATE + 3],NKI_lift(expected[3],l),l),
        ];
        let residual1 = [
            NKI_sub(opened[RATE + 4],NKI_lift(expected[4],l),l),
            NKI_sub(opened[RATE + 5],NKI_lift(expected[5],l),l),
            NKI_sub(opened[RATE + 6],NKI_lift(expected[6],l),l),
            NKI_sub(opened[RATE + 7],NKI_lift(NKI_tweak(expected[7],MERKLE_NODE_COMPRESSION_V3_TWEAK,l),l),l),
        ];
        (
            [
                NKI_mul(prepared_high,NKI_pack(&residual0,l),l),
                NKI_mul(prepared_high,NKI_pack(&residual1,l),l),
            ],
            [QM31::ZERO; DIGEST_ELEMS / 4],
        )
    } else {
        let residual0 = [
            NKI_sub(opened[0],NKI_lift(frontier[0],l),l),
            NKI_sub(opened[1],NKI_lift(frontier[1],l),l),
            NKI_sub(opened[2],NKI_lift(frontier[2],l),l),
            NKI_sub(opened[3],NKI_lift(frontier[3],l),l),
        ];
        let residual1 = [
            NKI_sub(opened[4],NKI_lift(frontier[4],l),l),
            NKI_sub(opened[5],NKI_lift(frontier[5],l),l),
            NKI_sub(opened[6],NKI_lift(frontier[6],l),l),
            NKI_sub(opened[7],NKI_lift(frontier[7],l),l),
        ];
        (
            [QM31::ZERO; DIGEST_ELEMS / 4],
            [
                NKI_mul(prepared_high,NKI_pack(&residual0,l),l),
                NKI_mul(prepared_high,NKI_pack(&residual1,l),l),
            ],
        )
    }
}
