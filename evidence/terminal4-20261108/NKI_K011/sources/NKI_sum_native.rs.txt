struct NKI_SumLog {hi:usize,lo:usize,high:[u32;4],low:[u32;4],ranges:[usize;4],calls:usize,reply:QM31}
fn NKI_sum(s:&Selectors,r:&[core::ops::Range<usize>],l:&mut NKI_SumLog)->QM31{l.high=NKI_words(s.high[l.hi]);l.low=NKI_words(s.low[l.lo]);l.ranges=[r[0].start,r[0].end,r[1].start,r[1].end];l.calls+=1;l.reply}
