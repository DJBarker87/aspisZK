/-- Ordered trace on the successful append-content writer's exact output.
This is a local model trace; no global rejected-prefix or ELF transport is
claimed. The actual constructed body supplies the checksum request. -/
theorem source {H : Type} (D : AspisR1Stage2.aspis_core.digest.Digest32 H) (hash : H)
    (total : ∀ parts, ∃ out, D.digest hash parts=ok out)
    (p : Prepared) (a : Accumulator) (proof pub : Slice U8) (saved : Slice U8)
    (size : a.weights.val.length=64) (variant : p.variant=.S2) (next : a.next.val=10)
    (execute : erase D hash (WAdm_SourceWriter.code p a proof pub)=ok saved) (before : Log) :
    ∃ (hp hu hc : Array U8 32#usize) (body : Slice U8),
      WAdm_SourceWriter.toSlice (WAdm_SourceBody.body p a hp hu)=ok body ∧
      run D hash (WAdm_SourceWriter.code p a proof pub) before=
        ok (saved,before++[(singletonParts proof,hp),(singletonParts pub,hu),(checksumParts body,hc)]) := by
  obtain ⟨hp,hu,hc,body,hpCall,huCall,bodyCall,hcCall,_len,_image⟩ :=
    JHc_WriterImage.source D hash total p a proof pub saved size variant next execute
  have finalCall := execute
  simp only [WAdm_SourceWriter.code,erase_bind,erase_read,erase_lift,hpCall,huCall,
    bodyCall,hcCall,bind_tc_ok] at finalCall
  refine ⟨hp,hu,hc,body,bodyCall,?_⟩
  simp only [WAdm_SourceWriter.code,run_bind,run_read,run_lift,run_pure,hpCall,huCall,
    bodyCall,hcCall,finalCall,bind_tc_ok,Aeneas.Std.uncurry,List.append_assoc]
#print axioms source
end AspisR1.JHc_WriterTrace
end
