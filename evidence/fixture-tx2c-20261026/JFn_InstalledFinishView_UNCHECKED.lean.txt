    (p : verifier.Prepared) (a : verifier.Accumulator) (b : verifier.BatchedWeights)
    (inactive : aspis_core.field.wide_exact.WideExact) :

      JFx_NativeBytes.proof p none inactive 7527#usize 7751#usize 7783#usize (a,b)=
    (do let (old,back) ← lift (alloc.vec.Vec.deref_mut a.weights)
        let flushed ← base.op9 b old
        Port4D_FinishInstall.finish finish (JFn_FinishView.view inactive) p {a with weights:=back flushed} none) := by
  exact JFn_FinishView.source (Tx2I_Install.stage base finish) p a b inactive
