/-- A list of per-site allowances, consumed in actual source order. -/
def Fits (impl : Implementation) (H : Bytes → State) {A : Type}
    (tree : Code A) (sites : List Nat) : Prop :=
  (eval H (interpret impl tree)).1.length ≤ sites.sum

theorem done {A : Type} (impl : Implementation) (H : Bytes → State)
    (a : A) (sites : List Nat) : Fits impl H (.done a) sites := Nat.zero_le _

theorem step {A B : Type} (impl : Implementation) (H : Bytes → State)
    (r : Result B) (k : B → Code A) (sites : List Nat)
    (tail : ∀ b, Fits impl H (k b) sites) : Fits impl H (.step r k) sites := by
  cases r with
  | fail e => exact Nat.zero_le _
  | div => exact Nat.zero_le _
  | ok b => exact tail b

theorem call {A B : Type} (impl : Implementation) (H : Bytes → State)
    (op : Call B) (k : B → Code A) (n : Nat) (sites : List Nat)
    (localCost : (eval H (impl op)).1.length ≤ n)
    (tail : ∀ b, Fits impl H (k b) sites) : Fits impl H (.call op k) (n::sites) := by
  unfold Fits
  simp only [interpret, eval_bind, List.length_append, List.sum_cons]
  have remaining : (eval H (continueWith (eval H (impl op)).2
      (fun a => interpret impl (k a)))).1.length ≤ sites.sum := by
    cases (eval H (impl op)).2 with
    | fail e => exact Nat.zero_le _
    | div => exact Nat.zero_le _
    | ok b => exact tail b
  omega

inductive Event : Type 1 where
  | returned {A : Type} : Call A → A → Event
abbrev Log := List Event
inductive Code (A : Type) : Type 1 where
  | done : A → Code A
  | step {B : Type} : Result B → (B → Code A) → Code A
  | call {B : Type} : Call B → (B → Code A) → Code A

def bind {A B : Type} : Code A → (A → Code B) → Code B
  | .done a, k => k a
  | .step r c, k => .step r (fun b => bind (c b) k)
  | .call op c, k => .call op (fun b => bind (c b) k)
instance : Monad Code where
  pure := Code.done
  bind := bind

def liftResult {A : Type} (r : Result A) : Code A := .step r .done
instance : MonadLift Result Code where
  monadLift := liftResult

def perform {A : Type} (op : Call A) : Code A := .call op .done

def erase (ops : VTop_Stages.Ops) {A : Type} : Code A → Result A
  | .done a => ok a
  | .step r k => do let b ← r; erase ops (k b)
  | .call op k => do let b ← invoke ops op; erase ops (k b)

def thenResult {A : Type u} {B : Type v} (r : Result A) (k : A → Result B) : Result B :=
  match r with
  | .ok a => k a
  | .fail e => .fail e
  | .div => .div

def forget {A : Type} (r : Result (A × Log)) : Result A :=
  thenResult r (fun x => ok x.1)

def run (ops : VTop_Stages.Ops) {A : Type} : Code A → Log → Result (A × Log)
  | .done a, log => ok (a,log)
  | .step r k, log => thenResult r (fun b => run ops (k b) log)
  | .call op k, log => thenResult (invoke ops op) (fun b =>
      run ops (k b) (log ++ [.returned op b]))

theorem erase_run (ops : VTop_Stages.Ops) {A : Type} (p : Code A) (log : Log) :
    forget (run ops p log) = erase ops p := by
  induction p generalizing log with
  | done a => rfl
  | step r k ih =>
    cases r with
    | fail e => rfl
    | div => rfl
    | ok b => exact ih b log
  | call op k ih =>
    cases h : invoke ops op with
    | fail e => simp only [run,thenResult,forget,erase,h,bind_tc_fail]
    | div => simp only [run,thenResult,forget,erase,h,bind_tc_div]
    | ok b => simpa only [run,thenResult,erase,h,bind_tc_ok] using ih b (log ++ [.returned op b])

theorem run_prefix (ops : VTop_Stages.Ops) {A : Type} (p : Code A)
    (before after : Log) (a : A) (success : run ops p before = ok (a,after)) :
    ∃ suffix, after = before ++ suffix := by
  induction p generalizing before after with
  | done b =>
    have eq := Result.ok.inj success
    exact ⟨[], by simpa using congrArg Prod.snd eq.symm⟩
  | step r k ih =>
    cases r with
    | fail e => cases success
    | div => cases success
    | ok b => exact ih b before after success
  | call op k ih =>
    cases h : invoke ops op with
    | fail e => simp only [run,thenResult,h]  at success; cases success
    | div => simp only [run,thenResult,h]  at success; cases success
    | ok b =>
      simp only [run,thenResult,h] at success
      obtain ⟨suffix,hs⟩ := ih b _ after success
      exact ⟨.returned op b :: suffix, by simpa only [List.append_assoc,List.singleton_append] using hs⟩

theorem erase_perform (ops : VTop_Stages.Ops) {A : Type} (op : Call A) :
    erase ops (perform op) = invoke ops op := by
  cases h : invoke ops op <;> simp only [perform,erase,h,bind_tc_ok,bind_tc_fail,bind_tc_div]

theorem erase_lift (ops : VTop_Stages.Ops) {A : Type} (r : Result A) :
    erase ops (liftResult r) = r := by cases r <;> rfl

theorem erase_bind (ops : VTop_Stages.Ops) {A B : Type} (p : Code A) (k : A → Code B) :
    erase ops (p >>= k) = (do let a ← erase ops p; erase ops (k a)) := by
  induction p with
  | done a => rfl
  | step r c ih =>
    change (do let b ← r; erase ops (bind (c b) k)) = _
    have h : ∀ b, erase ops (bind (c b) k) =
      (do let a ← erase ops (c b); erase ops (k a)) := ih
    simp only [h,erase]
    cases r <;> rfl
  | call op c ih =>
    change (do let b ← invoke ops op; erase ops (bind (c b) k)) = _
    have h : ∀ b, erase ops (bind (c b) k) =
      (do let a ← erase ops (c b); erase ops (k a)) := ih
    simp only [h,erase]
    cases invoke ops op <;> rfl

/-! Oracle interpretation of the extracted typed call tree. All call arguments
and returned mutable state are retained. No identification with dV is assumed
or claimed. Ordinary inner Rust Err is a value of the dependent result type. -/
set_option autoImplicit false
noncomputable section
namespace AspisR1.VTop_OracleCode
open Aeneas Aeneas.Std Result VTop_Trace
open AspisV8R19.DuplexFrames AspisV8R19.SourceDuplexStep
open AspisV8R19.MemoizedProgramLaw AspisV8R19.OracleProgramOps

abbrev Implementation := {A : Type} → Call A → Program Bytes State (Result A)

def continueWith {A B : Type} (r : Result A)
    (k : A → Program Bytes State (Result B)) : Program Bytes State (Result B) :=
  match r with
  | .ok a => k a
  | .fail e => .done (.fail e)
  | .div => .done .div

def interpret (impl : Implementation) {A : Type} : Code A → Program Bytes State (Result A)
  | .done a => .done (.ok a)
  | .step r k => continueWith r (fun a => interpret impl (k a))
  | .call op k => bind (impl op) (fun r => continueWith r (fun a => interpret impl (k a)))

/-- A local call law: complete ordered reads and the complete dependent return
are equal whenever that source call returns. This does not bind an arbitrary
Verifier to the tree, and does not identify a model implementation with dV. -/
def CallJoin (source model : Implementation) : Prop :=
  ∀ {A : Type} (call : Call A) (H : Bytes → State) (value : A),
    (eval H (source call)).2 = .ok value →
      eval H (source call) = eval H (model call)

/-- Symbolic substitution through the actual typed syntax, retaining source
branch selection and concatenation order. No large stage reduction is needed. -/
theorem accepted_interpret_eq (source model : Implementation) (joins : CallJoin source model)
    {A : Type} (tree : Code A) (H : Bytes → State) (value : A)
    (accepted : (eval H (interpret source tree)).2 = .ok value) :
    eval H (interpret source tree) = eval H (interpret model tree) := by
  induction tree with
  | done a => rfl
  | step r k ih =>
    cases r with
    | fail e => cases accepted
    | div => cases accepted
    | ok a => exact ih a accepted
  | call op k ih =>
    simp only [interpret, eval_bind] at accepted ⊢
    cases returned : (eval H (source op)).2 with
    | fail e => simp only [returned, continueWith, eval] at accepted; cases accepted
    | div => simp only [returned, continueWith, eval] at accepted; cases accepted
    | ok a =>
      have localJoin := joins op H a returned
      rw [← localJoin]
      simp only [returned, continueWith] at accepted ⊢
      rw [ih a accepted]

#print axioms accepted_interpret_eq
end AspisR1.VTop_OracleCode

/-- ALL inputs/outcomes. H occurs only in the pointwise native environment on
RHS, never as a parameter of the Program that is evaluated on LHS. -/
theorem reader_value (reserved : Bool) (unwrap : Unwrap) {Hash : Type}
    (D : aspis_core.digest.Digest32 Hash) (ck proof pub : Slice U8)
    (v : transport.Variant) (hash : Hash) (H : Bytes → State) :
    (eval H (reader reserved unwrap D ck proof pub v hash)).2 =
      Port6CK_Reader.reader reserved unwrap (native D H) ck proof pub v hash := by
  rw [reader,Port9OR_Code.value,← native_stage]
  simp only [Port8EFF_Bodies.readerCode,CheckpointBytes5Effect.erase_bind,CheckpointBytes5Effect.erase_pure]
  rw [Port8EFF_NativeReader.source]
  unfold Port6CK_Reader.reader
  apply (Aeneas.Std.bind_eq_iff _ _ _).mpr
  intro out _
  cases out <;> rfl

  simp only [Port8EFF_Bodies.writerCode,CheckpointBytes5Effect.erase_bind,CheckpointBytes5Effect.erase_pure]

/-! Natural two-stage source lift on the actual extracted carriers. Decode
uses adapter op47, validation op21 and the extracted semantic-public getter.
The physical source reads stored/public bytes; its preparation body owns its
transcript effects. A separate logical proof field is not a Rust argument.
This module deliberately does not claim the frozen2 model-order identity. -/
set_option autoImplicit false
noncomputable section
namespace AspisR1.Port1_Driver
open Aeneas Aeneas.Std Result VTop_Stages VTop_Trace VTop_OracleCode
open AspisV8R19.DuplexFrames AspisV8R19.SourceDuplexStep
open AspisV8R19.MemoizedProgramLaw AspisV8R19.OracleProgramOps
open OraclePortStoredExperimentC4 OraclePortRuntimeBindingC7
abbrev Semantic := aspis_statement.pool_v1.pair_forest_semantic_terminal.r0.Public

def asSlice (bytes : Bytes) : Option (Slice U8) :=
  if h : bytes.length ≤ Usize.max then
    some ⟨bytes.map (fun b => (UScalar.ofNatCore b.val b.isLt : U8)), by simpa using h⟩
  else none

/-- Complete success/error selection at the extracted decoder/validator cuts.
Their concrete implementations remain their existing source ports. -/
def decodeSemantic (adapter : AdapterRw.Ops) (pub : Slice U8) : Result (Option Semantic) := do
  let decoded ← adapter.op47 pub
  match decoded with
  | .Err _ => .ok none
  | .Ok statement =>
    let checked ← adapter.op21 statement
    match checked with
    | .Err _ => .ok none
    | .Ok () =>
      let semantic ← AdapterRw.aspis_statement.pool_v1.pair_forest_terminal.PoolV1PairForestTerminalStatementV1.semantic_public statement
      .ok (some semantic)

def verdict (result : Result (core.result.Result Unit verifier.Error)) : Bool :=
  match result with | .ok (.Ok ()) => true | _ => false

/-- TX2 consumes precisely the checkpoint returned by this TX1, with the same
public/proof slices and None observer in both calls. -/
def stages (impl : Implementation) {Hash : Type}
    (D : aspis_core.digest.Digest32 Hash) (hash : Hash) (semantic : Semantic)
    (pub proof : Slice U8) : Program Bytes State Bool :=
  bind (interpret impl (VTop_Flow.tx1 VTop_Flow.traceEffects D semantic pub proof 10#usize hash none))
    fun first => match first with
    | .ok (.Ok saved) =>
      bind (interpret impl (VTop_Flow.tx2_authenticated VTop_Flow.traceEffects D pub proof saved.deref hash none))
        fun second => .done (verdict second)
    | _ => .done false

/-- Explicit headline-typed lift, with no invented logical-proof reads.
This remains a source interpretation under the supplied body/decoder ports,
not a deployed account/runtime binding. -/
def rust (adapter : AdapterRw.Ops) (impl : Implementation) {Hash : Type}
    (D : aspis_core.digest.Digest32 Hash) (hash : Hash) : Verifier := fun i =>
  match asSlice i.statement.pubBytes, asSlice i.stored with
  | some pub, some proof =>
    match decodeSemantic adapter pub with
    | .ok (some semantic) => stages impl D hash semantic pub proof
    | _ => .done false
  | _, _ => .done false

/-- Source/model stage substitution now composes both entries with the same
returned checkpoint, rather than independent stage runs. -/
theorem stages_substitute (source model : Implementation)
    (joins : VTop_OracleStages.ConeJoins source model) {Hash : Type}
    (D : aspis_core.digest.Digest32 Hash) (hash : Hash) (semantic : Semantic)
    (pub proof : Slice U8) (H : Bytes → State)
    (accepted : (eval H (stages source D hash semantic pub proof)).2 = true) :
    eval H (stages source D hash semantic pub proof) =
      eval H (stages model D hash semantic pub proof) := by
  unfold stages at accepted ⊢
  simp only [eval_bind] at accepted ⊢
  cases first : (eval H (interpret source
    (VTop_Flow.tx1 VTop_Flow.traceEffects D semantic pub proof 10#usize hash none))).2 with
  | fail e => simp only [first, eval, Bool.false_eq_true] at accepted
  | div => simp only [first, eval, Bool.false_eq_true] at accepted
  | ok result =>
    cases result with
    | Err e => simp only [first, eval, Bool.false_eq_true] at accepted
    | Ok saved =>
      have one := VTop_OracleStages.accepted_tx1 source model joins D hash semantic pub proof 10#usize saved H first
      rw [← one]
      simp only [first] at accepted ⊢
      simp only [eval_bind, eval, List.append_nil] at accepted ⊢
      cases second : (eval H (interpret source
        (VTop_Flow.tx2_authenticated VTop_Flow.traceEffects D pub proof saved.deref hash none))).2 with
      | fail e => simp only [second, verdict, Bool.false_eq_true] at accepted
      | div => simp only [second, verdict, Bool.false_eq_true] at accepted
      | ok result =>
        cases result with
        | Err e => simp only [second, verdict, Bool.false_eq_true] at accepted
        | Ok unit =>
          cases unit
          have two := VTop_OracleStages.accepted_tx2_authenticated source model joins D hash pub proof saved.deref H second
          rw [← two, second]

#print axioms decodeSemantic
#print axioms stages_substitute
end AspisR1.Port1_Driver
