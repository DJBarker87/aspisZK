  M (transcript.control21.Transcript H P)
  := do
  let s ← lift (Array.to_slice self.state)

def block
  {H : Type} {P : Type} (effect : H → Parts → M Bytes32) (PolicyInst : transcript.control21.Policy P)
  (self : transcript.control21.Transcript H P) (row : Std.U8)
  (payload : Slice Std.U8) :
  M ((Array Std.U8 32#usize) × (transcript.control21.Transcript H P))
  := do
  let self1 ←
    absorb effect
      PolicyInst self row payload
  let s ← lift (Array.to_slice self1.state)
  let s1 ← lift (Array.to_slice (Array.make 1#usize [ 1#u8 ]))
  let s2 ← lift (Array.to_slice (Array.make 2#usize [ s, s1 ]))
  let block ← effect self1.hash s2
  let s3 ← lift (Array.to_slice self1.state)
  let s4 ← lift (Array.to_slice (Array.make 1#usize [ 2#u8 ]))
  let s5 ← lift (Array.to_slice (Array.make 2#usize [ s3, s4 ]))
  let a ← effect self1.hash s5
  pure (block, { self1 with state := a })


def scalar_k
  {H : Type} {P : Type} (effect : H → Parts → M Bytes32) (PolicyInst : transcript.control21.Policy P)
  (self : transcript.control21.Transcript H P) (row : Std.U8)
  (payload : Slice Std.U8) :
  M (aspis_core.field.QM31 × (transcript.control21.Transcript H P))
  := do
  let (a, self1) ←
    block effect
      PolicyInst self row payload
  let out ← PolicyInst.qm31 a
  let i ← lift (UScalar.cast .Usize row)
  let we ← aspis_core.field.wide_exact.WideExact.from_qm31 out
  let v ← alloc.vec.Vec.push self1.trace (4#u16, i, we)
  pure (out, { self1 with trace := v })


def scalar
  {H : Type} {P : Type} (effect : H → Parts → M Bytes32) (PolicyInst : transcript.control21.Policy P)
  (self : transcript.control21.Transcript H P) (row : Std.U8)
  (payload : Slice Std.U8) (nonzero : Bool) :
  M (aspis_core.field.wide_exact.WideExact ×
    (transcript.control21.Transcript H P))
  := do
  let (block, self1) ←
    block effect
      PolicyInst self row payload
  let out ←
    if nonzero
    then PolicyInst.gamma block
    else PolicyInst.ordinary block
  let i ← lift (UScalar.cast .Usize row)
  let v ← alloc.vec.Vec.push self1.trace (4#u16, i, out)
  pure (out, { self1 with trace := v })
