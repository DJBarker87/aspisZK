// Replacement for retired K028: record only the arbitrary selected byte at
// the finish call, leaving all other Prefix cells as a Lean frame partition.
struct WindowLog<'a>{prefix:Option<Prefix<'a>>,row:usize,position:usize,initial:([u8;32],Hash,u8),after:([u8;32],Hash,u8),error:Error,actual_byte:u8,actual_ptr:usize,actual_len:usize,entered:bool}
impl<'a> WindowLog<'a>{
 fn parse(&mut self,_:&'a[u8])->Result<Prefix<'a>,Error>{Ok(self.prefix.take().unwrap())}
 fn public_transcript(&mut self,_:Public<'a>,_:Hash,_:&'a[u8;32])->Result<SemanticTranscript<Hash>,Error>{Ok(crate::r1_wire::NLJ_factory::make(self.initial.1,self.initial.0,self.initial.2))}
 fn begin(&mut self,_:&Prefix<'a>,t:&mut SemanticTranscript<Hash>)->Result<Challenges,Error>{*t=crate::r1_wire::NLJ_factory::make(self.after.1,self.after.0,self.after.2);Ok(challenge())}
 fn finish(&mut self,_:Public<'a>,p:Prefix<'a>,_:SemanticTranscript<Hash>,_:Challenges)->Result<Verified<Hash>,Error>{let record=crate::r1_wire::NLJ_factory::record(&p,self.row);self.actual_len=record.len();self.actual_ptr=record.as_ptr() as usize;self.actual_byte=if record.is_empty(){0}else{record[self.position]};self.entered=true;Err(self.error)}
}
fn source_window<'a>(public:Public<'a>,bytes:&'a[u8],hash:Hash,log:&mut WindowLog<'a>)->Result<Verified<Hash>,Error>{
 let prefix=log.parse(bytes)?;super::mark(None,0);
 let mut transcript=log.public_transcript(public,hash,prefix.c1_root)?;
 let challenges=log.begin(&prefix,&mut transcript)?;super::mark(None,1);
 log.finish(public,prefix,transcript,challenges)
}
