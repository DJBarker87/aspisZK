def seed {C : Type} (c : DrvC_Installation.Configuration FIo_Install.W C) (H : Bytes → State) :=
  {c with adapter := {c.adapter with op43:=S2W_DigestExtraction.digest (F3U_Oracle.hashv H)}}
def configuration {C : Type} (c : DrvC_Installation.Configuration FIo_Install.W C) (err : C) (H : Bytes → State) :=
  FIp_Install.configuration err (seed c H) (.Ok ())
def first {C : Type} (c : DrvC_Installation.Configuration FIo_Install.W C) (err : C) (H : Bytes → State) :=
  AdapterRw.r1_pool_adapter.owned.stage1_owned (DrvC_Installation.sdk (configuration c err H))
def second {C : Type} (c : DrvC_Installation.Configuration FIo_Install.W C) (err : C) (H : Bytes → State) :=
  AdapterRw.r1_pool_adapter.owned.stage2_owned (DrvC_Installation.sdk (configuration c err H))

/-- Source-reference instantiation at the Rc10 installation; no wrapper acceptance or ELF claim. -/
theorem extraction {C : Type} (c : DrvC_Installation.Configuration FIo_Install.W C) (err : C) (H : Bytes → State) :
    S2W_DigestExtraction.AeneasExtraction (DrvC_Installation.sdk (configuration c err H))
      (first c err H) (second c err H) (F3U_Oracle.hashv H) :=
  ⟨by intros;rfl,by intros;rfl,by intros;rfl⟩

theorem oracle (H : Bytes → State) : S2W_ExtractionBoundary.P1 H H ∧
    S2W_ExtractionBoundary.RuntimeSyscall (F3U_Oracle.hashv H) H :=
  ⟨⟨fun _ => rfl⟩,F3U_Oracle.syscall H⟩

#print axioms extraction
#print axioms oracle
