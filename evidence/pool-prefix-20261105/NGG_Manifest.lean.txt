import Lean
import NFR_MemoryLimit
open Lean
unsafe def main (_args : List String) : IO UInt32 := do
 NFO_MemoryLimit.set
 initSearchPath (← findSysroot)
 enableInitializersExecution
 let env ← importModules #[{module := `NGG_COMBINED_IMPORT, importAll := true}] {}
   (loadExts := true) (level := .private)
 let h ← IO.FS.Handle.mk "NGG_MODULE_PATHS.jsonl" .write
 for idx in [0:env.header.moduleNames.size] do
  let name := env.header.moduleNames[idx]!
  let path ← findOLean name
  let data := env.header.moduleData[idx]!
  h.putStrLn (Json.mkObj [("module",toJson name.toString),("path",toJson path.toString),
    ("imports",toJson (data.imports.map fun i => i.module.toString))]).compress
 IO.println (Json.mkObj [("modules",toJson env.header.moduleNames.size),
   ("artifact",toJson "NGG_MODULE_PATHS.jsonl")]).compress
 return 0
