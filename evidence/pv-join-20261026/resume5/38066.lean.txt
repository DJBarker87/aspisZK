import R1Z.PV_Join_LoopInvariant5
import R1Z.PV_Join_ArrayInvariant5
import R1Z.PV_Join_ProducerCanonical5
set_option autoImplicit false
noncomputable section
namespace R1Z.PV_Join_AccumulateCanonical5
open Aeneas Aeneas.Std Result ProverRwFull PV_Join_Canonical5 PV_Join_ResultInvariant5
variable [ExternalTypes]
abbrev QA := Array aspis_core.field.QM31 28#usize
abbrev MA := Array aspis_core.field.M31 28#usize
def rank (s : QA × Usize) : Nat := 28-s.2.val
def inv (s : QA × Usize) : Prop := QArray s.1

theorem step (ops : ExternalFuns) (basis : MA) (scale : aspis_core.field.QM31)
    (out : QA) (k : Usize) (hout : QArray out) :
    Safe (fun cf => match cf with
      | .done ys => QArray ys
      | .cont ys => inv ys ∧ rank ys < rank (out,k))
      (@host_sumcheck.interpolation_accumulate_loop.body _ (ProverRwP6BodyStore.selected ops) basis scale out k) := by
  unfold host_sumcheck.interpolation_accumulate_loop.body
  split
  · rename_i hk
    change Safe _ (do
      let value ← Array.index_usize out k
      let m ← Array.index_usize basis k
      let delta ← PV_Join_ScaleCanonical5.scale scale m
      let next ← ProverRwDictionaryTower.qAdd value delta
      let out1 ← Array.update out k next
      let k1 ← lift (Std.Usize.wrapping_add k 1#usize)
      Result.ok (ControlFlow.cont (out1,k1)))
    apply PV_Join_ResultInvariant5.bind Q _ _ _
    · intro value hv
      exact PV_Join_ArrayInvariant5.read out hout k value hv
    · intro value hv
      apply PV_Join_ResultInvariant5.bind_any; intro m
      obtain ⟨delta,hd,hdelta⟩ := PV_Join_ScaleCanonical5.canonical scale m
      rw [hd]
      simp only [bind_tc_ok]
      obtain ⟨next,hn,hnext⟩ := PV_Join_ArithmeticCanonical5.add value delta hv hdelta
      rw [hn]
      simp only [bind_tc_ok]
      apply PV_Join_ResultInvariant5.bind QArray _ _ _
      · intro ys hys
        exact PV_Join_ArrayInvariant5.update out hout k next hnext ys hys
      · intro ys hys
        simp only [lift,bind_tc_ok]
        apply PV_Join_ResultInvariant5.ok
        refine ⟨hys,?_⟩
        have hval : (Std.Usize.wrapping_add k 1#usize).val = k.val+1 := by scalar_tac
        change 28-(Std.Usize.wrapping_add k 1#usize).val < 28-k.val
        rw [hval]
        have hbound : k.val<28 := hk
        omega
  · exact PV_Join_ResultInvariant5.ok _ _ hout

#print axioms step
end R1Z.PV_Join_AccumulateCanonical5
