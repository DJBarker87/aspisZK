set_option autoImplicit false
namespace AspisR1.NHU_ExpandFrame
/-- Two typed one-cell windows, composed using the prescribed copy value law.
Pointer validity, exact owner and bounds are native/caller obligations. -/
def two {A : Type} (before : Nat→A) (index : Nat) (left right : A) : Nat→A :=
 KMX_CopyMeaning.loop (fun _=>right)
  (KMX_CopyMeaning.loop (fun _=>left) before (2*index) 1) (2*index+1) 1

theorem value {A : Type} (before : Nat→A) (index cell : Nat) (left right : A) :
 two before index left right cell=
 if cell=2*index+1 then right else if cell=2*index then left else before cell := by
 unfold two
 rw [KMX_CopyMeaning.loop_spec,KMX_CopyMeaning.loop_spec]
 have l : (2*index≤cell ∧ cell<2*index+1)↔cell=2*index := by omega
 have r : (2*index+1≤cell ∧ cell<2*index+1+1)↔cell=2*index+1 := by omega
 rw [l,r]

theorem bounds (index : Nat) (inside : index<32) :
 2*index<64 ∧ 2*index+1<64 ∧ 2*index≠2*index+1 := by omega

theorem unread {A : Type} (before : Nat→A) (index cell : Nat) (left right : A)
 (prior : cell<index) : two before index left right cell=before cell := by
 rw [value,if_neg (by omega),if_neg (by omega)]

theorem outside {A : Type} (before : Nat→A) (index cell : Nat) (left right : A)
 (notLeft : cell≠2*index) (notRight : cell≠2*index+1) :
 two before index left right cell=before cell := by
 rw [value,if_neg notRight,if_neg notLeft]
