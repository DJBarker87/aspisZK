//! NLU split native windows. Fixed deployed SolanaDigest; no hash call.
use super::*;
use aspis_core::field::{M31,CM31,QM31};
use aspis_r1_cost_probe::r1_semantic::NLI_rounds::{source_high_iteration,source_construct,KMX_copy_nonoverlapping};
use aspis_r1_cost_probe::r1_wire::{NLI_factory,Error};
fn q()->QM31{let a:[u32;4]=kani::any();QM31{c0:CM31{a:M31(a[0]),b:M31(a[1])},c1:CM31{a:M31(a[2]),b:M31(a[3])}}}
fn word(q:QM31,j:usize)->u32{match j{0=>q.c0.a.0,1=>q.c0.b.0,2=>q.c1.a.0,_=>q.c1.b.0}}
fn high_window(frame:bool,bad:bool){
 let claims=Box::new(core::array::from_fn(|_|core::array::from_fn(|_|q())));
 let i:usize=kani::any();let j:usize=kani::any();kani::assume(i<10&&j<4);
 let expected=word(claims[0][16+i],j);
 let row:usize=kani::any();let col:usize=kani::any();kani::assume(row<3&&col<29);
 let before=word(claims[row][col],j);
 // Read and source frame are separate obligations. No completed-array premise.
 let mut slot=core::mem::MaybeUninit::uninit();source_high_iteration(&claims,i,&mut slot);
 let actual=unsafe{slot.assume_init()};
 let selected=if frame{word(claims[row][col],j)==before}else{word(actual,j)==expected};
 if bad{assert!(!selected)}else{assert!(selected)}
}
#[kani::proof] #[kani::unwind(90)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn high_read_cut(){high_window(false,false)}
#[kani::proof] #[kani::unwind(90)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn high_read_control(){high_window(false,true)}
#[kani::proof] #[kani::unwind(90)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn high_frame_cut(){high_window(true,false)}
#[kani::proof] #[kani::unwind(90)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn high_frame_control(){high_window(true,true)}
fn root_window(bad:bool){
 let data:[u8;32]=kani::any();let j:usize=kani::any();kani::assume(j<32);
 // Literal deployed try_into().unwrap() at payload length32. Other lengths
 // follow the pinned Result/unwrap abort law; not covered by this window.
 let root:[u8;32]=data.as_slice().try_into().unwrap();
 let selected=root[j]==data[j];if bad{assert!(!selected)}else{assert!(selected)}
}
#[kani::proof] #[kani::unwind(34)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn root_word_cut(){root_window(false)}
#[kani::proof] #[kani::unwind(34)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn root_word_control(){root_window(true)}
