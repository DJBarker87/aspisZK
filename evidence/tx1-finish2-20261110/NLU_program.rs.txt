fn high_window(frame:bool,bad:bool){
 let claims=Box::new(core::array::from_fn(|_|core::array::from_fn(|_|q())));
 let i:usize=kani::any();let j:usize=kani::any();kani::assume(i<10&&j<4);

 if bad{assert!(!selected)}else{assert!(selected)}
}

fn construct_window(mode:u8,bad:bool){
 let c1:[u8;32]=kani::any();let root:[u8;32]=kani::any();let ch=challenges();
 let alpha=core::array::from_fn(|_|q());let claims=Box::new(core::array::from_fn(|_|core::array::from_fn(|_|q())));
 let extra=core::array::from_fn(|_|q());let before=core::array::from_fn(|_|q());
 let state:[u8;32]=kani::any();let z0=aspis_core::circle::SecureCirclePoint{x:q(),y:q()};let z1=aspis_core::circle::SecureCirclePoint{x:q(),y:q()};
 let tstate:[u8;32]=kani::any();let next:u8=kani::any();let transcript=NLI_factory::make(SolanaDigest,tstate,next);
 let row:usize=kani::any();let col:usize=kani::any();let j:usize=kani::any();let byte:usize=kani::any();let ai:usize=kani::any();
 kani::assume(row<3&&col<29&&j<4&&byte<32&&ai<10);
 let owner=(&*claims as *const [[QM31;29];3]) as usize;
 let expected=word(claims[row][col],j);let ea=word(alpha[ai],j);
 let tag:u8=kani::any();kani::assume(tag<13);

 let out=source_construct(&c1,root,ch,alpha,claims,extra,before,state,z0,z1,transcript);
 let (after,_,after_next)=NLI_factory::snapshot(&out.transcript);
 let selected=match mode{
  0=>(&*out.claims as *const [[QM31;29];3]) as usize==owner,
  1=>word(out.claims[row][col],j)==expected,
  2=>word(out.alpha[ai],j)==ea,
  3=>{let actual=match tag{0=>word(out.extra[col],j),1=>word(out.before_z1[col],j),2=>word(out.challenges.lambda,j),3=>word(out.challenges.chi,j),4=>word(out.challenges.theta,j),5=>word(out.challenges.mu,j),6=>word(out.challenges.eta,j),7=>word(out.challenges.zc[ai],j),8=>word(out.z[0].x,j),9=>word(out.z[0].y,j),10=>word(out.z[1].x,j),11=>word(out.z[1].y,j),_=>out.state_before_z0[byte] as u32};actual==expected_misc},
  4=>{let choose:bool=kani::any();if choose{out.roots[0][byte]==c1[byte]}else{out.roots[1][byte]==root[byte]}},
  _=>after[byte]==tstate[byte]&&after_next==next,
 };
 if bad{assert!(!selected)}else{assert!(selected)}
}

// Literal scalar guards, retaining owned Claims through the surviving arm.
fn terminal_guard(running:QM31,masked_terminal:QM31,claims:Box<[[QM31;29];3]>)->Result<Box<[[QM31;29];3]>,Error>{
 if running != masked_terminal { return Err(Error::Terminal); }
 Ok(claims)
}
fn circle_guard(z0:aspis_core::circle::SecureCirclePoint,z1:aspis_core::circle::SecureCirclePoint,claims:Box<[[QM31;29];3]>)->Result<Box<[[QM31;29];3]>,Error>{
 if z0 == z1 { return Err(Error::EqualCirclePoints); }
 Ok(claims)
}
fn guard_window(circle:bool,bad:bool){
 let claims=Box::new(core::array::from_fn(|_|core::array::from_fn(|_|q())));
 let owner=(&*claims as *const [[QM31;29];3]) as usize;
 let row:usize=kani::any();let col:usize=kani::any();let j:usize=kani::any();kani::assume(row<3&&col<29&&j<4);
 let expected=word(claims[row][col],j);
 let x=q();let y=q();let z=q();let w=q();
 let p=aspis_core::circle::SecureCirclePoint{x,y};let r=aspis_core::circle::SecureCirclePoint{x:z,y:w};
 let reject=if circle{p==r}else{x!=y};
 let reply=if circle{circle_guard(p,r,claims)}else{terminal_guard(x,y,claims)};
 let selected=match reply{
  Err(e)=>reject&&if circle{matches!(e,Error::EqualCirclePoints)}else{matches!(e,Error::Terminal)},
  Ok(c)=>!reject&&(&*c as *const [[QM31;29];3]) as usize==owner&&word(c[row][col],j)==expected,
 };
 if bad{assert!(!selected)}else{assert!(selected)}
}

// Opaque-call-log form only: no field multiplication/addition is executed.
// Three replies remain independent until substituted with their Lean laws.
struct FieldLog{replies:[QM31;3],args:[(QM31,QM31);3],calls:usize}
impl FieldLog{
 fn call(&mut self,a:QM31,b:QM31)->QM31{
  let at=self.calls;self.calls+=1;self.args[at]=(a,b);self.replies[at]
 }
}
fn arithmetic_window(bad:bool){
 let eta=q();let terminal=q();let mask=q();let extra=q();let replies=[q(),q(),q()];
 let mut log=FieldLog{replies,args:[(q(),q());3],calls:0};
 // Source evaluation order of mask.add(eta.mul(terminal)).add(extra[28]).
 let product=log.call(eta,terminal);
 let partial=log.call(mask,product);
 let masked=log.call(partial,extra);
 let site:usize=kani::any();let limb:usize=kani::any();let side:bool=kani::any();kani::assume(site<3&&limb<4);
 let expected=match (site,side){(0,false)=>eta,(0,true)=>terminal,(1,false)=>mask,(1,true)=>replies[0],(2,false)=>replies[1],_=>extra};
 let actual=if side{log.args[site].1}else{log.args[site].0};
 let selected=log.calls==3&&word(actual,limb)==word(expected,limb)&&word(masked,limb)==word(replies[2],limb);
 if bad{assert!(!selected)}else{assert!(selected)}
}
