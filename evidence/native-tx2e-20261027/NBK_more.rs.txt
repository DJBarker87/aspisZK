use crate::pool_v1::pair_forest_terminal::PoolV1PairForestTerminalFormatErrorV1;
fn NBK_statement_prefix(master:[u8;32],checkpoint:[u8;32],selected:[u8;32],lane:u8)->Option<PoolV1PairForestTerminalFormatErrorV1>{
    if master == [0u8; 32]
        || checkpoint == [0u8; 32]
        || selected == [0u8; 32]
    {
        return Some(PoolV1PairForestTerminalFormatErrorV1::NonZeroBinding);
    }
    if usize::from(lane) >= 8 {
        return Some(PoolV1PairForestTerminalFormatErrorV1::InvalidOutputLane);
    }
None}

 assert_eq!(NBK_statement_prefix(m,c,s,l),expected);
}

fn NBK_after_prefix(index:u64,root:[u8;32])->[u8;688]{
    let mut output = [0u8; POOL_V1_PAIR_VERIFIED_AFTERSTATE_BYTES];
    output[..4].copy_from_slice(&POOL_V1_PAIR_VERIFIED_AFTERSTATE_MAGIC);
    output[4] = POOL_V1_PAIR_VERIFIER_TRANSPORT_VERSION;
    output[5] = crate::pool_v1::POOL_V1_PAIR_TREE_STORAGE_FORMAT_VERSION;
    output[6] = POOL_V1_DIGEST_ENCODING_VERSION;
    output[7] = 1;
    output[8..16].copy_from_slice(&index.to_le_bytes());
    output[16..48].copy_from_slice(&root);
output}
#[kani::proof]
fn NBK_after_prefix_gate(){
 let index:u64=kani::any();let root:[u8;32]=kani::any();let j:usize=kani::any();kani::assume(j<688);
 let out=NBK_after_prefix(index,root);
 let expected=if j<4{b"ASJA"[j]}else if j<8{1}else if j<16{index.to_le_bytes()[j-8]}else if j<48{root[j-16]}else{0};
 assert_eq!(out[j],expected);
}
fn NBK_live_prefix(output:&mut[u8;800],pool:[u8;32],domain:[u8;32],sequence:u64,index:u64,root:[u8;32]){
    output.fill(0);
    output[..8].copy_from_slice(&POOL_V1_PAIR_LIVE_SNAPSHOT_MAGIC);
    output[8] = POOL_V1_PAIR_LIVE_SNAPSHOT_VERSION;
    output[16..48].copy_from_slice(&POOL_V1_PAIR_TREE_FORMAT_BINDING);
    output[48..80].copy_from_slice(&pool);
    output[80..112].copy_from_slice(&domain);
    output[112..120].copy_from_slice(&sequence.to_le_bytes());
    output[120..128].copy_from_slice(&index.to_le_bytes());
    output[128..160].copy_from_slice(&root);
}
#[kani::proof]
fn NBK_live_prefix_gate(){
 let mut output:[u8;800]=kani::any();let pool:[u8;32]=kani::any();let domain:[u8;32]=kani::any();let sequence:u64=kani::any();let index:u64=kani::any();let root:[u8;32]=kani::any();let j:usize=kani::any();kani::assume(j<800);
 NBK_live_prefix(&mut output,pool,domain,sequence,index,root);
 let expected=if j<8{POOL_V1_PAIR_LIVE_SNAPSHOT_MAGIC[j]}else if j==8{1}else if j<16{0}else if j<48{POOL_V1_PAIR_TREE_FORMAT_BINDING[j-16]}else if j<80{pool[j-48]}else if j<112{domain[j-80]}else if j<120{sequence.to_le_bytes()[j-112]}else if j<128{index.to_le_bytes()[j-120]}else if j<160{root[j-128]}else{0};
 assert_eq!(output[j],expected);
}
