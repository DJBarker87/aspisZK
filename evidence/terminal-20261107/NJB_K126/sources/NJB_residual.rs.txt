fn canon(w:[u32;4])->bool{w[0]<2147483647&&w[1]<2147483647&&w[2]<2147483647&&w[3]<2147483647}
struct NJB_Log{left:[[u32;4];17],right:[[u32;4];17],reply:[[u32;4];17],kind:[u8;17],calls:usize}
trait NJB_Ops{fn NJB_mul(self,r:Self,l:&mut NJB_Log)->Self;fn NJB_add(self,r:Self,l:&mut NJB_Log)->Self;fn NJB_sub(self,r:Self,l:&mut NJB_Log)->Self;}

fn NJB_mul(self,r:Self,l:&mut NJB_Log)->Self{event(self,r,0,l)}
fn NJB_add(self,r:Self,l:&mut NJB_Log)->Self{event(self,r,1,l)}

fn NJB_copy_residual(row: CopyRowExtension, helper: QM31, chi: QM31, log:&mut NJB_Log) -> QM31 {
    let denominators = [
        chi.NJB_sub(row.producer_values[0],log),
        chi.NJB_sub(row.producer_values[1],log),
        chi.NJB_sub(row.consumer_values[0],log),
        chi.NJB_sub(row.consumer_values[1],log),
    ];
    let producer_denominator = denominators[0].NJB_mul(denominators[1],log);
    let consumer_denominator = denominators[2].NJB_mul(denominators[3],log);
    let producer_numerator = row.producer_weights[0]
        .NJB_mul(denominators[1],log)
        .NJB_add(row.producer_weights[1].NJB_mul(denominators[0],log),log);
    let consumer_numerator = row.consumer_weights[0]
        .NJB_mul(denominators[3],log)
        .NJB_add(row.consumer_weights[1].NJB_mul(denominators[2],log),log);
    producer_denominator
        .NJB_mul(helper.NJB_mul(consumer_denominator,log).NJB_add(consumer_numerator,log),log)
        .NJB_sub(consumer_denominator.NJB_mul(producer_numerator,log),log)
}
