fn NJB_path_head(path_selector:QM31,z:[QM31;16],log:&mut NJB_Log)->(PreparedQm31Multiplier,PreparedQm31Multiplier,[QM31;17]){
    let bit = z[0];
    let selected_bit = path_selector.NJB_mul(bit,log);
    let selected_empty = path_selector.NJB_sub(selected_bit,log);
    let left = NJB_new(selected_empty,log);
    let right = NJB_new(selected_bit,log);
    let mut path = [QM31::ZERO; 17];
    path[0] = selected_bit.NJB_mul(bit.NJB_sub(QM31::ONE,log),log);
    (left,right,path)}

fn NJB_occupancy_context(input_occupancy:QM31,output_occupancy:QM31,opened:&[QM31;16],log:&mut NJB_Log)->(QM31,QM31,QM31,PreparedQm31Multiplier){
    let occupied = opened[0];
    let inverse = opened[1];
    let one_minus = QM31::ONE.NJB_sub(occupied,log);
    let both = input_occupancy.NJB_add(output_occupancy,log);
    let occupied_selector = both.NJB_mul(occupied,log);
    let empty_selector = both.NJB_sub(occupied_selector,log);
    let empty = NJB_new(empty_selector,log);
    (one_minus,both,occupied_selector,empty)}
fn NJB_occupancy_head(occupied:QM31,inverse:QM31,both:QM31,occupied_selector:QM31,empty:PreparedQm31Multiplier,opened:&[QM31;16],log:&mut NJB_Log)->[QM31;12]{let mut occupancy = [QM31::ZERO; 12];
    occupancy[0] = occupied_selector.NJB_mul(occupied.NJB_sub(QM31::ONE,log),log);
    occupancy[1] = both.NJB_mul(opened[9].NJB_mul(inverse,log).NJB_sub(occupied,log),log);
    occupancy[2] = empty.NJB_mul(inverse,log);
    occupancy}
