fn words(q:QM31)->[u32;4]{[q.c0.a.0,q.c0.b.0,q.c1.a.0,q.c1.b.0]}
fn q(w:[u32;4])->QM31{QM31{c0:CM31{a:M31(w[0]),b:M31(w[1])},c1:CM31{a:M31(w[2]),b:M31(w[3])}}}
fn canon(w:[u32;4])->bool{w[0]<2147483647&&w[1]<2147483647&&w[2]<2147483647&&w[3]<2147483647}
struct NJB_Log{left:[[u32;4];17],right:[[u32;4];17],reply:[[u32;4];17],kind:[u8;17],calls:usize}
trait NJB_Ops{fn NJB_mul(self,r:Self,l:&mut NJB_Log)->Self;fn NJB_add(self,r:Self,l:&mut NJB_Log)->Self;fn NJB_sub(self,r:Self,l:&mut NJB_Log)->Self;}
fn event(a:QM31,b:QM31,kind:u8,l:&mut NJB_Log)->QM31{let j=l.calls;assert!(j<17);assert!(canon(words(a))&&canon(words(b)));l.left[j]=words(a);l.right[j]=words(b);l.kind[j]=kind;l.calls+=1;q(l.reply[j])}
impl NJB_Ops for QM31{
fn NJB_mul(self,r:Self,l:&mut NJB_Log)->Self{event(self,r,0,l)}
fn NJB_add(self,r:Self,l:&mut NJB_Log)->Self{event(self,r,1,l)}
fn NJB_sub(self,r:Self,l:&mut NJB_Log)->Self{event(self,r,2,l)}}
fn NJB_high_step(selectors:&Selectors,high_sums:&mut [QM31;7],mut block:usize,log:&mut NJB_Log)->usize{
            let mask = constants::ACTIVE_ROW_MASKS[block];
            let coordinate = match mask {
                6144 => 0,
                6145 => 1,
                4097 => 2,
                2048 => 3,
                2049 => 4,
                26214 => 5,
                1749 => 6,
                _ => unreachable!("generated Copy active mask outside frozen basis"),
            };
            high_sums[coordinate] = high_sums[coordinate].NJB_add(selectors.high[block],log);
            block += 1;
        block}
fn NJB_low_step(selectors:&Selectors,high_sums:&[QM31;7],mut sum:QM31,mut coordinate:usize,log:&mut NJB_Log)->(QM31,usize){
            let low_sum = match coordinate {
                0 => selectors.low[11].NJB_add(selectors.low[12],log),
                1 => selectors.low[0].NJB_add(selectors.low[11],log).NJB_add(selectors.low[12],log),
                2 => selectors.low[0].NJB_add(selectors.low[12],log),
                3 => selectors.low[11],
                4 => selectors.low[0].NJB_add(selectors.low[11],log),
                5 => selectors.low[1].NJB_add(selectors.low[2],log).NJB_add(selectors.low[5],log).NJB_add(selectors.low[6],log)
                    .NJB_add(selectors.low[9],log).NJB_add(selectors.low[10],log).NJB_add(selectors.low[13],log).NJB_add(selectors.low[14],log),
                6 => selectors.low[0].NJB_add(selectors.low[2],log).NJB_add(selectors.low[4],log).NJB_add(selectors.low[6],log)
                    .NJB_add(selectors.low[7],log).NJB_add(selectors.low[9],log).NJB_add(selectors.low[10],log),
                _ => unreachable!("coordinate outside frozen active-mask basis"),
            };
            sum = sum.NJB_add(high_sums[coordinate].NJB_mul(low_sum,log),log);
            coordinate += 1;
        (sum,coordinate)}
fn assume_all<const N:usize>(a:[[u32;4];N]){for w in a{kani::assume(canon(w));}}
#[kani::proof]#[kani::unwind(82)]#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn NJB_active_high(){
 // Selectors and QM31 contain only integer fields; every bit-pattern is valid.
 let selectors:Selectors=unsafe{kani::any_raw()};let mut sums:[QM31;7]=unsafe{kani::any_raw()};let hs=sums;let b:usize=kani::any();kani::assume(b<64);
 let c=match constants::ACTIVE_ROW_MASKS[b]{6144=>0,6145=>1,4097=>2,2048=>3,2049=>4,26214=>5,1749=>6,_=>unreachable!()};
 let reply:[u32;4]=kani::any();kani::assume(canon(words(hs[c]))&&canon(words(selectors.high[b]))&&canon(reply));let mut r=[[0;4];17];r[0]=reply;
 let mut log=NJB_Log{left:[[0;4];17],right:[[0;4];17],reply:r,kind:[0;17],calls:0};let next=NJB_high_step(&selectors,&mut sums,b,&mut log);
 let k:usize=kani::any();let limb:usize=kani::any();kani::assume(k<7&&limb<4);
 assert_eq!(next,b+1);assert_eq!(log.calls,1);assert_eq!(log.kind[0],1);assert_eq!(log.left[0][limb],words(hs[c])[limb]);assert_eq!(log.right[0][limb],words(selectors.high[b])[limb]);
 assert_eq!(words(sums[k])[limb],(if k==c{r[0][limb]}else{words(hs[k])[limb]}) ^ if cfg!(NJB_invert){1}else{0});
}
#[kani::proof]#[kani::unwind(82)]#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn NJB_active_low(){

}
