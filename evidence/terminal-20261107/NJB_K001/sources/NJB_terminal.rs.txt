#[kani::proof]
#[kani::unwind(2)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
fn NJB_semantic_public() {
 use crate::pool_v1::pair_forest_semantic_terminal::r0::Public;
 let s=if kani::any(){PoolV1PairForestTerminalStatementV1::PrivateTransfer{common:common(),public:transfer()}}else{PoolV1PairForestTerminalStatementV1::Withdrawal{common:common(),public:withdrawal()}};
 let reply=s.semantic_public();
 let correct=match (&s,reply) {
  (PoolV1PairForestTerminalStatementV1::PrivateTransfer{common,public},Public::Transfer(p,t))=>core::ptr::eq(p,public)&&core::ptr::eq(t,&common.lane_transition),
  (PoolV1PairForestTerminalStatementV1::Withdrawal{common,public},Public::Withdrawal(p,t))=>core::ptr::eq(p,public)&&core::ptr::eq(t,&common.lane_transition),
  _=>false
 };
 assert!(correct != cfg!(NJB_invert));
}
