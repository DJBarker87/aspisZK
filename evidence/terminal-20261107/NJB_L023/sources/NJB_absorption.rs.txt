fn NJB_abs_scales(low:QM31,fixed:QM31,chunk_two:QM31,chunk_eight:QM31,nodes:QM31,log:&mut NJB_Log)->[NJB_Prepared;3]{
    let scales = [
        NJB_Prepared::new(low.NJB_mul(fixed,log),log),
        NJB_Prepared::new(low.NJB_mul(fixed.NJB_add(chunk_two,log),log),log),
        NJB_Prepared::new(low.NJB_mul(fixed.NJB_add(chunk_two,log).NJB_add(chunk_eight,log).NJB_add(nodes,log),log),log),
    ];
    scales}
fn NJB_abs_lane(scales:[NJB_Prepared;3],openings:&[QM31;16],lane:usize,log:&mut NJB_Log)->QM31{
        let scale = if lane < 2 {
            scales[0]
        } else if lane < RATE {
            scales[1]
        } else {
            scales[2]
        };
        scale.NJB_mul(openings[lane],log)
    }
