struct Tail{count:usize,tag:[u8;20],a:[E;20],b:[E;20],q:[K;20],r:[K;20],er:[E;20],qr:[K;20],fp:*const K,pp:*const E}
fn tscalar(q:&[K;4],p:&[E;4],l:&mut Tail)->E{let i=l.count;l.tag[i]=0;l.fp=q.as_ptr();l.pp=p.as_ptr();l.count+=1;l.er[i]}
fn top(a:E,b:E,q:K,r:K,tag:u8,l:&mut Tail)->E{let i=l.count;l.tag[i]=tag;l.a[i]=a;l.b[i]=b;l.q[i]=q;l.r[i]=r;l.count+=1;l.er[i]}
fn tmix(a:E,q:K,l:&mut Tail)->E{top(a,E::ZERO,q,K::ZERO,1,l)}
fn tquarter(a:E,l:&mut Tail)->E{top(a,E::ZERO,K::ZERO,K::ZERO,2,l)}
fn tadd(a:E,b:E,l:&mut Tail)->E{top(a,b,K::ZERO,K::ZERO,3,l)}
fn tmul(a:E,b:E,l:&mut Tail)->E{top(a,b,K::ZERO,K::ZERO,4,l)}
fn tqmul(q:K,r:K,l:&mut Tail)->K{let i=l.count;l.tag[i]=5;l.q[i]=q;l.r[i]=r;l.count+=1;l.qr[i]}
fn tail_cut(ap0:&[E;4],ap1:&[E;4],q:&[K;4],high:&[K;4],q4:K,q16:K,side:usize,weights:&mut [E;4],l:&mut Tail){
 let low=tscalar(q,ap0,l);
 let term0=tquarter(tmix(ap1[PERM[0]],high[0],l),l);
 let term1=tquarter(tmix(ap1[PERM[1]],high[1],l),l);
 let term2=tquarter(tmix(ap1[PERM[2]],high[2],l),l);
 let term3=tquarter(tmix(ap1[PERM[3]],high[3],l),l);
 let terms=[term0,term1,term2,term3];
 let first2=tadd(terms[0],terms[1],l);let first3=tadd(first2,terms[2],l);let all=tadd(first3,terms[3],l);
 if side==0{weights[0]=tmul(low,all,l);weights[1]=tmix(tmul(low,first3,l),q16,l);}
 else{weights[1]=tadd(weights[1],tmul(low,tquarter(ap1[PERM[3]],l),l),l);weights[2]=tmix(tmul(low,all,l),q4,l);weights[3]=tmix(tmul(low,first2,l),tqmul(q16,q4,l),l);}
}
