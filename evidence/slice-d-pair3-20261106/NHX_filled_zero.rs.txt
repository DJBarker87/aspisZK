use crate::field::{QM31 as NativeQ,CM31 as NativeC,M31 as NativeM};
fn qraw(x:[u32;4])->NativeQ{NativeQ{c0:NativeC{a:NativeM(x[0]),b:NativeM(x[1])},c1:NativeC{a:NativeM(x[2]),b:NativeM(x[3])}}}
fn qwords(x:NativeQ)->[u32;4]{[x.c0.a.0,x.c0.b.0,x.c1.a.0,x.c1.b.0]}
#[kani::proof]#[kani::unwind(258)]
#[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
#[kani::stub(<[u8]>::copy_from_slice,KMX_copy_from_slice)]
fn zero64_window(){
 let mut out=Vec::<NativeQ>::new();
 if out.try_reserve_exact(64).is_err(){assert!(out.is_empty());return;}
 let ptr=out.as_ptr();let cap=out.capacity();assert!(cap>=64);
 let p:usize=kani::any();kani::assume(p<64);
 let before:[NativeQ;64]=core::array::from_fn(|_|qraw(kani::any()));
 for i in 0..64{if i<p{kani::assume(before[i]==NativeQ::ZERO);}}
 unsafe{core::ptr::copy_nonoverlapping(before.as_ptr(),out.as_mut_ptr(),64);out.set_len(p);}
 out.resize(p+1,NativeQ::ZERO.clone());
 let k:usize=kani::any();kani::assume(k<p+1);let word:usize=kani::any();kani::assume(word<4);
 if cfg!(NHX_corrupt){out[k]=qraw([1;4]);}
 assert!(out.as_ptr()==ptr&&out.capacity()==cap&&out.len()==p+1);assert!(qwords(out[k])[word]==0);
}
