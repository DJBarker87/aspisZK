struct RegControl{count:usize,kinds:[usize;5],stop:usize,error:Error}
fn reg_child_control(kind:usize,l:&mut RegControl)->Result<(),Error>{let i=l.count;l.kinds[i]=kind;l.count+=1;if i==l.stop{Err(l.error)}else{Ok(())}}
fn reg_control_cut(shared:bool,l:&mut RegControl)->Result<(),Error>{
 if !shared{reg_child_control(1,l)?;}reg_child_control(2,l)?;if !shared{reg_child_control(3,l)?;}
 reg_child_control(5,l)?;reg_child_control(6,l)?;Ok(())
}
fn regular_control(){let shared:bool=kani::any();let stop:usize=kani::any();kani::assume(stop<=5);let mut l=RegControl{count:0,kinds:[99;5],stop,error:err(kani::any())};let r=reg_control_cut(shared,&mut l);let total=if shared{3}else{5};let count=if stop<total{stop+1}else{total};let j:usize=kani::any();kani::assume(j<count);if cfg!(NHX_corrupt){l.kinds[j]=99;}
 let expected=if shared{[2,5,6,99,99]}else{[1,2,3,5,6]};assert!(l.count==count&&l.kinds[j]==expected[j]);assert!(r.map_err(code)==if stop<total{Err(code(l.error))}else{Ok(())});}
struct RegWindow{w:*const Weights,kind:usize,alpha:*const E,alen:usize,bytes:*const u8,len:usize,folds:usize,n:usize,ap0:*const E,ap1:*const E,vec:*const Vec<K>,scratch:*const K,width:usize,parts:*const E,calls:usize,postk:[K;64],poste:[E;7],ok:bool,error:Error}
fn reg_child(w:&Weights,kind:usize,alpha:&[E],f:&[u8],folds:usize,n:usize,ap0:&[E;4],ap1:&[E;4],scratch:&mut Vec<K>,parts:&mut [E;7],l:&mut RegWindow)->Result<(),Error>{
 l.calls+=1;l.w=w;l.kind=kind;l.alpha=alpha.as_ptr();l.alen=alpha.len();l.bytes=f.as_ptr();l.len=f.len();l.folds=folds;l.n=n;l.ap0=ap0.as_ptr();l.ap1=ap1.as_ptr();l.vec=scratch;l.scratch=scratch.as_ptr();l.width=scratch.len();l.parts=parts.as_ptr();
 scratch.as_mut_slice().copy_from_slice(&l.postk);parts.copy_from_slice(&l.poste);if l.ok{Ok(())}else{Err(l.error)}
}
fn regular_check(g:usize){
 let shared:bool=kani::any();let i:usize=kani::any();kani::assume(i<if shared{3}else{5});let kinds=if shared{[2usize,5,6,99,99]}else{[1usize,2,3,5,6]};let kind=kinds[i];
