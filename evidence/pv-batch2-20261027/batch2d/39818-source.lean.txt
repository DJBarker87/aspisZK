-- chain, recurrence or concrete generated table is evaluated.

  unfold reduce PV_B2_C_251FieldValue.aspis_core.field.ct.reduce64
    PV_B2_C_251FieldValue.aspis_core.field.ct.subtract_p F
    PV_B2_C_251FieldValue.aspis_core.field.P
  simp only [UScalar.lt_equiv, UScalar.le_equiv, UScalar.val,
    UScalar.cast, UScalar.cast_fromBool, UScalar.wrapping_shr,
    core.num.U32.wrapping_sub, U64.wrapping_add, U64.wrapping_shr,
    UScalar.wrapping_add, UScalar.wrapping_sub, UScalar.bv_and,
    UScalarTy.numBits, BitVec.zeroExtend_eq_setWidth, decide_eq_true_eq]
  trace_state
  bv_decide
#print axioms reduce_bound
