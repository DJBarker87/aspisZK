import R1Z.PV_B2_C_251FieldTotal
import Std.Tactic.BVDecide
set_option autoImplicit false
namespace PV_B2_D_ScalarCanonical
open Aeneas Aeneas.Std
abbrev F := PV_B2_C_251FieldValue.aspis_core.field.P
abbrev reduce := PV_B2_C_251FieldValue.aspis_core.field.ct.reduce64
-- One fixed-width reducer, no field multiplication chains.
def fold64 (x : BitVec 64) : BitVec 64 := (x &&& 2147483647#64) + (x >>> 31)
def word (x : BitVec 64) : BitVec 32 := (fold64 (fold64 x)).setWidth 32
def reduced (x : BitVec 64) : BitVec 32 :=
  word x - (2147483647#32 &&& (0#32 - (if (2147483647#32).ule (word x) then 1#32 else 0#32)))
theorem bitvec_bound (x : BitVec 64) : (reduced x).ult 2147483647#32 = true := by
  unfold reduced word fold64
  bv_decide
#print axioms bitvec_bound
theorem reduce_bound (x : U64) : reduce x < F := by
  unfold reduce PV_B2_C_251FieldValue.aspis_core.field.ct.reduce64
    PV_B2_C_251FieldValue.aspis_core.field.ct.subtract_p F
    PV_B2_C_251FieldValue.aspis_core.field.P
  simp only [UScalar.lt_equiv, UScalar.le_equiv, UScalar.val,
    UScalar.cast, UScalar.cast_fromBool, UScalar.wrapping_shr,
    core.num.U32.wrapping_sub, U64.wrapping_add, U64.wrapping_shr,
    UScalar.wrapping_add, UScalar.wrapping_sub, UScalar.bv_and,
    UScalarTy.numBits, BitVec.zeroExtend_eq_setWidth, decide_eq_true_eq]
  have h := bitvec_bound x.bv
  simp only [reduced,word,fold64,BitVec.ult,BitVec.ule,decide_eq_true_eq] at h
  simpa only [apply_ite] using h
#print axioms reduce_bound
end PV_B2_D_ScalarCanonical
