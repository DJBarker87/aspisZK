/-- Exact complete-Result numeric recurrence transport; arbitrary32-byte input. -/
theorem source (block : Array U8 32#usize) :
    PV_B2_Sampler.host_sampler.low_base_p_digits block =
      PV_B2_OrdinaryReference.sampler_owned.low_base_p_digits block := by
  have hb : PV_B2_Sampler.host_sampler.low_base_p_digits_loop0 =
      PV_B2_OrdinaryReference.sampler_owned.low_base_p_digits_loop0 := by
    funext b q i
    exact PV_B2_Bytes.loop b q i
  have hd : PV_B2_Sampler.host_sampler.low_base_p_digits_loop1 =
      PV_B2_OrdinaryReference.sampler_owned.low_base_p_digits_loop1 := by
    funext q d i
    exact PV_B2_Digits.loop q d i
  unfold PV_B2_Sampler.host_sampler.low_base_p_digits
    PV_B2_OrdinaryReference.sampler_owned.low_base_p_digits
  simp only [hb,hd]
  apply (Aeneas.Std.bind_eq_iff _ _ _).mpr
  intro q _
  apply (Aeneas.Std.bind_eq_iff _ _ _).mpr
  rintro ⟨q,d⟩ _
  have hc : PV_B2_Sampler.host_sampler.low_base_p_digits_loop2 1#usize = ok () :=
    PV_B2_ReleaseCheck.source
  dsimp only
  rw [hc]
  rfl
#print axioms source
end PV_B2_LowDigits
