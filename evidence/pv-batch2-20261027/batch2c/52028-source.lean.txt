import R1Z.PV_B2_C_ConstructorCompose
set_option autoImplicit false
noncomputable section
namespace PV_B2_C_ConstructorAssociation
open Aeneas Aeneas.Std Result
open PV_B2_C_ArrayTotal PV_B2_C_WeightsNative PV_B2_C_ConstructorRows

/-- Reassociation only: the complete emitted Result equals the five fixed-row
composition. No field operation or concrete constructor value is evaluated. -/
theorem new_eq_factored (variant : transport.Variant)
    (alpha : Array K 10#usize) (z : Array Point 2#usize) :
    host_weights.Weights.new variant alpha z =
      PV_B2_C_ConstructorCompose.factored variant alpha z := by
  simp only [host_weights.Weights.new,PV_B2_C_ConstructorCompose.factored,
    row0,row1,row2,row3,row4,bind_assoc,bind_tc_ok]

/-- Complete constructor success on arbitrary extracted carriers, including raw
noncanonical field limbs. Native allocation/effect equivalence is separate. -/
theorem new_total (variant : transport.Variant)
    (alpha : Array K 10#usize) (z : Array Point 2#usize) :
    RustOk (host_weights.Weights.new variant alpha z) := by
  rw [new_eq_factored]
  exact PV_B2_C_ConstructorCompose.factored_total variant alpha z
