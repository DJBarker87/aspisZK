def cutNew (variant : transport.Variant) (alpha : Array K 10#usize)
    (z : Array Point 2#usize) : Result (core.result.Result host_weights.Weights verifier.Error) := do
  let points ← host_weights.state_only_statement_points alpha
  let q ← Array.index_usize alpha 8#usize
  let q1 ← Array.index_usize alpha 9#usize
  let a := Array.repeat 4#usize aspis_core.field.QM31.ZERO
  let a1 := Array.repeat 5#usize a
  let factors := Array.repeat 5#usize a1
  flat0 variant points q q1 z factors

/-- Exact source prefix cut: the flat suffix is literally unchanged. -/
theorem source_eq (variant : transport.Variant) (alpha : Array K 10#usize)
    (z : Array Point 2#usize) :
    host_weights.Weights.new variant alpha z = cutNew variant alpha z := by
  rfl

theorem cutNew_total (variant : transport.Variant) (alpha : Array K 10#usize)
    (z : Array Point 2#usize) : RustOk (cutNew variant alpha z) := by
  unfold cutNew
  apply bind_rust _ _ (PV_B2_C_WeightsTotal.points alpha)
  intro points
  apply bind_rust _ _ (read _ _ (by decide))
  intro q
  apply bind_rust _ _ (read _ _ (by decide))
  intro q1
  exact flat0_total variant points q q1 z _

/-- Constructor success for every typed input, with no canonicality or success
premise. This is an extracted source law; native allocation transport remains separate. -/
theorem new_total (variant : transport.Variant) (alpha : Array K 10#usize)
    (z : Array Point 2#usize) : RustOk (host_weights.Weights.new variant alpha z) := by
  rw [source_eq]
  exact cutNew_total variant alpha z
