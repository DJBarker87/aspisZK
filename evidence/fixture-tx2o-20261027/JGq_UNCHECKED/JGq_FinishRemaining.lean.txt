set_option autoImplicit false

namespace AspisR1.JGq_FinishRemaining
open Aeneas Aeneas.Std Result AspisR1Stage2 AspisR1Stage2.Stage2Finish

def adapted : Result (core.result.Result E verifier.Error) := do
  let r ← JGq_PairAfterShared.remaining
  match r with
  | .Ok out=>ok (.Ok (FIo_Install.fromE out))
  | .Err e=>ok (.Err (FIo_Install.error e))

theorem pair : FIo_Install.pair (.Ok ()) JGo_TX2_WeightData.value JGo_TX2_FinishPrefix.alphas
    (JFw_Frames.window 8007 2048) JFo_Prepared.value.kappa=adapted := by
  rw [JGq_PairPrefix.source]
  unfold JGq_PairPrefix.adapted adapted
  rw [JGq_PairAfterShared.source]
#print axioms pair

/-- Exact final gates and native Pair remainder, with the SAME accumulator.
No successful Pair, dot, finish or wrapper is an input assumption. -/
def remaining (seed : FinishOps FIo_Install.W) (inactive : E) (a : verifier.Accumulator) :
    Result (core.result.Result Unit verifier.Error) :=
  let ops := Port4D_Round.native (FIo_Install.finishOps (.Ok ()) seed)
  let v := JGo_TX2_FinishPrefix.view inactive
  let p := JFo_Prepared.value
  let lastClaim := JGo_TX2_Horner1_6.after
  do
  let r3 ← adapted
  let cf3 ← core.result.Result.Insts.CoreOpsTry.branch r3
  match cf3 with
  | .Continue pair => do
    ops.op0 16#u16 0#usize pair
    ops.op0 16#u16 1#usize lastClaim
    let different ← ops.ne pair lastClaim
    if different then ok (.Err verifier.Error.Relation)
    else do
      ops.op1 none verifier.Phase.Relation
      let r4 ← ops.op4 v.final_message a.weights.deref
      let cf4 ← core.result.Result.Insts.CoreOpsTry.branch r4
      match cf4 with
      | .Continue dot => do
        ops.op0 17#u16 0#usize dot
        ops.op0 17#u16 1#usize a.rhs
        if a.next != 22#usize then ok (.Err verifier.Error.Query)
        else do
          let different ← ops.ne dot a.rhs
          if different then ok (.Err verifier.Error.Query)
          else do
            ops.op1 none verifier.Phase.Final
            ok (.Ok ())
      | .Break residual => core.result.Result.Insts.CoreOpsTryTraitFromResidualResultInfallible.from_residual Unit (core.convert.FromSame verifier.Error) residual
  | .Break residual => core.result.Result.Insts.CoreOpsTryTraitFromResidualResultInfallible.from_residual Unit (core.convert.FromSame verifier.Error) residual


theorem source (seed : FinishOps FIo_Install.W) (inactive : E) (a : verifier.Accumulator) :
    Port4D_FinishInstall.finish (FIo_Install.finishOps (.Ok ()) seed)
      (JGo_TX2_FinishPrefix.view inactive) JFo_Prepared.value a none=
      Port3_QueryInstall.result (remaining seed inactive a) := by
  rw [JGo_TX2_FinishPrefix.installed]
  congr 1
  unfold JGo_TX2_FinishPrefixTools.remaining remaining
  dsimp only
  have call : (Port4D_Round.native (FIo_Install.finishOps (.Ok ()) seed)).op2
      JGo_TX2_WeightData.value JGo_TX2_FinishPrefix.alphas
      (JGo_TX2_FinishPrefix.view inactive).final_message JFo_Prepared.value.kappa=adapted := by
    change FIo_Install.pair (.Ok ()) JGo_TX2_WeightData.value JGo_TX2_FinishPrefix.alphas
      (JFw_Frames.window 8007 2048) JFo_Prepared.value.kappa=adapted
    exact pair
  rw [call]
#print axioms source
end AspisR1.JGq_FinishRemaining
