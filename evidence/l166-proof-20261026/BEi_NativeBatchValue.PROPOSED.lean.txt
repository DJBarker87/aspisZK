      Port3_NativeBatchBody.verifier.candidate.batch_merkle
        (Port9OR_BatchStep.ops Port4B_Install.leaves) (SdkC_Source.dictionary hashv)
        input.proof input.prepared input.folded (ordinal slot) () := by
  exact (SdkC_Source.erasure hashv _ before).trans
    (Port9OR_Batch.source Port4B_Install.leaves (SdkC_Source.dictionary hashv)
      (SdkC_Source.dictionary hashv) () input.proof input.prepared input.folded (ordinal slot))
