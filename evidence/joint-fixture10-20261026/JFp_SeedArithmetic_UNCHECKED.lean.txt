/-- Strengthens the partial real-byte fixture with concrete SHA initialization
and the exact native initial running-value decode. Numeric challenges still
need to be identified with the sampled trace before acceptance follows. -/
theorem inhabited {Hash : Type} (D : AspisR1CurrentCaller.aspis_core.digest.Digest32 Hash)
    (hash : Hash) :
    let D0 := AspisR1CurrentCaller.InstalledPreparation.retainedDigest
      (F3V_OracleDictionary.current JFp_SHADigest.digest D)
    ∃ (expanded : Slice U8) (pre : r1_wire.Prefix),
      compact.expand_semantic nativeValues JFw_RealExpansion.stored JFw_RealExpansion.initial=ok (.Ok (),expanded) ∧
      PrefixAdapter.parseAccepted expanded=ok (.Ok pre) ∧
      r1_semantic.public_transcript D0 JFs_PublicSerialization.value hash pre.c1_root=ok (.Ok
        (⟨encodeState JFr_SeedBoundary.numeric,hash,0#u8⟩ : r1_wire.SemanticTranscript Hash)) ∧
      r1_wire.Prefix.validated_record pre 14#u8=ok (.Ok JFp_InitialValue.record) ∧
      AspisS1ExtractedRoundChildren.decodeMain 1#usize JFp_InitialValue.record 0#usize=ok (.Ok JFp_InitialValue.values) ∧
      Array.index_usize JFp_InitialValue.values 0#usize=ok JFp_InitialValue.running ∧
      (∀ r : Fin 10, ∃ vr,
        r1_wire.Prefix.validated_record pre (JFv_SemanticDecode.roundRow r)=ok (.Ok vr) ∧
        AspisS1ExtractedRoundChildren.decodeMain 28#usize vr 0#usize=ok (.Ok (JFr_RealRounds.polynomial r)) ∧
        AspisS1ExtractedRoundChildren.generatedMainBoundary (JFr_RealRounds.polynomial r)=ok (JFr_RealRounds.running r) ∧
        AspisS1ExtractedRoundChildren.generatedMainValue (JFr_RealRounds.polynomial r)
          (JFr_RealRounds.challenges r) (UScalar.ofNatCore r.val (by have := r.isLt;scalar_tac))=ok (JFr_RealRounds.output r)) := by
  obtain ⟨expanded,pre,expand,parse,rows⟩ := JFw_Prefix.inhabited
  obtain ⟨read,decode,index⟩ := JFp_InitialValue.source expanded pre expand parse
  refine ⟨expanded,pre,expand,parse,JFp_SourceSeed.source D hash expanded pre expand parse,read,decode,index,?_⟩
  intro r
  obtain ⟨vr,read,decode⟩ := JFr_RealRounds.decoded expanded pre expand parse r
  exact ⟨vr,read,decode,JFr_RealRounds.boundary r,JFr_RealRounds.value r⟩

#print axioms inhabited
end AspisR1.JFp_SeedArithmetic
