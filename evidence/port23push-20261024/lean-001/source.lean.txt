/-- [port23push_native::verifier::update_residue_row]:
    Source: 'src/update.rs', lines 6:0-16:1 -/
def verifier.update_residue_row (arrayIntoIter : ArrayIntoIter)
  (row : Array Std.U64 4#usize) (power : field.QM31) (basis : field.M31)
  (flush : Bool) :
  Result (Array Std.U64 4#usize)
  := do
  let i := power.c0.a
  let i1 := power.c0.b
  let i2 := power.c1.a
  let i3 := power.c1.b
  let row1 ←
    verifier.update_residue_row_loop0 { start := 0#usize, «end» := 4#usize }
      row basis (Array.make 4#usize [ i, i1, i2, i3 ])
  if flush
  then
    let (iter, into_iter_back) ←
      arrayIntoIter row1
    let back ← verifier.update_residue_row_loop1 iter (fun im => im)
    ok (into_iter_back back)
  else ok row1
