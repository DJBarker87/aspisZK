#![feature(proc_macro_hygiene,stmt_expr_attributes)]
#[kani::proof]
fn KNH_for_positive(){let a:[u8;4]=kani::any();let mut count:usize=0;
 #[kani::loop_invariant(count == kani::index && count <= 4)]
 for _x in a.iter().enumerate(){count+=1;}
 assert_eq!(count,4);
}
#[kani::proof]
fn KNH_for_negative(){let a:[u8;4]=kani::any();let mut count:usize=0;
 #[kani::loop_invariant(count == kani::index && count <= 4)]
 for _x in a.iter().enumerate(){count+=1;}
 assert_eq!(count,3);
}
struct KNH_Record {bytes:[u8;4]}
#[kani::proof]
fn KNH_borrowed_positive(){let record=KNH_Record{bytes:kani::any()};let borrowed=&record;let mut i=0usize;
 #[kani::loop_invariant(i<=borrowed.bytes.len())]
 while i<borrowed.bytes.len(){i+=1;}
 assert_eq!(i,4);
}
#[kani::proof]
fn KNH_borrowed_negative(){let record=KNH_Record{bytes:kani::any()};let borrowed=&record;let mut i=0usize;
 #[kani::loop_invariant(i<=borrowed.bytes.len())]
 while i<borrowed.bytes.len(){i+=1;}
 assert_eq!(i,3);
}
