/-- Both initial packed maps are fixed by the original owner context's sampler. -/
theorem value {F C : Type} (c : Configuration F C)
    (B : AspisR1PublicPreparation.FinishModel.Boundaries) (s : F3T_ByteCarrier.Submission)
    (H : Bytes → State) (e : Evidence c B s H) (i : F3_Model.Input)
    (p : OraclePortCompactDraft.Parsed) (q : Fin 22 → Fin 262144)
    (data : RecordedBinaryExperimentBin.SourceInputs)
    (atPrepared : FGl_PreparedAt.At c B s H e i p q data) (j : Fin 22) :
    OraclePortStoredDecisionDraftC4.packed p q 0 (q j).val=
      some ((OraclePortSelectedDraft.opening p j.val).take 480) ∧
    OraclePortStoredDecisionDraftC4.packed p q 1 (q j).val=
      some (OraclePortCompactDraft.slice (OraclePortSelectedDraft.opening p j.val) 480 128) :=
  FJo_PackedLeaves.value p q _ atPrepared.2.2.2.1 j
#print axioms value
