use super::{M31,CM31,QM31};

const MODULUS:u32=2147483647;
fn word(b:&[u8],i:usize)->u32 {
 (b[i] as u32)|((b[i+1] as u32)<<8)|((b[i+2] as u32)<<16)|((b[i+3] as u32)<<24)
}
fn m_ref(b:[u8;4])->Option<u32> {let v=word(&b,0);if v<MODULUS{Some(v)}else{None}}
fn c_ref(b:&[u8])->Option<[u32;2]> {
 let a=word(b,0);if a>=MODULUS{return None;}let z=word(b,4);if z>=MODULUS{return None;}Some([a,z])
}
fn q_ref(b:&[u8])->Option<[u32;4]> {
 let a=c_ref(&b[0..8])?;let z=c_ref(&b[8..16])?;Some([a[0],a[1],z[0],z[1]])
}
fn limbs_words_reference(x:[u32;8])->Option<[u32;8]> {
 if x[0]>=MODULUS||x[1]>=MODULUS||x[2]>=MODULUS||x[3]>=MODULUS||x[4]>=MODULUS||x[5]>=MODULUS||x[6]>=MODULUS||x[7]>=MODULUS {None}else{Some(x)}
}
fn wide_ref(b:&[u8])->Option<[u32;8]> {
 if b.len()!=32{return None;}
 limbs_words_reference([word(b,0),word(b,4),word(b,8),word(b,12),word(b,16),word(b,20),word(b,24),word(b,28)])
}
fn c_words(x:CM31)->[u32;2] {[x.a.0,x.b.0]}
fn q_words(x:QM31)->[u32;4] {[x.c0.a.0,x.c0.b.0,x.c1.a.0,x.c1.b.0]}
fn wide_words(x:WideExact)->[u32;8] {x.to_limbs()}
macro_rules! check {
 ($name:ident,$b:ident,$ty:ty,$actual:expr,$reference:expr,$neg:literal) => {
 #[kani::proof] #[kani::unwind(34)]
 #[kani::stub(core::ptr::copy_nonoverlapping,KMX_copy_nonoverlapping)]
 #[kani::stub(<[u8]>::copy_from_slice,KMX_copy_from_slice)]
 fn $name(){let $b:$ty=kani::any();let real=$actual;let expected=$reference;if $neg {assert_ne!(real,expected);}else{assert_eq!(real,expected);}}
 }
}
check!(m_parser,b,[u8;4],M31::from_le_bytes(b).map(|x|x.0),m_ref(b),false);
check!(m_parser_wrong,b,[u8;4],M31::from_le_bytes(b).map(|x|x.0),m_ref(b),true);
check!(c_parser,b,[u8;8],CM31::from_le_bytes(&b).map(c_words),c_ref(&b),false);
check!(c_parser_wrong,b,[u8;8],CM31::from_le_bytes(&b).map(c_words),c_ref(&b),true);
check!(q_parser,b,[u8;16],QM31::from_le_bytes(&b).map(q_words),q_ref(&b),false);
check!(q_parser_wrong,b,[u8;16],QM31::from_le_bytes(&b).map(q_words),q_ref(&b),true);
check!(limbs_parser,b,[u32;8],WideExact::from_limbs(b).map(wide_words),limbs_words_reference(b),false);
check!(limbs_parser_wrong,b,[u32;8],WideExact::from_limbs(b).map(wide_words),limbs_words_reference(b),true);
