theorem accepted {W A C Tx : Type} (L : Library W (Nat → Record))
    (copy : Heap (Nat → Record) → Nat → Payload → S2W_TrustedStd.Outcome (Nat → Record))
    (allocator : Allocator A) (c : DrvC_Installation.Configuration FIo_Install.W C)
    (err : C) (r : NP6_StdModel.Reservation)
    (native1 : S2W_ExtractionBoundary.First) (native2 : S2W_ExtractionBoundary.Second)
    (hashv : SdkC_Source.Parts → Result SdkC_Source.Digest)
    (extraction : S2W_DigestExtraction.AeneasExtraction
      (DrvC_Installation.sdk (FIp_Install.configuration err c r)) native1 native2 hashv)
    (a : Adversary A) (H : Bytes → State)
    (n : ARU_Support.At (Tx := Tx) L copy allocator c err r native1 native2 hashv a H)
    (std : S2W_TrustedStd.TrustedStd copy) (D : LiteralByteProtocol.Decoder)
    (words : CanonicalCutEnrichedTranscript.Words)
    (owners : FIp_GlobalOwners.Owners (FIp_Install.configuration err c r)
      (Rc2_Install.configuration (Rc5_Install.seed err (FIp_Install.select r c)) r).boundaries)
    (Q : Nat) (bound : ART_AdversaryWork.work (FIp_Install.configuration err c r)
      hashv copy allocator CPH_Program.verifier a H ≤ Q) :
    (eval H (F3T_Installed.source (FIp_Install.configuration err c r) hashv
      (input (FIp_Install.configuration err c r) (eval H (submission a)).2)) =
        eval H (F3_Model.verifier D (input (FIp_Install.configuration err c r) (eval H (submission a)).2)) ∧
      (eval H (F3T_Charged.projected D (F3T_Canon.canon (lift (FIp_Install.configuration err c r) (submission a))))).2.2 = true ∧
      OraclePortRefineJoin.RecoveredInputAgreement
        (ART_PhysicalView.physical (FIp_Install.configuration err c r) hashv copy allocator CPH_Program.verifier a H)
        (eval H (F3T_Charged.projected D (F3T_Canon.canon (lift (FIp_Install.configuration err c r) (submission a)))))) ∨
    BinRawByteCollisionTargets.event SelectedTranscriptParams.params.iv (Q+2965)
      (eval H (Hiq_RecordedJoint.program
        (ART_PartialAccount.account (FIp_Install.configuration err c r) hashv copy allocator CPH_Program.verifier a)
        ARQ_Observer.project (ARQ_Observer.select (FIp_Install.configuration err c r) CPH_Program.verifier))) := by
  have represented :=  ARU_PartialAccepted.accepted L copy allocator c err r native1 native2 hashv extraction a H n
  have replies := ARU_Replies.accepted L copy allocator c err r native1 native2 hashv extraction a H n
  have source := ARX_SourceAlternative.accepted L copy allocator c err r native1 native2 hashv extraction a H n std Q bound
  rcases source with sourceAccepted | bad
  · have branch := ARX_RecoveryBranch.accepted
      (Rc2_Install.configuration (Rc5_Install.seed err (FIp_Install.select r c)) r)
      (FIo_Install.finishOps r c.finish) (FIo_Nonvacuity.arithmetic_fixed r c.finish).symm ⟨rfl,rfl,rfl⟩
      hashv D words copy allocator CPH_Program.verifier a Q H bound
      (F3Y_InstalledLaws.prepare err (FIp_Install.select r c) r)
      (F3Y_InstalledLaws.primitives err (FIp_Install.select r c) r)
      (F3Y_InstalledLaws.canonical err (FIp_Install.select r c) r)
      (F3Y_InstalledLaws.domain err (FIp_Install.select r c) r)
      (F3Y_InstalledLaws.multiply err (FIp_Install.select r c) r)
      (F3Y_InstalledLaws.one err (FIp_Install.select r c) r)
      (F3Y_InstalledLaws.inverse err (FIp_Install.select r c) r)
      owners (ARQ_Installation.dictionary err c r) represented.2 replies.2.2 sourceAccepted
    rcases branch with ⟨raw,agreement⟩ | collision
    · apply Or.inl
      have physical := ART_PhysicalView.accepted (FIp_Install.configuration err c r) hashv copy allocator CPH_Program.verifier a H represented.1
      exact ⟨raw,ARQ_ModelAccepted.projected D _ H
        (ARQ_ModelAccepted.original _ D hashv (submission a) H sourceAccepted raw),by rw [physical]; exact agreement⟩
    · exact Or.inr collision
  · exact Or.inr bad

#print axioms accepted
end AspisR1.ARX_Join

theorem bounded
    (extraction : S2W_DigestExtraction.AeneasExtraction
      (DrvC_Installation.sdk (FIp_Install.configuration err c r)) native1 native2 hashv)
    (std : S2W_TrustedStd.TrustedStd copy) (D : LiteralByteProtocol.Decoder)
    (words : CanonicalCutEnrichedTranscript.Words)
    (owners : FIp_GlobalOwners.Owners (FIp_Install.configuration err c r)
      (Rc2_Install.configuration (Rc5_Install.seed err (FIp_Install.select r c)) r).boundaries)
    (Q0 Q1 : Nat)
    (queries : ∀ H, (ARW_Induced.issued native1 copy allocator CPH_Program.verifier a H).length ≤ Q0+Q1) :
    F3T_Runtime.deployedFailure (ARX_AdmittedRuntime.runtime (Tx := Tx) L copy allocator c err r native1 native2 hashv a) ≤
      lazyMean (F3T_Charged.projected D (F3T_Canon.canon
        (F3T_ByteCarrier.lift (FIp_Install.configuration err c r) (submission a)))) FS.emptyTable
        RawRecordedOracleExtractionMassBin.failureMass + F3T_Numeric.accountLoss (Q0+Q1) := by
  let cfg := FIp_Install.configuration err c r
  let source := ART_PartialAccount.account cfg hashv copy allocator CPH_Program.verifier a
  let capped := ARX_LocalCharge.bounded source (Q0+Q1)
  let model := F3T_Charged.projected D (F3T_Canon.canon (F3T_ByteCarrier.lift cfg (submission a)))
  let runtime := ARX_AdmittedRuntime.runtime (Tx := Tx) L copy allocator c err r native1 native2 hashv a
  let recorder := Hiq_RecordedJoint.program capped ARQ_Observer.project (ARQ_Observer.select cfg CPH_Program.verifier)
  let event : View Bytes State (View Bytes State (GExq_AccountExperiment.Outcome A)) → Prop := BinRawByteCollisionTargets.event SelectedTranscriptParams.params.iv (Q0+Q1+2965)
  have each : ∀ H, F3T_Runtime.runtimeMass (eval H runtime) ≤
      RawRecordedOracleExtractionMassBin.failureMass (eval H model) +
        AspisV8R19.CausalFirstHitUnionBound.indicator (event (eval H recorder)) := by
    intro H
    by_cases supported : Nonempty (ARU_Support.At (Tx := Tx) L copy allocator c err r native1 native2 hashv a H)
    · obtain ⟨n⟩ := supported
      have budget := ARW_AdmittedBudget.issued L copy allocator c err r native1 native2 hashv extraction a H n Q0 Q1 queries
      have same := ARX_LocalCharge.exact source (Q0+Q1) H budget.2
      have dominate := ARX_Domination.pointwise L copy allocator c err r native1 native2 hashv extraction a H n
        std D words owners (Q0+Q1) budget.1
      change F3T_Runtime.runtimeMass (eval H (ARX_AdmittedRuntime.runtime (Tx := Tx) L copy allocator c err r native1 native2 hashv a)) ≤ _
      rw [ARX_AdmittedRuntime.on_admitted L copy allocator c err r native1 native2 hashv a extraction H n,
        ARU_RuntimeMass.pointwise]
      dsimp only [recorder,capped,event,model]
      rw [ARX_LocalCharge.congr_source _ _ ARQ_Observer.project (ARQ_Observer.select cfg CPH_Program.verifier) H same]
      exact dominate
    · have no := ARX_AdmittedRuntime.outside L copy allocator c err r native1 native2 hashv a extraction H supported
      change F3T_Runtime.runtimeMass (eval H (ARX_AdmittedRuntime.runtime (Tx := Tx) L copy allocator c err r native1 native2 hashv a)) ≤ _
      unfold F3T_Runtime.runtimeMass
      rw [no]
      have positive := F3T_Runtime.nonnegative (eval H model)
      unfold AspisV8R19.CausalFirstHitUnionBound.indicator
      split_ifs <;> linarith
  have average := F3U_Averaging.charge runtime model recorder F3T_Runtime.runtimeMass
    RawRecordedOracleExtractionMassBin.failureMass event each
  have charge := F3T_AccountCharge.bound capped ARQ_Observer.project (ARQ_Observer.select cfg CPH_Program.verifier) (Q0+Q1)
    (fun H => (VerifierCallBudget.firstReads_le_length _ _).trans (ARX_LocalCharge.budget source (Q0+Q1) H))
  exact average.trans (add_le_add (le_refl _) charge)

    (native1 : S2W_ExtractionBoundary.First) (native2 : S2W_ExtractionBoundary.Second)

      (DrvC_Installation.sdk (FIp_Install.configuration err c r)) native1 native2 hashv)
    (a : Adversary A)
    (std : S2W_TrustedStd.TrustedStd copy) (D : LiteralByteProtocol.Decoder)
    (words : CanonicalCutEnrichedTranscript.Words)
    (owners : FIp_GlobalOwners.Owners (FIp_Install.configuration err c r)
      (Rc2_Install.configuration (Rc5_Install.seed err (FIp_Install.select r c)) r).boundaries)
    (Q0 Q1 : Nat)
    (queries : ∀ H, (ARW_Induced.issued native1 copy allocator CPH_Program.verifier a H).length ≤ Q0+Q1) :
    F3T_RuntimeBound.failureModel (ARX_AdmittedRuntime.runtime (Tx := Tx) L copy allocator c err r native1 native2 hashv a) ≤ F3T_Numeric.total (Q0+Q1) := by
  let cfg := FIp_Install.configuration err c r
  have mass := ARX_Mass.bounded (Tx := Tx) L copy allocator c err r native1 native2 hashv a extraction std D words owners Q0 Q1 queries
  have lifted : ∀ H : Bytes → State, (eval H (F3T_ByteCarrier.lift cfg (submission a))).1.length ≤ Q0+Q1 := by
    intro H
    rw [F3T_ByteCarrier.lift_eval]
    exact ARX_SubmissionBudget.bound native1 copy allocator CPH_Program.verifier a (Q0+Q1) queries H
  have model := add_le_add (F3T_ModelClaims.failure_bound D (F3T_ByteCarrier.lift cfg (submission a)) (Q0+Q1) lifted)
    (le_refl (F3T_Numeric.accountLoss (Q0+Q1)))
  rw [F3T_Combined.arithmetic] at model
  unfold F3T_RuntimeBound.failureModel F3T_ModelClaims.failure at *
  linarith
