struct ReaderLog{at:usize,new_at:usize,value:K,err:Option<Error>,calls:usize}
impl ReaderLog{fn read(&mut self,at:usize)->Result<(K,usize),Error>{assert_eq!(at,self.at);self.calls+=1;match self.err{Some(e)=>Err(e),None=>Ok((self.value,self.new_at))}}}
fn weight_body(slot:&mut K,at:&mut usize,log:&mut ReaderLog)->Result<(),Error>{
 let (value,reader_at)=log.read(*at)?;
 *at=reader_at;
 *slot=value;
 Ok(())
}

struct DigestLog<'a>{proof:&'a[u8],public:&'a[u8],checkpoint:&'a[u8],replies:[[u8;32];3],calls:usize}
impl DigestLog<'_>{fn digest(&mut self,parts:&[&[u8]])->[u8;32]{
 let n=self.calls;assert!(n<3);self.calls+=1;
 if n==0{assert_eq!(parts.len(),1);assert_eq!(parts[0].as_ptr(),self.proof.as_ptr());assert_eq!(parts[0].len(),self.proof.len());}
 else if n==1{assert_eq!(parts.len(),1);assert_eq!(parts[0].as_ptr(),self.public.as_ptr());assert_eq!(parts[0].len(),self.public.len());}
 else{assert_eq!(parts.len(),2);assert_eq!(parts[0],b"UNPROVED:R1:checkpoint");assert_eq!(parts[1].as_ptr(),self.checkpoint.as_ptr());assert_eq!(parts[1].len(),self.checkpoint.len()-32);}
 let mut reply=self.replies[n];if cfg!(NJJ_invert){reply[0]^=1;}reply
}}
// Exact frozen resume checksum conditional, CHILD digest calls as local oracles.
fn checksum_cut(bytes:&[u8],proof:&[u8],public:&[u8],log:&mut DigestLog)->Result<(),Error>{

 Ok(())
}
