fn NBH_live_iteration(frontier: &[Digest;20], mut level: usize) -> (usize, Option<PoolV1PairLiveSnapshotErrorV1>) {

        if !NBH_digest(&frontier[level]) {

}

fn NBH_live_prefix(

) -> Option<PoolV1PairLiveSnapshotErrorV1> {

        return Some(PoolV1PairLiveSnapshotErrorV1::SequenceIndexMismatch);

        return Some(PoolV1PairLiveSnapshotErrorV1::TreeFull);
    }
    if !NBH_digest(&snapshot.current_root) {
        return Some(PoolV1PairLiveSnapshotErrorV1::NonCanonicalDigest);
    }
 None}
