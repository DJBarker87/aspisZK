fn KMX_LPn_iterator(){
 let b:[u8;3717]=kani::any();
 let len:usize=kani::any();kani::assume(5<=len&&len<=3717);
 let record=ValidatedRecord{row:25,framed:&b[..len]};
 let n:usize=kani::any();kani::assume(n==1||n==28||n==29);
 let off:usize=kani::any();kani::assume(off<=116&&32*(off+n)<=len-5);
 let i:usize=kani::any();kani::assume(i<=n);
 let payload=record.payload();assert!(core::ptr::eq(payload.as_ptr(),b[5..].as_ptr()));
 let selected=&payload[32*off..32*(off+n)];
 let mut it=selected.chunks_exact(32).enumerate();
 if i>0 {let (j,_)=it.nth(i-1).unwrap();assert_eq!(j,i-1);}
 assert_eq!(it.len(),n-i);
 match it.next(){None=>assert_eq!(i,n),Some((value,part))=>{
  assert!(i<n);assert_eq!(value,i);assert_eq!(part.len(),32);
  assert!(core::ptr::eq(part.as_ptr(),b[5+32*(off+i)..].as_ptr()));
  assert_eq!(it.len(),n-i-1);assert!(off+value<116);
 }}
}

fn KMX_LPn_decoder(){
 let b:[u8;32]=kani::any();let row:u8=kani::any();
 let off:usize=kani::any();let i:usize=kani::any();kani::assume(off<=116&&i<29);kani::assume(off+i<116);
 let actual=QM31::from_le_bytes(&b[..16]).ok_or(Error::NonCanonical{row,value:off+i});
 let words=[le(&b[0..4]),le(&b[4..8]),le(&b[8..12]),le(&b[12..16])];
 let good=words.iter().all(|w|*w<2147483647);
 match actual { Ok(q)=>{assert!(good);assert_eq!(q.c0.a.0 as u64,words[0]);assert_eq!(q.c0.b.0 as u64,words[1]);assert_eq!(q.c1.a.0 as u64,words[2]);assert_eq!(q.c1.b.0 as u64,words[3]);},
 Err(e)=>{assert!(!good);assert_eq!(e,Error::NonCanonical{row,value:off+i});}}
}

fn KMX_LPn_guard(){

 let b:[u8;37]=kani::any();
 // This actual N29 call can only reject on this small carrier. It checks
 // all offset overflow cases and row precedence without any source iteration.
 let got=ValidatedRecord{row,framed:&b}.decode_k::<29>(off);
 assert_eq!(got,Err(if row<14{Error::InvalidRow}else{Error::Length{row}}));
}

 #[kani::unwind(70)]
fn KMX_LPn_scalar(){
 let b:[u8;3717]=kani::any();let len:usize=kani::any();kani::assume(5<=len&&len<=3717);
 let row:u8=kani::any();let off:usize=kani::any();let n:usize=kani::any();kani::assume(n==1||n==28||n==29);
 let native=(||->Result<(),Error>{
  if row<14{return Err(Error::InvalidRow);}
  let byte_offset=off.checked_mul(32).ok_or(Error::Length{row})?;
  let byte_len=n.checked_mul(32).ok_or(Error::Length{row})?;
  let end=byte_offset.checked_add(byte_len).ok_or(Error::Length{row})?;
  let _bytes=ValidatedRecord{row,framed:&b[..len]}.payload().get(byte_offset..end).ok_or(Error::Length{row})?;
  Ok(())
 })();
 let fit=(off as u128)*32+(n as u128)*32<=(len-5) as u128;
 assert_eq!(native,if row<14{Err(Error::InvalidRow)}else if fit{Ok(())}else{Err(Error::Length{row})});
}
fn arbitrary_q()->QM31{
 use aspis_core::field::{CM31,M31};
 let a:u32=kani::any();let b:u32=kani::any();let c:u32=kani::any();let d:u32=kani::any();
 // No canonicality restriction: only copying/storing the valid u32 representation.
 QM31{c0:CM31{a:M31(a),b:M31(b)},c1:CM31{a:M31(c),b:M31(d)}}
}

fn KMX_LPn_typed_write(){
 let b:[u8;37]=kani::any();let row:u8=kani::any();kani::assume(row>=14);
 let mut out=[arbitrary_q(),arbitrary_q(),arbitrary_q()];
 let j:usize=kani::any();kani::assume(j<3);let old=out[j];
 let got=ValidatedRecord{row,framed:&b}.decode_k::<1>(0);
 let expected=QM31::from_le_bytes(&b[5..21]);
 match(got,expected){
  (Ok([q]),Some(v))=>{out[1]=q;eq_q(q,v);eq_q(out[j],if j==1{v}else{old});},
  (Err(e),None)=>{assert_eq!(e,Error::NonCanonical{row,value:0});eq_q(out[j],old);},
  _=>panic!("typed write source decoder disagreement")
 }
}

 let records:[&[u8];27]=[&b[0..], &b[1..], &b[2..], &b[3..], &b[4..], &b[5..], &b[6..], &b[7..], &b[8..], &b[9..], &b[10..], &b[11..], &b[12..], &b[13..], &b[14..], &b[15..], &b[16..], &b[17..], &b[18..], &b[19..], &b[20..], &b[21..], &b[22..], &b[23..], &b[24..], &b[25..], &b[26..]];
 let p=Prefix{c1_root:&[0u8;32],records};
 match p.validated_record(row){
  Err(e)=>{assert!(row>=27);assert_eq!(e,Error::InvalidRow);},
  Ok(t)=>{assert!(row<27);assert_eq!(t.row(),row);assert_eq!(t.framed().len(),records[row as usize].len());assert!(core::ptr::eq(t.framed().as_ptr(),records[row as usize].as_ptr()));}
 }
}

fn KMX_LPn_nonvacuity(){
 let mut b=[0u8;37];let high:[u8;16]=kani::any();b[21..].copy_from_slice(&high);
 let got=ValidatedRecord{row:25,framed:&b}.decode_k::<1>(0).unwrap();eq_q(got[0],QM31::ZERO);
 b[5]=255;b[6]=255;b[7]=255;b[8]=127;
 assert_eq!(ValidatedRecord{row:25,framed:&b}.decode_k::<1>(0),Err(Error::NonCanonical{row:25,value:0}));
 assert_eq!(ValidatedRecord{row:0,framed:&b}.decode_k::<1>(usize::MAX),Err(Error::InvalidRow));
 assert_eq!(ValidatedRecord{row:25,framed:&b}.decode_k::<1>(usize::MAX),Err(Error::Length{row:25}));
}
