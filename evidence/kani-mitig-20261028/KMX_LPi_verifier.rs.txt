fn any_k()->K {let w:[u32;4]=kani::any();K{c0:CM31{a:M31(w[0]),b:M31(w[1])},c1:CM31{a:M31(w[2]),b:M31(w[3])}}}

fn kwords(x:K)->[u32;4] {[x.c0.a.0,x.c0.b.0,x.c1.a.0,x.c1.b.0]}
fn check_k(len:usize) {
 assert!(len<=1720);let x=any_k();let words=kwords(x);
 let initial:[u8;1736]=kani::any();let mut out=initial[..len].to_vec();
 put_k(&mut out,x);
 assert_eq!(out.len(),len+16);
 let at:usize=kani::any();kani::assume(at<out.len());
 assert_eq!(out[at],if at<len{initial[at]}else{let j=at-len;(words[j/4]>>((j%4)*8)) as u8});
 kani::cover!(at==len+15);
}
fn check_e(len:usize) {

}

// Joined source cuts: the serializer's whole-body association stays explicit/open.
#[derive(Clone,Copy)]
struct JoinedOracle<'a>{count:&'a core::cell::Cell<usize>,proof:&'a[u8],public:&'a[u8],body:&'a[u8;1736],selected:usize,answers:[[u8;32];3]}
impl Digest32 for JoinedOracle<'_>{
 fn digest(&self,parts:&[&[u8]])->[u8;32]{
  let n=self.count.get();assert!(n<3);
  match n {
   0=>{assert_eq!(parts.len(),1);assert_eq!(parts[0].len(),self.proof.len());assert_eq!(parts[0].as_ptr(),self.proof.as_ptr());}
   1=>{assert_eq!(parts.len(),1);assert_eq!(parts[0].len(),self.public.len());assert_eq!(parts[0].as_ptr(),self.public.as_ptr());}
   _=>{assert_eq!(parts.len(),2);assert_eq!(parts[0],b"UNPROVED:R1:checkpoint");assert_eq!(parts[1].len(),1736);assert_eq!(parts[1][self.selected],joined_byte(self.body,&self.answers,self.selected));}
  }
  self.count.set(n+1);self.answers[n]
 }
}
fn joined_byte(body:&[u8;1736],answers:&[[u8;32];3],at:usize)->u8 {
 if (16..48).contains(&at){answers[0][at-16]}
 else if (48..80).contains(&at){answers[1][at-48]}
 else if at<1736{body[at]}else{answers[2][at-1736]}
}
fn joined_digest_cut<const PROOF:usize,const PUBLIC:usize>() {
 let proof:[u8;PROOF]=kani::any();let public:[u8;PUBLIC]=kani::any();
 let body:[u8;1736]=kani::any();let answers:[[u8;32];3]=kani::any();
 let selected:usize=kani::any();kani::assume(selected<1736);
 let at:usize=kani::any();kani::assume(at<1768);
 let count=core::cell::Cell::new(0);let hash=JoinedOracle{count:&count,proof:&proof,public:&public,body:&body,selected,answers};
 let hp=hash.digest(&[&proof]);let hu=hash.digest(&[&public]);
 let mut out=Vec::from(body);out[16..48].copy_from_slice(&hp);out[48..80].copy_from_slice(&hu);
 let sum=hash.digest(&[b"UNPROVED:R1:checkpoint",&out]);out.extend_from_slice(&sum);
 assert_eq!(count.get(),3);assert_eq!(out.len(),1768);assert_eq!(out[at],joined_byte(&body,&answers,at));
 kani::cover!(at==1767);kani::cover!(selected==1735);
}
