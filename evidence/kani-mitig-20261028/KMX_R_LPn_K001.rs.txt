#[kani::unwind(33)]
fn KMX_LPn_step_frame() {
    let source: [u8;16] = kani::any();
    let mut backing: [u8;96] = kani::any();
    let slot: usize = kani::any(); kani::assume(slot < 3);
    let at = 32 * slot;
    let end: usize = kani::any(); kani::assume(at+32 <= end && end <= 96);
    let j: usize = kani::any(); kani::assume(j < end);
    let old = backing[j];
    let base = backing.as_mut_ptr();
    let window = &mut backing[..end][at..at+32];
    assert!(core::ptr::eq(window.as_ptr(),base.wrapping_add(at)));
    assert_eq!(window.len(),32);
    assert_eq!(expand_values(&source,window),Ok(()));
    assert_eq!(backing[j],if at<=j && j<at+16 { source[j-at] }
        else if at+16<=j && j<at+32 { 0 } else { old });
    assert!(core::ptr::eq(backing.as_ptr(),base));
}
