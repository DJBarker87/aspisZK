 #[kani::unwind(70)] fn KMX_LPg_check_step(){
 let b:[u8;16]=kani::any();let prior:bool=kani::any();
 let actual=if prior{check_values(&b)}else{Err(CodecError::NonCanonical)};
 let expect=if prior&&QM31::from_le_bytes(&b).is_some(){Ok(())}else{Err(CodecError::NonCanonical)};
 assert_eq!(actual,expect);
}
fn expansion<const LO:usize,const HI:usize>(){
 let b:[u8;16]=kani::any();let mut out:[u8;32]=kani::any();let old=out;
 let lane:usize=kani::any();kani::assume(lane>=LO&&lane<HI);
 assert_eq!(expand_values(&b,&mut out),Ok(()));
 assert_eq!(out[lane],if lane<16{b[lane]}else{0});let _=old;
}
