#[kani::unwind(1769)]
fn append32(){check_append::<32>();}
fn check_append<const N:usize>() {
 let initial:[u8;1736]=kani::any();let value:[u8;N]=kani::any();
 let len:usize=kani::any();kani::assume(len<=1736-N);
 let mut out=initial[..len].to_vec();out.extend_from_slice(&value);
 assert_eq!(out.len(),len+N);
 let at:usize=kani::any();kani::assume(at<out.len());
 assert_eq!(out[at],if at<len{initial[at]}else{value[at-len]});
 kani::cover!(len==0);kani::cover!(len==1736-N);kani::cover!(at==len+N-1);
}
