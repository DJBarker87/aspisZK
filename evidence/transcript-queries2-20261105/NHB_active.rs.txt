0 => NHB_contains_0(a,n,x),
1 => NHB_contains_1(a,n,x),
2 => NHB_contains_2(a,n,x),
3 => NHB_contains_3(a,n,x),
4 => NHB_contains_4(a,n,x),
5 => NHB_contains_5(a,n,x),
6 => NHB_contains_6(a,n,x),
7 => NHB_contains_7(*a,n,x),
8 => NHB_contains_8(*a,n,x),
9 => NHB_contains_9(*a,n,x),
10 => NHB_contains_10(*a,n,x),
11 => NHB_contains_11(*a,n,x),
12 => NHB_contains_12(*a,n,x),
13 => NHB_contains_13(*a,n,x),
14 => NHB_contains_14(*a,n,x),
15 => NHB_contains_15(*a,n,x),
16 => NHB_contains_16(*a,n,x),
17 => NHB_contains_17(*a,n,x),
18 => NHB_contains_18(*a,n,x),
19 => NHB_contains_19(*a,n,x),
20 => NHB_contains_20(*a,n,x),
21 => NHB_contains_21(*a,n,x),
22 => NHB_contains_22(*a,n,x),
_ => unreachable!(),
}}
impl QueryScan {
 fn NHB_word(&mut self,candidate:u32)->bool {
            if self.accepted_len == QUERY_COUNT || self.draws == QUERY_MAX_DRAWS {
                return true;
            }
            self.draws += 1;
            if !NHB_member(&self.accepted,self.accepted_len,candidate) {
                self.accepted[self.accepted_len] = candidate;
                self.accepted_len += 1;
            }
 false
 }
 fn NHB_cursor_word(&mut self,block:&[u8;32],cursor:&mut usize)->(bool,u32) {
  let words=query_words(block);
  let mut iterator=words.into_iter();

 }
