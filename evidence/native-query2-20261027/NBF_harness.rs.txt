static mut NBF_PTR: *mut [u64;4] = core::ptr::null_mut();
static mut NBF_POWER: [u32;4] = [0;4];
static mut NBF_BASIS:u32=0;
static mut NBF_FLUSH:bool=false;
static mut NBF_CALLS:usize=0;
// Context-only overapproximation. No row contents are read or modified here.
// NBB_Row.source supplies actual row semantics on its phase/canonical domain.
fn NBF_context(row:&mut[u64;4],power:K,basis:M31,flush:bool) {
 unsafe {NBF_PTR=row;NBF_POWER=[power.c0.a.0,power.c0.b.0,power.c1.a.0,power.c1.b.0];NBF_BASIS=basis.0;NBF_FLUSH=flush;NBF_CALLS+=1;}
}
#[kani::proof]
#[kani::stub(super::update_residue_row,NBF_context)]
fn NBF_index_context() {
 let backing:Box<[[u64;4];64]>=Box::new(kani::any());
 let slice:Box<[[u64;4]]>=backing;
 let raw=Vec::from(slice);
 let index:usize=kani::any();kani::assume(index<64);
 let terms:usize=kani::any();let p:[u32;4]=kani::any();let basis:u32=kani::any();
 let power=K{c0:CM31{a:M31(p[0]),b:M31(p[1])},c1:CM31{a:M31(p[2]),b:M31(p[3])}};
 let ptr=raw.as_ptr();let cap=raw.capacity();
 let mut weights=BatchedWeights{raw,terms};weights.push(index,power,M31(basis));
 assert_eq!(weights.raw.as_ptr(),ptr);assert_eq!(weights.raw.len(),64);assert_eq!(weights.raw.capacity(),cap);assert_eq!(weights.terms,terms);
 unsafe {assert_eq!(NBF_PTR,ptr.add(index) as *mut [u64;4]);assert_eq!(NBF_POWER,p);assert_eq!(NBF_BASIS,basis);assert_eq!(NBF_FLUSH,terms&3==3);assert_eq!(NBF_CALLS,1);}
 kani::cover!(index==0);kani::cover!(index==63);kani::cover!(terms&3==3);
}

static mut NBF_REPLY:[u64;4]=[0;4];
// One selected row write; every returned row is considered. No field multiply.
fn NBF_write(row:&mut[u64;4],_power:K,_basis:M31,_flush:bool) {
 let reply:[u64;4]=kani::any();unsafe {NBF_REPLY=reply;} *row=reply;
}
#[kani::proof]

// Native release row-loop iteration, factored verbatim at its loop boundary.
// The Vec push remains actual except its row leaf; no complete loop executes.
fn NBF_row_iteration(a:&Accumulator,weights:&mut BatchedWeights,basis:&[M31;256],j:usize) {
 let width=a.weights.len();
 weights.push(j,a.power,basis_at(basis,j));
 crate::trace::value(12,a.next*width+j,E::from_qm31(accumulated_weight_at(&a.weights,j)));
 crate::trace::value(13,a.next*width+j,E::from_m31(basis_at(basis,j)));
}
fn NBF_arbitrary_k()->K {
 let p:[u32;4]=kani::any();K{c0:CM31{a:M31(p[0]),b:M31(p[1])},c1:CM31{a:M31(p[2]),b:M31(p[3])}}
}
// Field carriers have no invalid bit patterns; map initialization is irrelevant
// to the checked iteration and is bounded explicitly.
#[kani::proof]
#[kani::unwind(257)]
#[kani::stub(super::update_residue_row,NBF_context)]
fn NBF_row_iteration_context() {
 let backing:Box<[[u64;4];64]>=Box::new(kani::any());
 let slice:Box<[[u64;4]]>=backing;let raw=Vec::from(slice);
 let mut basis=[M31::ZERO;256];
 let j:usize=kani::any();kani::assume(j<64);
 let b:u32=kani::any();basis[j]=M31(b);
 // Context gate observes no unselected weight. Arbitrary selected value and
 // arbitrary row contents; Vec width is the SAME native width used by traces.
 let mut output=alloc::vec![K::ZERO;64];output[j]=NBF_arbitrary_k();
 let next:usize=kani::any();kani::assume(next<10);

 assert_eq!(pending.terms,terms);assert_eq!(pending.raw.len(),64);
 assert_eq!(a.next,next);assert_eq!(a.weights.len(),64);
 kani::cover!(j==0);kani::cover!(j==63);kani::cover!(next==9);
}
