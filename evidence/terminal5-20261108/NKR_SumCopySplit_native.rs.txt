fn NKR_copy_split_endpoints(scratch:&mut[QM31],selectors:&Selectors,mut link_index:usize,weight:M31,l:&mut NKR_CopyLog)->usize{

}

fn NKR_copy_endpoints_split_cut(){
 let mut scratch:[QM31;103]=unsafe{core::mem::transmute::<[u32;412],[QM31;103]>(kani::any())};
 let selectors=Selectors{high:unsafe{core::mem::transmute::<[u32;256],[QM31;64]>(kani::any())},low:unsafe{core::mem::transmute::<[u32;64],[QM31;16]>(kani::any())}};
 let replies:[QM31;4]=unsafe{core::mem::transmute::<[u32;16],[QM31;4]>(kani::any())};let wr:u32=kani::any();let append:u64=kani::any();let private:bool=kani::any();let variant=if private{PoolV1PairForestCompiledVariantV1::PrivateTransfer}else{PoolV1PairForestCompiledVariantV1::Withdrawal};
 let index:usize=kani::any();let observe:usize=kani::any();let high_pick:usize=kani::any();let low_pick:usize=kani::any();let limb:usize=kani::any();kani::assume(index<136&&observe<103&&high_pick<64&&low_pick<16&&limb<4);
 let link=constants::COPY_LINKS[index];let p=link.producer;let c=link.consumer;
 let pg=usize::from(p.slot);let cg=2+usize::from(c.slot);
 let pw=COPY_SELECTOR_TENSOR_WEIGHT_OFFSET+usize::from(COPY_GROUP_LOCAL_COORDINATES[pg][usize::from(p.row)&15]);let pp=COPY_SELECTOR_TENSOR_PATTERN_OFFSET+usize::from(COPY_PATTERN_LOCAL_COORDINATES[pg][usize::from(p.pattern)][usize::from(p.row)&15]);
 let cw=COPY_SELECTOR_TENSOR_WEIGHT_OFFSET+usize::from(COPY_GROUP_LOCAL_COORDINATES[cg][usize::from(c.row)&15]);let cp=COPY_SELECTOR_TENSOR_PATTERN_OFFSET+usize::from(COPY_PATTERN_LOCAL_COORDINATES[cg][usize::from(c.pattern)][usize::from(c.row)&15]);
 let saved=scratch[observe];let middle=if observe==pp{replies[1]}else if observe==pw{replies[0]}else{saved};let after=if observe==cp{replies[3]}else if observe==cw{replies[2]}else{middle};
 let mut l=NKR_CopyLog{observe,high_pick,low_pick,calls:1,kind:[0,255,255],index:0,variant,tag:0,weight_kind:0,weight_level:0,weight_prow:0,weight_crow:0,weight_pslot:0,weight_cslot:0,weight_ppattern:0,weight_cpattern:0,side:[999;2],row:[0;2],slot:[0;2],pattern:[0;2],passed_tag:[0;2],passed_weight:[0;2],passed_scratch:[QM31::ZERO;2],high:[QM31::ZERO;2],low:[QM31::ZERO;2],weight_reply:M31(wr),replies};
 let next=NKR_copy_split_endpoints(&mut scratch,&selectors,index,M31(wr),&mut l);
 assert_eq!(l.calls,3);assert_eq!(l.kind,[0,1,1]);assert_eq!(next,index+1);
 assert_eq!(l.side,[0,2]);assert_eq!(l.row,[p.row,c.row]);assert_eq!(l.slot,[p.slot,c.slot]);assert_eq!(l.pattern,[p.pattern,c.pattern]);assert_eq!(l.passed_tag,[link.tag;2]);assert_eq!(l.passed_weight,[wr;2]);
 assert_eq!(NKR_copy_words(l.passed_scratch[0])[limb],NKR_copy_words(saved)[limb]);assert_eq!(NKR_copy_words(l.passed_scratch[1])[limb],NKR_copy_words(middle)[limb]);
 let call:usize=kani::any();kani::assume(call<2);assert_eq!(NKR_copy_words(l.high[call])[limb],NKR_copy_words(selectors.high[high_pick])[limb]);assert_eq!(NKR_copy_words(l.low[call])[limb],NKR_copy_words(selectors.low[low_pick])[limb]);

}
