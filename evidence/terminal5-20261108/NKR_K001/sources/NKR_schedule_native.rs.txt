fn NKR_initial(p:SemanticPublic<'_>,o:&StateOnlyPoseidonOpenings,s:&Selectors,l:&mut NKR_ScheduleLog)->([QM31;16],[QM31;16]){

fn NKR_preweighted(packed:&mut[QM31;24],start:usize,values:&[QM31;16],l:&mut NKR_ScheduleLog){let k=l.calls-1;assert!(k<2);l.starts[k]=start;l.before[k]=NKR_words(packed[l.pick])[l.limb];l.values[k]=NKR_words(values[l.pick%16])[l.limb];*packed=if k==0{l.middle}else{l.out};l.after[k]=NKR_words(packed[l.pick])[l.limb];l.calls+=1;}
fn NKR_schedule(packed:&mut[QM31;24],public:SemanticPublic<'_>,openings:&StateOnlyPoseidonOpenings,selectors:&Selectors,l:&mut NKR_ScheduleLog){

    NKR_preweighted(packed, 0, &initial,l);
    NKR_preweighted(packed, 16, &absorption,l);
}
