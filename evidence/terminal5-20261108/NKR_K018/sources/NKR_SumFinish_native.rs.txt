fn NKR_finish_words(q:QM31)->[u32;4]{[q.c0.a.0,q.c0.b.0,q.c1.a.0,q.c1.b.0]}
struct NKR_FinishLog{seek:usize,limb:usize,tag_calls:usize,tag_at:usize,tag_scratch:usize,tag_selectors:usize,high_pick:usize,low_pick:usize,scratch_pick:usize,tag_high:QM31,tag_low:QM31,tag_scratch_value:QM31,tag_replies:[QM31;4],calls:usize,kind:[u8;4],left:[QM31;4],right:[QM31;4],arity:[usize;4],op_replies:[QM31;4]}
fn NKR_finish_tag(scratch:&[QM31],coordinate:usize,selectors:&Selectors,l:&mut NKR_FinishLog)->QM31{
 let i=l.tag_calls;l.tag_calls+=1;if i==l.seek{l.tag_at=coordinate;l.tag_scratch=scratch.as_ptr()as usize;l.tag_selectors=selectors as*const Selectors as usize;l.tag_high=selectors.high[l.high_pick];l.tag_low=selectors.low[l.low_pick];l.tag_scratch_value=scratch[l.scratch_pick];}l.tag_replies[i]
}
fn NKR_finish_sum(left:[QM31;4],right:[QM31;4],count:usize,l:&mut NKR_FinishLog)->QM31{
 let i=l.calls;l.calls+=1;l.kind[i]=0;l.left[i]=left[l.seek];l.right[i]=right[l.seek];l.arity[i]=count;l.op_replies[i]
}
fn NKR_finish_add(left:QM31,right:QM31,l:&mut NKR_FinishLog)->QM31{
 let i=l.calls;l.calls+=1;l.kind[i]=1;l.left[i]=left;l.right[i]=right;l.op_replies[i]
}
fn NKR_finish_batch(scratch:&[QM31],selectors:&Selectors,mut values:[QM31;4],mut weights:[QM31;4],group:usize,mut coordinate:usize,l:&mut NKR_FinishLog)->([QM31;4],[QM31;4],usize){
 let end=COPY_GROUP_LOCAL_GROUP_OFFSETS[group+1];
 let count = core::cmp::min(4, end - coordinate);
 let tag_values: [QM31; 4] = core::array::from_fn(|index| {
     if index < count {
         NKR_finish_tag(scratch, coordinate + index, selectors,l)
     } else {
         QM31::ZERO
     }
 });
 let weight_values: [QM31; 4] = core::array::from_fn(|index| {
     if index < count {
         scratch[COPY_SELECTOR_TENSOR_WEIGHT_OFFSET + coordinate + index]
     } else {
         QM31::ZERO
     }
 });
 let lows: [QM31; 4] = core::array::from_fn(|index| {
     if index < count {
         selectors.low
             [usize::from(COPY_GROUP_LOCAL_COORDINATE_LOCALS[coordinate + index])]
     } else {
         QM31::ZERO
     }
 });
 values[group] =
     NKR_finish_add(values[group],NKR_finish_sum(tag_values, lows, count,l),l);
 weights[group] =
     NKR_finish_add(weights[group],NKR_finish_sum(weight_values, lows, count,l),l);
 coordinate += count;
 (values,weights,coordinate)
}
