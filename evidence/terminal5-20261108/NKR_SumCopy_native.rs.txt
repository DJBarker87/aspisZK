fn NKR_copy_link(scratch:&mut[QM31],selectors:&Selectors,append_index:u64,variant:PoolV1PairForestCompiledVariantV1,mut link_index:usize,l:&mut NKR_CopyLog)->usize{
 let link = constants::COPY_LINKS[link_index];
 let weight = NKR_copy_weight(link, append_index, variant,l);
 NKR_copy_endpoint(scratch,0,link.producer,link.tag,weight,selectors,l);
