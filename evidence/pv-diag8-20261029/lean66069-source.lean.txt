import PV_DIAG8_Messages
set_option autoImplicit false
namespace PV_DIAG8
/-! Sufficient request-discipline criterion for V400, not an instance of it.
Requests retain operation, route/name, offset and size. Payload is separate:
byte blindness must be proved for the deployed filesystem/allocator world. -/
structure Request where
 op : Nat
 route : List Nat
 offset : Nat
 size : Nat
 deriving DecidableEq
structure Reply where
 failed : Bool
 code : Nat
 count : Nat
 deriving DecidableEq
abbrev History := List (Request × Reply)
def replay (world : History → Request → Reply) : History → List Request → History
 | h, [] => h
 | h, q::qs => replay world (h ++ [(q,world h q)]) qs
def failures (h : History) : List Bool := h.map (fun x => x.2.failed)
def appendReq (h : History) (q : Request) (r : Reply) : History := h ++ [(q,r)]

theorem world_empty (world : History → Request → Reply) (h : History) : replay world h [] = h := rfl
#print axioms world_empty
theorem world_cons (world : History → Request → Reply) (h : History) (q : Request) (qs : List Request) : replay world h (q::qs) = replay world (appendReq h q (world h q)) qs := rfl
#print axioms world_cons
end PV_DIAG8
