/-! Existing extracted array-index theorem, specialized to reached fixed sizes.
These are index laws, not a claim that every native caller supplies the bound. -/

theorem array_no_bounds_failure {α : Type} {n : Usize} (a : Aeneas.Std.Array α n) (i : Usize) (h : i.val < n.val) : Aeneas.Std.Array.index_usize a i ≠ .fail .arrayOutOfBounds := by rw [R1Z.ProverCommitArrayIndex25.array_index_usize_ok a i h]; intro bad; cases bad
#print axioms array_no_bounds_failure
theorem array_4_safe {α : Type} (a : Aeneas.Std.Array α 4#usize) (i : Usize) (h : i.val < 4) : Aeneas.Std.Array.index_usize a i ≠ .fail .arrayOutOfBounds := array_no_bounds_failure a i (by simpa using h)
#print axioms array_4_safe
