import PV_DIAG8_World
set_option autoImplicit false
namespace PV_DIAG8
/-! Error-path capture requires the live stack at the capture point, not only
public formatted text or public backtrace configuration. No PCs are put into
option B's environment-failure input by this model. -/
inductive PathStep
 | enter (frame : Nat)
 | leave
 | capture (enabled : Bool)
 deriving DecidableEq
def stackStep (stack : List Nat) : PathStep → List Nat
 | .enter f => f::stack
 | .leave => stack.tail
 | .capture _ => stack
def captures (stack : List Nat) : List PathStep → List (List Nat)
 | [] => []
 | s::rest => (match s with | .capture true => [stack] | _ => []) ++ captures (stackStep stack s) rest

theorem frames_no_capture_empty (s : List Nat) : captures s []=[] := rfl
#print axioms frames_no_capture_empty
theorem frames_capture_here (s : List Nat) (rest : List PathStep) : captures s (.capture true::rest) = s::captures s rest := rfl
#print axioms frames_capture_here
end PV_DIAG8
