use super::*;
static mut NDM_PTR:*const u8=core::ptr::null();
static mut NDM_LEN:usize=0;
pub fn copy_cut(_account:&AccountInfo<'_>,data:&[u8],length1:usize,length2:usize) {
 let good=super::super::NDM_native::reference::snapshot_select_ref(data.len(),length1,length2)
  && unsafe{data.as_ptr()==NDM_PTR && data.len()==NDM_LEN};
 assert_eq!(good,!cfg!(NDM_negative));
 if !data.is_empty(){let k:usize=kani::any();kani::assume(k<data.len());
  let same=unsafe{data[k]==*NDM_PTR.add(k)};assert_eq!(same,!cfg!(NDM_negative));}
 // Observe the SAME to_vec receiver; allocation and the copy are separate obligations.
 kani::assume(false);
}
