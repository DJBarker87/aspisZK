pub fn common_ref<K,D,L,C>(master:K,checkpoint:K,lane:K,output:u8,sequence:u64,anchor:D,live:L,candidate:C)->(K,K,K,u8,u64,D,L,C) {
 (master,checkpoint,lane,output,sequence,anchor,live,candidate)
}
pub fn zero_tail_ref(byte:u8,empty:bool)->u8 { if empty {0} else if byte!=0 {1} else {2} }

pub fn identity_ref<T>(value:T)->T {value}

pub fn snapshot_select_ref(length:usize,limit1:usize,limit2:usize)->bool {length==limit1 || length==limit2}
pub fn snapshot_value_ref<K,B>(metadata:(K,K,bool,bool,bool),length:usize,admitted:bool,bytes:B)->((K,K,bool,bool,bool),usize,Option<B>) {
 (metadata,length,if admitted {Some(bytes)} else {None})
}
pub fn failed_snapshot_ref<T,E>(error:E)->Result<T,E> {Err(error)}
pub fn validation_ref<C,P>(common:C,public:P,valid:bool)->Result<(C,P),u8> {
 if valid {Ok((common,public))} else {Err(0)}
}

pub fn byte_word_ref(value:u8)->u64 {value as u64}
pub fn field_word_ref(value:u32)->u64 {value as u64}
pub fn next_gate_ref(live:u64,next:u64)->bool {
 match live.checked_add(1) {Some(incremented)=>incremented==next,None=>false}
}
pub fn reconstruction_gate_ref(live:u64,next:u64,asset:u32,requested:u32)->u8 {
 if !next_gate_ref(live,next) {1} else if asset!=requested {2} else {0}
}
