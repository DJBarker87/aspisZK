theorem zero_scan (n : Nat) : scan predicate true (List.replicate n 0#u32) 1#u32 = ok (false,n) := by
  induction n with
  | zero => rfl
  | succ n ih =>
    rw [List.replicate_succ,scan,zero_call]
    simp only [bind_tc_ok,Bool.false_eq_true,if_false]
    rw [ih]
    rfl
#print axioms zero_scan

  change iterScan predicate true ⟨Array.to_slice zero,0⟩ 1#u32 = _

             ok (r.1,{slice := Array.to_slice zero,i := 0+r.2})) = _
