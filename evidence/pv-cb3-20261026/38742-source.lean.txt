import R1Z.PV_CB3_MaskCast
set_option autoImplicit false
namespace R1Z.PV_CB3_MaskShift
open Aeneas Aeneas.Std

theorem four_mod : 4 % UScalarTy.numBits .Usize = 4 := by
  rcases System.Platform.numBits_eq with h|h <;>
    simp only [UScalarTy.numBits,h] <;> rfl

theorem block_value (row : Usize) :
    (Usize.wrapping_shr row 4#u32).val = row.val / 16 := by
  change (row.bv.ushiftRight (4 % UScalarTy.numBits .Usize)).toNat = _
  rw [four_mod,BitVec.toNat_ushiftRight,Nat.shiftRight_eq_div_pow]
  rfl

theorem in_bounds (row : Usize) :
    (Usize.wrapping_shr row 4#u32).val < 64 ↔ row.val < 1024 := by
  rw [block_value]
  omega

theorem out_of_bounds (row : Usize) (h : 1024 ≤ row.val) :
    64 ≤ (Usize.wrapping_shr row 4#u32).val := by
  rw [block_value]
  omega

theorem bit_value (row : Usize) :
    (U16.wrapping_shl 1#u16 (UScalar.cast .U32 (PV_CB3_MaskCast.low row))).val =
      2^(PV_CB3_MaskCast.low row).val := by
  change ((1#u16).bv.shiftLeft ((UScalar.cast .U32 (PV_CB3_MaskCast.low row)).val % 16)).toNat = _
  rw [PV_CB3_MaskCast.cast_value,Nat.mod_eq_of_lt (PV_CB3_MaskCast.low_bound row),
    BitVec.toNat_shiftLeft,Nat.shiftLeft_eq]
  have hp : 2^(PV_CB3_MaskCast.low row).val < 2^16 :=
    Nat.pow_lt_pow_right (by decide) (PV_CB3_MaskCast.low_bound row)
  simpa only [Nat.one_mul,Nat.mod_eq_of_lt hp]
