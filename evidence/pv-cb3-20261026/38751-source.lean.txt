theorem slice_length {T : Type} (xs : Slice T) : (Slice.len xs).val = xs.val.length := by
  rfl

theorem slice_read {T : Type} (xs : Slice T) (i : Usize) (h : i < Slice.len xs) :
    Slice.index_usize xs i = ok (xs.val[i.val]'(by
      have hi : i.val < (Slice.len xs).val := h
      simpa only [slice_length] using hi)) := by
  change (match xs.val[i.val]? with | none => _ | some x => _) = _
  rw [List.getElem?_eq_getElem]

theorem increment (row : Usize) (h : row.val < 1024) :
    (Usize.wrapping_add row 1#usize).val = row.val+1 := by
  change (row.val + (1#usize).val) % 2^(UScalarTy.numBits .Usize) = row.val+1
  have hone : (1#usize).bv.toNat = 1 := by scalar_tac
  rw [hone]
  apply Nat.mod_eq_of_lt
  rcases System.Platform.numBits_eq with hbits|hbits <;>
    simp only [UScalarTy.numBits,hbits] <;> omega
