abbrev Columns := Array (alloc.vec.Vec U32) 16#usize
abbrev State := Array U32 16#usize

def rowIndex (block row : Usize) : Usize :=
  Usize.wrapping_add (Usize.wrapping_mul block 16#usize) row

theorem ne_self (x : U32) : core.cmp.PartialEq.ne.trait_default
    aspis_core.field.M31.Insts.CoreCmpPartialEqM31 x x = ok false := by
  simp only [core.cmp.PartialEq.ne.trait_default,
    aspis_core.field.M31.Insts.CoreCmpPartialEqM31]
  with_unfolding_all simp
#print axioms ne_self

/-- Source checker completeness for a row whose actual Vec reads have the
specified values. No vector length/read premise is manufactured here. -/
theorem loop (columns : Columns) (block row : Usize) (expected : State)
    (reads : ∀ lane : Usize, lane.val < 16 →
      alloc.vec.Vec.index (core.slice.index.SliceIndexUsizeSlice U32)
        columns.val[lane.val]! (rowIndex block row) = ok expected.val[lane.val]!)
    (iter : core.ops.range.Range Usize) (he : iter.end = 16#usize)
    (hi : iter.start.val ≤ 16) :
    pair_forest_trace.validate_forest_poseidon_row_loop iter columns block row expected = ok (.Ok ()) := by
  generalize hremaining : 16-iter.start.val = remaining
  induction remaining using Nat.strong_induction_on generalizing iter with
  | h remaining ih =>
    rw [pair_forest_trace.validate_forest_poseidon_row_loop]
    by_cases hlt : iter.start.val < 16
    · obtain ⟨after,ha,hv,hend⟩ := PV_Src_Forest_BankRange.next_some iter (by simpa [he] using hlt)
      rw [ha]
      simp only [bind_tc_ok,lift]
      rw [index_ok_gen columns iter.start (by simpa using hlt)]
      simp only [bind_tc_ok]
      have hr := reads iter.start hlt
      unfold rowIndex at hr
      with_unfolding_all rw [hr,index_ok_gen expected iter.start (by simpa using hlt)]
      simp only [bind_tc_ok,ne_self,Bool.false_eq_true,if_false]
      apply ih (16-after.start.val) (by omega) after (hend.trans he) (by omega) rfl
    · rw [PV_Src_Forest_BankRange.next_none iter (by simpa [he] using (Nat.le_of_not_gt hlt))]
      rfl
#print axioms loop

theorem row (columns : Columns) (block row : Usize) (expected : State)
    (reads : ∀ lane : Usize, lane.val < 16 →
      alloc.vec.Vec.index (core.slice.index.SliceIndexUsizeSlice U32)
        columns.val[lane.val]! (rowIndex block row) = ok expected.val[lane.val]!) :
    pair_forest_trace.validate_forest_poseidon_row columns block row expected = ok (.Ok ()) :=
  loop columns block row expected reads ⟨0#usize,16#usize⟩ rfl (by simp)
#print axioms row
end R1Z.PV_RunJoin_BankRow17
