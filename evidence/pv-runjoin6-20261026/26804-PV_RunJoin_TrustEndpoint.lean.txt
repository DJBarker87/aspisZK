/-- Remaining P work, kept outside TrustItems. The first three clauses are
candidate source/control theorems once their actual data are installed. The
last is extraction-to-security-model complete trace refinement, NOT L3 trust.
No field assumes a privacy inequality. -/
structure ProofObligations : Prop where
  source :
    (∀ w : (h).Instance, (h).public w = x.1 →
      PV_RunJoin_UnifiedRetry.BodySourceLaw (boundary.sourceBody h x w) (runtime.body h x w)) ∧
    (∀ (s : State) k nonce next,
      runtime.account.call_mut s k = .ok (nonce,next) → next.provider = s.provider) ∧
    (∀ (w : (h).Instance), (h).public w = x.1 → ∀ secret (s next : State) status,
      runtime.body h x w secret s = .ok (status,next) → next.provider = s.provider)
  fill : ∀ (p : ProverRwRngCaller.Tape), p.used.val < p.replies.val.length →
    boundary.draw.fill p (Aeneas.Std.Array.repeat 64#usize 0#u8) =
      ProverRwRngCaller.Tape.Insts.Prover_rw_rng_callerDraw64.fill p (Aeneas.Std.Array.repeat 64#usize 0#u8)
  ledger : ∀ (w : (h).Instance), (h).public w = x.1 →
    ∀ secret (s next : State) status, runtime.body h x w secret s = .ok (status,next) →
      boundary.ledger next =
        match boundary.reservation secret s with
        | none => boundary.ledger s
        | some path => (ProverCommitLedger36.reserve boundary.schedule boundary.unix path (boundary.ledger s)).2
  byte_blind :
    (∀ secret (s : State) kind,
      boundary.fault secret s kind = boundary.schedule (boundary.ledger s).history.length kind (boundary.ledger s).history) ∧
    (∀ t u rest, boundary.ledger (PV_RunJoin_Deployed.initialState runtime (boundary.coordinates.symm (t,rest))) =
      boundary.ledger (PV_RunJoin_Deployed.initialState runtime (boundary.coordinates.symm (u,rest)))) ∧
    (∀ (i : Fin 128) (t u : Fin 128 → ProverCommitPublicNonce40.Raw) (rest : Rest), (∀ j : Fin 128, j.val < i.val → t j = u j) →
      boundary.frame i (t,rest) = boundary.frame i (u,rest))
  trace : ∀ (adv : (rust).Adversary) (w : (h).Instance), (h).public w = x.1 →
    ∀ r : World, ¬ (∃ i, ProverCommitFuelJoin22.bad (oldBad w) (queries adv w) i r.coins) →
      extracted r.coins adv w =
        (source).real h B outputs (RomObserverBin.observerMap rust observer_program adv) x w r.coins.1.1

/-- Native-to-extraction trust composed with the separate P trace theorem.
No equality with the security experiment was placed in the trust structure. -/
theorem complete_view (adv : (rust).Adversary) (w : (h).Instance)
    (hw : (h).public w = x.1) (r : World)
    (hn : ¬ (∃ i, ProverCommitFuelJoin22.bad (oldBad w) (queries adv w) i r.coins)) :
    (rust).real h B outputs adv x w r =
      (source).real h B outputs (RomObserverBin.observerMap rust observer_program adv) x w r.coins.1.1 := by
  have same := congrFun (congrFun (T.observation r.coins) adv) w
  change (rust).real h B outputs adv x w
      (PV_RunJoin_FiniteGraph.create (PV_RunJoin_Deployed.initialState runtime) r.coins) =
    extracted r.coins adv w at same
  have graph := (PV_RunJoin_FiniteGraph.coordinates
    (PV_RunJoin_Deployed.initialState runtime)).left_inv r
  change PV_RunJoin_FiniteGraph.create (PV_RunJoin_Deployed.initialState runtime) r.coins = r at graph
  rw [graph] at same
  exact same.trans (P.trace adv w hw r hn)
