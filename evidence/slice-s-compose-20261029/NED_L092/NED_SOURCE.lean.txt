def mapOuter {α E F : Type} (f : E → F) : Result (core.result.Result α E) → Result (core.result.Result α F)
 | .ok r => .ok (mapResult f r)
 | .fail e => .fail e
 | .div => .div

theorem mapOuter_bind {α β E F : Type} (f : E → F) (m : Result α)
    (k : α → Result (core.result.Result β E)) :
    mapOuter f (m >>= k) = (m >>= fun x => mapOuter f (k x)) := by
  cases m <;> rfl
