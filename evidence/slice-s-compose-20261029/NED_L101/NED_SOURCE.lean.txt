def mapOuter {α E F : Type} (f : E → F) : Result (core.result.Result α E) → Result (core.result.Result α F)
 | .ok r => .ok (mapResult f r)
 | .fail e => .fail e
 | .div => .div

theorem mapOuter_bind {α β E F : Type} (f : E → F) (m : Result α)
    (k : α → Result (core.result.Result β E)) :
    mapOuter f (m >>= k) = (m >>= fun x => mapOuter f (k x)) := by
  cases m <;> rfl

theorem mapOuter_ok {α E F : Type} (f : E → F) (x : α) :
    mapOuter f (ok (core.result.Result.Ok x)) = ok (.Ok x) := by rfl

theorem mapOuter_split {α β γ E F : Type} (f : E → F) (p : α × β)
    (k : α → β → Result (core.result.Result γ E)) :
    mapOuter f (let (a,b) := p; k a b) = (let (a,b) := p; mapOuter f (k a b)) := by
  cases p; rfl

theorem mapOuter_bind_pair {α β γ E F : Type} (f : E → F) (m : Result (α × β))
    (k : α → β → Result (core.result.Result γ E)) :
    mapOuter f (do let (a,b) ← m; k a b) = (do let (a,b) ← m; mapOuter f (k a b)) := by
  cases m with
  | ok p => cases p; rfl
  | fail e => rfl
  | div => rfl

theorem bind_same {α β : Type} (m : Result α) (f g : α → Result β)
    (point : ∀ x, f x = g x) : (m >>= f) = (m >>= g) := by
  exact congrArg (fun k => m >>= k) (funext point)
