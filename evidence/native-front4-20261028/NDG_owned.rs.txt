#[kani::proof]
#[kani::unwind(3)]
fn zero_window(){let b:[u8;4]=kani::any();let i:usize=kani::any();kani::assume(i<4);let r=require_zero(&b[i..i+1]);assert_eq!(r.is_ok()==(b[i]==0),!cfg!(NDG_negative));}
#[kani::proof]
#[kani::unwind(2)]
fn zero_empty(){assert_eq!(require_zero(&[]).is_ok(),!cfg!(NDG_negative));}
#[kani::proof]
#[kani::unwind(2)]
fn zero_reject(){let b:[u8;32]=kani::any();kani::assume(b[0]!=0);assert_eq!(matches!(require_zero(&b),Err(ProgramError::AccountAlreadyInitialized)),!cfg!(NDG_negative));}

#[kani::proof]
#[kani::stub(core::ptr::copy_nonoverlapping,crate::NDG_kmx::KMX_copy_nonoverlapping)]
#[kani::unwind(34)]
fn snapshot_one(){
 let key=Pubkey::new_from_array(kani::any());let owner=Pubkey::new_from_array(kani::any());
 let mut lamports=0u64;let b:u8=kani::any();let mut data=[b];
 let signer=kani::any();let writable=kani::any();let executable=kani::any();
 let a=AccountInfo::new(&key,signer,writable,&mut lamports,&mut data,&owner,executable,0);
 let r=snapshot(&a,1,2);
 let good=r.key==key && r.owner==owner && r.is_signer==signer && r.is_writable==writable && r.executable==executable && r.length==Ok(1) && r.bytes.as_ref().unwrap().as_slice()==[b];
 assert_eq!(good,!cfg!(NDG_negative));
}
