import Aeneas
set_option autoImplicit false
noncomputable section
namespace PV_DIAG9
open Aeneas Aeneas.Std Result

theorem ok_excludes_panic {A : Type} {r : Result A} {a : A}
 (h : r=.ok a) : r ≠ .fail .panic := by rw [h]; intro e; cases e
#print axioms ok_excludes_panic
theorem ok_excludes_div {A : Type} {r : Result A} {a : A}
 (h : r=.ok a) : r ≠ .div := by rw [h]; intro e; cases e
#print axioms ok_excludes_div
theorem ok_excludes_failure {A : Type} {r : Result A} {a : A}
 (h : r=.ok a) (e : Error) : r ≠ .fail e := by rw [h]; intro x; cases x
#print axioms ok_excludes_failure
end PV_DIAG9
