type R=Result<(Accumulator,BatchedWeights),Error>;
enum Step { Done(R), Continue(R) }
struct Oracle { error:Option<Error>, rhs:E,power:K,next:usize,terms:usize,
 row:usize,raw:[u64;4],called:bool,seen:[usize;9] }
impl Oracle {
 fn call(&mut self,proof:&[u8],p:&Prepared,mut a:Accumulator,g:&[E;29],geometry:&QueryGeometry,mut pending:BatchedWeights)->R {
  assert!(!self.called);self.called=true;
  self.seen=[proof.as_ptr() as usize,proof.len(),p as *const _ as usize,g as *const _ as usize,
   geometry as *const _ as usize,a.next,a.weights.as_ptr() as usize,pending.raw.as_ptr() as usize,pending.terms];
  #[cfg(NFG_bad_argument)] {self.seen[5]^=1;}
  if let Some(error)=self.error {return Err(error);}
  a.rhs=self.rhs;a.power=self.power;a.next=self.next;pending.terms=self.terms;for cell in &mut pending.raw { *cell=kani::any(); } pending.raw[self.row]=self.raw;
  Ok((a,pending))
 }
}
// Exact source match-body, replacing only returns by Step and the reached child by its local oracle.
fn NFG_step(proof:&[u8],p:&Prepared,g:&[E;29],geometry:&QueryGeometry,end:usize,state:R,oracle:&mut Oracle)->Step {
 match state {
  Err(error) => Step::Done(Err(error)),
  Ok((a, pending)) => {
   if a.next >= end {return Step::Done(Ok((a, pending)));}
   let state=oracle.call(proof,p,a,g,geometry,pending);
   Step::Continue(state)
  }
 }
}
