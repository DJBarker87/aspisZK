theorem returned (buffer : Array U8 64#usize) (pos : Usize) (suffix : Array U8 8#usize)
    (hp : pos.val<64) (first retained : Array U8 64#usize)
    (second : Option (Array U8 64#usize)) (newpos : Usize)
    (he : PV_CBL_ShaPadFixed.sha_pad_owned_split8 buffer pos suffix = ok (first,second,retained,newpos)) :
    retained=first ∧ newpos=0#usize ∧
      first.val=PV_CBL_PaddingModel.first buffer.val pos.val suffix.val ∧
      second.map (fun a => a.val)=PV_CBL_PaddingModel.second pos.val suffix.val := by
  obtain ⟨f,s,hr,hf,hs⟩ := PV_CBL_PadContentSource.source buffer pos suffix hp
  rw [he] at hr
  cases hr
  exact ⟨rfl,rfl,hf,hs⟩

    (hp : pos.val<64) (error : Error) :
    PV_CBL_ShaPadFixed.sha_pad_owned_split8 buffer pos suffix ≠ fail error := by
  obtain ⟨f,s,hr,_⟩ := PV_CBL_PadContentSource.source buffer pos suffix hp
  rw [hr]
  intro h
  cases h
