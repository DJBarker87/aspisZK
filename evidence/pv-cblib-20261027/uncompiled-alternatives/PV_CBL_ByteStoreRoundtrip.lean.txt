import R1Z.PV_CBL_ByteCodecValue
import Aeneas.Tactic.Bvify
import Std.Tactic.BVDecide
open Aeneas Aeneas.Std Result
namespace R1Z.PV_CBL_ByteStoreRoundtrip
set_option autoImplicit false
theorem word (w : U32) :
    PV_CBL_ByteCodecValue.load
      (UScalar.cast .U8 (U32.wrapping_shr w 24#u32))
      (UScalar.cast .U8 (U32.wrapping_shr w 16#u32))
      (UScalar.cast .U8 (U32.wrapping_shr w 8#u32))
      (UScalar.cast .U8 w) = w := by
  unfold PV_CBL_ByteCodecValue.load
  bvify
  bv_decide
#print axioms word
end R1Z.PV_CBL_ByteStoreRoundtrip
