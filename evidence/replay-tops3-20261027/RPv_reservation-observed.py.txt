#!/usr/bin/env python3
from pathlib import Path
import json,time,sys,subprocess
# lead 2026-10-26: cross-stream quiet window. If /home/dombarker/project-offloads/QUIET_WINDOW.json exists with {"until": <epoch seconds>}
# and now < until, refuse admission (workers wait-and-retry under local-lean rule 6).
import sys as _qw_sys, time as _qw_time, json as _qw_json, pathlib as _qw_pathlib
_qw = _qw_pathlib.Path("/home/dombarker/project-offloads/QUIET_WINDOW.json")
if _qw.exists():
    try:
        _qw_until = float(_qw_json.loads(_qw.read_text()).get("until", 0))
    except Exception:
        _qw_until = 0
    if _qw_time.time() < _qw_until:
        _qw_r = {"allowed": False, "reason": "quiet window until %d (cross-stream timing gate)" % int(_qw_until),
                 "quiet_window_until": int(_qw_until), "bounded_reserved_bytes": None, "requested_bytes": None}
        if len(_qw_sys.argv) > 1:
            _qw_pathlib.Path(_qw_sys.argv[1]).write_text(_qw_json.dumps(_qw_r) + "\n")
        print(_qw_json.dumps(_qw_r))
        raise SystemExit(78)  # lead 2026-10-27: refusal uses the contention exit code (was 0: wrappers read exit 0 as admission)

root=Path('/sys/fs/cgroup/user.slice/user-1000.slice/user@1000.service/app.slice');rows=[]
for p in root.glob('*.scope'):
 try:
  if not (p/'cgroup.procs').read_text().strip():continue
  limit=(p/'memory.max').read_text().strip()
  rows.append(dict(scope=str(p),memory_max=int(limit) if limit!='max' else None,processes=[subprocess.run(['ps','-p',pid,'-o','args='],text=True,capture_output=True).stdout.strip() for pid in (p/'cgroup.procs').read_text().split()],memory_high=(p/'memory.high').read_text().strip(),memory_swap_max=(p/'memory.swap.max').read_text().strip(),memory_current=int((p/'memory.current').read_text())))
 except FileNotFoundError:continue  # lead 2026-10-26: a scope that vanished between glob and read is not populated; skip it (race with short-lived scopes)
assert all(r['memory_max'] is not None for r in rows), 'Unbounded populated scope'
total=sum(r['memory_max'] or 0 for r in rows)
out=dict(requested_bytes=8*2**30,allowed=total+8*2**30<=55*2**30,utc=time.time(),populated_scopes=rows,bounded_reserved_bytes=total,all_populated_scopes_bounded=all(r['memory_max'] is not None for r in rows),safe_limit_bytes=55*2**30,print_axioms='N/A: read-only resource preflight')
Path(sys.argv[1]).write_text(json.dumps(out,indent=2)+'\n');print(json.dumps({k:out[k] for k in ["allowed","bounded_reserved_bytes","requested_bytes"]}))

if not out["allowed"]: raise SystemExit(78)
