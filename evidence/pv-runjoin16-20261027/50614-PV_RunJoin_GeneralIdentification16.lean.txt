theorem with_roots (empty : Array (Array U32 8#usize) 21#usize)
    (i : pair_forest_trace.PoolV1PairForestInputNoteWitnessV1)
    (out : U32) (rec : Option payment_relation.PoolV1OutputNoteWitnessV1)
    (chg : payment_relation.PoolV1OutputNoteWitnessV1)
    (snap : pair_tree_profile.PoolV1PairLiveSnapshotV1) (asset : U32) :
    @Records.call (PV_RunJoin_RecordsAdapter16.R_shipped empty) i out rec chg snap asset =
      PV_Src_RecordReference.call empty i out rec chg snap asset := by
  change (do
    let (w,p) ← PV_RunJoin_Records.pool_v1.pair_forest_copy_builder.records
      (PV_RunJoin_RecordsAdapter16.input i) out (rec.map PV_RunJoin_RecordsAdapter16.note)
      (PV_RunJoin_RecordsAdapter16.note chg) (PV_RunJoin_RecordsAdapter16.snapshot snap) asset empty
    ok (PV_RunJoin_RecordsAdapter16.witness w, PV_RunJoin_RecordsAdapter16.publicRecord p)) = _
  unfold PV_Src_RecordReference.call
  rw [BuilderRecordsReference.records]
  simp only [bind_tc_ok]
  rcases i with ⟨⟨sk, salt, value, ⟨first, occ, second, inv⟩, selected,
    ⟨siblings, index⟩⟩, supers, directions⟩
  rcases chg with ⟨changeOwner, changeSalt, changeValue⟩
  rcases snap with ⟨pool, domain, sequence, next, current, frontier⟩
  rw [shape20 siblings, shape3 supers, shape3 directions, shape20 frontier, shape21 empty]
  cases selected <;> cases rec
  all_goals
    with_unfolding_all
      simp only [PV_RunJoin_RecordsAdapter16.R_shipped, Records.call,
        PV_RunJoin_RecordsAdapter16.input, PV_RunJoin_RecordsAdapter16.note,
        PV_RunJoin_RecordsAdapter16.snapshot, PV_RunJoin_RecordsAdapter16.witness,
        PV_RunJoin_RecordsAdapter16.publicRecord,
        PV_RunJoin_Records.pool_v1.pair_forest_copy_builder.records,
        PV_RunJoin_Records.pool_v1.pair_forest_copy_builder.records.owned,
        PV_RunJoin_RecordsAdapter16.raw_owned_all_words, lift, bind_tc_ok] <;> rfl

theorem general
    (i : pair_forest_trace.PoolV1PairForestInputNoteWitnessV1)
    (out : U32) (rec : Option payment_relation.PoolV1OutputNoteWitnessV1)
    (chg : payment_relation.PoolV1OutputNoteWitnessV1)
    (snap : pair_tree_profile.PoolV1PairLiveSnapshotV1) (asset : U32) :
    @Records.call (PV_RunJoin_RecordsAdapter16.R_shipped PV_Src_CanonicalRoots.table)
      i out rec chg snap asset =
    @PV_Src_RecordDynamic.call PV_RunJoin_ShippedExternal.E_shipped
      i out rec chg snap asset := by
  rw [with_roots]
  unfold PV_Src_RecordDynamic.call
  rw [show @pair_trace.pair_empty_roots PV_RunJoin_ShippedExternal.E_shipped =
    ok PV_Src_CanonicalRoots.table from PV_Src_RootHelper.source]
  rfl

#print axioms with_roots
#print axioms general
end R1Z.PV_RunJoin_GeneralIdentification16
