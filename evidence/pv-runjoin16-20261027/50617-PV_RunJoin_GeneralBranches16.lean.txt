theorem none_first (empty : Array (Array U32 8#usize) 21#usize)
    (i : pair_forest_trace.PoolV1PairForestInputNoteWitnessV1)
    (out : U32)
    (chg : payment_relation.PoolV1OutputNoteWitnessV1)
    (snap : pair_tree_profile.PoolV1PairLiveSnapshotV1) (asset : U32)
    (hselected : i.pair.selected_second = false) :
    @Records.call (PV_RunJoin_RecordsAdapter16.R_shipped empty) i out none chg snap asset =
      PV_Src_RecordReference.call empty i out none chg snap asset := by
  change (do
    let (w,p) ← PV_RunJoin_Records.pool_v1.pair_forest_copy_builder.records
      (PV_RunJoin_RecordsAdapter16.input i) out (none.map PV_RunJoin_RecordsAdapter16.note)
      (PV_RunJoin_RecordsAdapter16.note chg) (PV_RunJoin_RecordsAdapter16.snapshot snap) asset empty
    ok (PV_RunJoin_RecordsAdapter16.witness w, PV_RunJoin_RecordsAdapter16.publicRecord p)) = _
  unfold PV_Src_RecordReference.call
  rw [BuilderRecordsReference.records]
  simp only [bind_tc_ok]
  rcases i with ⟨⟨sk, salt, value, ⟨first, occ, second, inv⟩, selected,
    ⟨siblings, index⟩⟩, supers, directions⟩
  change selected = false at hselected
  subst selected
  rcases chg with ⟨changeOwner, changeSalt, changeValue⟩
  rcases snap with ⟨pool, domain, sequence, next, current, frontier⟩
  rw [shape20 siblings, shape3 supers, shape3 directions, shape20 frontier, shape21 empty]
  dsimp only [PV_RunJoin_RecordsAdapter16.input, PV_RunJoin_RecordsAdapter16.note,
    PV_RunJoin_RecordsAdapter16.snapshot]
  unfold PV_RunJoin_Records.pool_v1.pair_forest_copy_builder.records
  unfold PV_RunJoin_Records.pool_v1.pair_forest_copy_builder.records.owned
  repeat' first
    | rw [PV_RunJoin_RecordsAdapter16.raw_owned_all_words]
    | rw [bind_tc_ok]
  with_unfolding_all rfl

#print axioms none_first
