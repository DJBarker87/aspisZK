// All references point into an initialized allocator object; no fake fat pointer.
unsafe fn NCF_reject_slice(n:usize, f:impl FnOnce(&[u8])) {
    let layout=core::alloc::Layout::array::<u8>(n).unwrap();
    let p=alloc::alloc::alloc_zeroed(layout);
    kani::assume(!p.is_null());
    let j:usize=kani::any();kani::assume(j<n);
    *p.add(j)=kani::any();
    let input=core::slice::from_raw_parts(p,n);
    f(input);
    alloc::alloc::dealloc(p,layout);
}
