    let p=alloc::alloc::alloc_zeroed(layout);
    kani::assume(!p.is_null());
    let j:usize=kani::any();kani::assume(j<n);
    *p.add(j)=kani::any();
    let input=core::slice::from_raw_parts(p,n);
    f(input);
    alloc::alloc::dealloc(p,layout);
}

fn NCF_le(bytes:&[u8],o:usize)->u64 {
    (bytes[o] as u64) | ((bytes[o+1] as u64)<<8) | ((bytes[o+2] as u64)<<16) |
    ((bytes[o+3] as u64)<<24) | ((bytes[o+4] as u64)<<32) |
    ((bytes[o+5] as u64)<<40) | ((bytes[o+6] as u64)<<48) | ((bytes[o+7] as u64)<<56)
}
