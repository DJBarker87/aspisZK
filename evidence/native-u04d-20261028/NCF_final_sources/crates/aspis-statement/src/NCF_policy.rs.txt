pub fn NCF_spec(p:&VerifierPolicyV1)->Result<(),PoolV1FormatError>{
 if p.flags & !3 != 0 {Err(PoolV1FormatError::UnsupportedFlags)}
 else if p.registry_program==[0;32] || p.policy_binding==[0;32] {Err(PoolV1FormatError::ZeroRequiredBinding)}
 else if (p.flags & 2 != 0 && p.flags & 1 == 0) || ((p.flags & 1 != 0) != (p.registry_authority==[0;32])) {Err(PoolV1FormatError::InvalidAuthorityMode)}
 else {Ok(())}
}
