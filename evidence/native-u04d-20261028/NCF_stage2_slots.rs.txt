fn NCF_le(bytes:&[u8],o:usize)->u64 {
    (bytes[o] as u64) | ((bytes[o+1] as u64)<<8) | ((bytes[o+2] as u64)<<16) |
    ((bytes[o+3] as u64)<<24) | ((bytes[o+4] as u64)<<32) |
    ((bytes[o+5] as u64)<<40) | ((bytes[o+6] as u64)<<48) | ((bytes[o+7] as u64)<<56)
}
