// Valid byte generation via a deliberately weak *verified* no-op contract.
// This helper is harness input generation, never an implementation replacement.
#[kani::requires(!input.is_empty())]
#[kani::modifies(input)]
#[kani::ensures(|_: &()| true)]
fn NCF_payload(input:&mut [u8]) {}
#[kani::requires(!input.is_empty())]
#[kani::modifies(input)]
#[kani::ensures(|_: &()| false)]
fn NCF_payload_bad(input:&mut [u8]) {}
#[kani::proof_for_contract(NCF_payload)]
#[kani::unwind(3)]
fn NCF_payload_contract(){let n:usize=kani::any();kani::assume(n>0 && n<=isize::MAX as usize);unsafe{let layout=core::alloc::Layout::array::<u8>(n).unwrap();let p=alloc::alloc::alloc_zeroed(layout);kani::assume(!p.is_null());let input=core::slice::from_raw_parts_mut(p,n);NCF_payload(input);alloc::alloc::dealloc(p,layout);}}
