/-- Carrier identification on the SAME native, bank-supported response-selected
entry. Pointwise only: no all-world factory or independent runtime surjection. -/
theorem on_entry {A C Tx : Type}
    (base : DrvC_Installation.Configuration FIo_Install.W C) (err : C)
    (reservation : NP6_StdModel.Reservation)
    (hashv : SdkC_Source.Parts → Result SdkC_Source.Digest)
    (copy : Heap (Nat → Record) → Nat → Payload → S2W_TrustedStd.Outcome (Nat → Record))
    (std : S2W_TrustedStd.TrustedStd copy) (allocator : Allocator A)
    (native1 : S2W_ExtractionBoundary.First) (native2 : S2W_ExtractionBoundary.Second)
    (extraction : S2W_DigestExtraction.AeneasExtraction
      (DrvC_Installation.sdk (FIp_Install.configuration err base reservation)) native1 native2 hashv)
    (a : Adversary A) (H : Bytes → State)
    (saved : alloc.vec.Vec U8) (post1 post2 : Heap (Nat → Record)) (final1 final2 : A)
    (accepted1 : first native1 copy allocator (eval H a.first).2.entry CPH_Program.verifier
      (eval H a.first).2.data (eval H a.first).2.state = ok (.Ok (saved,post1,final1)))
    (accepted2 : let j := (eval H (a.second (eval H a.first).2 saved post1 final1)).2
      second native2 allocator j.entry CPH_Program.verifier j.data j.slot j.state = ok (.Ok (post2,final2)))
    (p : CPG_Projection.Observations Tx A) (t u : Tx) (actions : List (CPP_BankState.Action A))
    (game : CPH_Game.DeployedPairExecution p t u (eval H a.first).2.entry
      (eval H (a.second (eval H a.first).2 saved post1 final1)).2.entry saved actions
      native1 native2 copy allocator CPH_Program.verifier)
    (RuntimePersistence : CPR_DataPersistence.TwoTransactionLaw
      (CPG_Projection.adjacent p) (CPG_Projection.ordinary p (Hist_Nonvacuity.key 0#u8))
      (CPG_Projection.before p) (CPG_Projection.tentative p) (CPG_Projection.success p) (CPG_Projection.loaded p))
    (f : FirstCut) (s : SecondCut)
    (cuts : SourceCuts (SCH_IdealSdk.dictionary (FIp_Install.configuration err base reservation) hashv H)
      (eval H a.first).2.entry.first
      ((eval H (a.second (eval H a.first).2 saved post1 final1)).2.entry.second secondSizes) saved f s)
    (run : F3T_Experiment.RunView)
    (observed : (eval H (runtime (FIp_Install.configuration err base reservation)
      hashv copy allocator CPH_Program.verifier a)).2 = some run) :
    run.2.1.stored = bytesOf f.proof.val ∧
    run.2.1.physical.bytes = bytesOf f.pub.val ∧
    bytesOf s.checkpoint.val = bytesOf saved.val := by
  have submitted := SCE_FirstCutBytes.submitted_cut (eval H a.first).2
    (SCH_IdealSdk.dictionary (FIp_Install.configuration err base reservation) hashv H)
    _ saved f s cuts
  refine ⟨?_,?_,?_⟩
  · exact (SCE_Projection.stored _ hashv copy allocator CPH_Program.verifier a H run observed).trans
      ((congrArg (fun x : List (Bytes × State) × F3T_ByteCarrier.Submission => x.2.stored)
        (submission_eval a H)).trans submitted.2)
  · exact (SCE_PublicProjection.bytes _ hashv copy allocator CPH_Program.verifier a H run observed).trans
      ((congrArg (fun x : List (Bytes × State) × F3T_ByteCarrier.Submission => x.2.publicBytes)
        (submission_eval a H)).trans submitted.1)
  · exact ARQ_BankCut.checkpoint base err reservation hashv copy std allocator native1 native2 extraction
      a H saved post1 post2 final1 final2 accepted1 accepted2 p t u actions game RuntimePersistence f s cuts

#print axioms on_entry
end AspisR1.ARR_Carrier
