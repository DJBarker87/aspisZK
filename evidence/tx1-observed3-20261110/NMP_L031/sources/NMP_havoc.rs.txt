// Weak frame contract: full slice havoc, with one recorded arbitrary cell.
// It is a nondeterministic-generator utility, NEVER a callee's functional law.
#[kani::requires(!dst.is_empty() && dst.len()<=13831 && probe<dst.len())]
#[kani::modifies(dst)]
#[kani::ensures(|_| dst[probe]==word)]
pub(crate) fn point(dst:&mut[u8],probe:usize,word:u8,_side:&mut u8){dst[probe]=word;}
#[kani::requires(!dst.is_empty() && dst.len()<=13831 && probe<dst.len())]
#[kani::modifies(dst)]
#[kani::ensures(|_| dst[probe]!=word)]
fn inverse(dst:&mut[u8],probe:usize,word:u8,_side:&mut u8){dst[probe]=word;}
#[kani::requires(!dst.is_empty() && dst.len()<=13831 && probe<dst.len())]
#[kani::modifies(dst)]
#[kani::ensures(|_| true)]
fn weak(dst:&mut[u8],probe:usize,_word:u8,_side:&mut u8){dst[probe]=0;}
#[kani::requires(!dst.is_empty() && dst.len()<=13831 && probe<dst.len())]
#[kani::modifies(dst)]
#[kani::ensures(|_| true)]
fn bad_modifies(dst:&mut[u8],probe:usize,word:u8,side:&mut u8){dst[probe]=word;*side=side.wrapping_add(1);}
