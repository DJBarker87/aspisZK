#[derive(Clone,Copy)]struct Meta{ptr:*const u8,len:usize}

#[repr(C)]struct Frame<'a>{proof:[u8;57607],log:Log<'a>}
struct Log<'a>{

unsafe fn from_source<'a>(bytes:&[u8],offset:usize)->&'a mut Log<'a>{
 let start=unsafe{bytes.as_ptr().sub(offset)};
 unsafe{&mut *(start.add(core::mem::offset_of!(Frame<'static>,log)) as *mut Log<'a>)}
}
unsafe fn from_hash<'a,H:Digest32>(h:H)->&'a mut Log<'a>{
 assert_eq!(core::mem::size_of::<H>(),core::mem::size_of::<LoggedHash>());
 unsafe{&mut *((*(&h as *const H as *const LoggedHash)).0 as *mut Log<'a>)}
}
fn same_public(a:Public<'_>,b:Public<'_>)->bool{match(a,b){
 (Public::Transfer(p,t),Public::Transfer(q,u))=>core::ptr::eq(p,q)&&core::ptr::eq(t,u),
 (Public::Withdrawal(p,t),Public::Withdrawal(q,u))=>core::ptr::eq(p,q)&&core::ptr::eq(t,u),_=>false}}
