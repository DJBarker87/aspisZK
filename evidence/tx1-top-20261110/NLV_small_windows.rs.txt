// Isolated per-occurrence candidate windows, NOT a physical TX1 producer.
// No callee internals are credited. Field/digest arithmetic is unreachable.
fn small_window(mode:u8,bad:bool){
 let j=usize::from(kani::any::<u16>());let tag=usize::from(kani::any::<u16>());
 let good=match mode{
  0=>{let k=usize::from(kani::any::<u16>());let error=err();let mut called=false;
   let source=||->Result<(),Error>{if k!=SPLIT{return Err(Error::Parse)}called=true;Err(error)};
   let mut source=source;let out=source();matches!(out,Err(e)if e==if k==SPLIT{error}else{Error::Parse})&&called==(k==SPLIT)},
  1=>{let g=[E::ZERO;29];g[j%29]==E::ZERO},
  2=>{let proof:[u8;57607]=kani::any();let slice=&proof[compact::OFF_FOLDED_ROOT..compact::OFF_ROUND1];slice.len()==32&&slice[j%32]==proof[compact::OFF_FOLDED_ROOT+j%32]&&slice.as_ptr()as usize==proof.as_ptr()as usize+compact::OFF_FOLDED_ROOT},
  3=>{let bytes:[u8;32]=kani::any();let root:Result<[u8;32],_>=bytes.as_slice().try_into();matches!(root,Ok(r)if r[j%32]==bytes[j%32])},
  4=>{let bytes:[u8;32]=kani::any();let root:Result<[u8;32],core::array::TryFromSliceError>=Ok(bytes);let root=root.unwrap();root[j%32]==bytes[j%32]},
  5=>{let queries:[u32;22]=kani::any();let before=queries[j%22];let k=10;let start=0;let end=k;let borrowed=&queries;start==0&&end==10&&borrowed[j%22]==before&&borrowed.as_ptr()==queries.as_ptr()},
  6=>{let a=accumulator();let before=acell(&a,4*(tag%64),j);let pointer=a.weights.as_ptr();let weights=&a.weights;weights.len()==64&&coord(weights[tag%64],j)as u64==before&&weights.as_ptr()==pointer},
  7=>{let g=core::array::from_fn::<E,29,_>(|_|e());let before=g[j%29];let borrowed=&g;borrowed[j%29]==before&&borrowed.as_ptr()==g.as_ptr()},
  8=>{let p=prepared();let before=pcell(&p,tag,j);let ptr=&*p as *const Prepared;let borrowed=&p;pcell(borrowed,tag,j)==before&&&**borrowed as *const Prepared==ptr},
  9=>{let a=accumulator();let before=acell(&a,tag,j);let pointer=a.weights.as_ptr();let moved=a;acell(&moved,tag,j)==before&&moved.weights.as_ptr()==pointer},
  10=>{let geom=geometry();let before=geom.inverses[tag%10][j%5];let pointer=geom.inverses.as_ptr();let borrowed=&geom;borrowed.inverses[tag%10][j%5]==before&&borrowed.inverses.as_ptr()==pointer},
  11=>{let p=make_pending();let before=p.raw[tag%64][j%4];let terms=p.terms;let pointer=p.raw.as_ptr();let moved=p;moved.raw[tag%64][j%4]==before&&moved.terms==terms&&moved.raw.as_ptr()==pointer},
  // One arbitrary native write iteration; full-vector loop association is OPEN.
  12=>{let mut a=accumulator();let rhs=a.rhs;let power=a.power;let next=a.next;let after=q();let index=tag%64;let other=j%64;let before=a.weights[other];let weights=&mut a.weights;weights[index]=after;
   a.rhs==rhs&&a.power==power&&a.next==next&&a.weights[index]==after&&(other==index||a.weights[other]==before)},
  13=>{let bytes=Vec::from(kani::any::<[u8;1768]>());let before=bytes[j%1768];let pointer=bytes.as_ptr();let out:Result<Vec<u8>,Error>=Ok(bytes);
   matches!(out,Ok(ref b)if b.len()==1768&&b[j%1768]==before&&b.as_ptr()==pointer)},
  _=>unreachable!()
 };
 if bad{assert!(!good,"NLV matched corruption")}else{assert!(good,"NLV isolated candidate window")}
}
