struct MaskLog{cell:usize,low:QM31,high:QM31,blind:QM31,calls:usize}
impl MaskLog{
 fn mask(&mut self,low:&[QM31;16],high:&[QM31;10],blind:QM31,alpha:&[QM31;10])->QM31{
  self.low=low[self.cell];self.high=high[self.cell%10];self.blind=blind;self.calls+=1;alpha[self.cell%10]
 }
}
fn mask_cut(claims:&[[QM31;29];3],alpha:&[QM31;10],log:&mut MaskLog)->QM31{
    let mask = log.mask(
        &core::array::from_fn(|i| claims[0][i]),
        &core::array::from_fn(|i| claims[0][16 + i]),
        claims[0][27],
        &alpha,
    );
    mask
}

 let claims=core::array::from_fn(|_|core::array::from_fn(|_|q()));let alpha=core::array::from_fn(|_|q());let i:usize=kani::any();kani::assume(i<16);

 let (actual,expected)=match mode{0=>(log.low,claims[0][i]),1=>(log.high,claims[0][16+i%10]),2=>(log.blind,claims[0][27]),_=>(out,alpha[i%10])};
 assert_eq!(log.calls,1);if bad{assert_ne!(actual,expected)}else{assert_eq!(actual,expected)}
}
fn terminal_guard(running:QM31,masked_terminal:QM31)->Result<(),Error>{
    if running != masked_terminal {
        return Err(Error::Terminal);
    }
    Ok(())
}

fn circle_guard(z0:SecureCirclePoint,z1:SecureCirclePoint)->Result<(),Error>{
    if z0 == z1 {
        return Err(Error::EqualCirclePoints);
    }
    Ok(())
}
