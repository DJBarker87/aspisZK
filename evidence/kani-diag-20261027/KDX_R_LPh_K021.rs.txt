#[kani::proof]
#[kani::unwind(33)]
fn KDX_LPh_live_after(){live_window::<3>()}
fn live_window<const PART:usize>(){
 let input:[u8;16]=kani::any();let mut out:[u8;96]=kani::any();let j:usize=kani::any();kani::assume(j<96);
 kani::assume(if PART==0{j<32}else if PART==1{j>=32&&j<48}else if PART==2{j>=48&&j<64}else{j>=64});
 let old=out[j];assert_eq!(expand_values(&input,&mut out[32..64]),Ok(()));
 assert_eq!(out[j],if j>=32&&j<48{input[j-32]}else if j>=48&&j<64{0}else{old});
}
