inductive Event : Type 1 where

abbrev Log := List Event
inductive Code (A : Type) : Type 1 where
  | done : A → Code A
  | step {B : Type} : Result B → (B → Code A) → Code A
  | call {B : Type} : Call B → (B → Code A) → Code A

def bind {A B : Type} : Code A → (A → Code B) → Code B
  | .done a, k => k a
  | .step r c, k => .step r (fun b => bind (c b) k)
  | .call op c, k => .call op (fun b => bind (c b) k)
instance : Monad Code where
  pure := Code.done
  bind := bind

def liftResult {A : Type} (r : Result A) : Code A := .step r .done
instance : MonadLift Result Code where
  monadLift := liftResult

def perform {A : Type} (op : Call A) : Code A := .call op .done

def erase (ops : VTop_Stages.Ops) {A : Type} : Code A → Result A
  | .done a => ok a
  | .step r k => do let b ← r; erase ops (k b)
  | .call op k => do let b ← invoke ops op; erase ops (k b)

def thenResult {A : Type u} {B : Type v} (r : Result A) (k : A → Result B) : Result B :=
  match r with
  | .ok a => k a
  | .fail e => .fail e
  | .div => .div

def forget {A : Type} (r : Result (A × Log)) : Result A :=
  thenResult r (fun x => ok x.1)

def run (ops : VTop_Stages.Ops) {A : Type} : Code A → Log → Result (A × Log)
  | .done a, log => ok (a,log)
  | .step r k, log => thenResult r (fun b => run ops (k b) log)
  | .call op k, log => thenResult (invoke ops op) (fun b =>
      run ops (k b) (log ++ [.returned op b]))

theorem erase_run (ops : VTop_Stages.Ops) {A : Type} (p : Code A) (log : Log) :
    forget (run ops p log) = erase ops p := by
  induction p generalizing log with
  | done a => rfl
  | step r k ih =>
    cases r with
    | fail e => rfl
    | div => rfl
    | ok b => exact ih b log
  | call op k ih =>
    cases h : invoke ops op with
    | fail e => simp only [run,thenResult,forget,erase,h,bind_tc_fail]
    | div => simp only [run,thenResult,forget,erase,h,bind_tc_div]
    | ok b => simpa only [run,thenResult,erase,h,bind_tc_ok] using ih b (log ++ [.returned op b])

theorem run_prefix (ops : VTop_Stages.Ops) {A : Type} (p : Code A)
    (before after : Log) (a : A) (success : run ops p before = ok (a,after)) :
    ∃ suffix, after = before ++ suffix := by
  induction p generalizing before after with
  | done b =>
    have eq := Result.ok.inj success
    exact ⟨[], by simpa using congrArg Prod.snd eq.symm⟩
  | step r k ih =>
    cases r with
    | fail e => cases success
    | div => cases success
    | ok b => exact ih b before after success
  | call op k ih =>
    cases h : invoke ops op with
    | fail e => simp only [run,thenResult,h]  at success; cases success
    | div => simp only [run,thenResult,h]  at success; cases success
    | ok b =>
      simp only [run,thenResult,h] at success
      obtain ⟨suffix,hs⟩ := ih b _ after success
      exact ⟨.returned op b :: suffix, by simpa only [List.append_assoc,List.singleton_append] using hs⟩

theorem erase_perform (ops : VTop_Stages.Ops) {A : Type} (op : Call A) :
    erase ops (perform op) = invoke ops op := by
  cases h : invoke ops op <;> simp only [perform,erase,h,bind_tc_ok,bind_tc_fail,bind_tc_div]

theorem erase_lift (ops : VTop_Stages.Ops) {A : Type} (r : Result A) :
    erase ops (liftResult r) = r := by cases r <;> rfl

theorem erase_bind (ops : VTop_Stages.Ops) {A B : Type} (p : Code A) (k : A → Code B) :
    erase ops (p >>= k) = (do let a ← erase ops p; erase ops (k a)) := by
  induction p with
  | done a => rfl
  | step r c ih =>
    change (do let b ← r; erase ops (bind (c b) k)) = _
    have h : ∀ b, erase ops (bind (c b) k) =
      (do let a ← erase ops (c b); erase ops (k a)) := ih
    simp only [h,erase]
    cases r <;> rfl
  | call op c ih =>
    change (do let b ← invoke ops op; erase ops (bind (c b) k)) = _
    have h : ∀ b, erase ops (bind (c b) k) =
      (do let a ← erase ops (c b); erase ops (k a)) := ih
    simp only [h,erase]
    cases invoke ops op <;> rfl

/-! Oracle interpretation of the extracted typed call tree. All call arguments
and returned mutable state are retained. No identification with dV is assumed
or claimed. Ordinary inner Rust Err is a value of the dependent result type. -/
set_option autoImplicit false
noncomputable section
namespace AspisR1.VTop_OracleCode
open Aeneas Aeneas.Std Result VTop_Trace
open AspisV8R19.DuplexFrames AspisV8R19.SourceDuplexStep
open AspisV8R19.MemoizedProgramLaw AspisV8R19.OracleProgramOps

abbrev Implementation := {A : Type} → Call A → Program Bytes State (Result A)

def continueWith {A B : Type} (r : Result A)
    (k : A → Program Bytes State (Result B)) : Program Bytes State (Result B) :=
  match r with
  | .ok a => k a
  | .fail e => .done (.fail e)
  | .div => .done .div

def interpret (impl : Implementation) {A : Type} : Code A → Program Bytes State (Result A)
  | .done a => .done (.ok a)
  | .step r k => continueWith r (fun a => interpret impl (k a))
  | .call op k => bind (impl op) (fun r => continueWith r (fun a => interpret impl (k a)))

/-- A local call law: complete ordered reads and the complete dependent return
are equal whenever that source call returns. This does not bind an arbitrary
Verifier to the tree, and does not identify a model implementation with dV. -/
def CallJoin (source model : Implementation) : Prop :=
  ∀ {A : Type} (call : Call A) (H : Bytes → State) (value : A),
    (eval H (source call)).2 = .ok value →
      eval H (source call) = eval H (model call)

/-- Symbolic substitution through the actual typed syntax, retaining source
branch selection and concatenation order. No large stage reduction is needed. -/
theorem accepted_interpret_eq (source model : Implementation) (joins : CallJoin source model)
    {A : Type} (tree : Code A) (H : Bytes → State) (value : A)
    (accepted : (eval H (interpret source tree)).2 = .ok value) :
    eval H (interpret source tree) = eval H (interpret model tree) := by
  induction tree with
  | done a => rfl
  | step r k ih =>
    cases r with
    | fail e => cases accepted
    | div => cases accepted
    | ok a => exact ih a accepted
  | call op k ih =>
    simp only [interpret, eval_bind] at accepted ⊢
    cases returned : (eval H (source op)).2 with
    | fail e => simp only [returned, continueWith, eval] at accepted; cases accepted
    | div => simp only [returned, continueWith, eval] at accepted; cases accepted
    | ok a =>
      have localJoin := joins op H a returned
      rw [← localJoin]
      simp only [returned, continueWith] at accepted ⊢
      rw [ih a accepted]

#print axioms accepted_interpret_eq
end AspisR1.VTop_OracleCode

/-! Explicit unproved execution and runtime joins for the draft headline.
Full stored inputs survive in Runtime, through the probability experiment. -/
set_option autoImplicit false
noncomputable section
namespace AspisR1.OraclePortRuntimeBindingC7
open FS FS2
open AspisV8R19.DuplexFrames AspisV8R19.SourceDuplexStep
open AspisV8R19.MemoizedProgramLaw AspisV8R19.OracleProgramOps

abbrev Verifier := Input → Program Bytes State Bool
abbrev Runtime := Program Bytes State (Option RunView)

def originalWith (rust : Verifier) (P : Prover) : Program Bytes State (Input × Bool) :=
  bind P fun i => bind (rust i) fun ok => .done (i,ok)

def annotatedWith (D : LiteralByteProtocol.Decoder) (rust : Verifier) (P : Prover) :=
  originalWith rust (bind (OracleProgramRecording.record (originalWith rust P)) fun v =>
    .done (enrich (build D (view v)) v.2.1))

/-- Complete oracle-program equality, including every read, rejection and
returned Bool. This is the frozen Rust/source join, still a hypothesis. -/
def ExecutionJoin (D : LiteralByteProtocol.Decoder) (rust : Verifier) : Prop :=
  ∀ i : Input, rust i = verifier D dV i

theorem annotatedWith_eq (D : LiteralByteProtocol.Decoder) (rust : Verifier)
    (join : ExecutionJoin D rust) (P : Prover) :
    annotatedWith D rust P = annotated D dV P := by
  have hr : rust = verifier D dV := funext join
  rw [hr]
  rfl

/-- Runtime must bind the whole stored carrier and trace on every accepted
resumed run, in the same oracle world; this is stronger than a mass premise. -/
def StoredCarrierBinding (D : LiteralByteProtocol.Decoder) (rust : Verifier)
    (P : Prover) (runtime : Runtime) : Prop :=
  ∀ H : Bytes → State, ∀ run : RunView,
    (eval H runtime).2 = some run → run.2.2 = true →
      run = eval H (annotatedWith D rust (canon P))

def runtimeMass (v : View Bytes State (Option RunView)) : ℚ :=
  match v.2 with
  | none => 0
  | some run => failureMass (view run)

def deployedFailure (runtime : Runtime) : ℚ := lazyMean runtime emptyTable runtimeMass

theorem failure_nonnegative (run : RawCutProverCompiler.RunView) : 0 ≤ failureMass run := by
  unfold failureMass
  have h := mean_mono (fun coins : GSConcreteByteDecoderBank.RawCoins =>
    indicator_nonneg (run.2.2 = true ∧ RawRecordedPublicSpendExtractorBin.extract run coins = none))
  simpa only [mean_const] using h

theorem transport (D : LiteralByteProtocol.Decoder) (rust : Verifier) (P : Prover)
    (runtime : Runtime) (join : ExecutionJoin D rust)
    (binding : StoredCarrierBinding D rust P runtime) :
    deployedFailure runtime ≤ lazyMean (projected D P) emptyTable failureMass := by
  apply OracleProgramLawExt.lazyMean_le_of_eval
  intro H
  unfold runtimeMass
  cases hr : (eval H runtime).2 with
  | none => exact failure_nonnegative _
  | some run =>
      change failureMass (view run) ≤ failureMass (eval H (projected D P))
      by_cases accepted : run.2.2 = true
      · have same := binding H run hr accepted
        rw [annotatedWith_eq D rust join] at same
        rw [same,projected_eval]
      · have zero : failureMass (view run) = 0 := by
          unfold failureMass
          simp only [view] at *
          simp [accepted,indicator,mean_const]
        rw [zero]
        exact failure_nonnegative _

/-! Compose actual memoized-oracle averaging with fresh private extractor
coins. This does not identify shared-ROM sampling with fresh row sampling.
The classifier hit mass and concrete accepted-model refinement remain named
inputs; no query factor or collision term is discarded. -/
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency true
set_option backward.isDefEq.respectTransparency.types true
noncomputable section
namespace AspisR1.RawRecordedOracleExtractionMassBin
open FS FS2
open AspisR1.RawCutWordsFromRunExecutableBin
open AspisR1.LiteralFreshBankAbortComposition
open AspisV8R19.DuplexFrames AspisV8R19.SourceDuplexStep
open AspisV8R19.MemoizedProgramLaw AspisV8R19.OracleResampling
open AspisV8R19.CausalFirstHitUnionBound AspisV8R19.AdaptiveFirstReadLaw

attribute [local instance] Classical.propDecidable
attribute [local irreducible] RawRecordedPublicSpendExtractorBin.extract
  GSConcreteByteDecoderBank.failed LiteralClassifier.bad

abbrev Experiment := Program Bytes State ((LiteralTranscript.Public ×
  (Nat → LiteralTranscript.Message)) × Bool)

/-- Acceptance in the chosen concrete experiment must establish terminal
checks and the matching no-hit statement for this exact recorded run. The
observable `hit` must be refined to that classifier event; it is not given
fresh-round probabilities by this interface. -/
def CleanAcceptedModel (decoder : LiteralByteProtocol.Decoder)
    (exp : Experiment) (hit : RawCutProverCompiler.RunView → Prop) : Prop :=
  ∀ H : Bytes → State,
    let run := eval H exp
    run.2.2 = true → ¬ hit run →
      ∃ _checks : LiteralAcceptanceTerminal.Checks run.2.1.1
        ((LiteralByteProtocol.protocol decoder).transcript H run.2.1.1
          (RawCutTranscriptRun.proof (wordsFromRun run) run.2.1.2) 33).rounds,
        ¬ GenericNoHit.hitFrom LiteralClassifier.bad run.2.1.1 []
          ((LiteralByteProtocol.protocol decoder).transcript H run.2.1.1
            (RawCutTranscriptRun.proof (wordsFromRun run) run.2.1.2) 33).rounds

def failureMass (run : RawCutProverCompiler.RunView) : ℚ :=
  mean (fun coins : GSConcreteByteDecoderBank.RawCoins =>
    indicator (run.2.2 = true ∧ RawRecordedPublicSpendExtractorBin.extract run coins = none))

private theorem failureMass_le_one (run : RawCutProverCompiler.RunView) :
    failureMass run ≤ 1 := by
  unfold failureMass
  apply le_trans (mean_mono ?_) (by rw [mean_const])
  intro coins
  by_cases h : run.2.2 = true ∧ RawRecordedPublicSpendExtractorBin.extract run coins = none
  <;> simp [indicator, h]

/-- The terminal private-coin failure probability is bounded under the real
lazy memoized oracle law by the classifier hit mass plus the uniform actual
byte-bank abort envelope. This is conditional on CleanAcceptedModel; it makes
no complete source-acceptance or classifier-density claim. -/
theorem shared_oracle_failure_le_hit_add_abort
    (decoder : LiteralByteProtocol.Decoder) (exp : Experiment)
    (hit : RawCutProverCompiler.RunView → Prop)
    (hmodel : CleanAcceptedModel decoder exp hit) :
    lazyMean exp emptyTable failureMass ≤
      lazyMean exp emptyTable (fun run => indicator (hit run)) +
        (1 / (2 : ℚ)^126) := by
  have hpoint : ∀ H : Bytes → State,
      failureMass (eval H exp) ≤ indicator (hit (eval H exp)) +
        (1 / (2 : ℚ)^126) := by
    intro H
    let run := eval H exp
    change failureMass run ≤ indicator (hit run) + (1 / (2 : ℚ)^126)
    by_cases hh : hit run
    · have hle := failureMass_le_one run
      have hpos : (0 : ℚ) ≤ 1 / (2 : ℚ)^126 := by positivity
      simpa only [indicator, if_pos hh] using
        hle.trans (le_add_of_nonneg_right hpos)
    · by_cases ha : run.2.2 = true
      · obtain ⟨checks, hno⟩ := hmodel H ha hh
        have hle : failureMass run ≤ bankAbortMass
            (AspisWide.GSConcreteReceivedWordInputs.fromWords (initialWordsFromRun run)
              (fun _ => finalWordFromRun run)) := by
          apply mean_mono
          intro coins
          apply indicator_mono
          rintro ⟨_, hf⟩
          by_contra hb
          obtain ⟨w, hw, _⟩ :=
            RawRecordedPublicSpendExtractorBin.extract_of_checks_noHit_bank
              decoder H run checks hno coins hb
          rw [hf] at hw
          cases hw
        have hb := bankAbortMass_le
          (AspisWide.GSConcreteReceivedWordInputs.fromWords (initialWordsFromRun run)
            (fun _ => finalWordFromRun run))
        simpa only [indicator, if_neg hh, zero_add] using hle.trans hb
      · have hz : failureMass run = 0 := by
          unfold failureMass
          simp [ha, indicator, mean_const]
        rw [show failureMass (eval H exp) = 0 from hz]
        simp only [show ¬ hit (eval H exp) from hh, indicator, if_false, zero_add]
        positivity
  calc
    _ ≤ lazyMean exp emptyTable (fun run => indicator (hit run) +
        (1 / (2 : ℚ)^126)) :=
      AllAddressFiatShamir.lazyMean_mono_of_eval exp _ _ hpoint
    _ = _ := by rw [lazyMean_add, lazyMean_const]

/-- Any actual shared-ROM classifier-hit bound transfers unchanged, with
only the independent extractor-abort envelope added. In particular this
lemma does not replace a query-dependent bound by a 33-round bound. -/
theorem shared_oracle_failure_le_supplied_hit_bound
    (decoder : LiteralByteProtocol.Decoder) (exp : Experiment)
    (hit : RawCutProverCompiler.RunView → Prop)
    (hmodel : CleanAcceptedModel decoder exp hit) (bound : ℚ)
    (hhit : lazyMean exp emptyTable (fun run => indicator (hit run)) ≤ bound) :
    lazyMean exp emptyTable failureMass ≤ bound + (1 / (2 : ℚ)^126) :=
  (shared_oracle_failure_le_hit_add_abort decoder exp hit hmodel).trans
    (add_le_add hhit le_rfl)
