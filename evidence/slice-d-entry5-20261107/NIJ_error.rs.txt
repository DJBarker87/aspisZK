fn writer_parse(w:&mut String,s:&str)->core::fmt::Result {
 let old=w.len(); assert!(old<2); assert!(w.capacity()>=2);
 if old==0 {assert!(s=="r1:error:");}else{assert!(s=="Parse");}
 unsafe {w.as_mut_vec().as_mut_ptr().add(old).write(if old==0{1}else{2});w.as_mut_vec().set_len(old+1);}
 Ok(())
}
