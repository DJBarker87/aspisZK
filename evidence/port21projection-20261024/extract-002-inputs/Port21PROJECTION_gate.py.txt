#!/usr/bin/env python3
"""Read-only, all-hierarchy admission. Missing/raced data denies admission.

Finite populated ancestors reserve once; inactive groups reserve nothing.
Uncapped process RSS is separate from hierarchical memory.current/page cache.
No service, cache or cgroup is mutated. Run twice immediately before launch.
"""
import hashlib, json, os, pathlib, sys, time, select

G = 1024**3
ROOT = pathlib.Path('/sys/fs/cgroup')

def hierarchy(root=ROOT):
    rows = []
    def visit(p, cover=None):
        rel = str(p.relative_to(root))
        events = dict(x.split() for x in (p/'cgroup.events').read_text().splitlines()) if p != root else {'populated':'1'}
        pop = int(events['populated'])
        assert pop in (0, 1)
        maximum = (p/'memory.max').read_text().strip() if (p/'memory.max').exists() else 'max'
        if p != root and 'memory' in (p.parent/'cgroup.subtree_control').read_text().split():
            assert (p/'memory.max').is_file(), rel
        cap = None if maximum == 'max' else int(maximum)
        row = dict(path=rel, populated=pop, maximum=maximum, covering_ancestor=cover)
        if cap is not None and pop and cover is None:
            assert cap > 0
            row['reservation_bytes'] = cap
            cover = rel
        row['effective_cover'] = cover
        # Read direct IDs even for groups with no memory controller.
        kind = (p/'cgroup.type').read_text().strip() if p != root else 'domain'
        row['direct_ids'] = (p/('cgroup.threads' if kind == 'threaded' else 'cgroup.procs')).read_text().split()
        if (p/'memory.current').exists():
            row['memory_current_diagnostic_only'] = int((p/'memory.current').read_text())
        rows.append(row)
        for child in sorted(p.iterdir()):
            if child.is_dir(): visit(child, cover)
    visit(root)
    return rows

def topology(rows):
    return [(r['path'], r['populated'], r['maximum'], r['covering_ancestor']) for r in rows]

def processes(rows):
    groups = {r['path']:r for r in rows}
    out, exited = [], []
    pids = {p.name for p in pathlib.Path('/proc').iterdir() if p.name.isdecimal()}
    pids.update(pid for r in rows for pid in r['direct_ids'])
    for pid in sorted(pids, key=int):
        p = pathlib.Path('/proc', pid)
        # Pin the kernel identity before reading any /proc fields. A readable
        # pidfd is authoritative EXITED evidence, not a missing-live-data waiver.
        try: fd = os.pidfd_open(int(p.name), 0)
        except OSError as e:
            if e.errno == 3:
                # Enumerated PID vanished before pinning. Revalidate both the
                # kernel identity lookup and /proc. A replacement LIVE/UNKNOWN
                # identity is not excluded.
                try: replacement = os.pidfd_open(int(p.name), 0)
                except ProcessLookupError:
                    assert not p.exists(), (p.name, 'ESRCH but proc identity exists')
                    exited.append(dict(pid=int(p.name), start=None,
                        evidence='enumerated PID; two kernel pidfd_open ESRCH; proc absent; exited before stat'))
                    continue
                else:
                    os.close(replacement)
                    raise RuntimeError((p.name,'replacement identity appeared'))
            if e.errno != 22: raise  # ESRCH/EPERM still deny admission
            status = dict(l.split(':',1) for l in (p/'status').read_text().splitlines() if ':' in l)
            leader = status['Tgid'].strip()
            assert leader != p.name and leader in pids, (p.name, 'unknown thread leader')
            out.append(dict(pid=int(p.name), group=next(l[3:].lstrip('/') or '.' for l in (p/'cgroup').read_text().splitlines() if l.startswith('0::')),
                            rss=0, swap=0, capped=True, thread_leader=int(leader), accounting='address space counted at leader'))
            continue
        poll = select.poll(); poll.register(fd, select.POLLIN)
        first = None
        try:
            try:
                stat = (p/'stat').read_text()
                fields = stat[stat.rindex(')')+2:].split()
                first = fields[19]
                group = next(l[3:].lstrip('/') or '.' for l in (p/'cgroup').read_text().splitlines() if l.startswith('0::'))
                assert group in groups, (p.name, group)
                status = dict(l.split(':',1) for l in (p/'status').read_text().splitlines() if ':' in l)
                # Kernel threads/zombies have no user address space. Other live
                # identities require both metrics; never default unreadable data.
                no_mm = fields[0] in ('Z','X') or status.get('Kthread','').strip() == '1'
                rss = int(status['VmRSS'].split()[0])*1024 if 'VmRSS' in status else (0 if no_mm else None)
                swap = int(status['VmSwap'].split()[0])*1024 if 'VmSwap' in status else (0 if no_mm else None)
                assert rss is not None and swap is not None, (p.name, 'missing live memory metrics')
                last = (p/'stat').read_text()
                assert last[last.rindex(')')+2:].split()[19] == first, (p.name,'identity changed')
            except FileNotFoundError:
                assert poll.poll(0), (p.name, 'unreadable LIVE/UNKNOWN identity')
                exited.append(dict(pid=int(p.name), start=first, evidence='pinned pidfd POLLIN: EXITED'))
                continue
            if poll.poll(0):
                exited.append(dict(pid=int(p.name), start=first, evidence='pinned pidfd POLLIN: EXITED after reads'))
                continue
            out.append(dict(pid=int(p.name), start=first, group=group, rss=rss, swap=swap,
                            capped=groups[group]['effective_cover'] is not None))
        finally: os.close(fd)
    return out, exited

def snapshot(new_bytes=2*G):
    before = hierarchy()
    procs, exited = processes(before)
    after = hierarchy()
    if topology(before) != topology(after):
        b={r[0]:r[1:] for r in topology(before)}
        a={r[0]:r[1:] for r in topology(after)}
        raise RuntimeError(json.dumps(dict(reason='cgroup topology/limits/population raced',
            changed={k:dict(before=b.get(k),after=a.get(k)) for k in b.keys()|a.keys() if b.get(k)!=a.get(k)})))
    # Reconcile cgroup IDs with a fresh identity-pinned process scan; new or
    # unreadable live identities deny admission. Threads use the same /proc ID path.
    known = {str(p['pid']) for p in procs} | {str(p['pid']) for p in exited}
    for row in after:
        for pid in row['direct_ids']: assert pid in known, ('unscanned cgroup identity', pid)
    mem = {l.split(':')[0]:int(l.split()[1])*1024 for l in pathlib.Path('/proc/meminfo').read_text().splitlines() if l.split()[1].isdigit()}
    reserved = sum(r.get('reservation_bytes',0) for r in after)
    uncapped = sum(p['rss'] for p in procs if not p['capped'])
    # Shared RSS can overcount; never add parent/child memory.current values.
    resident = sum(p['rss'] for p in procs)
    admitted = (reserved+new_bytes < 55*G and
                reserved+new_bytes+uncapped+7*G < mem['MemTotal'] and
                resident+new_bytes+7*G < mem['MemTotal'] and
                mem['MemAvailable'] >= new_bytes+7*G)
    return dict(schema='Port21PROJECTION-admission-v2',timestamp=time.time(),pid=os.getpid(),
                all_cgroups=after,all_processes=procs,authoritatively_exited=exited,
                reserved_bytes=reserved, resident_rss_bytes=resident,
                uncapped_rss_bytes=uncapped,os_reserve_bytes=7*G,new_bytes=new_bytes,
                meminfo=mem,admitted=admitted,print_axioms='N/A: read-only admission')

if __name__ == '__main__':
    try: result=snapshot()
    except Exception as e: result=dict(admitted=False,fail_closed=True,error=repr(e))
    print(json.dumps(result,indent=2))
    sys.exit(0 if result['admitted'] else 2)
