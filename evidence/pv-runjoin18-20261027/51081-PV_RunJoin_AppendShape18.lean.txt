def history : root_history.RootHistoryLocationV1 := { page_number := 0#u64,slot := 1#u16 }

theorem history_one : root_history.root_history_location 1#u64 = ok history := by
  rw [root_history.root_history_location,root_history.POOL_V1_ROOT_HISTORY_CAPACITY,
    root_history.POOL_V1_ROOT_HISTORY_CAPACITY_LOG2]
  have hcap : (1#usize <<< 8#u8) = ok 256#usize := by
    change UScalar.shiftLeft 1#usize 8 = _
    rw [UScalar.shiftLeft,if_pos (by change 8 < System.Platform.numBits; have := System.Platform.le_numBits; omega)]
    apply congrArg Result.ok
    apply UScalar.eq_of_val_eq
    change (1#usize.bv <<< 8).toNat = 256
    rw [BitVec.toNat_shiftLeft]
    change 256 % 2^UScalarTy.Usize.numBits = 256
    apply Nat.mod_eq_of_lt
    scalar_tac
  rw [hcap]
  simp only [bind_tc_ok,lift]
  have hc : UScalar.cast .U64 256#usize = 256#u64 := by
    apply UScalar.eq_of_val_eq
    rw [UScalar.cast_val_eq]
    rfl
  rw [hc]
  have hd : (1#u64 / 256#u64) = ok 0#u64 := by
    obtain ⟨d,hd,hv⟩ := UScalar.div_spec 1#u64 (y := 256#u64) (by decide)
    have he : d = 0#u64 := UScalar.eq_of_val_eq hv
    exact hd.trans (congrArg Result.ok he)
  have hm : (1#u64 % 256#u64) = ok 1#u64 := by
    obtain ⟨d,hd,hv⟩ := WP.spec_imp_exists (UScalar.rem_spec 1#u64 (y := 256#u64) (by decide))
    have he : d = 1#u64 := UScalar.eq_of_val_eq hv
    exact hd.trans (congrArg Result.ok he)
  rw [hd]
  simp only [bind_tc_ok]
  rw [hm]
  rfl
#print axioms history_one

    (hr : incremental_merkle.IncrementalMerkleTreeV1.reconstruct_nonfull_root_from_level
      (after leaf (Array.repeat 8#usize aspis_core.field.M31.ZERO))
        PV_Src_CanonicalRoots.table 0#usize = ok (.Ok root)) :
    incremental_merkle.tree_reference.OwnedTree.append_one owned leaf =
      ok (.Ok ({ inner := after leaf root, empty := PV_Src_CanonicalRoots.table },receipt root)) := by
  rw [incremental_merkle.tree_reference.OwnedTree.append_one,PV_RunJoin_ZeroConstructor18.capacity]
  simp only [bind_tc_ok]
  rw [if_neg (by decide)]
  simp only [owned,PV_RunJoin_ZeroConstructor18.tree]
  have hz : PV_RunJoin_FixtureArguments.snapshot.next_pair_index = 0#u64 := rfl
  rw [hz,carry]
  simp only [bind_tc_ok,lift,format.POOL_V1_TREE_DEPTH]
  rw [if_neg (by decide)]
  have hu : Array.update PV_RunJoin_FixtureArguments.snapshot.frontier 0#usize leaf = ok (frontier leaf) := by
    simp [Array.update,frontier]
  rw [hu]
  simp only [bind_tc_ok]
  change (do
    let r ← incremental_merkle.IncrementalMerkleTreeV1.reconstruct_nonfull_root_from_level
      (after leaf (Array.repeat 8#usize aspis_core.field.M31.ZERO)) PV_Src_CanonicalRoots.table 0#usize

    match cf with
    | core.ops.control_flow.ControlFlow.Continue val => do
      let loc ← root_history.root_history_location 1#u64
      ok (core.result.Result.Ok ({ inner := after leaf val,empty := PV_Src_CanonicalRoots.table },
        { leaf_index := 0#u64,root_sequence := 1#u64,root := val,history := loc }))
    | core.ops.control_flow.ControlFlow.Break residual =>
      core.result.Result.Insts.CoreOpsTryTraitFromResidualResultInfallible.from_residual _
        (core.convert.FromSame incremental_merkle.PoolV1TreeError) residual) = _
  rw [hr]
  simp only [bind_tc_ok,core.result.Result.Insts.CoreOpsTry.branch]
  rw [history_one]
  rfl
