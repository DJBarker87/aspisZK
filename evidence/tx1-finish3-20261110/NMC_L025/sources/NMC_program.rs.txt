fn run<const TAG:u8>(site:u8,bad:bool){
 let mut p=parent();let tag:u8=if TAG==255{kani::any()}else{TAG};let i:usize=kani::any();let j:usize=kani::any();kani::assume(tag<23&&i<87&&j<4);
 let before=observe(&p,tag,i,j);
 match site{
 0=>{let good:bool=kani::any();let value=q();let reply=if good{Ok(value)}else{Err(public_error())};
 let expected=if tag==3{word(value,j)as u64}else{before};let out=S::map_public(&mut p,reply);
 let law=match out{Ok(())=>good&&observe(&p,tag,i,j)==expected,Err(e)=>!good&&e==Error::Public&&observe(&p,tag,i,j)==before};check(law,bad)},
 1=>{let masked=q();let reject=p.running!=masked;let out=S::terminal_guard(&p,masked);let law=match out{Ok(())=>!reject&&observe(&p,tag,i,j)==before,Err(e)=>reject&&e==Error::Terminal&&observe(&p,tag,i,j)==before};check(law,bad)},
 2=>{let reject=p.z0==p.z1;let out=S::circle_guard(&p);let law=match out{Ok(())=>!reject&&observe(&p,tag,i,j)==before,Err(e)=>reject&&e==Error::EqualCirclePoints&&observe(&p,tag,i,j)==before};check(law,bad)},
 3|4=>{let at:usize=kani::any();kani::assume(at<if site==3{16}else{10});let expected=word(p.claims[0][if site==3{at}else{16+at}],j);let mut slot=core::mem::MaybeUninit::uninit();if site==3{S::low(&p,at,&mut slot)}else{S::high(&p,at,&mut slot)};let actual=unsafe{slot.assume_init()};check(word(actual,j)==expected&&observe(&p,tag,i,j)==before,bad)},
 5|6=>{let expected=word(if site==5{p.claims[0][27]}else{p.extra[28]},j);let value=if site==5{S::blind(&p)}else{S::extra28(&p)};check(word(value,j)==expected&&observe(&p,tag,i,j)==before,bad)},
 7=>{let expected=if tag==7{NLI_factory::snapshot(&p.transcript).0[i%32]as u64}else{before};S::state_before(&mut p);check(observe(&p,tag,i,j)==expected,bad)},
 8|9=>{let good:bool=kani::any();let values=core::array::from_fn(|_|q());let err=error();let reply=if good{Ok(values)}else{Err(err)};let expected=if tag==if site==8{5}else{6}{word(values[i%29],j)as u64}else{before};let out=S::decoded(&mut p,reply,site==9);let law=match out{Ok(())=>good&&observe(&p,tag,i,j)==expected,Err(e)=>!good&&e==err&&observe(&p,tag,i,j)==before};check(law,bad)},
 _=>unreachable!(),
 }
}
