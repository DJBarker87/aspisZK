
// Exact source index>>9, versus the library's unsigned literal type.
// This checks translator-literal fidelity, not another Slice-Q function row.
#[kani::proof]
fn NEW_shift_literal(){
 let index:usize=kani::any();let mut actual=index>>9;
 #[cfg(NET_bad_reply)] {actual^=1;}
 assert_eq!(actual,index>>9u32);
}
