fn NET_gather(flat:&mut Vec<M31>,lines:&mut Vec<M31>,values:[M31;5])->Result<(),Error>{
 if values.iter().any(|x|*x==M31::ZERO){
  #[cfg(NET_bad_option)] {return Ok(());}
  return Err(Error::Query);
 }
 flat.extend_from_slice(&values);
 lines.push(values[4]);

}

#[derive(Clone,Copy)]
struct NET_BackEvent{a:u32,b:u32,reply:u32}
struct NET_BackLog{events:[NET_BackEvent;2],next:usize}
fn NET_back_product(a:M31,b:M31,reply:M31,log:&mut NET_BackLog)->M31{
 let n=log.next;log.events[n]=NET_BackEvent{a:a.0,b:b.0,reply:reply.0};log.next+=1;reply
}
fn NET_back_step(flat:&Vec<M31>,prefixes:&Vec<M31>,inverse:&mut Vec<M31>,suffix:M31,i:usize,reply:[u32;2],log:&mut NET_BackLog)->M31{
 inverse[i]=NET_back_product(suffix,prefixes[i],M31(reply[0]),log);
 let suffix=NET_back_product(suffix,flat[i],M31(reply[1]),log);
 suffix
}

// One native std Iterator::next and the frozen zero callback on arbitrary pre-state.
#[derive(Clone,Copy)]
struct NEW_ZeroReply { kind:u8, stop:bool, word:u32, rest:usize, pointer:*const M31 }
fn NEW_zero_iteration(values:&[M31;5],i:usize)->NEW_ZeroReply{
 let mut it=values[i..].iter();
 let (kind,stop,word)=match it.next(){
  None=>(0,false,0),
  Some(value)=>{let stop=*value==M31::ZERO;(if stop{2}else{1},stop,value.0)}
 };
 NEW_ZeroReply{kind,stop,word,rest:it.len(),pointer:it.as_slice().as_ptr()}
}
fn NEW_zero_iteration_case(mode:u8){
 let raw:[u32;5]=kani::any();let mut values=[M31(raw[0]),M31(raw[1]),M31(raw[2]),M31(raw[3]),M31(raw[4])];
 let i:usize=kani::any();kani::assume(i<=5);
 if mode==1{kani::assume(i==5);}
 if mode==2{kani::assume(i<5);kani::assume(raw[i]!=0);}
 if mode==3{kani::assume(i<5);kani::assume(raw[i]==0);}
 let j:usize=kani::any();kani::assume(j<5);
 let mut reply=NEW_zero_iteration(&values,i);
 #[cfg(NET_bad_reply)] {reply.stop=!reply.stop;}
 #[cfg(NET_bad_argument)] {reply.rest^=1;}
 #[cfg(NET_bad_frame)] {values[j].0^=1;}
 #[cfg(NEW_wrong_getter)] {if i<5{reply.word^=1;}}
 #[cfg(NEW_wrong_wrapper)] {reply.kind^=4;}
 let taken=usize::from(i<5);let next=i+taken;
 let word=if i<5{raw[i]}else{0};let stop=i<5&&word==0;
 let kind=if i==5{0}else if stop{2}else{1};
 assert_eq!(reply.stop,stop);assert_eq!(reply.kind,kind);assert_eq!(reply.word,word);
 assert_eq!(reply.rest,5-next);assert_eq!(reply.pointer,unsafe{values.as_ptr().add(next)});
 assert_eq!(values[j].0,raw[j]);
}
