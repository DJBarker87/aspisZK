fn NET_gather(flat:&mut Vec<M31>,lines:&mut Vec<M31>,values:[M31;5])->Result<(),Error>{
 if values.iter().any(|x|*x==M31::ZERO){
  #[cfg(NET_bad_option)] {return Ok(());}
  return Err(Error::Query);
 }
 flat.extend_from_slice(&values);
 lines.push(values[4]);
 Ok(())
}

#[derive(Clone,Copy)]
struct NET_BackEvent{a:u32,b:u32,reply:u32}
struct NET_BackLog{events:[NET_BackEvent;2],next:usize}
fn NET_back_product(a:M31,b:M31,reply:M31,log:&mut NET_BackLog)->M31{
 let n=log.next;log.events[n]=NET_BackEvent{a:a.0,b:b.0,reply:reply.0};log.next+=1;reply
}
fn NET_back_step(flat:&Vec<M31>,prefixes:&Vec<M31>,inverse:&mut Vec<M31>,suffix:M31,i:usize,reply:[u32;2],log:&mut NET_BackLog)->M31{
 inverse[i]=NET_back_product(suffix,prefixes[i],M31(reply[0]),log);
 let suffix=NET_back_product(suffix,flat[i],M31(reply[1]),log);
 suffix
}
