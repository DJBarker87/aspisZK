#[derive(Clone,Copy,Debug,PartialEq,Eq)]
struct State {sums:[u64;8],count:usize}

fn state(d:&EbyKDot)->State{let (sums,count)=carrier::get(d);State{sums,count}}
fn words(q:K)->[u32;4]{[q.c0.a.0,q.c0.b.0,q.c1.a.0,q.c1.b.0]}
fn q(w:[u32;4])->K{K{c0:CM31{a:M31(w[0]),b:M31(w[1])},c1:CM31{a:M31(w[2]),b:M31(w[3])}}}

// All state and both Result arms are arbitrary parameters; no child success premise.
fn oracle(d:&mut EbyKDot,v:E,s:K,post:State,reply:bool,log:&mut Log)->Result<(),()>{
 *log=Log{value:v.to_limbs(),scalar:words(s),before:state(d),after:post,reply,calls:1};
 #[cfg(NGY_corrupt)] {log.scalar[0]^=1;}
 *d=carrier::from(post.sums,post.count);
 if reply{Ok(())}else{Err(())}
}
fn reference_word(bytes:&[u8],at:usize)->u32{u32::from_le_bytes([bytes[at],bytes[at+1],bytes[at+2],bytes[at+3]])}
fn reference_scalar(bytes:&[u8],l:usize)->K{let at=32*l;q([reference_word(bytes,at),reference_word(bytes,at+4),reference_word(bytes,at+8),reference_word(bytes,at+12)])}
// CLOSED pure field child, full eight-word natural spec; paired NGY_FieldSpec.source.
fn field_spec(bytes:&[u8],l:usize)->E{

 for j in 0..8{kani::assume(w[j]==reference_word(bytes,32*l+4*j));}

}

}
