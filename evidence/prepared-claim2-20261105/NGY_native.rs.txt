#[derive(Clone,Copy,Debug,PartialEq,Eq)]
struct State {sums:[u64;8],count:usize}
#[derive(Clone,Copy,Debug,PartialEq,Eq)]
struct Log {value:[u32;8],scalar:[u32;4],before:State,after:State,reply:bool,calls:u8}
#[derive(Clone,Copy,Debug,PartialEq,Eq)]
struct Out {result:Result<(),Error>,parse:bool,next:usize,state:State,log:Log}
fn state(d:&EbyKDot)->State{let (sums,count)=carrier::get(d);State{sums,count}}
fn words(q:K)->[u32;4]{[q.c0.a.0,q.c0.b.0,q.c1.a.0,q.c1.b.0]}
fn q(w:[u32;4])->K{K{c0:CM31{a:M31(w[0]),b:M31(w[1])},c1:CM31{a:M31(w[2]),b:M31(w[3])}}}
fn empty()->Log{Log{value:[0;8],scalar:[0;4],before:State{sums:[0;8],count:0},after:State{sums:[0;8],count:0},reply:false,calls:0}}
// All state and both Result arms are arbitrary parameters; no child success premise.
fn oracle(d:&mut EbyKDot,v:E,s:K,post:State,reply:bool,log:&mut Log)->Result<(),()>{
 *log=Log{value:v.to_limbs(),scalar:words(s),before:state(d),after:post,reply,calls:1};
 #[cfg(NGY_corrupt)] {log.scalar[0]^=1;}
 *d=carrier::from(post.sums,post.count);
 if reply{Ok(())}else{Err(())}
}
// Hand reference: selected operands + full poststate + Parse/continue/control/log.
fn f_ref<H:Digest32>(sem:&crate::r1_semantic::Verified<H>,y1:&[u8],g:&[E;29],point:usize,l:usize,pre:State,post:State,reply:bool)->Out{
 let scalar=if point<3{sem.claims[point][l]}else if point==3{sem.extra[l]}else if point==4{sem.before_z1[l]}else{field(y1,l).c0()};
 Out{result:if reply{Ok(())}else{Err(Error::Parse)},parse:!reply,next:if reply{l+1}else{l},state:post,log:Log{value:g[l].to_limbs(),scalar:words(scalar),before:pre,after:post,reply,calls:1}}
}
