theorem loop (is : List Usize) (bound : ∀ i ∈ is,i.val<84) :
    ∃ xs,OraclePortStdBuilder.fromFnLoop fn is claims=ok (xs,claims) ∧
      xs.length=is.length ∧ ∀ q ∈ xs,JFu_TerminalCanonical.Canon q := by
  induction is with
  | nil => exact ⟨[],rfl,rfl,by simp⟩
  | cons i is ih =>
    obtain ⟨q,call,canonical⟩ := JFu_TerminalCallbackCanonical.source i (bound i (by simp))
    obtain ⟨xs,tail,length,allCanonical⟩ := ih (by intro j hj;exact bound j (by simp only [List.mem_cons];exact Or.inr hj))
    refine ⟨q::xs,?_,by simp only [List.length_cons,length],?_⟩
    · simp only [OraclePortStdBuilder.fromFnLoop,fn,call,bind_tc_ok,Aeneas.Std.uncurry,tail]
    · intro x hx
      rcases List.mem_cons.mp hx with same|mem
      · exact same ▸ canonical
      · exact allCanonical x mem
