theorem bindB_assoc {A B C : Type} (r : BRun A)
 (f : A → Buffer → Cursor → BRun B) (g : B → Buffer → Cursor → BRun C) :
 bindB (bindB r f) g=bindB r (fun a b c => bindB (f a b c) g) := by
 cases r with
 | mk ex b c =>
  cases ex with
  | mk es re =>
   cases re with
   | ok a =>
    cases h : f a b c with
    | mk fex fb fc =>
     cases fex with
     | mk fs fre => cases fre <;> simp only [bindB,h,List.append_assoc]
   | error e => rfl
   | cut => rfl
#print axioms bindB_assoc

theorem jsonStream_append (sig : Signal) (fuel : Nat) (xs ys : List Span) :
 ∀ b c, jsonStream sig fuel (xs++ys) b c=
 bindB (jsonStream sig fuel xs b c) (fun _ b c => jsonStream sig fuel ys b c) := by
 induction xs with
 | nil => intro b c; simp only [List.nil_append,jsonStream,bindB_pure]
 | cons s xs ih =>
  intro b c
  simp only [List.cons_append,jsonStream,bindB_assoc]
  apply congrArg (bindB (lineAll 1 sig fuel s b c))
  funext u b c
  exact ih b c
#print axioms jsonStream_append

theorem onePrintln_chunks_append (sig : Signal) (fuel : Nat) (xs ys : List Span)
 (newline : Span) (b : Buffer) (c : Cursor) :
 onePrintln sig fuel (xs++ys) newline b c=
 bindB (jsonStream sig fuel xs b c) (fun _ b c => onePrintln sig fuel ys newline b c) := by
 rw [onePrintln,jsonStream_append,bindB_assoc]
 rfl

end PV_DIAG12
