def AspisR1.S2Trust.Valid : {F : Type} → AspisR1.S2Trust.Heap F → Prop :=
fun {F} s =>
  List.Pairwise (fun g h => g.token ≠ h.token ∧ g.owner ≠ h.owner) s.guards ∧
    (∀ (c : ℕ),
        AspisR1.S2Trust.minCounter ≤ s.counter c ∧
          s.counter c ≤ AspisR1.S2Trust.maxCounter ∧
            s.counter c =
                s.external c + AspisR1.S2Trust.count s c AspisR1.S2Trust.Kind.shared -
                  AspisR1.S2Trust.count s c AspisR1.S2Trust.Kind.exclusive ∧
              (AspisR1.S2Trust.count s c AspisR1.S2Trust.Kind.exclusive > 0 →
                  s.external c = 0 ∧
                    AspisR1.S2Trust.count s c AspisR1.S2Trust.Kind.shared = 0 ∧
                      AspisR1.S2Trust.count s c AspisR1.S2Trust.Kind.exclusive = 1) ∧
                (s.external c < 0 →
                  AspisR1.S2Trust.count s c AspisR1.S2Trust.Kind.shared = 0 ∧
                    AspisR1.S2Trust.count s c AspisR1.S2Trust.Kind.exclusive = 0)) ∧
      (∀ g ∈ s.guards, s.allocated g.cell = Bool.true ∧ g.descriptor = (s.payload g.cell).descriptor) ∧
        ∀ (a : ℕ),
          s.ownerLive a = Bool.true → (s.accounts a).1 ≠ (s.accounts a).2 ∧ s.allocated (s.accounts a).1 = Bool.true
def AspisR1.S2Trust.Enabled : {F : Type} → AspisR1.S2Trust.Heap F → AspisR1.S2Trust.Op → Prop :=
fun {F} s x =>
  match x with
  | AspisR1.S2Trust.Op.shared a t o => s.ownerLive a = Bool.true ∧ AspisR1.S2Trust.Fresh s t o
  | AspisR1.S2Trust.Op.exclusive a t o => s.ownerLive a = Bool.true ∧ AspisR1.S2Trust.Fresh s t o
  | AspisR1.S2Trust.Op.deref t => ∃ g, AspisR1.S2Trust.lookup s t = Option.some g
  | AspisR1.S2Trust.Op.drop t => ∃ g, AspisR1.S2Trust.lookup s t = Option.some g
  | AspisR1.S2Trust.Op.write t i value =>
    ∃ g,
      AspisR1.S2Trust.lookup s t = Option.some g ∧
        g.kind = AspisR1.S2Trust.Kind.exclusive ∧ i < (s.payload g.cell).descriptor.length
  | AspisR1.S2Trust.Op.move t o => (∃ g, AspisR1.S2Trust.lookup s t = Option.some g) ∧ ∀ g ∈ s.guards, g.owner ≠ o
  | AspisR1.S2Trust.Op.errorMoveDrop => True
