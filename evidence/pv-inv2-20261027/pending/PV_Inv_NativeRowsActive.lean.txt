import R1Z.PV_Inv_RegistryInput
import R1Z.PV_Inv_ActiveRegistry
set_option autoImplicit false
namespace R1Z.PV_Inv_NativeRowsActive
open R0P PV_Inv_RegistryInput
variable {K : Type} [Field K]

def rows (registry : List (SourceLink K)) : List (Fin 1024) :=
  registry.flatMap (fun l => [l.producer.row,l.consumer.row])

theorem active (registry : List (SourceLink K)) (enumeration : P_registry registry)
    (r : Fin 1024) (hr : r ∈ rows registry) : CopyActiveRow r := by
  obtain ⟨l,hl,he⟩ := List.mem_flatMap.mp hr
  obtain ⟨m,hm,eq⟩ := PV_Inv_RegistryInput.member registry enumeration l hl
  subst l
  simp only [List.mem_cons,List.not_mem_nil,or_false] at he
  rcases he with rfl|rfl
  · exact (PV_Inv_ActiveRegistry.active m hm).1
  · exact (PV_Inv_ActiveRegistry.active m hm).2

#print axioms active
end R1Z.PV_Inv_NativeRowsActive
